11 /**
12  * Conservative linear interpolation that does not mix cells of different types, in the spirit
13  * of WRF's land/water-masked nest interpolation (erf-model/ERF#4390). Every cell is
14  * interpolated; a cell's type decides only which neighbours its slopes and limiter may see.
15  * The types are whatever the caller encodes: land and water (SLM's state), a valid value and
16  * the lsm_undefined sentinel (Noah-MP), or both at once (SLM's fluxes).
17  *
18  * The coarse data carries nfield fields followed by nind type indicators (one value per type;
19  * NaN, no type, outside the domain). With nind == 1 every field shares the one indicator; with
20  * nind == nfield field n has indicator n. In each direction the slope of a coarse cell is taken
21  * only if both of its neighbours are of the same type as the cell; otherwise it is zero, as it
22  * is at a domain boundary. The min/max limiter likewise looks only at neighbours of the same
23  * type. Where every neighbour is of the same type this is exactly AMReX's cell_cons_interp
24  * (CellConservativeLinear without linear limiting), so the result changes only next to a type
25  * boundary. Each coarse cell's mean is kept, and the indicator is copied (piecewise constant),
26  * so a cell whose neighbours are all of other types (an isolated sentinel, say) comes through
27  * exactly.
28  */

61  * Slopes of the field (component nu) of coarse cell (i,j,k), from neighbours of its own type only
62  * (indicator in component nm). Follows mf_cell_cons_lin_interp_mcslope term for term; the slope
63  * in direction d goes to component ns + d*nslope, the layout mf_cell_cons_lin_interp reads.
64  */
65 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
66 void typed_mcslope (int i, int j, int k, amrex::Array4<amrex::Real> const& slope,
67  int ns, int nslope,
68  amrex::Array4<amrex::Real const> const& u, int nu, int nm,
69  amrex::Box const& domain, amrex::IntVect const& ratio,
70  amrex::BCRec const& bc) noexcept
71 {
72  using amrex::Real;
73  // NaN (outside the domain) is never of the same type
74  const Real m0 = u(i,j,k,nm);
75  auto same = [&] (int ii, int jj, int kk) { return u(ii,jj,kk,nm) == m0; };
76 
77  Real sx = Real(0.);
78  Real sy = Real(0.);
79  Real sz = Real(0.);
80 
81  if (ratio[0] > 1 && same(i-1,j,k) && same(i+1,j,k)) {
82  Real dc = amrex::mf_compute_slopes_x(i, j, k, u, nu, domain, bc);
83  Real df = Real(2.0) * (u(i+1,j,k,nu) - u(i ,j,k,nu));
84  Real db = Real(2.0) * (u(i ,j,k,nu) - u(i-1,j,k,nu));
85  sx = (df*db >= Real(0.0)) ? amrex::min(std::abs(df),std::abs(db)) : Real(0.);
86  sx = std::copysign(Real(1.),dc)*amrex::min(sx,std::abs(dc));
87  }
88  if (ratio[1] > 1 && same(i,j-1,k) && same(i,j+1,k)) {
89  Real dc = amrex::mf_compute_slopes_y(i, j, k, u, nu, domain, bc);
90  Real df = Real(2.0) * (u(i,j+1,k,nu) - u(i,j ,k,nu));
91  Real db = Real(2.0) * (u(i,j ,k,nu) - u(i,j-1,k,nu));
92  sy = (df*db >= Real(0.0)) ? amrex::min(std::abs(df),std::abs(db)) : Real(0.);
93  sy = std::copysign(Real(1.),dc)*amrex::min(sy,std::abs(dc));
94  }
95  if (ratio[2] > 1 && same(i,j,k-1) && same(i,j,k+1)) {
96  Real dc = amrex::mf_compute_slopes_z(i, j, k, u, nu, domain, bc);
97  Real df = Real(2.0) * (u(i,j,k+1,nu) - u(i,j,k ,nu));
98  Real db = Real(2.0) * (u(i,j,k ,nu) - u(i,j,k-1,nu));
99  sz = (df*db >= Real(0.0)) ? amrex::min(std::abs(df),std::abs(db)) : Real(0.);
100  sz = std::copysign(Real(1.),dc)*amrex::min(sz,std::abs(dc));
101  }
102 
103  Real alpha = Real(1.0);
104  if (sx != Real(0.0) || sy != Real(0.0) || sz != Real(0.0)) {
105  Real dumax = std::abs(sx) * Real(ratio[0]-1)/Real(2*ratio[0])
106  + std::abs(sy) * Real(ratio[1]-1)/Real(2*ratio[1])
107  + std::abs(sz) * Real(ratio[2]-1)/Real(2*ratio[2]);
108  Real umax = u(i,j,k,nu);
109  Real umin = u(i,j,k,nu);
110  const int ilim = ratio[0] > 1 ? 1 : 0;
111  const int jlim = ratio[1] > 1 ? 1 : 0;
112  const int klim = ratio[2] > 1 ? 1 : 0;
113  for (int koff = -klim; koff <= klim; ++koff) {
114  for (int joff = -jlim; joff <= jlim; ++joff) {
115  for (int ioff = -ilim; ioff <= ilim; ++ioff) {
116  if (same(i+ioff,j+joff,k+koff)) {
117  umin = amrex::min(umin, u(i+ioff,j+joff,k+koff,nu));
118  umax = amrex::max(umax, u(i+ioff,j+joff,k+koff,nu));
119  }
120  }}}
121  // dumax > 0 here; this only keeps an if-converted divide below from dividing by zero
122  dumax = dumax == Real(0.) ? Real(1.) : dumax;
123  if (dumax * alpha > (umax - u(i,j,k,nu))) {
124  alpha = (umax - u(i,j,k,nu)) / dumax;
125  }
126  if (dumax * alpha > (u(i,j,k,nu) - umin)) {
127  alpha = (u(i,j,k,nu) - umin) / dumax;
128  }
129  }
130 
131  slope(i,j,k,ns ) = sx * alpha;
132  slope(i,j,k,ns+ nslope) = sy * alpha;
133  slope(i,j,k,ns+2*nslope) = sz * alpha;
134 }

225  * component it applies to every field, with src.nComp() components component n applies to field

228  * keep their values in dst. Every field takes the boundary condition bcs[bcscomp].
