21 /**
22  * Turbulent gradient mixing of spectral particles. One coefficient acts on
23  * every member of every bin group, and particle molecular (Brownian)
24  * diffusion is zero under every policy.
25  *
26  * HostPassiveScalar consumes ERF's Scalar_h/Scalar_v eddy channels, which are
27  * already density weighted and Schmidt scaled. Under ERF's LES closures they
28  * equal the Q channels that diffuse moisture, but the PBL schemes other than
29  * SHOC fill Q_v and not Scalar_v, so this policy is for LES closures only and
30  * enabling it must reject a PBL scheme. ConstantReference multiplies
31  * kinematic reference diffusivities by the face-averaged dry-air density.
32  */
33 enum class ParticleMixingPolicy { None, ConstantReference, HostPassiveScalar };

160 /**
161  * Terrain-fitted mixing only: the fraction theta in [0, 1] of one matched
162  * coordinate's nominal low-order diffusive rate `low` that a face carries,
163  * given that coordinate's native rate `high`: 1 for low = 0, 0 where high is
164  * zero or opposes low, |high| / |low| where it is smaller, and 1 otherwise,
165  * so theta low lies between zero and high. A nonfinite rate gives NaN, which
166  * the caller rejects. A face's group takes the least fraction of its
167  * coordinates.
168  */
169 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE amrex::Real
170 matched_diffusion_fraction (const amrex::Real high, const amrex::Real low) noexcept
171 {
172  if (!amrex::Math::isfinite(high) || !amrex::Math::isfinite(low)) {
173  return std::numeric_limits<amrex::Real>::quiet_NaN();
174  }
175  if (low == amrex::Real(0.0)) { return amrex::Real(1.0); }
176  if (high == amrex::Real(0.0) || (high > amrex::Real(0.0)) != (low > amrex::Real(0.0))) {
177  return amrex::Real(0.0);
178  }
179  const amrex::Real h = amrex::Math::abs(high);
180  const amrex::Real l = amrex::Math::abs(low);
181  return h < l ? h / l : amrex::Real(1.0);
182 }
