143 /**
144  * Read the init type that wrote a checkpoint, from the last erf.init_type assignment in its
145  * job_info. Before the marker, WRFInput stored SST and TSK as potential temperature and
146  * Metgrid as absolute temperature, so for a checkpoint without one this tells which of the
147  * two it holds. For one with a version 1 marker, which holds potential temperature, it
148  * tells which surface pressure they were converted with.
149  */
150 inline LegacyInitType
151 parse_legacy_init_type_from_job_info (std::istream& input)
152 {
153  constexpr const char* key = "erf.init_type";
154  constexpr std::size_t key_length = 13;
155  LegacyInitType result = LegacyInitType::Unknown;
156  std::string line;
157 
158  while (std::getline(input, line)) {
159  std::size_t pos = 0;
160  while (pos < line.size() &&
161  std::isspace(static_cast<unsigned char>(line[pos]))) {
162  ++pos;
163  }
164  if (line.compare(pos, key_length, key) != 0) {
165  continue;
166  }
167  pos += key_length;
168  if (pos < line.size() && !std::isspace(static_cast<unsigned char>(line[pos])) &&
169  line[pos] != '=') {
170  continue;
171  }
172  while (pos < line.size() &&
173  std::isspace(static_cast<unsigned char>(line[pos]))) {
174  ++pos;
175  }
176  if (pos == line.size() || line[pos] != '=') {
177  result = LegacyInitType::Unknown;
178  continue;
179  }
180  ++pos;
181  while (pos < line.size() &&
182  std::isspace(static_cast<unsigned char>(line[pos]))) {
183  ++pos;
184  }
185 
186  std::string value;
187  bool malformed = (pos == line.size());
188  if (!malformed && line[pos] == '"') {
189  ++pos;
190  const std::size_t end_quote = line.find('"', pos);
191  if (end_quote == std::string::npos) {
192  malformed = true;
193  } else {
194  value = line.substr(pos, end_quote - pos);
195  pos = end_quote + 1;
196  }
197  } else if (!malformed) {
198  const std::size_t value_begin = pos;
199  while (pos < line.size() &&
200  !std::isspace(static_cast<unsigned char>(line[pos]))) {
201  ++pos;
202  }
203  value = line.substr(value_begin, pos - value_begin);
204  }
205 
206  while (pos < line.size() &&
207  std::isspace(static_cast<unsigned char>(line[pos]))) {
208  ++pos;
209  }
210  if (malformed || pos != line.size()) {
211  result = LegacyInitType::Unknown;
212  continue;
213  }
214 
215  result = parse_legacy_init_type_value(value);
216  }
217 
218  return result;
219 }

316 /**
317  * Decide how to bring the SST_0 and TSK_0 of a checkpoint to absolute temperature.
318  *
319  * Arrays that hold potential temperature were converted with the surface pressure of the
320  * input they were read from, which only the init type that wrote them identifies. A WRFInput
321  * restart reads the first init file's PSFC at level 0 and, at finer levels, uses the level's
322  * own PSFC where covered and the recovered parent pressure elsewhere. The checkpoint does
323  * not record whether RemakeLevel replaced fine values with parent-interpolated potential
324  * temperature, so fine-level migration is approximate wherever that may have happened:
325  * pressure-dependent conversion and spatial interpolation do not commute. A Metgrid run
326  * converted them with its met_em surface pressure, which a restart does not read. Metgrid
327  * migration therefore uses restarted ground pressure on boxes that reach the ground and
328  * parent ground pressure for aloft-only boxes. Both old formats lack enough metadata to
329  * recover the original absolute temperatures exactly.
330  *
331  * @param provenance what the checkpoint holds and which init type wrote it
332  * @param restart_reads_wrfinput whether the restart is a WRFInput run, in a build with NetCDF
333  * @param checkpointname name of the checkpoint, for the reason of a rejection
334  */
335 inline SurfaceTemperatureMigrationPlan
336 plan_surface_temperature_migration (const SurfaceTemperatureProvenance& provenance,
337  const bool restart_reads_wrfinput,
338  const std::string& checkpointname)
339 {
340  SurfaceTemperatureMigrationPlan plan;
341  plan.provenance = provenance;
342  if (provenance.stored == StoredSurfaceTemperature::Absolute) {
343  return plan;
344  }
345 
346  const std::string holds = "Restart checkpoint '" + checkpointname + "' holds SST_0/TSK_0 " +
347  "(from AMR level " + std::to_string(provenance.first_level) + ")";
348  const std::string set_assertion = "Set erf." + std::string(legacy_init_type_key) +
349  " to the erf.init_type of the run that wrote it: ";
350  plan.method = SurfaceTemperatureMigration::Reject;
351  if (provenance.stored == StoredSurfaceTemperature::UnknownProvenance) {
352  plan.reason = holds + " without a surface-temperature contract marker, and its job_info "
353  "does not record the erf.init_type that wrote them. " + set_assertion +
354  "WRFInput, which stored them as potential temperature, or Metgrid, which "
355  "stored them as absolute temperature.";
356  return plan;
357  }
358 
359  switch (provenance.source) {
360  case LegacyInitType::WRFInput:
361  if (restart_reads_wrfinput) {
362  plan.method = SurfaceTemperatureMigration::InputPressure;
363  } else {
364  plan.reason = holds + " as potential temperature, which a WRFInput run converted with "
365  "the surface pressure (PSFC) of its wrfinput files. Recovering them "
366  "needs that PSFC: restart with erf.init_type = WRFInput, in a build "
367  "with NetCDF, and the erf.nc_init_file_<lev> of the original run.";
368  }
369  break;
370  case LegacyInitType::Metgrid:
371  plan.method = SurfaceTemperatureMigration::RestartPressure;
372  break;
373  case LegacyInitType::Unknown:
374  plan.reason = holds + " as potential temperature (surface-temperature contract version " +
375  std::to_string(provenance.marker_version) + "), but its job_info does not "
376  "record the erf.init_type that converted them, so the surface pressure "
377  "they were converted with is not known. " + set_assertion +
378  "WRFInput, to use available wrfinput PSFC (fine-level regrids may be "
379  "approximate), or Metgrid, to migrate them approximately with restarted "
380  "ground pressure and parent ground pressure for aloft-only levels.";
381  break;
382  }
383  return plan;
384 }

537 /**
538  * Convert a surface field (SST or TSK) from potential to absolute temperature with the
539  * surface pressure it was converted with, T = theta (p/p0)^(R_d/c_p). A ghost cell takes the
540  * pressure of the nearest valid cell of its box. If the ghost cannot be converted, a usable
541  * nearest valid temperature is converted to kelvin and copied into it before FillBoundary.
542  *
543  * Every cell that holds a finite, positive potential temperature is converted, whether or not
544  * the surface layer takes the field there, so that no value the run may come to use (after a
545  * regrid or an interpolation, say) is left as a potential temperature; that takes a finite,
546  * positive surface pressure. A cell that holds none is left as it is. Of the valid cells
547  * left unconverted:
548  *
549  * - one where the surface layer takes the field (see SurfaceTemperatureUse), or that holds a
550  * potential temperature it has no pressure to convert with, is counted, and the first one
551  * described, for abort_on_legacy_conversion_failure;
552  * - one where the surface layer does not take the field, and that holds no temperature, is
553  * a placeholder, such as a zero SST over land, or a zero TSK at a level without an SST:
554  * it is only counted as preserved. It is not made to look like a temperature, so should
555  * the field ever be taken there, the surface layer rejects it then.
556  *
557  * An invalid unused ghost placeholder is left as it is. A required ghost that cannot be
558  * reconstructed is counted and aborts the migration, just like an unconvertible valid cell.
559  *
560  * @param[in,out] field potential temperature in, absolute temperature [K] out
561  * @param[in] source which field it is: SurfaceTemperatureSource::sst or ::tsk
562  * @param[in] use where the surface layer takes it; the land mask, if any, on the
563  * layout of field
564  * @param[in] pressure surface pressure [Pa] on the valid cells of the layout of field
565  * @param[in] rdOcp R_d/c_p
566  * @param[in] period periodicity of the level
567  * @return the cells of this rank that were not converted or reconstructed
568  */
569 inline LegacyConversionFailure
570 potential_to_absolute_temperature (amrex::MultiFab& field,
571  const erf_surface_temperature::SurfaceTemperatureSource source,
572  const SurfaceTemperatureUse& use,
573  const amrex::MultiFab& pressure,
574  const amrex::Real rdOcp,
575  const amrex::Periodicity& period)
576 {
577  AMREX_ALWAYS_ASSERT(source != erf_surface_temperature::SurfaceTemperatureSource::prescribed);

596  : amrex::Array4<const int>{};

604  f(i,j,k) = temperature;

606  return;

611  land != 0, have_tsk, ignore_sst) == source;
612  if (!used && !erf_surface_temperature::valid_temperature(theta)) {

622  fail->ghost = 0;

632  const amrex::IntVect hi = vbx.bigEnd();

634  const amrex::Array4<const amrex::Real> p = pressure.const_array(mfi);
635  const amrex::Array4<const int> mask = (use.lmask) ? use.lmask->const_array(mfi)
636  : amrex::Array4<const int>{};

639  {

641  k >= lo[2] && k <= hi[2]) {
642  return;
643  }

645  const int lj = amrex::min(amrex::max(j, lo[1]), hi[1]);
646  const int lk = amrex::min(amrex::max(k, lo[2]), hi[2]);

648  const amrex::Real p_s = p(li,lj,lk);
649  amrex::Real temperature = theta;

651  f(i,j,k) = temperature;
652  was_converted(i,j,k) = 1;
653  return;
654  }

656  // An invalid temperature is not a positive potential temperature. If an
657  // authoritative neighbor was converted, copy its kelvin value into this
658  // physical-boundary/nest-edge ghost; otherwise preserve the invalid sentinel.
659  // Active invalid valid cells are reported by the valid-cell pass below/above.
660  if (!erf_surface_temperature::valid_temperature(theta)) {
661  if (was_converted(li,lj,lk) != 0) {
662  f(i,j,k) = f(li,lj,lk);
663  was_converted(i,j,k) = 1;
664  }
665  return;
666  }
667 
668  const int land = (mask) ? mask(li,lj,lk) : 1;
669  const bool used = have_sst &&
670  erf_surface_temperature::surface_temperature_source(
671  land != 0, have_tsk, ignore_sst) == source;
672  if (was_converted(li,lj,lk) != 0) {
673  f(i,j,k) = f(li,lj,lk);
674  was_converted(i,j,k) = 1;
675  return;
676  }

678  // A ghost that merely repeats a positive potential temperature from an
679  // unconvertible valid cell is covered by that valid-cell failure. Counting
680  // each duplicate physical-boundary ghost separately obscures the actionable
681  // owner location and cannot let a mixed-unit field proceed past the abort.
682  const amrex::Real owner_theta = f(li,lj,lk);
683  if (erf_surface_temperature::valid_temperature(owner_theta) &&
684  theta == owner_theta) {
685  return;
686  }

689  amrex::Gpu::Atomic::Add(&fail->ghost_count, 1);
690  if (amrex::Gpu::Atomic::CAS(&fail->claimed, 0, 1) == 0) {
691  fail->i = i;
692  fail->j = j;
693  fail->land = land;
694  fail->used = used ? 1 : 0;
695  fail->ghost = 1;
696  fail->theta = theta;
697  fail->pressure = p_s;
698  }
699  });
700  }
701  amrex::Gpu::streamSynchronize();

703  return d_fail.dataValue();

706 /**
707  * Say how many placeholders of a converted surface field were left as they are, and abort the
708  * restart if any rank could not convert a valid cell, after each rank that could not
709  * describes its first such cell. Collective: every rank must call this.
710  *
711  * @param[in] fail this rank's record, from potential_to_absolute_temperature
712  * @param[in] source which field was converted: SurfaceTemperatureSource::sst or ::tsk
713  * @param[in] what what was being converted, to open the messages with
714  */
715 inline void
716 abort_on_legacy_conversion_failure (const LegacyConversionFailure& fail,
717  const erf_surface_temperature::SurfaceTemperatureSource source,
718  const std::string& what)
719 {
720  const std::string name = surface_field_name(source);
721  int counts[3] = {fail.count, fail.preserved, fail.ghost_count};
722  amrex::ParallelDescriptor::ReduceIntSum(counts, 3);
723  const int count = counts[0];
724  const int preserved = counts[1];
725  const int ghosts = counts[2];
726  if (preserved > 0) {
727  amrex::Print() << what << ": kept " << preserved << " cells where the surface layer "
728  << "does not take the " << name << " as they are, placeholders without "
729  << "a finite, positive temperature\n";
730  }
731  if (count == 0) {
732  return;
733  }
734  if (fail.count > 0) {
735  // Assemble the whole report before writing it, so that the ranks do not interleave
736  std::ostringstream msg;
737  msg << std::setprecision(10) << what << " [rank " << amrex::ParallelDescriptor::MyProc()
738  << "]: " << fail.count << " cells cannot be converted; the first, (" << fail.i
739  << "," << fail.j << "), a " << ((fail.ghost != 0) ? "ghost " : "")
740  << ((fail.land != 0) ? "land" : "water") << " cell "
741  << "where the surface layer " << ((fail.used != 0) ? "takes" : "does not take")
742  << " the " << name << ", has theta = " << fail.theta << ", surface pressure = "
743  << fail.pressure << " Pa\n";
744  amrex::AllPrint() << msg.str() << std::flush;
745  }
746  amrex::ParallelDescriptor::Barrier();

748  std::to_string(ghosts) + " ghosts) could not be converted to "
749  "absolute temperature (see above). A cell where the surface layer takes the " +
750  name + " needs a finite, positive potential temperature, and any cell that "
751  "holds one a finite, positive surface pressure to convert it with");
752 }

780  }
781  }
782  });
