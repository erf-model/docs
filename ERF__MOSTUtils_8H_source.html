39 /**
40  * Lower bound of the log-law bracket ln(zref/z0) - psi of the MOST flux kernels: C/2 held
41  * between 1/2 and 1.
42  *
43  * The bound is 1 for zref >= e^2 z0 and C/2 down to zref = e z0, so it does not exceed the
44  * neutral bracket C wherever zref >= e^(1/2) z0. The lower limit of 1/2 keeps the bracket,
45  * and so u*, positive and finite when z0 approaches or exceeds zref (C <= 0).
46  *
47  * The bound is set by the under-relaxed u* iteration, not by the physics. At an unstable
48  * fixed point the relaxed map has slope (1 - alpha) - alpha s / bracket, with
49  * s = 3 (z/L) dpsi_m/d(z/L) = 3 (1 - phi_m) between 1.3 (z/L = -0.5) and 2.4 (z/L = -50) for
50  * Businger-Dyer.
51  * With alpha = 0.5 the iteration converges only for bracket > s/3 (0.4 to 0.8), so a bound of
52  * 1 keeps the switch between the bounded and free branches inside the convergent range, while
53  * 0.1 C (0.3 at C = 3) does not and the iteration oscillates between two values. If alpha
54  * changes, the bound needs revisiting; see PR #4866.
55  *
56  * @param[in] C ln(zref/z0)
57  */
58 AMREX_GPU_HOST_DEVICE
59 AMREX_FORCE_INLINE
60 amrex::Real
61 most_bracket_floor (amrex::Real C)
62 {
63  return amrex::max(amrex::min(amrex::Real(0.5) * C, amrex::Real(1.0)), amrex::Real(0.5));
64 }

228  /**
229  * Stable z/L at which the prescribed-flux problem folds.
230  *
231  * With L fixed by u* and the flux, the wind a given u* supports is
232  * u* max(C - psi_m, b) / kappa, with b = most_bracket_floor(C). Below the smallest such
233  * wind the prescribed stable flux has no solution with z/L below most_zeta_lim (Basu et
234  * al., BLM 2008; with QNSE the supported wind rises again past its first minimum to a
235  * local maximum at the second zero of f below, at z/L of order 1e3 to 1e4, and only then
236  * tends to zero with u*; the kernels never reach that range). The physical
237  * (large u*) branch has z/L below this value, so bounding z/L by it leaves every point that
238  * has a solution unchanged. The result is positive for every C.
239  *
240  * Businger-Dyer (psi_m = -beta_m z/L): for C > 0 the minimum is at z/L = C / (2 beta_m);
241  * for small or negative C (z0 near or above zref) the floor binds first and the minimum is
242  * at the kink z/L = (b - C) / beta_m.
243  *
244  * QNSE (psi_m = -c ln(1 + b_m z/L), c = (a_m - b_m)/b_m): with u* proportional to
245  * (z/L)^(-1/3), the supported wind falls with z/L while
246  * f(z/L) = C + c ln(1 + b_m z/L) - 3 c b_m (z/L) / (1 + b_m z/L) is positive, and f falls
247  * from C at z/L = 0 to its minimum at z/L = 2 / b_m and rises without bound after it. The
248  * fold is the first zero of f beyond the kink where the floor stops binding
249  * (C - psi_m = b), found by Newton's method on the bracket [kink, 2 / b_m] where f is
250  * convex and decreasing (f' = c b_m (b_m z/L - 2) / (1 + b_m z/L)^2 < 0, f'' > 0 for
251  * b_m z/L < 5): after the first step the iterates approach the root from the left, a step
252  * that would leave the bracket is replaced by its midpoint, and the iteration stops at a
253  * relative change of 1e-10 (a handful of steps against the 60 passes of a bisection to
254  * the same precision; the roughness kernels call this once per u* iteration). It is the
255  * kink itself when f is already negative there, and most_zeta_lim when f has no zero
256  * (C > c (2 - ln 3), 10.4 with the default coefficients): the wind then falls
257  * monotonically with z/L up to the bound, where the kernels hold z/L and reduce the flux
258  * as at a fold. Every return value is at most most_zeta_lim.
259  *
260  * @param[in] C ln(zref/z0)
261  */
262  AMREX_GPU_HOST_DEVICE
263  AMREX_FORCE_INLINE
264  amrex::Real
265  calc_zeta_fold (amrex::Real C) const
266  {
267  if (!m_qnse) {
268  return amrex::max(C / (two * beta_m), (most_bracket_floor(C) - C) / beta_m);
269  }
270  // A non-finite C (bad zref or z0) is returned as it is, as most_rib_zeta_bracketed does,
271  // so that it shows up in the kernel's results instead of a bisection artefact
272  if (!amrex::Math::isfinite(C)) { return C; }
273  const amrex::Real b = m_qnse_bm;
274  const amrex::Real c = (m_qnse_am - m_qnse_bm) / m_qnse_bm;
275  if (c <= zero) { return most_zeta_lim; } // a_m <= b_m is refused by the reader; psi_m = 0 would have no fold
276  const amrex::Real bf = most_bracket_floor(C);
277  amrex::Real lo = (C < bf) ? (std::exp((bf - C) / c) - one) / b : zero;
278  auto f = [&] (amrex::Real z) {
279  const amrex::Real s = one + b * z;
280  return C + c * std::log(s) - three * c * b * z / s;
281  };
282  auto df = [&] (amrex::Real z) {
283  const amrex::Real s = one + b * z;
284  return c * b * (b * z - two) / (s * s);
285  };
286  const amrex::Real zlim = most_zeta_lim; // by value: a namespace constant must not bind to min's reference in device code
287  if (f(lo) <= zero) { return amrex::min(lo, zlim); }
288  amrex::Real hi = two / b;
289  if (lo >= hi || f(hi) >= zero) { return zlim; }
290  // Safeguarded Newton on [lo, hi]: f(lo) > 0 > f(hi), f convex and decreasing, so from
291  // the second step on the iterates rise monotonically to the root; a step outside the
292  // bracket (the first one can be, from the midpoint) falls back to the midpoint
293  amrex::Real z = myhalf * (lo + hi);
294  constexpr amrex::Real rtol = amrex::Real(1.0e-10);
295  for (int n = 0; n < 30; ++n) {
296  const amrex::Real fz = f(z);
297  if (fz > zero) { lo = z; } else if (fz < zero) { hi = z; } else { break; }
298  const amrex::Real d = df(z);
299  amrex::Real zn = (d < zero) ? z - fz / d : myhalf * (lo + hi);
300  if (!(zn > lo && zn < hi)) { zn = myhalf * (lo + hi); }
301  const bool done = std::abs(zn - z) <= rtol * zn;
302  z = zn;
303  if (done) { break; }
304  }
305  return amrex::min(z, zlim);
306  }

485 /**
486  * Solve zeta = Rib (C - psi_m)^2 / (C - psi_h) of the prescribed-temperature kernels by
487  * bracketing and bisection, with the Jimenez functions and the brackets of most_bracket.
488  *
489  * The relaxed fixed-point iteration stalls where zeta - Rib F(zeta) nearly touches zero
490  * without crossing it, which the Jimenez stable functions produce when z0 is comparable to or
491  * above zref (C below about 1): each pass then moves zeta by about the tolerance. From zeta0
492  * this steps, with doubling steps, in the direction the iteration moves until the residual
493  * changes sign, and bisects that interval, so it returns the root the iteration was heading
494  * to, or the next one past a near-touch. The first step is limited so that two nearby roots
495  * are not stepped over together. The search stays within +-most_zeta_lim, and returns that
496  * bound when no root lies inside it, as the prescribed-flux kernels bound zeta.
497  *
498  * @param[in] sfuns similarity functions
499  * @param[in] C ln(zref/z0)
500  * @param[in] Rib bulk Richardson number
501  * @param[in] zeta0 z/L to start from
502  * @param[in] tol width of the final interval
503  */
504 AMREX_GPU_HOST_DEVICE
505 AMREX_FORCE_INLINE
506 amrex::Real
507 most_rib_zeta_bracketed (const similarity_funs& sfuns, amrex::Real C, amrex::Real Rib,
508  amrex::Real zeta0, amrex::Real tol)
509 {
510  auto resid = [&] (amrex::Real zeta) { return most_rib_resid(sfuns, C, Rib, zeta); };
511 
512  amrex::Real a = zeta0;
513  amrex::Real ra = resid(a);
514  if (ra == zero) { return a; }
515 
516  // A non-finite residual (a NaN or infinite C, Rib or zeta0) cannot be bracketed: with it the
517  // step below is NaN and the search never reaches zlim. Return it, so the bad input shows up
518  // in the caller's results rather than as a hang.
519  if (!amrex::Math::isfinite(ra)) { return ra; }
520 
521  // Step against the residual, within +-most_zeta_lim, until it changes sign
522  const amrex::Real dir = (ra > zero) ? amrex::Real(-1.0) : amrex::Real(1.0);
523  const amrex::Real zlim = dir * most_zeta_lim;
524  auto step = [&] (amrex::Real z, amrex::Real dz) {
525  return (dir > zero) ? amrex::min(z + dz, zlim) : amrex::max(z - dz, zlim);
526  };
527  amrex::Real h = amrex::max(amrex::min(std::abs(ra),
528  amrex::Real(0.1) * amrex::max(amrex::Real(one), std::abs(a))),
529  tol);
530  amrex::Real b = step(a, h);
531  amrex::Real rb = resid(b);
532  // With finite values the doubling step reaches zlim in well under 200 steps (from tol to
533  // 2 most_zeta_lim); the cap only guarantees that this loop ends
534  int nstep = 0;
535  while ((rb > zero) == (ra > zero) && b != zlim && nstep < 200) {
536  a = b;
537  ra = rb;
538  h *= two;
539  b = step(a, h);
540  rb = resid(b);
541  ++nstep;
542  }
543  // No root within the bound: hold zeta there
544  if ((rb > zero) == (ra > zero)) { return zlim; }
545 
546  // Bisect [a, b]
547  int n = 0;
548  while (std::abs(b - a) > tol && n < 60) {
549  const amrex::Real m = myhalf * (a + b);
550  const amrex::Real rm = resid(m);
551  if ((rm > zero) == (ra > zero)) {
552  a = m;
553  ra = rm;
554  } else {
555  b = m;
556  }
557  ++n;
558  }
559  return myhalf * (a + b);
560 }

590  *
591  * The virtual heat flux is tflux_s + tflux_q, with tflux_s = surf_temp_flux (1 + epsv qv) and

598  * not taken below 0, so when a diagnosed condensation flux alone is stronger than the fold

607  * @param[in] tflux_q moisture part of the virtual heat flux
608  * @param[in] spec_qflux whether the moisture flux is prescribed
609  * @param[in] kappa von Karman constant

695 /**
696  * Aitken delta-squared acceleration of the relaxed u* iteration in the prescribed-flux kernels.
697  *
698  * Near the stable fold the slope of the fixed-point map approaches 1 and the plain iteration
699  * needs hundreds of passes. From pass `start` on, every third iterate is replaced by the
700  * Aitken extrapolation of the last three (kept only within a factor of 2 of them), so points
701  * that converge before `start` are unchanged. The map is not stationary when w* or a
702  * u*-dependent z0 also changes from pass to pass, so the kernels do not stop on an
703  * extrapolated pass: the plain pass after it, with zeta and psi from the same u*, decides
704  * convergence.
705  */
706 struct most_aitken
707 {
708  /**
709  * Return the iterate to continue from.
710  *
711  * @param[in] u_new iterate after pass `iter`
712  * @param[in] iter number of passes done
713  */
714  AMREX_GPU_HOST_DEVICE
715  AMREX_FORCE_INLINE
716  amrex::Real
717  update (amrex::Real u_new, int iter)
718  {
719  extrapolated = false;
720  if (iter < start) { return u_new; }
721  u0 = u1;
722  u1 = u2;
723  u2 = u_new;
724  if (++n < 3) { return u_new; }
725  const amrex::Real d = u2 - two * u1 + u0;
726  const amrex::Real ua = (d != zero) ? u2 - (u2 - u1) * (u2 - u1) / d : u_new;
727  const amrex::Real lo = myhalf * amrex::min(u0, amrex::min(u1, u2));
728  const amrex::Real hi = two * amrex::max(u0, amrex::max(u1, u2));
729  if (d == zero || !(ua > lo && ua < hi)) {
730  n = 0;
731  return u_new;
732  }
733  // The extrapolated value starts the next triple
734  u2 = ua;
735  n = 1;
736  extrapolated = true;
737  return ua;
738  }
739 
740  int start{25};
741  bool extrapolated{false}; ///< whether the last update returned an extrapolated value
742  int n{0};

744  amrex::Real u1{zero};
745  amrex::Real u2{zero};
