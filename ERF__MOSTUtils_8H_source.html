35 /**
36  * Lower bound of the log-law bracket ln(zref/z0) - psi of the MOST flux kernels: C/2 held
37  * between 1/2 and 1.
38  *
39  * The bound is 1 for zref >= e^2 z0 and C/2 down to zref = e z0, so it does not exceed the
40  * neutral bracket C wherever zref >= e^(1/2) z0. The lower limit of 1/2 keeps the bracket,
41  * and so u*, positive and finite when z0 approaches or exceeds zref (C <= 0).
42  *
43  * The bound is set by the under-relaxed u* iteration, not by the physics. At an unstable
44  * fixed point the relaxed map has slope (1 - alpha) - alpha s / bracket, with
45  * s = -3 (z/L) dpsi_m/d(z/L) between 1.3 (z/L = -0.5) and 2.4 (z/L = -50) for Businger-Dyer.
46  * With alpha = 0.5 the iteration converges only for bracket > s/3 (0.4 to 0.8), so a bound of
47  * 1 keeps the switch between the bounded and free branches inside the convergent range, while
48  * 0.1 C (0.3 at C = 3) does not and the iteration oscillates between two values. If alpha
49  * changes, the bound needs revisiting; see PR #4866.
50  *
51  * @param[in] C ln(zref/z0)
52  */
53 AMREX_GPU_HOST_DEVICE
54 AMREX_FORCE_INLINE
55 amrex::Real
56 most_bracket_floor (amrex::Real C)
57 {
58  return amrex::max(amrex::min(amrex::Real(0.5) * C, amrex::Real(1.0)), amrex::Real(0.5));
59 }

68  // Similarity functions from Jimenez (2012)

70  /**
71  * Compute the Jimenez momentum stability correction.
72  *
73  * @param[in] zeta stability coordinate z/L
74  */
75  AMREX_GPU_HOST_DEVICE
76  AMREX_FORCE_INLINE
77  amrex::Real
78  calc_psi_m2 (amrex::Real zeta) const
79  {
80  if (zeta > 0) {
81  amrex::Real x = std::pow(one + std::pow(zeta, amrex::Real(2.5)), one/amrex::Real(2.5));
82  return ( -amrex::Real(6.1)*std::log(zeta + x) );
83  } else {
84  amrex::Real x = std::pow(one - amrex::Real(16.0)*zeta, fourth);
85  amrex::Real psi_k_m = two * std::log(myhalf * (one + x)) + std::log(myhalf * (one + x * x)) -
86  two * std::atan(x) + PIoTwo;
87  amrex::Real y = std::pow(one - amrex::Real(10.0)*zeta, one/three);
88  amrex::Real psi_c_m = (three/two)*std::log((y*y + y + one)/three)
89  - std::sqrt(three)*std::atan((two*y + one)/std::sqrt(three))
90  + PI/std::sqrt(three);
91  return ( (psi_k_m + zeta*zeta*psi_c_m) / (one + zeta*zeta) );
92  }
93  }

95  /**
96  * Compute the Jimenez heat stability correction.
97  *
98  * @param[in] zeta stability coordinate z/L
99  */
100  AMREX_GPU_HOST_DEVICE
101  AMREX_FORCE_INLINE
102  amrex::Real
103  calc_psi_h2 (amrex::Real zeta) const
104  {
105  if (zeta > 0) {
106  amrex::Real x = std::pow(one + std::pow(zeta, amrex::Real(1.1)), one/amrex::Real(1.1));
107  return ( -amrex::Real(5.3)*std::log(zeta + x) );
108  } else {
109  amrex::Real x = std::sqrt(one - amrex::Real(16.0)*zeta);
110  amrex::Real psi_k_h = two * std::log(myhalf * (one + x));
111  amrex::Real y = std::pow(one - amrex::Real(34.0)*zeta, one/three);
112  amrex::Real psi_c_h = (three/two)*std::log((y*y + y + one)/three)
113  - std::sqrt(three)*std::atan((two*y + one)/std::sqrt(three))
114  + PI/std::sqrt(three);
115  return ( (psi_k_h + zeta*zeta*psi_c_h) / (one + zeta*zeta) );
116  }
117  }

160  //
161  // Gradient (phi) functions
162  //
163  /**
164  * Compute the Businger-Dyer momentum gradient function.
165  *
166  * This is the exact companion of calc_psi_m: the two are tied by
167  * psi_m(zeta) = int_0^zeta (1 - phi_m(x))/x dx, so differentiating the MOST
168  * wind profile u(z) = (u_star/kappa)[ln(z/z0) - psi_m(z/L)] gives
169  * du/dz = u_* phi_m(zeta) / (kappa z).
170  *
171  * NOTE: the surface_temp* flux calculators use the Jimenez psi pair
172  * (calc_psi_m2/calc_psi_h2) rather than this one; their companions are
173  * not implemented yet, so a prescribed-surface-temperature run gets the
174  * Businger-Dyer gradient here.
175  *
176  * @param[in] zeta stability coordinate z/L
177  */
178  AMREX_GPU_HOST_DEVICE
179  AMREX_FORCE_INLINE
180  amrex::Real
181  calc_phi_m (amrex::Real zeta) const
182  {
183  if (zeta > 0) {
184  return one + beta_m * zeta;
185  } else {
186  return one / std::sqrt(std::sqrt(one - gamma_m * zeta));
187  }
188  }

190  /**
191  * Compute the Businger-Dyer heat gradient function, the exact companion of
192  * calc_psi_h: dtheta/dz = theta_* phi_h(zeta) / (kappa z).
193  *
194  * See the note in calc_phi_m about the Jimenez pair.
195  *
196  * @param[in] zeta stability coordinate z/L
197  */
198  AMREX_GPU_HOST_DEVICE
199  AMREX_FORCE_INLINE
200  amrex::Real
201  calc_phi_h (amrex::Real zeta) const
202  {
203  if (zeta > 0) {
204  return one + beta_h * zeta;
205  } else {
206  return one / std::sqrt(one - gamma_h * zeta);
207  }
208  }

210  /**
211  * Stable z/L at which the prescribed-flux problem folds.
212  *
213  * With the Businger-Dyer psi_m = -beta_m z/L and L fixed by u* and the flux, the wind a
214  * given u* supports is u* max(C - psi_m, b) / kappa, with b = most_bracket_floor(C). For
215  * C > 0 it is smallest at z/L = C / (2 beta_m); for small or negative C (z0 near or above
216  * zref) the floor binds first and the minimum is at the kink z/L = (b - C) / beta_m. Below
217  * that wind the prescribed stable flux has no solution (Basu et al., BLM 2008). The
218  * physical (large u*) branch has z/L below this value, so bounding z/L by it leaves every
219  * point that has a solution unchanged. The result is positive for every C.
220  *
221  * @param[in] C ln(zref/z0)
222  */
223  AMREX_GPU_HOST_DEVICE
224  AMREX_FORCE_INLINE
225  amrex::Real
226  calc_zeta_fold (amrex::Real C) const
227  {
228  return amrex::max(C / (two * beta_m), (most_bracket_floor(C) - C) / beta_m);
229  }

233  amrex::Real beta_h{amrex::Real(5.0)}; ///< https://doi.org/amrex::Real(10.1007)/BF00240838
234  amrex::Real gamma_m{amrex::Real(16.0)};
235  amrex::Real gamma_h{amrex::Real(16.0)};
236 };

240  * Log-law bracket ln(zref/z0) - psi of the MOST flux kernels, held at or above
241  * most_bracket_floor(C).
242  *
243  * In strongly unstable air psi exceeds C = ln(zref/z0) and the bracket changes sign, which
244  * turns u* negative and puts the diagnosed surface temperature on the wrong side of the air
245  * for a prescribed flux. With the floor the bracket is at least 1/2, so u* stays positive and
246  * finite for every C.

251 AMREX_GPU_HOST_DEVICE
252 AMREX_FORCE_INLINE
253 amrex::Real
254 most_bracket (amrex::Real C, amrex::Real psi)
255 {
256  return amrex::max(C - psi, most_bracket_floor(C));
257 }

264 /**
265  * Residual zeta - Rib (C - psi_m)^2 / (C - psi_h) of the prescribed-temperature kernels'
266  * z/L equation, with the Jimenez functions and the brackets of most_bracket.
267  *
268  * @param[in] sfuns similarity functions
269  * @param[in] C ln(zref/z0)
270  * @param[in] Rib bulk Richardson number
271  * @param[in] zeta z/L
272  */
273 AMREX_GPU_HOST_DEVICE
274 AMREX_FORCE_INLINE
275 amrex::Real
276 most_rib_resid (const similarity_funs& sfuns, amrex::Real C, amrex::Real Rib, amrex::Real zeta)
277 {
278  const amrex::Real CmPsim = most_bracket(C, sfuns.calc_psi_m2(zeta));
279  const amrex::Real CmPsih = most_bracket(C, sfuns.calc_psi_h2(zeta));
280  return zeta - Rib * CmPsim * CmPsim / CmPsih;
281 }

307 /**
308  * Solve zeta = Rib (C - psi_m)^2 / (C - psi_h) of the prescribed-temperature kernels by
309  * bracketing and bisection, with the Jimenez functions and the brackets of most_bracket.
310  *
311  * The relaxed fixed-point iteration stalls where zeta - Rib F(zeta) nearly touches zero
312  * without crossing it, which the Jimenez stable functions produce when z0 is comparable to or
313  * above zref (C below about 1): each pass then moves zeta by about the tolerance. From zeta0
314  * this steps, with doubling steps, in the direction the iteration moves until the residual
315  * changes sign, and bisects that interval, so it returns the root the iteration was heading
316  * to, or the next one past a near-touch. The first step is limited so that two nearby roots
317  * are not stepped over together. The search stays within +-most_zeta_lim, and returns that
318  * bound when no root lies inside it, as the prescribed-flux kernels bound zeta.
319  *
320  * @param[in] sfuns similarity functions
321  * @param[in] C ln(zref/z0)
322  * @param[in] Rib bulk Richardson number
323  * @param[in] zeta0 z/L to start from
324  * @param[in] tol width of the final interval
325  */
326 AMREX_GPU_HOST_DEVICE
327 AMREX_FORCE_INLINE
328 amrex::Real
329 most_rib_zeta_bracketed (const similarity_funs& sfuns, amrex::Real C, amrex::Real Rib,
330  amrex::Real zeta0, amrex::Real tol)
331 {
332  auto resid = [&] (amrex::Real zeta) { return most_rib_resid(sfuns, C, Rib, zeta); };
333 
334  amrex::Real a = zeta0;
335  amrex::Real ra = resid(a);
336  if (ra == zero) { return a; }
337 
338  // Step against the residual, within +-most_zeta_lim, until it changes sign
339  const amrex::Real dir = (ra > zero) ? amrex::Real(-1.0) : amrex::Real(1.0);
340  const amrex::Real zlim = dir * most_zeta_lim;
341  auto step = [&] (amrex::Real z, amrex::Real dz) {
342  return (dir > zero) ? amrex::min(z + dz, zlim) : amrex::max(z - dz, zlim);
343  };
344  amrex::Real h = amrex::max(amrex::min(std::abs(ra),
345  amrex::Real(0.1) * amrex::max(amrex::Real(one), std::abs(a))),
346  tol);
347  amrex::Real b = step(a, h);
348  amrex::Real rb = resid(b);
349  while ((rb > zero) == (ra > zero) && b != zlim) {
350  a = b;
351  ra = rb;
352  h *= two;
353  b = step(a, h);
354  rb = resid(b);
355  }
356  // No root within the bound: hold zeta there
357  if ((rb > zero) == (ra > zero)) { return zlim; }
358 
359  // Bisect [a, b]
360  int n = 0;
361  while (std::abs(b - a) > tol && n < 60) {
362  const amrex::Real m = myhalf * (a + b);
363  const amrex::Real rm = resid(m);
364  if ((rm > zero) == (ra > zero)) {
365  a = m;
366  ra = rm;
367  } else {
368  b = m;
369  }
370  ++n;
371  }
372  return myhalf * (a + b);
373 }

401 /**
402  * Part of a prescribed stable flux the wind can carry at the fold.
403  *
404  * The virtual heat flux is tflux_s + tflux_q, with tflux_s = surf_temp_flux (1 + epsv qv) and
405  * tflux_q = qflux epsv theta. When z/L is held at the fold (see
406  * similarity_funs::calc_zeta_fold), the wind is below the minimum the prescribed flux needs and
407  * carries only -u*^3 thv zeta_fold / (kappa g zref). Only the prescribed components that are
408  * downward drive the fold, so only they are reduced, by a common factor, until the net virtual
409  * flux equals that value. An upward component, or a moisture flux diagnosed from q_surf, is
410  * left unchanged; if no prescribed component is downward, nothing is reduced. The factor is
411  * not taken below 0, so when a diagnosed condensation flux alone is stronger than the fold
412  * value, the downward prescribed components are removed and the net flux remains stronger
413  * than the fold value. Returns 1 for both factors away from the fold.
414  *
415  * @param[in] at_fold whether z/L was held at the fold
416  * @param[in] zeta_fold z/L at the fold
417  * @param[in] ustar friction velocity
418  * @param[in] thv virtual potential temperature at zref
419  * @param[in] tflux_s sensible part of the virtual heat flux
420  * @param[in] tflux_q moisture part of the virtual heat flux
421  * @param[in] spec_qflux whether the moisture flux is prescribed
422  * @param[in] kappa von Karman constant
423  * @param[in] gravity gravitational acceleration
424  * @param[in] zref reference height
425  */
426 AMREX_GPU_HOST_DEVICE
427 AMREX_FORCE_INLINE
428 most_flux_scale
429 most_fold_flux_scale (bool at_fold, amrex::Real zeta_fold, amrex::Real ustar,
430  amrex::Real thv, amrex::Real tflux_s, amrex::Real tflux_q,
431  bool spec_qflux, amrex::Real kappa, amrex::Real gravity,
432  amrex::Real zref)
433 {
434  most_flux_scale scale;
435  const amrex::Real tflux = tflux_s + tflux_q;
436  if (!at_fold || tflux >= zero) { return scale; }
437  const amrex::Real tflux_fold = -ustar * ustar * ustar * thv * zeta_fold / (kappa * gravity * zref);
438  if (tflux_fold <= tflux) { return scale; }
439  const bool down_s = (tflux_s < zero);
440  const bool down_q = (spec_qflux && tflux_q < zero);
441  const amrex::Real down = (down_s ? tflux_s : zero) + (down_q ? tflux_q : zero);
442  if (down >= zero) { return scale; }
443  const amrex::Real r = amrex::max((down + tflux_fold - tflux) / down, amrex::Real(0.0));
444  if (down_s) { scale.temp = r; }
445  if (down_q) { scale.moist = r; }
446  return scale;
447 }

454 /**
455  * Surface state of a prescribed-flux kernel once its u* iteration has converged.
456  *
457  * Sets t_surf, t_star and q_star (and q_surf when the moisture flux is prescribed). At the
458  * fold the prescribed fluxes are reduced by most_fold_flux_scale and the Obukhov length is set
459  * to the fold value. u* is floored at most_min_ustar here, so every kernel applies the same floor.
460  *
461  * @param[in] mdata MOST data (prescribed fluxes, kappa, gravity)
462  * @param[in] spec_qflux whether the moisture flux is prescribed
463  * @param[in] at_fold whether z/L was held at the fold on the last pass
464  * @param[in] zeta_fold z/L at the fold
465  * @param[in] C ln(zref/z0)
466  * @param[in] psi_h heat stability correction of the last pass
467  * @param[in] ustar friction velocity (floored at most_min_ustar)
468  * @param[in] zref reference height
469  * @param[in] tm potential temperature at zref
470  * @param[in] tvm virtual potential temperature at zref
471  * @param[in] qvm water vapor at zref
472  * @param[in] qflux moisture flux of the last pass
473  * @param[inout] olen Obukhov length
474  * @param[out] t_surf surface potential temperature
475  * @param[out] t_star temperature scale
476  * @param[inout] q_surf surface water vapor (read when the moisture flux is diagnosed)
477  * @param[out] q_star moisture scale
478  */
479 AMREX_GPU_HOST_DEVICE
480 AMREX_FORCE_INLINE
481 void
482 most_flux_outputs (const most_data& mdata, bool spec_qflux, bool at_fold,
483  amrex::Real zeta_fold, amrex::Real C, amrex::Real psi_h,
484  amrex::Real ustar, amrex::Real zref, amrex::Real tm,
485  amrex::Real tvm, amrex::Real qvm, amrex::Real qflux,
486  amrex::Real& olen, amrex::Real& t_surf, amrex::Real& t_star,
487  amrex::Real& q_surf, amrex::Real& q_star)
488 {
489  ustar = amrex::max(ustar, amrex::Real(most_min_ustar));
490  const most_flux_scale scale = most_fold_flux_scale(at_fold, zeta_fold, ustar, tvm,
491  mdata.surf_temp_flux * (one + epsv * qvm),
492  qflux * epsv * tm, spec_qflux,
493  mdata.kappa, mdata.gravity, zref);
494  if (at_fold) { olen = zref / zeta_fold; }
495  const amrex::Real CmPsih = most_bracket(C, psi_h);
496  const amrex::Real hflux = scale.temp * mdata.surf_temp_flux;
497  t_surf = hflux * CmPsih / (ustar * mdata.kappa) + tm;
498  t_star = -hflux / ustar;
499  if (spec_qflux) {
500  const amrex::Real mflux = scale.moist * mdata.surf_moist_flux;
501  q_surf = mflux * CmPsih / (ustar * mdata.kappa) + qvm;
502  q_star = -mflux / ustar;
503  } else {

508 /**
509  * Aitken delta-squared acceleration of the relaxed u* iteration in the prescribed-flux kernels.
510  *
511  * Near the stable fold the slope of the fixed-point map approaches 1 and the plain iteration
512  * needs hundreds of passes. From pass `start` on, every third iterate is replaced by the
513  * Aitken extrapolation of the last three (kept only within a factor of 2 of them), so points
514  * that converge before `start` are unchanged. The map is not stationary when w* or a
515  * u*-dependent z0 also changes from pass to pass, so the kernels do not stop on an
516  * extrapolated pass: the plain pass after it, with zeta and psi from the same u*, decides
517  * convergence.
518  */
519 struct most_aitken
520 {
521  /**
522  * Return the iterate to continue from.
523  *
524  * @param[in] u_new iterate after pass `iter`
525  * @param[in] iter number of passes done
526  */
527  AMREX_GPU_HOST_DEVICE
528  AMREX_FORCE_INLINE
529  amrex::Real
530  update (amrex::Real u_new, int iter)
531  {
532  extrapolated = false;
533  if (iter < start) { return u_new; }
534  u0 = u1;
535  u1 = u2;
536  u2 = u_new;
537  if (++n < 3) { return u_new; }
538  const amrex::Real d = u2 - two * u1 + u0;
539  const amrex::Real ua = (d != zero) ? u2 - (u2 - u1) * (u2 - u1) / d : u_new;
540  const amrex::Real lo = myhalf * amrex::min(u0, amrex::min(u1, u2));
541  const amrex::Real hi = two * amrex::max(u0, amrex::max(u1, u2));
542  if (d == zero || !(ua > lo && ua < hi)) {
543  n = 0;
544  return u_new;
545  }
546  // The extrapolated value starts the next triple
547  u2 = ua;
548  n = 1;
549  extrapolated = true;
550  return ua;
551  }
552 
553  int start{25};
554  bool extrapolated{false}; ///< whether the last update returned an extrapolated value
555  int n{0};
556  amrex::Real u0{zero};
557  amrex::Real u1{zero};
558  amrex::Real u2{zero};
559 };

714 /// Lower bound of the roughness recovered by Compute_roughness; shared with
715 /// the guard in Compute_roughness so the two cannot drift apart.
716 constexpr amrex::Real inverted_roughness_lo = amrex::Real(1.0e-7);

718 /**
719  * Bounds on the roughness recovered by Compute_roughness. The inversion is
720  * unbounded above: in weak wind under strong stability (u* from an LSM or bulk
721  * flux large relative to kappa*U, psi_m << 0) it returns z0 >> zref. The
722  * upper bound is above every tabulated land-cover roughness in NoahmpTable.TBL
723  * (Z0MVT <= 1.10 m) and below the 2 m diagnostic height; z0 is also kept below
724  * half of zref so that ln(zref/z0) stays positive.
725  */
726 AMREX_GPU_HOST_DEVICE
727 AMREX_FORCE_INLINE
728 amrex::Real
729 bound_inverted_roughness (amrex::Real z0, amrex::Real zref) noexcept
730 {
731  const amrex::Real z0_lo = inverted_roughness_lo;
732  const amrex::Real z0_hi = amrex::min(amrex::Real(1.5), amrex::Real(0.5) * zref);
733  return amrex::min(amrex::max(z0, z0_lo), z0_hi);
734 }

736 /**
737  * Invert log-law to get roughness
738  *
739  * @param[in] zref reference height for the wind speed
740  * @param[in] Olen Obukhov length
741  * @param[in] umm horizontal wind speed magnitude at zref
742  * @param[in] ustar friction velocity
743  */
744 AMREX_GPU_DEVICE
745 AMREX_FORCE_INLINE
746 amrex::Real
747 Compute_roughness (amrex::Real zref,
748  amrex::Real Olen,
749  amrex::Real umm,
750  amrex::Real ustar)
751 {
752  similarity_funs sfuns;
753  amrex::Real zeta = zref / Olen;
754  amrex::Real psi_m = sfuns.calc_psi_m2(zeta);
755  // z0 = zref / exp(arg) is below the lower bound once arg > log(zref / z0_lo),
756  // so the bound is returned without evaluating exp(arg), which overflows when
757  // u* sits at its floor of machine epsilon (zero stress: bulk_coeff with
758  // Cd = 0, or an LSM that returns none). At or below the threshold exp(arg)
759  // is finite and the inversion runs as before, so no other input changes.
760  const amrex::Real arg = KAPPA*umm/ustar + psi_m;
761  if (arg > std::log(zref / inverted_roughness_lo)) {
762  return bound_inverted_roughness(inverted_roughness_lo, zref);
763  }
764  amrex::Real z0 = zref / std::exp(arg);
765  return bound_inverted_roughness(z0, zref);
766 }
