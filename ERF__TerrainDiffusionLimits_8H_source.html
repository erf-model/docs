20  * Terrain drop across one cell of a terrain-fitted mesh.
21  *
22  * dx_drop and dy_drop are the height differences across the cell along the
23  * coordinate surfaces, each the mean of the absolute differences on the four
24  * cell edges in that direction; dz is the physical thickness of the cell, the

26  * are in metres and read only the eight nodes of the cell.
27  */

35 /**
36  * Compute the terrain drops across cell (i,j,k) from the nodal heights.
37  *
38  * @param[in] i,j,k cell-centered indices
39  * @param[in] z_nd nodal physical heights
40  */
41 AMREX_GPU_HOST_DEVICE
42 AMREX_FORCE_INLINE
43 TerrainCellDrops
44 ComputeTerrainCellDrops (int i, int j, int k,
45  const amrex::Array4<const amrex::Real>& z_nd) noexcept
46 {
47  using amrex::Math::abs;
48  TerrainCellDrops d;
49  d.dx_drop = fourth * ( abs(z_nd(i+1,j ,k ) - z_nd(i,j ,k ))
50  + abs(z_nd(i+1,j+1,k ) - z_nd(i,j+1,k ))
51  + abs(z_nd(i+1,j ,k+1) - z_nd(i,j ,k+1))
52  + abs(z_nd(i+1,j+1,k+1) - z_nd(i,j+1,k+1)) );
53  d.dy_drop = fourth * ( abs(z_nd(i ,j+1,k ) - z_nd(i ,j,k ))
54  + abs(z_nd(i+1,j+1,k ) - z_nd(i+1,j,k ))
55  + abs(z_nd(i ,j+1,k+1) - z_nd(i ,j,k+1))
56  + abs(z_nd(i+1,j+1,k+1) - z_nd(i+1,j,k+1)) );
57  d.dz = fourth * ( (z_nd(i ,j ,k+1) - z_nd(i ,j ,k))
58  + (z_nd(i+1,j ,k+1) - z_nd(i+1,j ,k))
59  + (z_nd(i ,j+1,k+1) - z_nd(i ,j+1,k))
60  + (z_nd(i+1,j+1,k+1) - z_nd(i+1,j+1,k)) );
61  return d;
62 }

64 /**
65  * Slope factor alpha = max( sqrt(dx_drop^2 + dy_drop^2) / dz, 1 ) of a cell:
66  * the terrain drop across one cell measured in cell thicknesses, which is
67  * h * Delta_x / Delta_z with h the physical coordinate slope. This is the
68  * factor of WRF's Smagorinsky-2D slope limiter (dyn_em/module_diffusion_em.F,
69  * smag2d_km, diff_opt = 2). WRF averages |z_x| at the two x-faces and the two
70  * w-levels of the cell and multiplies by dx / msftx; here the drop is taken on
71  * the four x-edges of the cell (ERF heights are nodal) and carries no map
72  * factor, so that alpha is exactly h * Delta_x / Delta_z with the physical
73  * spacing. The two agree when the map factor is 1.
74  */
75 AMREX_GPU_HOST_DEVICE
76 AMREX_FORCE_INLINE
77 amrex::Real
78 TerrainSlopeFactor (const TerrainCellDrops& d) noexcept
79 {
80  const amrex::Real a = std::sqrt(d.dx_drop*d.dx_drop + d.dy_drop*d.dy_drop) / d.dz;
81  return amrex::max(a, amrex::Real(1.0));
82 }

89 /**
90  * Apply the opt-in limits of WRF's smag2d_km to a Smagorinsky-2D horizontal
91  * kinematic eddy viscosity, in WRF's order:
92  * 1. the cap nu <= kh_cap * Delta_h (only if kh_cap > 0; WRF uses 10 m/s);
93  * 2. the slope limiter (only if use_slope):
94  * nu /= alpha^2 if |D_h| > max(10/Delta_h, 1e-3),
95  * nu /= alpha otherwise.
96  * |D_h| is the horizontal deformation sqrt((u_x - v_y)^2 + (u_y + v_x)^2),
97  * which equals ERF's sqrt(2 S_mn S_mn) of ComputeSmnSmn2D and WRF's
98  * sqrt(def2). Delta_h = sqrt(dx/m_x * dy/m_y), WRF's mlen_h.
99  *
100  * @param[in] nu Cs^2 Delta_h^2 |D_h| (m^2/s)
101  * @param[in] strain |D_h| (1/s)
102  * @param[in] DeltaH horizontal filter width (m)
103  * @param[in] alpha slope factor from TerrainSlopeFactor (>= 1)
104  * @param[in] kh_cap cap speed (m/s); <= 0 leaves nu uncapped
105  * @param[in] use_slope apply the slope limiter
106  */
107 AMREX_GPU_HOST_DEVICE
108 AMREX_FORCE_INLINE
109 amrex::Real
110 Smag2DLimitedViscosity (amrex::Real nu, amrex::Real strain, amrex::Real DeltaH,
111  amrex::Real alpha, amrex::Real kh_cap, bool use_slope) noexcept
112 {
113  if (kh_cap > zero) {
114  const amrex::Real nu_max = kh_cap * DeltaH;
115  nu = (nu < nu_max) ? nu : nu_max;
116  }
117  if (use_slope) {
118  const amrex::Real def_limit = amrex::max(amrex::Real(smag2d_def_limit_speed) / DeltaH,
119  amrex::Real(smag2d_def_limit_min));
120  nu = (strain > def_limit) ? nu / (alpha*alpha) : nu / alpha;
121  }
122  return nu;
123 }

125 /**
126  * Explicit diffusion rates of one cell, for the diffusive time-step estimate
127  * dt * rate <= Fourier limit (1/2 for forward Euler in 3-D; the three-stage
128  * Runge-Kutta scheme's real-axis bound is about 2.5 / 4 = 0.63).
129  *
130  * Momentum: the largest of the three component rows,
131  * u: mu_h (2 a + b) + e_uv mu_v / dz^2 + mu_v sqrt(a c)
132  * v: mu_h (a + 2 b) + e_uv mu_v / dz^2 + mu_v sqrt(b c)
133  * w: mu_v (a + b) + 2 e_w mu_v / dz^2 + mu_v (sqrt(a c) + sqrt(b c)),
134  * with c = 1/dz^2, plus 2 mu_p h^2 / dz^2 for the terrain-metric part. The horizontal
135  * diffusion of w uses mu_v: tau31 and tau32 carry the same edge-averaged mu_v as tau13 and
136  * tau23. The sqrt terms are the u-w and v-w cross derivatives (dw/dx in tau13, du/dz in
137  * tau31, ...), which stay explicit whether or not the implicit solve takes the second
138  * vertical derivative; they are always counted, since with dz > dx/m they exceed the
139  * vertical term (dx = 100 m, dz = 500 m: five times). (The expansion
140  * rate in tau11 also couples u to dw/dz, with mu_h; that is covered by "an estimate when
141  * mu_h != mu_v" below.) The metric term is an upper bound: the
142  * projected stresses of the terrain stress (ComputeStressVarVisc_T) carry mu_v today (ERF
143  * issue #4214) and would carry mu_h once that is changed, hence mu_p = max(mu_h, mu_v);
144  * with mu_v there, the K_h part of that coupling enters only through the mixed xi-zeta
145  * terms, so the bound can overstate that coupling by about alpha / 2 on steep slopes.
146  *
147  * Scalar: K_h (a + b) + K_h h^2 / dz^2 + e K_v / dz^2; the K_h h^2 part
148  * (TerrainDiffusionGz) is explicit.
149  *
150  * a = (m_x / dx)^2, b = (m_y / dy)^2 and h^2 = (m_x dx_drop / dx)^2 +
151  * (m_y dy_drop / dy)^2 is the squared physical slope; on a mesh without
152  * terrain pass h2 = 0. e is the explicit fraction of the vertical diffusion
153  * (ExplicitVerticalFraction): 0 when the implicit solve keeps every stage bounded,
154  * 1 otherwise. All diffusivities are dynamic (rho * nu); rates are in 1/s.
155  * The caller passes the largest diffusivities and the smallest density around the
156  * cell (NeighbourhoodMax, NeighbourhoodMinPositive), which bounds the row whatever
157  * face or edge averaging the operator uses.
158  */
159 AMREX_GPU_HOST_DEVICE
160 AMREX_FORCE_INLINE
161 amrex::Real
162 MomentumDiffusiveRate (amrex::Real rho, amrex::Real mu_h, amrex::Real mu_v,
163  amrex::Real a, amrex::Real b, amrex::Real h2, amrex::Real dzinv2,
164  amrex::Real e_uv, amrex::Real e_w) noexcept
165 {
166  const amrex::Real xa = std::sqrt(a * dzinv2);
167  const amrex::Real xb = std::sqrt(b * dzinv2);
168  const amrex::Real r_u = mu_h * (two*a + b) + e_uv * mu_v * dzinv2 + mu_v * xa;
169  const amrex::Real r_v = mu_h * (a + two*b) + e_uv * mu_v * dzinv2 + mu_v * xb;
170  const amrex::Real r_w = mu_v * (a + b) + two * e_w * mu_v * dzinv2 + mu_v * (xa + xb);
171  const amrex::Real r_uv = (r_u > r_v) ? r_u : r_v;
172  const amrex::Real row = (r_uv > r_w) ? r_uv : r_w;
173  const amrex::Real mu_p = (mu_h > mu_v) ? mu_h : mu_v;
174  return ( row + two * mu_p * h2 * dzinv2 ) / rho;
175 }

177 /**
178  * Largest value of component n over cell (i,j,k) and its 26 neighbours. The diffusion
179  * operators average the cell diffusivities to faces and edges, so a row of the operator
180  * never carries more than this. With NeighbourhoodMinPositive for the density and the
181  * coefficients frozen, the scalar rate is a Gershgorin bound on the centred interior stencil
182  * of the Cartesian operator (the higher-order one-sided gradients next to Dirichlet
183  * boundaries, ERF_ScalarDiffusion.H, whose row is about 4/3 larger, are not covered); for
184  * momentum it is a bound for
185  * uniform coefficients when mu_h = mu_v (tight on
186  * cubic cells) and an estimate otherwise, since the operator is then not symmetric.
187  * Needs one ghost cell.
188  */
189 AMREX_GPU_HOST_DEVICE
190 AMREX_FORCE_INLINE
191 amrex::Real
192 NeighbourhoodMax (int i, int j, int k, int n, const amrex::Array4<const amrex::Real>& a) noexcept
193 {
194  amrex::Real m = a(i,j,k,n);
195  for (int kk = -1; kk <= 1; ++kk) {
196  for (int jj = -1; jj <= 1; ++jj) {
197  for (int ii = -1; ii <= 1; ++ii) {
198  const amrex::Real v = a(i+ii,j+jj,k+kk,n);
199  m = (v > m) ? v : m;
200  }}}
201  return m;
202 }

205  * Smallest positive value of component n over cell (i,j,k) and its 26 neighbours; ghost
206  * cells that hold no density (<= 0) are skipped. Returns `fallback` if none is positive.

212  amrex::Real fallback) noexcept
213 {
214  amrex::Real m = fallback;

252  * @param[in] s_arr conserved state (Rho_comp), one ghost cell
253  * @param[in] k_arr eddy diffusivities (EddyDiff components), one ghost cell
254  * @param[in] z_nd nodal heights (read only when set.variable_dz)
255  * @param[in] mf_mx, mf_my map factors of the cell

258 AMREX_GPU_HOST_DEVICE
259 AMREX_FORCE_INLINE
260 void
261 CellDiffusiveRates (int i, int j, int k,
262  const amrex::Array4<const amrex::Real>& s_arr,
263  const amrex::Array4<const amrex::Real>& k_arr,
264  const amrex::Array4<const amrex::Real>& z_nd,
265  amrex::Real mf_mx, amrex::Real mf_my,
266  const amrex::GpuArray<amrex::Real, AMREX_SPACEDIM>& dxinv,
267  const DiffusiveRateSettings& set,
268  amrex::Real& rate_mom, amrex::Real& rate_scal) noexcept
269 {
270  const amrex::Real ax = mf_mx * dxinv[0];
271  const amrex::Real ay = mf_my * dxinv[1];
272  amrex::Real dzinv = dxinv[2];
273  amrex::Real h2 = zero;
274  if (set.variable_dz) {
275  const TerrainCellDrops d = ComputeTerrainCellDrops(i,j,k,z_nd);
276  dzinv = one / d.dz;
277  const amrex::Real hx = ax * d.dx_drop;
278  const amrex::Real hy = ay * d.dy_drop;
279  h2 = hx*hx + hy*hy;
280  }

282  if (!(rho > zero)) { rate_mom = zero; rate_scal = zero; return; }
283  const amrex::Real rdiv = NeighbourhoodMinPositive(i,j,k,Rho_comp,s_arr,rho);
284  const amrex::Real a2 = ax*ax;
285  const amrex::Real b2 = ay*ay;
286  const amrex::Real dzinv2 = dzinv*dzinv;
287  rate_mom = MomentumDiffusiveRate(rdiv, NeighbourhoodMax(i,j,k,EddyDiff::Mom_h,k_arr),
288  NeighbourhoodMax(i,j,k,EddyDiff::Mom_v,k_arr),
289  a2, b2, h2, dzinv2, set.e_uv, set.e_w);
290  amrex::Real rs = ScalarDiffusiveRate(rdiv, NeighbourhoodMax(i,j,k,EddyDiff::Theta_h,k_arr),
291  NeighbourhoodMax(i,j,k,EddyDiff::Theta_v,k_arr),
292  a2, b2, h2, dzinv2, set.e_th);
293  if (set.has_q) {
294  rs = amrex::max(rs, ScalarDiffusiveRate(rdiv, NeighbourhoodMax(i,j,k,EddyDiff::Q_h,k_arr),
295  NeighbourhoodMax(i,j,k,EddyDiff::Q_v,k_arr),
296  a2, b2, h2, dzinv2, set.e_q));
297  }
298  if (set.has_ke) {
299  rs = amrex::max(rs, ScalarDiffusiveRate(rdiv, NeighbourhoodMax(i,j,k,EddyDiff::KE_h,k_arr),
300  NeighbourhoodMax(i,j,k,EddyDiff::KE_v,k_arr),
301  a2, b2, h2, dzinv2, set.e_ke));
302  }
303  if (set.has_scalar) {
304  rs = amrex::max(rs, ScalarDiffusiveRate(rdiv, NeighbourhoodMax(i,j,k,EddyDiff::Scalar_h,k_arr),
305  NeighbourhoodMax(i,j,k,EddyDiff::Scalar_v,k_arr),
306  a2, b2, h2, dzinv2, one));
307  }

309 }

332 /**
333  * Explicit fraction of the vertical diffusion for the diffusive time-step check: 0 if every
334  * stage of the partly implicit Runge-Kutta scheme stays bounded for the vertical diffusion,
335  * 1 otherwise (including every pattern for which that has not been shown).
336  *
337  * Each stage k starts from S^n with stage step c_k dt; its explicit part (1 - f_k) acts on
338  * the previous stage's state and its implicit part f_k on its own, so for dS/dt = lambda S
339  * and y = -lambda dt >= 0, S_k = (1 - (1 - f_k) c_k y S_{k-1}) / (1 + f_k c_k y), S_0 = 1.
340  * Each stage's state feeds the next stage's explicit terms, so every |S_k| must stay <= 1,
341  * not only the last. The result is decided analytically:
342  *
343  * - three stages (c = 1/3, 1/2, 1; compressible): bounded for f_1 >= 1/2, f_2 = 1 and any
344  * f_3 in [0,1]. |S_1| <= 1 iff f_1 >= 1/2: |1 - (1-f) x| <= 1 + (1-f) x <= 1 + f x for
345  * f >= 1/2, and for f < 1/2, S_1 -> -(1-f)/f < -1 as x -> infinity;
346  * S_2 = 1 / (1 + y/2); and S_3 = (1 + (f_3 - 1/2) y) / ((1 + y/2)(1 + f_3 y)), whose
347  * numerator is at most 1 + y/2 in size. The default 1 1 0 is Crank-Nicolson, 1 1 1
348  * backward Euler.
349  * - two stages (c = 1/2, 1; anelastic MidPoint): bounded for f_1 = 1 and any f_2 in [0,1]
350  * (the same S_3 form); with f_1 < 1 and f_2 = 0, as MidPoint forces, the second stage
351  * grows without bound.
352  *
353  * Every other pattern returns 1: some are not bounded (0 0 1, 1 0 0, and 0.995 0.1 0.0666,
354  * whose last stage reaches 1.0008 near y = 23.5), and the rest are not shown to be.
355  *
356  * @param[in] nstages 3 for the compressible scheme, 2 for anelastic MidPoint
357  * @param[in] fac implicit factor f_k of each stage (erf.vert_implicit_fac)
358  * @param[in] implicit whether the implicit solve treats this component at all
359  */
360 template <typename V>
361 amrex::Real
362 ExplicitVerticalFraction (int nstages, const V& fac, bool implicit)
363 {
364  if (!implicit || static_cast<int>(fac.size()) < nstages) { return one; }
365  // The proof assumes factors in [0,1]; anything else (the inputs are not range-checked,
366  // and 1 1.5 0 amplifies by 4/3) counts as explicit rather than being clamped.
367  for (int k = 0; k < nstages; ++k) {
368  const double fk = static_cast<double>(fac[k]);
369  if (!(fk >= 0.0 && fk <= 1.0)) { return one; }
370  }
371  auto f = [&] (int k) { return static_cast<double>(fac[k]); };
372  if (nstages == 3) {
373  return (f(0) >= 0.5 && f(1) >= 1.0) ? zero : one;
374  }
375  if (nstages == 2) {
376  return (f(0) >= 1.0) ? zero : one;
377  }
378  return one;
379 }
