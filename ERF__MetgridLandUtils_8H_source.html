25  * beyond the shallowest and deepest midpoints.

31 inline amrex::Real
32 interp_in_depth (const amrex::Vector<amrex::Real>& z_src,
33  const amrex::Vector<amrex::Real>& v_src,
34  amrex::Real z)
35 {
36  const int n = z_src.size();
37  if (z <= z_src[0]) { return v_src[0]; }
38  if (z >= z_src[n-1]) { return v_src[n-1]; }
39  int m = 0;
40  while (z > z_src[m+1]) { ++m; }
41  const amrex::Real w = (z - z_src[m]) / (z_src[m+1] - z_src[m]);
42  return (amrex::Real(1.0) - w) * v_src[m] + w * v_src[m+1];
43 }
