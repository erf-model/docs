20 //
21 // Placing a point given by latitude and longitude on the model grid.
22 //
23 // The model has no map projection of its own: a run initialized from a WRF or
24 // metgrid file carries the latitude and longitude of every level-0 mass point
25 // (lat_m / lon_m), whatever projection produced them. A requested point is
26 // found by searching those arrays for the nearest mass point and then inverting
27 // the map locally with one linear solve, which is exact for a map that is
28 // linear over a cell and well under a cell width in error for any smooth one.
29 // Longitude differences are taken the short way round, so a grid that crosses
30 // the antimeridian is handled like any other.
31 //
32 // The same arrays give the orientation of the grid: the direction in which the
33 // i index grows, measured counterclockwise from true east, is the angle alpha
34 // by which an earth-relative (east, north) wind has to be rotated to become a
35 // grid-relative (u, v) wind.
36 //
37 
38 //
39 // Invert the latitude/longitude map locally. Given the difference between the
40 // requested lat/lon and the lat/lon of a grid point, and the gradients of
41 // latitude and longitude with respect to the two horizontal index directions at
42 // that point, return the offset in cells from the grid point to the request.
43 //
44 // Returns false when the two gradients are parallel and the map is therefore not
45 // locally invertible. "Parallel" is judged relative to the gradients' size and
46 // to the precision: in single precision the determinant of an exactly
47 // degenerate map comes out of rounding at about 1e-8 of its scale squared, far
48 // above a fixed 1e-12, so the threshold is at least 64 machine epsilons (which
49 // leaves double precision at 1e-12).
50 //
51 inline bool
52 invert_latlon_offset (amrex::Real dlat, amrex::Real dlon,
53  amrex::Real dlat_di, amrex::Real dlon_di,
54  amrex::Real dlat_dj, amrex::Real dlon_dj,
55  amrex::Real& di, amrex::Real& dj)
56 {
57  const amrex::Real scale = std::max(std::max(std::abs(dlat_di), std::abs(dlon_di)),
58  std::max(std::abs(dlat_dj), std::abs(dlon_dj)));
59  const amrex::Real det = dlat_di*dlon_dj - dlat_dj*dlon_di;
60  const amrex::Real rel = std::max(amrex::Real(1.0e-12),
61  amrex::Real(64) * std::numeric_limits<amrex::Real>::epsilon());
62  if (scale <= amrex::Real(0.0) || std::abs(det) <= rel*scale*scale) {
63  return false;
64  }
65  di = ( dlat*dlon_dj - dlon*dlat_dj) / det;
66  dj = (-dlat*dlon_di + dlon*dlat_di) / det;
67  return true;
68 }

70 //
71 // A longitude difference in degrees, brought into [-180, 180] so that two
72 // points either side of the antimeridian are close rather than 360 degrees apart.
73 //
74 inline amrex::Real
75 wrap_longitude_difference (amrex::Real dlon)
76 {
77  if (dlon > amrex::Real(180.0)) { return dlon - amrex::Real(360.0); }
78  if (dlon < -amrex::Real(180.0)) { return dlon + amrex::Real(360.0); }
79  return dlon;
80 }
