22 /**
23  * An array filled on the host and read by the kernels: the host side is a std::vector (clear() keeps its
24  * capacity), copied through pinned memory to the device with one asynchronous copy. The pinned and device
25  * sides only grow. A call that uploads must synchronise the stream before it returns, so that the next
26  * call may refill the host side.
27  */
28 template <class T>
29 struct StagedBuffer
30 {
31  std::vector<T> host; //!< what the next upload() sends
32  amrex::Gpu::PinnedVector<T> pinned; //!< staging, and the host side of download_async()
33  amrex::Gpu::DeviceVector<T> device;
34 
35  /** Copy host to the device, asynchronously. */
36  void upload ()
37  {
38  grow_buffer(pinned, host.size());
39  grow_buffer(device, host.size());
40  if (host.empty()) { return; }
41  std::copy(host.begin(), host.end(), pinned.begin());
42  amrex::Gpu::copyAsync(amrex::Gpu::hostToDevice, pinned.begin(), pinned.begin() + host.size(), device.begin());
43  }
44  /** Copy the first n device values to pinned, asynchronously: they are there after the next stream synchronisation. */
45  void download_async (std::size_t n)
46  {
47  grow_buffer(pinned, n);
48  if (n == 0) { return; }
49  amrex::Gpu::copyAsync(amrex::Gpu::deviceToHost, device.begin(), device.begin() + n, pinned.begin());
50  }
51  T* data () { return device.data(); }
52 };
