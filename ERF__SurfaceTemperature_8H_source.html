96 /**
97  * Why a surface conversion could not be made, in the order the kernels test for it.
98  *
99  * Every kernel that converts between the atmospheric state and a surface temperature or
100  * humidity guards the same sequence of ways that state can be unusable. Which guard
101  * tripped is the difference between "the surface layer gave up" and knowing that the
102  * state handed to it was already bad -- so the guards are named rather than merged into
103  * a single flag.
104  */

113  bad_pressure,

115  bad_q_air,

119 /**
120  * The first failing cell on this rank, the state that produced it, and how many cells
121  * failed.
122  *
123  * The count is the part that is easy to leave out and expensive to do without: one bad
124  * column and a whole plane of NaN state are different problems with different causes,
125  * and nothing else in the report tells them apart. Cells the fill is not authoritative
126  * for are counted separately -- they do not abort the run, but a halo full of failures
127  * beside one bad owned cell says where to look.
128  */
129 struct ConversionFailure
130 {
131  int code = static_cast<int>(ConversionError::none);
132  int count = 0;
133  int halo_count = 0;
134  int i = 0, j = 0, k = 0;
135  amrex::Real rho = amrex::Real(0.0);
136  amrex::Real rho_theta = amrex::Real(0.0);
137  amrex::Real rho_qv = amrex::Real(0.0);
138  amrex::Real t_surf = amrex::Real(0.0);
139  amrex::Real pressure = amrex::Real(0.0);
140  amrex::Real delta_z = amrex::Real(0.0);
141  amrex::Real q_air = amrex::Real(0.0);

144 /**
145  * Record one cell's failure.
146  *
147  * Every failing cell adds to a count; the first authoritative one to arrive also claims
148  * the description, so what gets reported is a failure that matters rather than whichever
149  * thread happened to finish last. Pass the values the caller has reached by this point:
150  * the report prints the ones that mean something for the code it is given.
151  *
152  * @param[in,out] fail This rank's record
153  * @param[in] fatal Whether this is a cell the fill is authoritative for
154  * @param[in] code Which guard tripped
155  * @param[in] i,j,k The cell whose state was read
156  */
157 AMREX_GPU_DEVICE AMREX_FORCE_INLINE
158 void
159 record_conversion_failure (ConversionFailure* fail,
160  const bool fatal,
161  const ConversionError code,
162  const int i, const int j, const int k,
163  const amrex::Real rho = amrex::Real(0.0),
164  const amrex::Real rho_theta = amrex::Real(0.0),
165  const amrex::Real rho_qv = amrex::Real(0.0),
166  const amrex::Real t_surf = amrex::Real(0.0),
167  const amrex::Real pressure = amrex::Real(0.0),
168  const amrex::Real delta_z = amrex::Real(0.0),
169  const amrex::Real q_air = amrex::Real(0.0))
170 {
171  if (!fatal) {
172  amrex::Gpu::Atomic::Add(&fail->halo_count, 1);
173  return;
174  }
175 
176  amrex::Gpu::Atomic::Add(&fail->count, 1);
177 
178  // The first failure to arrive owns the description; the rest only add to the count.
179  constexpr int unclaimed = static_cast<int>(ConversionError::none);
180  if (amrex::Gpu::Atomic::CAS(&fail->code, unclaimed, static_cast<int>(code)) != unclaimed) {
181  return;
182  }
183 
184  fail->i = i; fail->j = j; fail->k = k;
185  fail->rho = rho;
186  fail->rho_theta = rho_theta;
187  fail->rho_qv = rho_qv;
188  fail->t_surf = t_surf;
189  fail->pressure = pressure;
190  fail->delta_z = delta_z;
191  fail->q_air = q_air;
192 }

194 AMREX_FORCE_INLINE
195 const char*
196 conversion_error_text (const ConversionError code)
197 {
198  switch (code) {
199  case ConversionError::bad_density:
200  return "the density or (rho theta) of the state cell is not usable";
201  case ConversionError::no_moisture_comp:
202  return "the state carries no water vapor component to convert";
203  case ConversionError::bad_rho_qv:
204  return "the (rho qv) of the state cell is not usable";
205  case ConversionError::bad_qv:
206  return "the water vapor mixing ratio derived from the state cell is not usable";
207  case ConversionError::bad_theta_to_temperature:
208  return "the surface potential temperature could not be converted to absolute temperature";
209  case ConversionError::bad_temperature_to_theta:
210  return "the surface temperature could not be converted to potential temperature";
211  case ConversionError::bad_pressure:
212  return "the surface pressure derived from the state cell is not usable";
213  case ConversionError::bad_skin_temperature:
214  return "the skin temperature is not usable";
215  case ConversionError::bad_q_air:
216  return "the reference-height water vapor mixing ratio is not usable";
217  case ConversionError::bad_q_surf:
218  return "the surface mixing ratio came out non-finite";
219  default:
220  return "the conversion failed for a reason that was not recorded";
221  }

224 /**
225  * Describe what this rank recorded, then abort if any rank recorded anything.
226  *
227  * Collective: every rank must call this. Each rank that failed describes its own first
228  * failure before the abort, so a failure confined to one rank still names the cell and
229  * the numbers rather than only the check that tripped.
230  *
231  * @param[in] who Name of the fill, used to open the message
232  * @param[in] lev Level being filled
233  * @param[in] fail This rank's record
234  */
235 inline void
236 abort_on_conversion_failure (const char* who,
237  const int lev,
238  ConversionFailure const& fail)
239 {
240  int any_failed = (fail.code != static_cast<int>(ConversionError::none)) ? 1 : 0;
241  amrex::ParallelDescriptor::ReduceIntMax(any_failed);
242  if (any_failed == 0) { return; }
243 
244  if (fail.code != static_cast<int>(ConversionError::none)) {
245  const auto code = static_cast<ConversionError>(fail.code);
246 
247  const bool is_conversion = (code == ConversionError::bad_theta_to_temperature ||
248  code == ConversionError::bad_temperature_to_theta);
249  const bool has_state = (code != ConversionError::no_moisture_comp);
250  const bool has_vapor = (is_conversion ||
251  code == ConversionError::bad_rho_qv ||
252  code == ConversionError::bad_qv ||
253  code == ConversionError::bad_q_air ||
254  code == ConversionError::bad_q_surf);
255  const bool has_surface = (is_conversion ||
256  code == ConversionError::bad_pressure ||
257  code == ConversionError::bad_skin_temperature ||
258  code == ConversionError::bad_q_surf);
259  const bool has_q_air = (code == ConversionError::bad_q_air ||
260  code == ConversionError::bad_q_surf);
261 
262  // Assemble the whole report before writing it, so that ranks that failed do not
263  // interleave their lines with one another
264  std::ostringstream msg;
265  msg << who << ": surface conversion failed at level " << lev
266  << " [rank " << amrex::ParallelDescriptor::MyProc() << "]\n"
267  << " " << fail.count << " of the cells this fill owns failed";
268  if (fail.halo_count > 0) {
269  msg << " (and " << fail.halo_count << " halo cells, which are not fatal)";
270  }
271  msg << "\n first failure at (" << fail.i << "," << fail.j << "," << fail.k << "): "
272  << conversion_error_text(code) << "\n";
273 
274  if (has_state) {
275  msg << std::setprecision(10) << " rho = " << fail.rho
276  << " (rho theta) = " << fail.rho_theta;
277  if (has_vapor) { msg << " (rho qv) = " << fail.rho_qv; }
278  if (has_q_air) { msg << " q_air = " << fail.q_air; }
279  if (has_surface) {
280  msg << "\n surface temperature = " << fail.t_surf
281  << " pressure = " << fail.pressure
282  << " delta_z = " << fail.delta_z;
283  }
284  msg << "\n";
285  }
286 
287  amrex::AllPrint() << msg.str() << std::flush;
288  }
289 
290  // Let every rank finish writing before one of them tears the job down
291  amrex::ParallelDescriptor::Barrier();
292  amrex::Abort(std::string(who) + ": could not convert the surface state (see above)");
293 }
