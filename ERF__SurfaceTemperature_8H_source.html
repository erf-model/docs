36 /**
37  * Diagnose pressure at a boundary face from the adjacent atmospheric cell.
38  *
39  * The state pressure is diagnosed at the cell center. Adding rho*g*delta_z
40  * gives the pressure at the boundary face, where delta_z is the signed
41  * distance from the face to the cell center. The moisture component is read
42  * only when moisture is enabled and present.
43  */
44 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
45 SurfacePressureResult
46 diagnose_surface_pressure (const amrex::Real rho,
47  const amrex::Real rho_theta,
48  const amrex::Real rho_qv,
49  const bool moist,
50  const bool have_rho_qv,
51  const amrex::Real delta_z)
52 {
53  SurfacePressureResult result{};
54  if (!amrex::Math::isfinite(rho) || rho <= amrex::Real(0.0)) {
55  result.error = SurfacePressureError::bad_density;
56  return result;
57  }
58  if (!amrex::Math::isfinite(rho_theta) || rho_theta <= amrex::Real(0.0)) {
59  result.error = SurfacePressureError::bad_rho_theta;
60  return result;
61  }
62  if (!amrex::Math::isfinite(delta_z)) {
63  result.error = SurfacePressureError::bad_delta_z;
64  return result;
65  }
66  if (moist) {
67  if (!have_rho_qv) {
68  result.error = SurfacePressureError::no_moisture_comp;
69  return result;
70  }
71  if (!amrex::Math::isfinite(rho_qv)) {
72  result.error = SurfacePressureError::bad_rho_qv;
73  return result;
74  }
75  result.qv = rho_qv / rho;
76  if (!amrex::Math::isfinite(result.qv)) {
77  result.error = SurfacePressureError::bad_qv;
78  return result;
79  }
80  // Advection can produce finite negative undershoots. Treat them as dry for this
81  // thermodynamic diagnostic; the prognostic rho_qv field is left untouched.
82  result.qv = amrex::max(result.qv, amrex::Real(0.0));
83  }
84  result.pressure = getPgivenRTh(rho_theta, result.qv) +
85  rho * CONST_GRAV * delta_z;
86  if (!amrex::Math::isfinite(result.pressure) ||
87  result.pressure <= amrex::Real(0.0)) {
88  result.error = SurfacePressureError::bad_pressure;
89  }
90  return result;
91 }

283 /**
284  * Record one cell's failure.
285  *
286  * Every failing cell adds to a count; the first authoritative one to arrive also claims
287  * the description, so what gets reported is a failure that matters rather than whichever
288  * thread happened to finish last. Pass the values the caller has reached by this point:
289  * the report prints the ones that mean something for the code it is given.
290  *
291  * @param[in,out] fail This rank's record
292  * @param[in] fatal Whether this is a cell the fill is authoritative for
293  * @param[in] code Which guard tripped
294  * @param[in] i,j,k The cell whose state was read
295  */
296 AMREX_GPU_DEVICE AMREX_FORCE_INLINE
297 void
298 record_conversion_failure (ConversionFailure* fail,
299  const bool fatal,
300  const ConversionError code,
301  const int i, const int j, const int k,
302  const amrex::Real rho = amrex::Real(0.0),
303  const amrex::Real rho_theta = amrex::Real(0.0),
304  const amrex::Real rho_qv = amrex::Real(0.0),
305  const amrex::Real t_surf = amrex::Real(0.0),
306  const amrex::Real pressure = amrex::Real(0.0),
307  const amrex::Real delta_z = amrex::Real(0.0),
308  const amrex::Real q_air = amrex::Real(0.0))
309 {
310  if (!fatal) {
311  amrex::Gpu::Atomic::Add(&fail->halo_count, 1);
312  return;
313  }
314 
315  amrex::Gpu::Atomic::Add(&fail->count, 1);
316 
317  // The first failure to arrive owns the description; the rest only add to the count.
318  constexpr int unclaimed = static_cast<int>(ConversionError::none);
319  if (amrex::Gpu::Atomic::CAS(&fail->code, unclaimed, static_cast<int>(code)) != unclaimed) {
320  return;
321  }
322 
323  fail->i = i; fail->j = j; fail->k = k;
324  fail->rho = rho;
325  fail->rho_theta = rho_theta;
326  fail->rho_qv = rho_qv;
327  fail->t_surf = t_surf;
328  fail->pressure = pressure;
329  fail->delta_z = delta_z;
330  fail->q_air = q_air;
331 }

391 /**
392  * Describe what this rank recorded, then abort if any rank recorded anything.
393  *
394  * Collective: every rank must call this. Each rank that failed describes its own first
395  * failure before the abort, so a failure confined to one rank still names the cell and
396  * the numbers rather than only the check that tripped.
397  *
398  * @param[in] who Name of the fill, used to open the message
399  * @param[in] lev Level being filled
400  * @param[in] fail This rank's record
401  */
402 inline void
403 abort_on_conversion_failure (const char* who,
404  const int lev,
405  ConversionFailure const& fail)
406 {
407  int any_failed = (fail.code != static_cast<int>(ConversionError::none)) ? 1 : 0;
408  amrex::ParallelDescriptor::ReduceIntMax(any_failed);
409  if (any_failed == 0) { return; }
410 
411  if (fail.code != static_cast<int>(ConversionError::none)) {
412  const auto code = static_cast<ConversionError>(fail.code);
413 
414  const bool is_conversion = (code == ConversionError::bad_theta_to_temperature ||
415  code == ConversionError::bad_temperature_to_theta);
416  const bool has_state = (code != ConversionError::no_moisture_comp);
417  const bool has_vapor = (is_conversion ||
418  code == ConversionError::bad_rho_qv ||
419  code == ConversionError::bad_qv ||
420  code == ConversionError::bad_q_air ||
421  code == ConversionError::bad_q_surf);
422  const bool has_surface = (is_conversion ||
423  code == ConversionError::bad_pressure ||
424  code == ConversionError::bad_delta_z ||
425  code == ConversionError::bad_skin_temperature ||
426  code == ConversionError::bad_q_surf);
427  const bool has_q_air = (code == ConversionError::bad_q_air ||
428  code == ConversionError::bad_q_surf);
429 
430  // Assemble the whole report before writing it, so that ranks that failed do not
431  // interleave their lines with one another
432  std::ostringstream msg;
433  msg << who << ": surface conversion failed at level " << lev
434  << " [rank " << amrex::ParallelDescriptor::MyProc() << "]\n"
435  << " " << fail.count << " of the cells this fill owns failed";
436  if (fail.halo_count > 0) {
437  msg << " (and " << fail.halo_count << " halo cells, which are not fatal)";
438  }
439  msg << "\n first failure at (" << fail.i << "," << fail.j << "," << fail.k << "): "
440  << conversion_error_text(code) << "\n";
441 
442  if (has_state) {
443  msg << std::setprecision(10) << " rho = " << fail.rho
444  << " (rho theta) = " << fail.rho_theta;
445  if (has_vapor) { msg << " (rho qv) = " << fail.rho_qv; }
446  if (has_q_air) { msg << " q_air = " << fail.q_air; }
447  if (has_surface) {
448  msg << "\n surface temperature = " << fail.t_surf
449  << " pressure = " << fail.pressure
450  << " delta_z = " << fail.delta_z;
451  }
452  msg << "\n";
453  }
454 
455  amrex::AllPrint() << msg.str() << std::flush;
456  }
457 
458  // Let every rank finish writing before one of them tears the job down
459  amrex::ParallelDescriptor::Barrier();
460  amrex::Abort(std::string(who) + ": could not convert the surface state (see above)");
461 }
