18 //
19 // Station time-series output.
20 //
21 // A station is a named set of geographic points at which a user-chosen set of
22 // variables is written to an ASCII time series, for comparison against
23 // meteorological tower and surface-station observations. The inputs follow the
24 // same shape as erf.refinement_indicators:
25 //
26 // erf.station_names = Lake1 Forests
27 // erf.Lake1.field = rain_accum
28 // erf.Lake1.lat = 45.13
29 // erf.Lake1.long = -122.34
30 // erf.Forests.field = magvel local_helicity
31 // erf.Forests.lat = 45.20 45.41
32 // erf.Forests.long = -122.10 -122.02
33 // erf.Forests.height_agl = 10.0 80.0
34 //
35 // lat/long are paired positionally, so the example above is two locations, not
36 // four. A station may instead be placed with .x/.y in domain coordinates, which
37 // is the only option for a run that has no lat/lon arrays. The heights apply to
38 // every location of the station and are given one way or the other, never both:
39 // .height_agl is metres above the local terrain, .height_abs is metres in the
40 // model's own vertical coordinate, the same z as geometry.prob_lo/prob_hi.
41 //
42 // The variable names accepted are the 3D plotfile variables and the built-in 2D
43 // diagnostics, and the values come from the same fill routines
44 // (ERF::FillPlot3DVars / ERF::FillPlot2DVars), so a station column and the
45 // corresponding plotfile component cannot disagree. A 2D plotfile can also
46 // carry sampled-level fields, named for the field and the level such as
47 // 'theta_z100m'; those are not accepted here, since a station asks for the 3D
48 // variable and a height directly.
49 //
50 // Four more names are accepted, which neither catalog holds as a 3D variable:
51 // the earth-relative winds u_east, v_north, wind_speed and wind_dir. They are
52 // formed from the station's interpolated x_velocity and y_velocity, rotated by
53 // the grid orientation at the station (see ERF_EarthRelativeWind.H), so that a
54 // column compares directly with a tower on a rotated (e.g. Lambert) grid.
55 //
56 
57 // Which earth-relative wind a station variable is; None for every other variable
58 enum class StationEarthWind { None, UEast, VNorth, WindSpeed, WindDir };
59 
60 inline StationEarthWind
61 station_earth_wind_from_name (const std::string& name)
62 {
63  if (name == "u_east") { return StationEarthWind::UEast; }
64  if (name == "v_north") { return StationEarthWind::VNorth; }
65  if (name == "wind_speed") { return StationEarthWind::WindSpeed; }
66  if (name == "wind_dir") { return StationEarthWind::WindDir; }

87 //
88 // Bilinear interpolation of the four values around a station, with weights
89 // (1-wx)(1-wy), wx(1-wy), (1-wx)wy and wx*wy. A value equal to missing_value, or
90 // not finite, is left out and the weights of the others are renormalised, so a
91 // missing neighbour cannot be blended in as a number; missing_value is returned
92 // only when every neighbour with a nonzero weight is missing.
93 //
94 inline amrex::Real
95 station_bilinear_skip_missing (amrex::Real v00, amrex::Real v10,
96  amrex::Real v01, amrex::Real v11,
97  amrex::Real wx, amrex::Real wy,
98  amrex::Real missing_value)
99 {
100  const amrex::Real v[4] = {v00, v10, v01, v11};
101  const amrex::Real w[4] = {(amrex::Real(1.0)-wx)*(amrex::Real(1.0)-wy),
102  wx*(amrex::Real(1.0)-wy),
103  (amrex::Real(1.0)-wx)*wy,
104  wx*wy};
105  amrex::Real sum = amrex::Real(0.0);
106  amrex::Real wsum = amrex::Real(0.0);
107  for (int n = 0; n < 4; ++n) {
108  if (w[n] > amrex::Real(0.0) && std::isfinite(v[n]) && v[n] != missing_value) {
109  sum += w[n] * v[n];
110  wsum += w[n];
111  }
112  }
113  return (wsum > amrex::Real(0.0)) ? sum / wsum : missing_value;
114 }
