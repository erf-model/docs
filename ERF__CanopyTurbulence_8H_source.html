17 /**
18  * Canopy terms of the one-equation k RANS model (erf.rans_type = kEqn).
19  *
20  * The forest drag removes mean kinetic energy from the flow. Part of it becomes
21  * turbulence in the wakes of leaves and branches, at scales far below the mesh,
22  * and dissipates quickly (the spectral short cut). With
23  * erf.forest_canopy_tke = true, two terms act where the forest drag field
24  * f = C_d LAD of ForestDrag is positive:
25  *
26  * - a TKE source, S = rho f (beta_p |U|^3 - beta_d |U| k)
27  * (Green 1992, Liu et al. 1996, Sanz 2003). Production and sink are
28  * integrated together and exactly over each RK stage, from the k at the
29  * start of the step (see tke_source), so the canopy sink never removes more
30  * than that k;
31  * - a cap on the turbulent length scale, L <= alpha / f (Harman and Finnigan
32  * 2007, alpha = 2 beta^3 with beta = u_* / U_h), applied before the
33  * dissipation, the Richardson number and the stability functions are
34  * evaluated from L. The cap needs erf.implicit_tke_dissipation = true
35  * (CanopyTurbulence::check_config): with L = alpha / f the explicit
36  * dissipation rate Cmu0^3 sqrt(k) f / alpha can exceed 1/dt.
37  *
38  * Input names, defaults and formulas follow the ForestDrag.canopy_* inputs of
39  * the KLAxell model in kynema-sgf (branch f/winsent of rthedin/amr-wind). In
40  * neutral air both codes run the same canopy model; Forest.rst lists where the
41  * time integration and the stratified closure differ.
42  */

45 /**
46  * Turbulent length scale capped in the canopy, min(length, alpha / f).
47  * Cells with f <= 0 are unchanged, and alpha = 0 disables the cap. The test
48  * alpha < length f avoids comparing with a large alpha / f. The cap is
49  * computed with a denominator that is never zero, because an optimised build
50  * may evaluate it before the test (as for 1/theta_ref in ERF_TurbKESources.H);
51  * adding the smallest normal Real leaves any f above min / epsilon unchanged.
52  */
53 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
54 amrex::Real limited_length (amrex::Real length, amrex::Real f_drag, amrex::Real alpha)
55 {
56  const amrex::Real cap = alpha / (amrex::max(f_drag, amrex::Real(0.0))
57  + std::numeric_limits<amrex::Real>::min());
58  const bool capped = (alpha > amrex::Real(0.0)) && (f_drag > amrex::Real(0.0))
59  && (alpha < length * f_drag);
60  return capped ? cap : length;
61 }

63 /**
64  * Canopy source of rho k per unit volume and time for an RK stage that updates
65  * rho k from its value at the start of the step, (rho k)^n, over dt:
66  * rho beta_p f |U|^3 phi1(x) - (rho k)^n (1 - exp(-x)) / dt,
67  * with x = beta_d f |U| dt and phi1(x) = (1 - exp(-x)) / x. Taken alone, the
68  * stage update (rho k)^n + dt S is the exact solution of
69  * d(rho k)/dt = rho P - c rho k with P = beta_p f |U|^3 and c = beta_d f |U|
70  * frozen over the stage,
71  * (rho k)^n exp(-x) + rho (P / c) (1 - exp(-x)),
72  * so the canopy terms balance at rho P / c at any dt, and the canopy sink never
73  * removes more than (rho k)^n. The RK3 stages (dt/3, dt/2, dt from the start of
74  * the step) and the trapezoidal anelastic RK2 (both stages over dt) both update
75  * from (rho k)^n.
76  * The closure's own dissipation is a separate term: the explicit form adds
77  * -dt rho Cmu0^3 k^1.5 / L to that update, and the implicit form divides the
78  * update by 1 + dt Cmu0^3 sqrt(k) / L (positive, so k stays non-negative when
79  * the other terms do). With MovingFittedMesh the update is also scaled by the
80  * old and new detJ. In these cases the stage value is not the expression above,
81  * and the steady k is the balance of all the terms, not P / c.
82  * For f = 0 the source is zero (x = 0, phi1 = 1).
83  * @param rho density of the stage
84  * @param rhok_old rho k at the start of the step
85  * @param f_drag forest drag field f = C_d LAD (1/m), not negative
86  * @param wind cell-centred wind speed |U| of the stage
87  * @param beta_p wake production coefficient
88  * @param beta_d short-circuit dissipation coefficient
89  * @param dt time step of the update the source enters (> 0)
90  */
91 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
92 amrex::Real tke_source (amrex::Real rho, amrex::Real rhok_old, amrex::Real f_drag,
93  amrex::Real wind, amrex::Real beta_p, amrex::Real beta_d,
94  amrex::Real dt)
95 {
96  const amrex::Real x = beta_d * f_drag * wind * dt;
97  // 1 - exp(-x), in [0, 1) for x >= 0
98  const amrex::Real decay = -std::expm1(-x);
99  // phi1 with a divisor that is never zero (an optimised build may evaluate it
100  // for x = 0); at the smallest normal Real, phi1 rounds to its limit 1
101  const amrex::Real x_safe = amrex::max(x, std::numeric_limits<amrex::Real>::min());
102  const amrex::Real phi1 = -std::expm1(-x_safe) / x_safe;
103  return rho * beta_p * f_drag * wind * wind * wind * phi1 - rhok_old * decay / dt;
104 }
