156  * independent face-flux view. Horizontal map-factor numerator and denominator
157  * remain separate to preserve the native arithmetic association. Physical
158  * Dirichlet flags select the legacy one-sided stencil; ext_dir_upwind selection
159  * stays in the adapter.
160  */
161 template <int Direction>
162 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE amrex::Real
163 ScalarDiffusionFlux_N(
164  const amrex::Array4<const amrex::Real>& scalar,
165  const int scalar_comp,
166  const int i,
167  const int j,
168  const int k,
169  const amrex::Real coeff,
170  const amrex::Real inv_spacing,
171  const amrex::Real map_numerator,
172  const amrex::Real map_denominator,
173  const bool ext_dir_lo,
174  const bool ext_dir_hi) noexcept
175 {
176  static_assert(Direction >= 0 && Direction < 3);
177  amrex::Real grad;
178  if constexpr (Direction == 0) {
179  if (ext_dir_lo) {
180  grad = -(amrex::Real(8.0) / amrex::Real(3.0)) *
181  scalar(i - 1, j, k, scalar_comp) +
182  amrex::Real(3.0) * scalar(i, j, k, scalar_comp) -
183  (amrex::Real(1.0) / amrex::Real(3.0)) *
184  scalar(i + 1, j, k, scalar_comp);
185  } else if (ext_dir_hi) {
186  grad =
187  (amrex::Real(8.0) / amrex::Real(3.0)) * scalar(i, j, k, scalar_comp) -
188  amrex::Real(3.0) * scalar(i - 1, j, k, scalar_comp) +
189  (amrex::Real(1.0) / amrex::Real(3.0)) *
190  scalar(i - 2, j, k, scalar_comp);
191  } else {
192  grad = scalar(i, j, k, scalar_comp) - scalar(i - 1, j, k, scalar_comp);
193  }
194  } else if constexpr (Direction == 1) {
195  if (ext_dir_lo) {
196  grad = -(amrex::Real(8.0) / amrex::Real(3.0)) *
197  scalar(i, j - 1, k, scalar_comp) +
198  amrex::Real(3.0) * scalar(i, j, k, scalar_comp) -
199  (amrex::Real(1.0) / amrex::Real(3.0)) *
200  scalar(i, j + 1, k, scalar_comp);
201  } else if (ext_dir_hi) {
202  grad =
203  (amrex::Real(8.0) / amrex::Real(3.0)) * scalar(i, j, k, scalar_comp) -
204  amrex::Real(3.0) * scalar(i, j - 1, k, scalar_comp) +
205  (amrex::Real(1.0) / amrex::Real(3.0)) *
206  scalar(i, j - 2, k, scalar_comp);
207  } else {
208  grad = scalar(i, j, k, scalar_comp) - scalar(i, j - 1, k, scalar_comp);
209  }
210  } else {
211  if (ext_dir_lo) {
212  grad = -(amrex::Real(8.0) / amrex::Real(3.0)) *
213  scalar(i, j, k - 1, scalar_comp) +
214  amrex::Real(3.0) * scalar(i, j, k, scalar_comp) -
215  (amrex::Real(1.0) / amrex::Real(3.0)) *
216  scalar(i, j, k + 1, scalar_comp);
217  } else if (ext_dir_hi) {
218  grad =
219  (amrex::Real(8.0) / amrex::Real(3.0)) * scalar(i, j, k, scalar_comp) -
220  amrex::Real(3.0) * scalar(i, j, k - 1, scalar_comp) +
221  (amrex::Real(1.0) / amrex::Real(3.0)) *
222  scalar(i, j, k - 2, scalar_comp);
223  } else {
224  grad = scalar(i, j, k, scalar_comp) - scalar(i, j, k - 1, scalar_comp);
225  }
226  }
227  if constexpr (Direction == 2) {
228  return -coeff * grad * inv_spacing;
229  } else {
230  return -coeff * grad * inv_spacing * map_numerator / map_denominator;
231  }
232 }

525  xbar = amrex::Real(1.5) * xlo - myhalf * xhi;
526  const amrex::Real ylo = myhalf * (raw_yflux(i, j, klo, flux_comp) +
527  raw_yflux(i, j + 1, klo, flux_comp));
528  const amrex::Real yhi = myhalf * (raw_yflux(i, j, klo + 1, flux_comp) +
529  raw_yflux(i, j + 1, klo + 1, flux_comp));
530  ybar = amrex::Real(1.5) * ylo - myhalf * yhi;
531  } else if (kface == khi) {
532  const amrex::Real xlo = myhalf * (raw_xflux(i, j, khi - 2, flux_comp) +
533  raw_xflux(i + 1, j, khi - 2, flux_comp));
534  const amrex::Real xhi = myhalf * (raw_xflux(i, j, khi - 1, flux_comp) +
535  raw_xflux(i + 1, j, khi - 1, flux_comp));
536  xbar = amrex::Real(1.5) * xhi - myhalf * xlo;
537  const amrex::Real ylo = myhalf * (raw_yflux(i, j, khi - 2, flux_comp) +
538  raw_yflux(i, j + 1, khi - 2, flux_comp));
539  const amrex::Real yhi = myhalf * (raw_yflux(i, j, khi - 1, flux_comp) +
540  raw_yflux(i, j + 1, khi - 1, flux_comp));
541  ybar = amrex::Real(1.5) * yhi - myhalf * ylo;
542  } else {
543  xbar = fourth * (raw_xflux(i, j, kface, flux_comp) +
544  raw_xflux(i + 1, j, kface, flux_comp) +
545  raw_xflux(i, j, kface - 1, flux_comp) +
546  raw_xflux(i + 1, j, kface - 1, flux_comp));
547  ybar = fourth * (raw_yflux(i, j, kface, flux_comp) +
548  raw_yflux(i, j + 1, kface, flux_comp) +
549  raw_yflux(i, j, kface - 1, flux_comp) +
550  raw_yflux(i, j + 1, kface - 1, flux_comp));
551  }
552 
553  const amrex::Real mx = mf_mx(i, j, 0);
554  const amrex::Real my = mf_my(i, j, 0);
555  const amrex::Real h_xi = Compute_h_xi_AtKface(i, j, kface, cellSizeInv, z_nd);
556  const amrex::Real h_eta =
557  Compute_h_eta_AtKface(i, j, kface, cellSizeInv, z_nd);
558  const amrex::Real Gz = TerrainDiffusionGz(
559  raw_zflux(i, j, kface, flux_comp), mx, h_xi, xbar, my, h_eta, ybar);
560  return Gz;
561 }

597 /**
598  * Return the positive geometry-weighted mapped flux divergence. For
599  * detJ_cell <= 0, return zero without dividing. The diffusion array wrapper
600  * subtracts this value from the existing RHS.
601  */
602 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE amrex::Real
603 ScalarDiffusionMappedFluxDivergence(
604  const amrex::Real fx_hi,
605  const amrex::Real fx_lo,
606  const amrex::Real fy_hi,
607  const amrex::Real fy_lo,
608  const amrex::Real fz_hi,
609  const amrex::Real fz_lo,
610  const amrex::Real dx_inv,
611  const amrex::Real dy_inv,
612  const amrex::Real dzeta_inv,
613  const amrex::Real mx,
614  const amrex::Real my,
615  const amrex::Real detJ_cell) noexcept
616 {
617  if (detJ_cell <= amrex::Real(0.0))
618  return amrex::Real(0.0);
619  const amrex::Real map_over_jacobian = (mx * my) / detJ_cell;
620  return map_over_jacobian *
621  ((fx_hi - fx_lo) * dx_inv + (fy_hi - fy_lo) * dy_inv +
622  (fz_hi - fz_lo) * dzeta_inv);
623 }

625 /** Native terrain divergence in the legacy arithmetic association. */
626 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE amrex::Real
627 TerrainDiffusionDivergence_T(
628  const amrex::Real mapped_Fx_hi,
629  const amrex::Real mapped_Fx_lo,
630  const amrex::Real mapped_Fy_hi,
631  const amrex::Real mapped_Fy_lo,
632  const amrex::Real Gz_hi,
633  const amrex::Real Gz_lo,
634  const amrex::Real dx_inv,
635  const amrex::Real dy_inv,
636  const amrex::Real dz_inv,
637  const amrex::Real mfsq,
638  const amrex::Real detJ_cell) noexcept

644  stateContrib /= detJ_cell;
645  return stateContrib;
