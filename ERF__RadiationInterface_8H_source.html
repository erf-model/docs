12 // True when every box of ba spans dom in z. The radiation column sweeps need a whole column
13 // inside ONE box, so this tests each box rather than the level's bounding box: a level with a
14 // full-column box and a box aloft elsewhere has a bounding box reaching both domain ends while
15 // not every box holds a column. The one definition of the test, shared by every radiation model
16 // and by ERF::rad_level_needs_interpolation, so they cannot disagree about a level.
17 inline bool
18 rad_boxes_span_column (const amrex::BoxArray& ba, const amrex::Box& dom)
19 {
20  for (int ib = 0; ib < ba.size(); ++ib) {
21  if (ba[ib].smallEnd(2) != dom.smallEnd(2) || ba[ib].bigEnd(2) != dom.bigEnd(2)) {
22  return false;
23  }
24  }
25  return true;
26 }
