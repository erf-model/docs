6 /**
7  * Restriction of a quantity held in the ground layer of cells (a surface accumulation) from a
8  * fine level to the coarse level below it. Each coarse ground cell under a fine box that stands
9  * on the ground gets a scale times the sum of the rr[0]*rr[1] fine ground cells above it: the
10  * inverse area for a depth, 1 for an amount per cell. The coarse ground boxes, distributed with
11  * the fine boxes they lie under, are built once and serve every quantity restricted with them.
12  */

20  */

54  const auto c = m_tmp.array(mfi);
55  amrex::ParallelFor(mfi.validbox(), [=] AMREX_GPU_DEVICE (int ic, int jc, int kc) noexcept

61  }
62  }
