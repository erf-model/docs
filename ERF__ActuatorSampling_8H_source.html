25 /**
26  * Sample the velocity at a set of points from the face-centred velocity fields.
27  *
28  * Each component is interpolated trilinearly from its own staggered grid: u from the x faces,
29  * v from the y faces, w from the z faces, so a field linear in x, y and z is reproduced
30  * exactly. The vertical weight uses the physical face heights (z_phys_nd on a stretched or
31  * terrain-following mesh, prob_lo + k dz otherwise), so the same holds on those meshes as long
32  * as the field is linear in the physical height.
33  *
34  * Every rank interpolates the points whose containing cell lies in one of its valid boxes,
35  * reading the neighbours from the ghost cells, and the results are summed across ranks; the
36  * velocity fields must therefore have at least one filled ghost cell, and z_phys_nd one ghost
37  * node, on the velocities' boxes and distribution. The ghost cells across the domain's lateral
38  * boundaries are read too (a point within half a cell of a non-periodic face takes up to half
39  * its value from them), so the fields must be filled there as well (FillPatch, not only
40  * FillBoundary). In a periodic direction a point is moved by whole periods into the domain; a
41  * point on the domain's upper x, y or z face belongs to the last cell. A point outside the
42  * domain otherwise, or one found by no rank or by more than one, aborts
43  * with its coordinates: the caller owns the points and must keep them inside the domain. Every
44  * rank must pass the same points: the reduction is collective and the call stops when the ranks
45  * disagree on their number. A model whose ranks hold different points (those of the bodies
46  * each owns) calls sample_velocity_local() instead, which has no such check.
47  *
48  * @param[in] U,V,W face-centred velocities on one level, ghost cells filled (m/s)
49  * @param[in] z_phys_nd nodal physical heights (m), or nullptr on a uniform-dz mesh
50  * @param[in] geom the level's geometry
51  * @param[in] pos 3*n point coordinates (x0,y0,z0, x1,y1,z1, ...) (m, ERF domain coordinates)
52  * @param[out] vel 3*n sampled velocities (m/s), the same on every rank on return

122 /**
123  * Return whether every point, with the cells within `reach` metres around it, lies inside the
124  * union of boxes of a level (the anchor level's grids): the sampler needs the neighbouring cells
125  * and the spreader the kernel's reach, and on a refined level that is not the whole domain. In a
126  * periodic direction the needed cells wrap across the seam to the domain's other side; in a
127  * non-periodic one they are clamped to the domain. Returns false and names the first point
128  * outside in `first_outside`. On a terrain-following or stretched mesh a point's k is not its
129  * height over dz, so z is checked as `z` says: Reach (the cells within reach of the height over dz,
130  * for a flat, uniform mesh), Footprint (the reach across only, at any height: the sampler then finds
131  * a point above or below the grids by itself), or Column (for the spreading, whose kernel would
132  * otherwise be cut where the grids stop: the reach across over every cell the heights within reach
133  * may lie in, over the point's own columns from `cells` (column_cell_ranges, one range per point),
134  * else whatever the column from the mesh's bounds `zb` (mesh_z_bounds), or over the whole height
135  * without either). With `cells` a point whose range is empty (kb > kt) is not covered.
136  * Not with a periodic z.
137  */

159  /**
160  * The cells k that heights from zl to zh may lie in, whatever the column: [kb, kt], clamped to
161  * [klo - 1, khi + 1] (so a range outside the domain stays outside it); [klo, khi] without bounds.
162  * With the table: exactly the boxes' cells k whose lowest bottom node is not above zh and whose
163  * highest top node is not below zl, or kb > kt when there is none. zl or zh may be infinite (not NaN):
164  * heights from -infinity take every cell up to zh's, and a range at one infinity takes no cell of
165  * the domain.
166  */
167  void k_range (amrex::Real zl, amrex::Real zh, int klo, int khi, int& kb, int& kt) const
168  {
169  kb = klo;
170  kt = khi;
171  if (has_table()) {
172  kb = khi + 1;
173  kt = klo - 1;
174  // a relative 1e-5 more each way: the caller's zl and zh are rounded sums (a point's height and
175  // its reach), and a face at a node level's own height h just past them still has weight
176  // From the finite ends only: an infinite end needs none. An infinite end is replaced by 0 before
177  // the difference is taken, not skipped by a test, so that inf - inf is never computed (a compiler
178  // may evaluate both sides of a test, and the invalid operation would trap)
179  const amrex::Real fl = std::isinf(zl) ? amrex::Real(0.0) : zl;
180  const amrex::Real fh = std::isinf(zh) ? amrex::Real(0.0) : zh;
181  const amrex::Real slack = amrex::Real(1.0e-5) * std::max(std::abs(fl), std::max(std::abs(fh), std::abs(fh - fl)));
182  const amrex::Real lo = zl - slack, hi = zh + slack;
183  for (std::size_t n = 0; n < has_cell.size(); ++n) {
184  if (has_cell[n] == 0 || node_lo[n] > hi || node_hi[n+1] < lo) { continue; }
185  const int k = k0 + static_cast<int>(n);
186  kb = std::min(kb, k);
187  kt = std::max(kt, k);
188  }
189  return;
190  }
191  if (!has_ground()) { return; }
192  // counted from the ground's cell klo. A cell that only touches the range at a node is not always
193  // within [kb, kt]: the spreader takes one face level more each way (reach_box_k())
194  const double b = klo + std::floor((static_cast<double>(zl) - ground_hi) / dz_hi);
195  const double t = klo + std::floor((static_cast<double>(zh) - ground_lo) / dz_lo);
196  kb = static_cast<int>(std::min<double>(khi + 1, std::max<double>(klo - 1, b)));
197  kt = static_cast<int>(std::min<double>(khi + 1, std::max<double>(klo - 1, t)));
198  }
199 };

264  * in some node column the reach across touches (a periodic x or y wrapping across the seam, as

267  * a box's lowest node in a column (the box's bottom face, unless it is the domain's floor), the cell under
