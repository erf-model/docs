35 amrex::Real compute_relative_humidity(amrex::Real qv, amrex::Real T, amrex::Real P)
36 {
37  // Guard against invalid inputs
38  if (T <= 0.0 || P <= 0.0 || qv < 0.0 || !std::isfinite(qv) ||
39  !std::isfinite(T) || !std::isfinite(P)) {
40  return 0.0;
41  }
42 
43  // Simple saturation mixing ratio approximation (Magnus formula variant)
44  // qsat = 0.622 * e_sat / (P - e_sat)
45  // where e_sat = e0 * exp(a * (T - T0) / (T - b))
46  // For simplicity, use approximate saturation vapor pressure
47  const amrex::Real e0 = 611.2; // Pa (saturation vapor pressure at 0°C)
48  const amrex::Real T0 = 273.15; // K
49  const amrex::Real a = 17.62;
50  const amrex::Real b = 243.12; // K
51 
52  // Compute saturation vapor pressure (Magnus formula)
53  amrex::Real arg = a * (T - T0) / (T - b);
54  if (!std::isfinite(arg) || arg > 100.0) {
55  // Protect against overflow
56  arg = 100.0;
57  }
58  amrex::Real e_sat = e0 * std::exp(arg);
59 
60  // Saturation mixing ratio: qsat = epsilon * e_sat / (P - e_sat)
61  const amrex::Real epsilon = 0.622; // ratio of molecular weights
62  amrex::Real qsat = epsilon * e_sat / (P - e_sat);
63  if (qsat < 0.0 || !std::isfinite(qsat)) {
64  qsat = 1.0e-6; // fallback
65  }
66 
67  // RH = qv / qsat, clamped to [0, 1]
68  amrex::Real rh = qv / qsat;
69  if (rh < 0.0) rh = 0.0;
70  if (rh > 1.0) rh = 1.0;
71 
72  return rh;
73 }

105  !std::isfinite(qc_scale)) {
106  return 0.0; // Safe fallback on invalid input
107  }
108 
109  // Clamp RH and qc to valid ranges
110  if (rh < 0.0) rh = 0.0;
111  if (rh > 1.0) rh = 1.0;
112  if (qc < 0.0) qc = 0.0;
113 
114  amrex::Real cf = 0.0;
115 
116  // RH contribution: linear ramp from 0 at rh_min to 1 at rh_max
117  if (rh_max > rh_min) {

119  amrex::Real cf_rh = (rh - rh_min) / (rh_max - rh_min);
120  if (cf_rh < 0.0) cf_rh = 0.0;
121  if (cf_rh > 1.0) cf_rh = 1.0;
122  cf += cf_rh;
123  } else if (rh >= rh_min) {
124  // rh_max == rh_min: step function
125  cf += 1.0;
126  }
127 
128  // qc contribution: qc_scale is the cloud water [kg/kg] at which the
129  // liquid-water term alone saturates, so cf_qc = qc / qc_scale, capped at 1.
130  if (qc_scale > 0.0) {

132  if (cf_qc > 1.0) cf_qc = 1.0;
133  cf += cf_qc;
134  }
135 
136  // Saturate combined cloud fraction to [0, 1]
137  if (cf < 0.0) cf = 0.0;
138  if (cf > 1.0) cf = 1.0;

140  // Final sanity check

142  cf = 0.0;
143  }
