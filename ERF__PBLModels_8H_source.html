301 AMREX_GPU_DEVICE
302 AMREX_FORCE_INLINE
303 void
304 ComputeVerticalDerivativesPBL (int i, int j, int k,
305  const amrex::Array4<const amrex::Real>& uvel,
306  const amrex::Array4<const amrex::Real>& vvel,
307  const amrex::Array4<const amrex::Real>& cell_data,
308  const int izmin,
309  const int izmax,
310  const PBLDerivativeDzInv& dz_inv,
311  const bool c_ext_dir_on_zlo,
312  const bool c_ext_dir_on_zhi,
313  const bool u_ext_dir_on_zlo,
314  const bool u_ext_dir_on_zhi,
315  const bool v_ext_dir_on_zlo,
316  const bool v_ext_dir_on_zhi,
317  amrex::Real& dthetadz,
318  amrex::Real& dudz,
319  amrex::Real& dvdz,
320  const MoistureComponentIndices& moisture_indices,
321  const StratType strat = StratType::thetav)
322 {
323  // ext_dir ghost values hold the value at the physical boundary, which is
324  // half a ghost-center spacing from the adjacent cell center. Use a
325  // two-point, first-order boundary-to-center difference with inverse
326  // distance 2/dz.
327  if ( k==izmax && c_ext_dir_on_zhi ) {
328  dthetadz = ( GetPBLStratVar(i,j,k+1,strat,cell_data,moisture_indices)
329  -GetPBLStratVar(i,j,k ,strat,cell_data,moisture_indices) )*dz_inv.dirichlet_hi;
330  } else if ( k==izmin && c_ext_dir_on_zlo ) {
331  dthetadz = ( GetPBLStratVar(i,j,k ,strat,cell_data,moisture_indices)
332  -GetPBLStratVar(i,j,k-1,strat,cell_data,moisture_indices) )*dz_inv.dirichlet_lo;
333  } else {
334  dthetadz = ( GetPBLStratVar(i,j,k+1,strat,cell_data,moisture_indices)
335  -GetPBLStratVar(i,j,k-1,strat,cell_data,moisture_indices) )*dz_inv.interior;
336  }
337 
338  if ( k==izmax && u_ext_dir_on_zhi ) {
339  dudz = myhalf*( uvel(i ,j,k+1) - uvel(i ,j,k)
340  +uvel(i+1,j,k+1) - uvel(i+1,j,k) )*dz_inv.dirichlet_hi;
341  } else if ( k==izmin && u_ext_dir_on_zlo ) {
342  dudz = myhalf*( uvel(i ,j,k) - uvel(i ,j,k-1)
343  +uvel(i+1,j,k) - uvel(i+1,j,k-1) )*dz_inv.dirichlet_lo;
344  } else {
345  dudz = myhalf*( uvel(i ,j,k+1) - uvel(i ,j,k-1)
346  +uvel(i+1,j,k+1) - uvel(i+1,j,k-1) )*dz_inv.interior;
347  }
348 
349  if ( k==izmax && v_ext_dir_on_zhi ) {
350  dvdz = myhalf*( vvel(i,j ,k+1) - vvel(i,j ,k)
351  +vvel(i,j+1,k+1) - vvel(i,j+1,k) )*dz_inv.dirichlet_hi;
352  } else if ( k==izmin && v_ext_dir_on_zlo ) {
353  dvdz = myhalf*( vvel(i,j ,k) - vvel(i,j ,k-1)
354  +vvel(i,j+1,k) - vvel(i,j+1,k-1) )*dz_inv.dirichlet_lo;
355  } else {
356  dvdz = myhalf*( vvel(i,j ,k+1) - vvel(i,j ,k-1)
357  +vvel(i,j+1,k+1) - vvel(i,j+1,k-1) )*dz_inv.interior;
358  }
359 }

373 /**

376  *
377  * @param[in] t_star temperature scale
378  * @param[in] q_star moisture scale

384 AMREX_FORCE_INLINE
385 amrex::Real
386 ComputeVirtualTStarPBL (amrex::Real t_star,

388  amrex::Real theta,
389  amrex::Real qv,
390  bool use_moisture)
391 {
392  amrex::Real tstar_v = t_star * (one + epsv*qv);

397 /**
398  * Overwrite the vertical gradients in the first cell above the surface with the
399  * MOST profile gradients,
400  * |dU/dz| = u_* phi_m(zeta) / (kappa z)
401  * dthetav/dz = thetav_* phi_h(zeta) / (kappa z)
402  *
403  * With zlo.type = surface_layer the state carries foextrap and the velocities
404  * hoextrap at the lower boundary, so the centered stencil in
405  * ComputeVerticalDerivativesPBL reads ghost values that hold no surface
406  * information there: the potential-temperature gradient comes out half the
407  * one-sided value and the velocity gradient has no dependence on the friction
408  * velocity at all (ERF #4037). The MOST profile used here is the one the surface
409  * stress and heat flux were themselves derived from.
410  *
411  * Every PBL scheme that consumes these gradients uses only the shear magnitude
412  * dudz^2 + dvdz^2, so that magnitude is carried in dudz and dvdz is left at zero
413  * rather than being split by wind direction.
414  *
415  * @param[in] sl Monin-Obukhov inputs for this column
416  * @param[out] dthetadz vertical derivative of virtual potential temperature
417  * @param[out] dudz vertical derivative of x-velocity (carries the shear magnitude)
418  * @param[out] dvdz vertical derivative of y-velocity (set to zero)
419  */
420 AMREX_GPU_DEVICE
421 AMREX_FORCE_INLINE
422 void
423 ApplySurfaceLayerGradientsPBL (const PBLSurfaceLayerGradient& sl,
424  amrex::Real& dthetadz,
425  amrex::Real& dudz,
426  amrex::Real& dvdz)
427 {
428  const similarity_funs sfuns{};
429  const amrex::Real inv_kappa_z = one / (KAPPA * sl.zval);
430  dudz = sl.u_star * sfuns.calc_phi_m(sl.zeta) * inv_kappa_z;
431  dvdz = zero;
432  dthetadz = sl.tstar_v * sfuns.calc_phi_h(sl.zeta) * inv_kappa_z;
433 }

492  // - Eddy diffusivities are updated at the beginning of each time step only.
493 
494  const bool use_surface_fluxes = ( surface_layer_on_zlo && (k == izmin) &&
495  tau13 && tau23 && hfx_z );
496 
497  if (use_surface_fluxes)
498  {
499  // The first cell center lies inside the unresolved surface layer. With
500  // zlo.type = surface_layer the state carries foextrap and the velocities
501  // hoextrap at the lower boundary, so the stencil used below reads ghost
502  // values that hold no surface information: the resolved gradients there
503  // are not a discretization of anything (ERF #4037). Build the production
504  // terms from the fluxes the model actually carries instead:
505  //
506  // tau13, tau23 = rho<u'w'>, rho<v'w'> (z-faces, staggered in x,y)
507  // hfx_z, qfx1_z = rho<w'theta'>, rho<w'qv'> (z-faces)
508  //
509  // Two different heights are wanted here, so the two are kept apart:
510  // - u_*, thetav_* and L are surface-layer parameters and have no
511  // cell-centered counterpart, so they come from the zlo face alone;
512  // - the production terms belong at the cell center, so their flux
513  // factors average the two z-faces bounding this cell.
514  // Only the profile shape, phi(zeta_1)/(kappa z_1), ties the two together.
515 
516  const amrex::Real theta = cell_data(i,j,k,RhoTheta_comp) / rho;
517  const amrex::Real qv = (moisture_indices.qv >= 0) ?
518  cell_data(i,j,k,moisture_indices.qv) / rho : zero;
519 
520  // ---- Surface-layer parameters, on the zlo face ----
521 
522  // |tau|/rho = u_*^2
523  const amrex::Real t13_s = myhalf * ( tau13(i,j,k) + tau13(i+1,j,k) );
524  const amrex::Real t23_s = myhalf * ( tau23(i,j,k) + tau23(i,j+1,k) );
525  const amrex::Real tau_s = std::sqrt(t13_s*t13_s + t23_s*t23_s);
526  const amrex::Real ustar2 = tau_s / rho;
527  const amrex::Real ustar = std::sqrt(ustar2);
528 
529  // Surface virtual heat flux rho<w'theta_v'> (Stull Eqn. 4.4.5d)
530  amrex::Real hfxv_s = hfx_z(i,j,k) * (one + epsv*qv);
531  if (qfx1_z) { hfxv_s += epsv * theta * qfx1_z(i,j,k); }
532 
533  // Height of this cell center above the surface. dz_inv.dirichlet_lo is by
534  // construction the inverse of the boundary-to-first-center distance, so it
535  // carries the right height for uniform, stretched and terrain-fitted grids.
536  const amrex::Real z1 = one / dz_inv.dirichlet_lo;
537 
538  // Obukhov length, from the surface fluxes. A vanishing stress (a building
539  // face, say) leaves the shear production zero whatever zeta is, so the
540  // neutral value is the safe one to carry into phi_m there.
541  const amrex::Real thetav = theta * (one + epsv*qv);
542  const amrex::Real eps = std::numeric_limits<amrex::Real>::epsilon();
543  amrex::Real zeta = zero;
544  if (std::abs(hfxv_s) > eps && ustar > eps) {
545  const amrex::Real l_obukhov = -( thetav * rho * ustar2 * ustar )
546  / ( KAPPA * CONST_GRAV * hfxv_s );
547  zeta = z1 / l_obukhov;
548  }
549 
550  // phi_m, the companion of the psi_m that SurfaceLayer uses
551  const similarity_funs sfuns{};
552  const amrex::Real phi_m = sfuns.calc_phi_m(zeta);
553 
554  // ---- Production terms, at the cell center ----
555 
556  // Cell-centered stress: average the two z-faces bounding this cell
557  const amrex::Real t13_c = fourth * ( tau13(i,j,k ) + tau13(i+1,j,k )
558  + tau13(i,j,k+1) + tau13(i+1,j,k+1) );
559  const amrex::Real t23_c = fourth * ( tau23(i,j ,k) + tau23(i,j ,k+1)
560  + tau23(i,j+1,k) + tau23(i,j+1,k+1) );
561 
562  // Cell-centered virtual heat flux. Both faces are from this stage: the
563  // QKE sources are added after RhoQ1 is diffused (see AddTurbKESources).
564  amrex::Real hfxv_c = myhalf * ( hfx_z(i,j,k) + hfx_z(i,j,k+1) ) * (one + epsv*qv);
565  if (qfx1_z) {
566  hfxv_c += epsv * theta * myhalf * ( qfx1_z(i,j,k) + qfx1_z(i,j,k+1) );
567  }
568 
569  // Shear production, -<u'w'> dU/dz - <v'w'> dV/dz. MOST sets the direction
570  // of the mean shear anti-parallel to the surface stress and its magnitude
571  // to u_* phi_m/(kappa z_1), so the dot product is the cell-centered stress
572  // projected onto the surface stress. This reduces to |tau| u_* phi_m /
573  // (kappa z_1) when the stress does not turn across the cell.
574  if (tau_s > eps) {
575  const amrex::Real tau_proj = (t13_c*t13_s + t23_c*t23_s) / tau_s;
576  source_term += std::max(tau_proj,amrex::Real(0.)) * ustar * phi_m / (KAPPA * z1);
577  }
578 
579  // Buoyancy production, which needs no gradient at all
580  source_term += (CONST_GRAV/theta_mean) * hfxv_c;
581  }
582  else
583  {
584  // Compute some relevant derivatives
585  amrex::Real dthetadz, dudz, dvdz;
586  ComputeVerticalDerivativesPBL(i, j, k,
587  uvel, vvel, cell_data, izmin, izmax, dz_inv,
588  c_ext_dir_on_zlo, c_ext_dir_on_zhi,
589  u_ext_dir_on_zlo, u_ext_dir_on_zhi,
590  v_ext_dir_on_zlo, v_ext_dir_on_zhi,
591  dthetadz, dudz, dvdz,
592  moisture_indices);
593 
594  // Second-order turbulent fluxes, e.g.:
595  // -<uw> = L q SM dU/dz (NN09, Eqn. 18)
596  // = Kmv/rho dU/dz
597 
598  // Shear Production
599  source_term += K_turb(i,j,k,EddyDiff::Mom_v) * (dudz*dudz + dvdz*dvdz);
600 
601  // Buoyancy Production
602  source_term -= (CONST_GRAV/theta_mean)*K_turb(i,j,k,EddyDiff::Theta_v)*dthetadz;
603  }
604 
605  // Dissipation (NN09, Eqn. 12)
606  amrex::Real qke = two * cell_prim(i,j,k,PrimKE_comp);
607  if (std::abs(qke) > zero) {
608  source_term -= rho * std::pow(qke,amrex::Real(1.5)) /
609  (pbl_mynn_B1_l * K_turb(i,j,k,EddyDiff::Turb_lengthscale));
610  }
611 
612  return source_term;
613 }

682 /**
683  * @brief Apply spatial smoothing to PBLH field using 5-point stencil.
684  *
685  * Applies a simple 5-point spatial averaging to reduce grid-to-grid noise in
686  * the diagnosed PBLH field. This is standard post-processing practice for
687  * discrete Rib-crossing detection as described in Seibert et al. (2000):
688  * "Review and intercomparison of operational methods for the determination of
689  * the mixing height", Atmospheric Environment, 34, 1001-1027.
690  *
691  * Stencil: PBLH_smooth(i,j) = w * PBLH(i,j) + (1-w)/4 * [PBLH(i+-1,j) + PBLH(i,j+-1)]
692  *
693  * The stencil reads one column outside the region it writes, so the result on a
694  * cell depends on columns that a single tile -- or a single box -- does not own.
695  * The caller therefore names the region it wants smoothed (xybx_valid, normally
696  * the planar tile box) inside a work array carrying `passes` columns of halo
697  * around it, filled from the same inputs. Pass p writes the region grown by
698  * passes-1-p and reads the one grown by passes-p, so each pass consumes one
699  * column of halo and the last lands exactly on xybx_valid. Every read stays
700  * inside the array, and the result does not depend on how the domain is divided
701  * into boxes or tiles.
702  *
703  * At a non-periodic domain edge there is no column to read, so the stencil reuses
704  * the cell's own value (reflective/Neumann). Across a periodic edge the halo holds
705  * the wrapped columns and is read directly, which is why the periodicity has to be
706  * passed in: clamping there would fold an edge into a domain that has none.
707  *
708  * @param[in,out] pblh_fab FArrayBox holding the 2D PBLH field (component 0),
709  * sized to contain xybx_valid grown by `passes` in x and y
710  * @param[in] xybx_valid Region to smooth (x,y indices only)
711  * @param[in] weight Center cell weight in stencil (must be in [0,1])
712  * @param[in] passes Number of smoothing iterations
713  * @param[in] domain Domain box, for the reflective edge condition
714  * @param[in] periodicity Domain periodicity, so periodic edges read the halo
715  */
716 inline void
717 ApplyPBLHSmoothing (amrex::FArrayBox& pblh_fab,
718  const amrex::Box& xybx_valid,
719  const amrex::Real weight,
720  const int passes,
721  const amrex::Box& domain,
722  const amrex::Periodicity& periodicity)
723 {
724  if (passes <= 0) { return; }
725 
726  // The caller must supply the halo the passes will consume; see the note above.
727  AMREX_ALWAYS_ASSERT_WITH_MESSAGE(
728  pblh_fab.box().contains(amrex::grow(xybx_valid, amrex::IntVect(passes,passes,0))),
729  "ApplyPBLHSmoothing: the PBLH work array needs `passes` columns of halo "
730  "around the region being smoothed");
731 
732  // Use a temporary to hold intermediate results
733  amrex::FArrayBox pblh_temp(pblh_fab.box(), 1, amrex::The_Async_Arena());
734  auto pblh = pblh_fab.array();
735  auto pblh_tmp = pblh_temp.array();
736 
737  const auto& dom_lo = amrex::lbound(domain);
738  const auto& dom_hi = amrex::ubound(domain);
739 
740  const bool per_x = periodicity.isPeriodic(0);
741  const bool per_y = periodicity.isPeriodic(1);
742 
743  const amrex::Real wt_side = (one - weight) / four; // Weight for each of 4 neighbors
744 
745  // Apply smoothing passes
746  for (int pass = 0; pass < passes; ++pass) {
747  const int halo = passes - 1 - pass;
748  const amrex::Box bx = amrex::grow(xybx_valid, amrex::IntVect(halo,halo,0));
749 
750  amrex::ParallelFor(bx, [=] AMREX_GPU_DEVICE(int i, int j, int) noexcept
751  {
752  // Reflective (Neumann) at a non-periodic domain edge: reuse the cell's
753  // own value. Everywhere else -- including across a periodic edge, where
754  // the halo holds the wrapped column -- read the neighbour.
755  const int i_xlo = (per_x || i > dom_lo.x) ? (i - 1) : i;
756  const int i_xhi = (per_x || i < dom_hi.x) ? (i + 1) : i;
757  const int j_ylo = (per_y || j > dom_lo.y) ? (j - 1) : j;
758  const int j_yhi = (per_y || j < dom_hi.y) ? (j + 1) : j;
759 
760  // 5-point stencil
761  pblh_tmp(i, j, 0) = weight * pblh(i, j, 0)
762  + wt_side * (pblh(i_xlo, j, 0) + pblh(i_xhi, j, 0) +
763  pblh(i, j_ylo, 0) + pblh(i, j_yhi, 0));
764  });
765 
766  // Copy back only what this pass computed: outside bx the array still holds
767  // the previous pass, which no later pass reads.
768  pblh_fab.copy<amrex::RunOn::Device>(pblh_temp, bx, 0, bx, 0, 1);
769  }
770 }
