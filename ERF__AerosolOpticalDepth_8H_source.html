56 /**
57  * @brief Diagnose aerosol optical depth for Exponential profile.
58  *
59  * Returns the optical depth of one layer of an exponentially decaying aerosol
60  * profile, i.e. the profile integrated across that layer,
61  * tau_aerosol(k) = (tau_surface / scale_height) * dz * exp(-z(k) / scale_height)
62  * so the column sum approaches tau_surface once the domain is much deeper
63  * than the scale height. This is a per-layer contribution, not the column
64  * value above z(k).
65  *
66  * Clamped to [0, 100] for physical reasonableness.
67  * Handles invalid parameters safely with fallback to 0.
68  *
69  * @param[in] z_level Height at level k [m]
70  * @param[in] dz Thickness of layer k [m]
71  * @param[in] tau_surface Total-column aerosol optical depth at surface [dimensionless]
72  * @param[in] scale_height_m Decay scale height [m]
73  * @return Clamped aerosol optical depth [0, 100]
74  */
75 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
76 amrex::Real diagnose_tau_aerosol_exponential(amrex::Real z_level,
77  amrex::Real dz,
78  amrex::Real tau_surface,
79  amrex::Real scale_height_m)
80 {
81  if (!std::isfinite(z_level) || !std::isfinite(dz) ||
82  !std::isfinite(tau_surface) || !std::isfinite(scale_height_m)) {
83  return 0.0;
84  }
85  if (scale_height_m <= 0.0 || dz <= 0.0) {
86  return 0.0;
87  }
88  if (tau_surface < 0.0) {
89  return 0.0;
90  }
91  if (z_level < 0.0) {
92  z_level = 0.0;
93  }
94 
95  amrex::Real arg = -z_level / scale_height_m;
96  if (arg < -100.0) {
97  return 0.0;
98  }
99 
100  // Per-layer contribution = (tau_surface / scale_height) * dz * exp(-z/H)
101  // so the vertical integral over all layers approaches tau_surface as domain height >> H
102  amrex::Real tau_aerosol = (tau_surface / scale_height_m) * dz * std::exp(arg);
103 
104  if (tau_aerosol < 0.0) tau_aerosol = 0.0;
105  if (tau_aerosol > 100.0) tau_aerosol = 100.0;
106 
107  return tau_aerosol;
108 }
