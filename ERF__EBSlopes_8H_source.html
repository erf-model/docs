14  *
15  * @param[in] dx Mesh spacing in x.
16  * @param[in] dy Mesh spacing in y.
17  * @param[in] dz Mesh spacing in z.
18  * @param[in] i Cell index in x.
19  * @param[in] j Cell index in y.
20  * @param[in] k Cell index in z.
21  * @param[in] bcent_eb EB boundary centroid.
22  * @param[in] state_eb Dirichlet state at the EB boundary.
23  * @param[in] state Cell-centered state array.
24  * @param[in] ccent Cell centroid coordinates.
25  * @param[in] flag Cell-centered EB flags and connectivity.
26  */
27 AMREX_GPU_DEVICE AMREX_FORCE_INLINE
28 amrex::GpuArray<amrex::Real,AMREX_SPACEDIM>
29 erf_calc_slopes_eb_Dirichlet (
30  amrex::Real dx, amrex::Real dy, amrex::Real dz,
31  int i, int j, int k,
32  amrex::RealVect const& bcent_eb,
33  amrex::Real const state_eb,
34  amrex::Array4<amrex::Real const> const& state,
35  amrex::Array4<amrex::Real const> const& ccent,
36  amrex::Array4<amrex::EBCellFlag const> const& flag)
37 {
38  amrex::EBSlopeNormalEquation normal_eq;
39 
40  // Prevent loop unrolling to reduce register pressure.
41 #ifdef AMREX_USE_GPU
42 #pragma unroll 1
43 #endif
44  for (int kk = -1; kk <= 1; ++kk) {
45 #ifdef AMREX_USE_GPU
46 #pragma unroll 1
47 #endif
48  for (int jj = -1; jj <= 1; ++jj) {
49 #ifdef AMREX_USE_GPU
50 #pragma unroll 1
51 #endif
52  for (int ii = -1; ii <= 1; ++ii) {
53  if (flag(i,j,k).isConnected(ii,jj,kk) && !(ii == 0 && jj == 0 && kk == 0)) {
54  amrex::Real offset_x = (amrex::Real(ii) + ccent(i+ii,j+jj,k+kk,0) - bcent_eb[0]) * dx;
55  amrex::Real offset_y = (amrex::Real(jj) + ccent(i+ii,j+jj,k+kk,1) - bcent_eb[1]) * dy;
56  amrex::Real offset_z = (amrex::Real(kk) + ccent(i+ii,j+jj,k+kk,2) - bcent_eb[2]) * dz;
57  amrex::Real du = state(i+ii,j+jj,k+kk) - state_eb;
58  normal_eq.add(offset_x, offset_y, offset_z, du);
59  }
60  }}}
61 
62  return normal_eq.solve();
63 }

107  // Set bias in the index space between the two staggered grids

109  bias[GridQuery-1] = myhalf;

111

113 
114  // Prevent loop unrolling to reduce register pressure.
115 #ifdef AMREX_USE_GPU
116 #pragma unroll 1

118  for (int kk = kk_lo; kk <= kk_hi; ++kk) {
119 #ifdef AMREX_USE_GPU
120 #pragma unroll 1
121 #endif
122  for (int jj = jj_lo; jj <= jj_hi; ++jj) {
123 #ifdef AMREX_USE_GPU

133  }
