11 /**
12  * Conservative linear interpolation of a land-surface field that does not mix land and water
13  * (erf-model/ERF#4390), in the spirit of WRF's land/water-masked nest interpolation.
14  *
15  * The coarse data carries two components: the field, and a land/water indicator (one value for
16  * water, another for land; NaN outside the domain). In each direction the slope of a coarse cell
17  * is taken only if both of its neighbours are of the same type as the cell; otherwise it is zero,
18  * as it is at a domain boundary. The min/max limiter likewise looks only at neighbours of the
19  * same type. Where every neighbour is of the same type this is exactly AMReX's cell_cons_interp
20  * (CellConservativeLinear without linear limiting), so the result changes only next to a
21  * coastline. Each coarse cell's mean is kept, and the indicator is copied (piecewise constant).
22  */

49 /**
50  * Slopes of the field (component nu) of coarse cell (i,j,k), from neighbours of its own type only
51  * (indicator in component nm). Follows mf_cell_cons_lin_interp_mcslope term for term.
52  */
53 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
54 void land_water_mcslope (int i, int j, int k, amrex::Array4<amrex::Real> const& slope,
55  amrex::Array4<amrex::Real const> const& u, int nu, int nm,
56  amrex::Box const& domain, amrex::IntVect const& ratio,
57  amrex::BCRec const& bc) noexcept
58 {
59  using amrex::Real;
60  // NaN (outside the domain) is never of the same type
61  const Real m0 = u(i,j,k,nm);
62  auto same = [&] (int ii, int jj, int kk) { return u(ii,jj,kk,nm) == m0; };
63 
64  Real sx = Real(0.);
65  Real sy = Real(0.);
66  Real sz = Real(0.);
67 
68  if (ratio[0] > 1 && same(i-1,j,k) && same(i+1,j,k)) {
69  Real dc = amrex::mf_compute_slopes_x(i, j, k, u, nu, domain, bc);
70  Real df = Real(2.0) * (u(i+1,j,k,nu) - u(i ,j,k,nu));
71  Real db = Real(2.0) * (u(i ,j,k,nu) - u(i-1,j,k,nu));
72  sx = (df*db >= Real(0.0)) ? amrex::min(std::abs(df),std::abs(db)) : Real(0.);
73  sx = std::copysign(Real(1.),dc)*amrex::min(sx,std::abs(dc));
74  }
75  if (ratio[1] > 1 && same(i,j-1,k) && same(i,j+1,k)) {
76  Real dc = amrex::mf_compute_slopes_y(i, j, k, u, nu, domain, bc);
77  Real df = Real(2.0) * (u(i,j+1,k,nu) - u(i,j ,k,nu));
78  Real db = Real(2.0) * (u(i,j ,k,nu) - u(i,j-1,k,nu));
79  sy = (df*db >= Real(0.0)) ? amrex::min(std::abs(df),std::abs(db)) : Real(0.);
80  sy = std::copysign(Real(1.),dc)*amrex::min(sy,std::abs(dc));
81  }
82  if (ratio[2] > 1 && same(i,j,k-1) && same(i,j,k+1)) {
83  Real dc = amrex::mf_compute_slopes_z(i, j, k, u, nu, domain, bc);
84  Real df = Real(2.0) * (u(i,j,k+1,nu) - u(i,j,k ,nu));
85  Real db = Real(2.0) * (u(i,j,k ,nu) - u(i,j,k-1,nu));
86  sz = (df*db >= Real(0.0)) ? amrex::min(std::abs(df),std::abs(db)) : Real(0.);
87  sz = std::copysign(Real(1.),dc)*amrex::min(sz,std::abs(dc));
88  }
89 
90  Real alpha = Real(1.0);
91  if (sx != Real(0.0) || sy != Real(0.0) || sz != Real(0.0)) {
92  Real dumax = std::abs(sx) * Real(ratio[0]-1)/Real(2*ratio[0])
93  + std::abs(sy) * Real(ratio[1]-1)/Real(2*ratio[1])
94  + std::abs(sz) * Real(ratio[2]-1)/Real(2*ratio[2]);
95  Real umax = u(i,j,k,nu);
96  Real umin = u(i,j,k,nu);
97  const int ilim = ratio[0] > 1 ? 1 : 0;
98  const int jlim = ratio[1] > 1 ? 1 : 0;
99  const int klim = ratio[2] > 1 ? 1 : 0;
100  for (int koff = -klim; koff <= klim; ++koff) {
101  for (int joff = -jlim; joff <= jlim; ++joff) {
102  for (int ioff = -ilim; ioff <= ilim; ++ioff) {
103  if (same(i+ioff,j+joff,k+koff)) {
104  umin = amrex::min(umin, u(i+ioff,j+joff,k+koff,nu));
105  umax = amrex::max(umax, u(i+ioff,j+joff,k+koff,nu));
106  }
107  }}}
108  // dumax > 0 here; this only keeps an if-converted divide below from dividing by zero
109  dumax = dumax == Real(0.) ? Real(1.) : dumax;
110  if (dumax * alpha > (umax - u(i,j,k,nu))) {
111  alpha = (umax - u(i,j,k,nu)) / dumax;
112  }
113  if (dumax * alpha > (u(i,j,k,nu) - umin)) {
114  alpha = (u(i,j,k,nu) - umin) / dumax;
115  }
116  }
117 
118  slope(i,j,k,0) = sx * alpha;
119  slope(i,j,k,1) = sy * alpha;
120  slope(i,j,k,2) = sz * alpha;
121 }
