858  void init_params (bool two_stream_selected, std::string pp_prefix)
859  {
860  amrex::ParmParse pp(pp_prefix);
861  enabled = two_stream_selected;
862 
863  // A refined hierarchy is supported: every level runs its own column sweep
864  // and writes its own heating rates. The one requirement is that each
865  // level's grids span that level's domain in z, because the sweep imposes
866  // top-of-atmosphere and surface boundary conditions at the ends of a box.
867  // That is checked where the grids exist, not here: ERF_RefineBox.cpp for
868  // an explicit erf.boxN refinement box, and TwoStreamRadiation::define_level
869  // for every path including tagged refinement and regrid.
870 
871  // Only query sub-options if the two-stream model is selected
872  if (enabled) {
873  pp.query("radiation.sw_enabled", sw_enabled);
874  pp.query("radiation.lw_enabled", lw_enabled);
875  pp.query("radiation.tau_per_layer", tau_per_layer);
876  pp.query("radiation.tau_lw_per_layer", tau_lw_per_layer);
877 
878  // The sun, the site and the surface temperature come from the keys the
879  // RRTMGP interface reads, so the two models share one set of inputs.
880  // ParmParse ignores a key nobody reads, so a deck still carrying one of
881  // the earlier two-stream-only keys would run with silently different
882  // settings: stop and name the replacement instead.
883  static const char* retired_keys[][2] = {
884  {"radiation.solar_zenith",
885  "erf.fixed_solar_zenith_angle, which is the cosine of the angle (60 degrees is 0.5)"},
886  {"radiation.solar_zenith_deg",
887  "erf.fixed_solar_zenith_angle, which is the cosine of the angle (60 degrees is 0.5)"},
888  {"radiation.S0", "erf.fixed_total_solar_irradiance"},
889  {"radiation.earth_sun_distance_enable",
890  "nothing: leave erf.fixed_total_solar_irradiance unset and the irradiance follows the date"},
891  {"radiation.surface_temp_k", "erf.rad_t_sfc"},
892  {"radiation.solar_geometry_dynamic_enable",
893  "nothing: leave erf.fixed_solar_zenith_angle unset and the sun follows start_datetime"},
894  {"radiation.latitude_deg", "erf.rad_cons_lat"},
895  {"radiation.longitude_deg", "erf.rad_cons_lon"},
896  {"radiation.day_of_year", "start_datetime"},
897  {"radiation.time_zone_offset_hours",
898  "nothing: start_datetime is UTC and the longitude sets the local solar time"},
899  };
900  for (const auto& entry : retired_keys) {
901  if (pp.contains(entry[0])) {
902  amrex::Abort(std::string("erf.") + entry[0] + " is no longer read by the "
903  "two-stream model; use " + entry[1] + ".");
904  }
905  }
906  pp.query("fixed_solar_zenith_angle", fixed_solar_zenith_angle);
907  if (fixed_solar_zenith_angle > 1.0) {
908  amrex::Abort("erf.fixed_solar_zenith_angle = " + std::to_string(fixed_solar_zenith_angle) +
909  " is the cosine of the solar zenith angle (as for RRTMGP) and cannot "
910  "exceed 1; 60 degrees is 0.5.");
911  }
912  pp.query("fixed_total_solar_irradiance", fixed_total_solar_irradiance);
913  if (!pp.query("rad_t_sfc", rad_t_sfc)) {
914  amrex::Abort("erf.radiation_model = TwoStream needs erf.rad_t_sfc, the surface "
915  "temperature [K] used where no land-surface model or surface layer "
916  "supplies one (the same input RRTMGP requires).");
917  }
918  if (!std::isfinite(rad_t_sfc) || rad_t_sfc <= 0.0) {
919  amrex::Abort("erf.rad_t_sfc = " + std::to_string(rad_t_sfc) + " must be a positive temperature [K].");
920  }
921  pp.query("rad_cons_lat", rad_cons_lat);
922  pp.query("rad_cons_lon", rad_cons_lon);
923  if (!std::isfinite(rad_cons_lat) || rad_cons_lat < -90.0 || rad_cons_lat > 90.0) {
924  amrex::Abort("erf.rad_cons_lat = " + std::to_string(rad_cons_lat) + " must lie in [-90, 90] degrees.");
925  }
926  if (!std::isfinite(rad_cons_lon) || rad_cons_lon < -180.0 || rad_cons_lon > 180.0) {
927  amrex::Abort("erf.rad_cons_lon = " + std::to_string(rad_cons_lon) + " must lie in [-180, 180] degrees.");
928  }
929  pp.query("rad_orbital_year", rad_orbital_year);
930  pp.query("rad_orbital_eccentricity", rad_orbital_eccentricity);
931  pp.query("rad_orbital_obliquity", rad_orbital_obliquity);
932  pp.query("rad_orbital_mvelp", rad_orbital_mvelp);
933  if (pp.contains("rad_freq_in_steps") && amrex::ParallelDescriptor::IOProcessor()) {
934  amrex::Print() << "WARNING: erf.rad_freq_in_steps is not read by the two-stream model, "
935  "which runs every step.\n";
936  }
937  std::string tau_model_str = "per_layer";
938  pp.query("radiation.tau_model", tau_model_str);
939  if (tau_model_str == "per_layer") {
940  tau_model = TauModel::PerLayer;
941  } else if (tau_model_str == "mass") {
942  tau_model = TauModel::Mass;
943  } else {
944  // The two models give very different column optical depths, so say so
945  // rather than let a misspelling quietly pick the default.
946  tau_model = TauModel::PerLayer;
947  if (amrex::ParallelDescriptor::IOProcessor()) {
948  amrex::Print() << "WARNING: unknown erf.radiation.tau_model '"
949  << tau_model_str << "'; valid values are per_layer "
950  "and mass. Using per_layer.\n";
951  }
952  }
953  pp.query("radiation.sw_kabs_dry", sw_kabs_dry);
954  pp.query("radiation.sw_kscat_dry", sw_kscat_dry);
