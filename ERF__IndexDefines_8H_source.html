145 /**
146  * The boundary-condition slot (BCVars) of conserved component n: every passive
147  * scalar shares BCVars::RhoScalar_bc_comp, and each component above the scalars
148  * has its own slot, NSCALARS - 1 below its component index. Boundary planes are
149  * stored by these slots too (ReadBndryPlanes::plane_slot).
150  */
151 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
152 int cons_bc_comp (int n) noexcept
153 {
154  if (n >= RhoScalar_comp && n < RhoScalar_comp + NSCALARS) { return BCVars::RhoScalar_bc_comp; }
155  return (n >= RhoScalar_comp + NSCALARS) ? n - (NSCALARS - 1) : n;
156 }
