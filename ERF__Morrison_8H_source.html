60 void morrison_copy_state_to_micro_cell (
61  const amrex::Array4<const amrex::Real>& states,
62  const amrex::Array4<const amrex::Real>& base,
63  const amrex::Array4<amrex::Real>& rho,
64  const amrex::Array4<amrex::Real>& theta,
65  const amrex::Array4<amrex::Real>& qv,
66  const amrex::Array4<amrex::Real>& qc,
67  const amrex::Array4<amrex::Real>& qi,
68  const amrex::Array4<amrex::Real>& qn,
69  const amrex::Array4<amrex::Real>& qt,
70  const amrex::Array4<amrex::Real>& qpr,
71  const amrex::Array4<amrex::Real>& qps,
72  const amrex::Array4<amrex::Real>& qpg,
73  const amrex::Array4<amrex::Real>& qp,
74  const amrex::Array4<amrex::Real>& tabs,
75  const amrex::Array4<amrex::Real>& pres,
76  const amrex::Real rdOcp,
77  const bool use_anelastic_reference_pressure,
78  const int i, const int j, const int k) noexcept
79 {
80  const amrex::Real rho_value = states(i,j,k,Rho_comp);
81  const amrex::Real rho_theta = states(i,j,k,RhoTheta_comp);
82  rho(i,j,k) = rho_value;
83  theta(i,j,k) = rho_theta / rho_value;
84  qv(i,j,k) = amrex::max(amrex::Real(0.0), states(i,j,k,RhoQ1_comp) / rho_value);
85  qc(i,j,k) = amrex::max(amrex::Real(0.0), states(i,j,k,RhoQ2_comp) / rho_value);
86  qi(i,j,k) = amrex::max(amrex::Real(0.0), states(i,j,k,RhoQ3_comp) / rho_value);
87  qn(i,j,k) = qc(i,j,k) + qi(i,j,k);
88  qt(i,j,k) = qv(i,j,k) + qn(i,j,k);
89  qpr(i,j,k) = amrex::max(amrex::Real(0.0), states(i,j,k,RhoQ4_comp) / rho_value);
90  qps(i,j,k) = amrex::max(amrex::Real(0.0), states(i,j,k,RhoQ5_comp) / rho_value);
91  qpg(i,j,k) = amrex::max(amrex::Real(0.0), states(i,j,k,RhoQ6_comp) / rho_value);
92  qp(i,j,k) = qpr(i,j,k) + qps(i,j,k) + qpg(i,j,k);
93 
94  const amrex::Real p0 = use_anelastic_reference_pressure
95  ? base(i,j,k,BaseState::p0_comp) : amrex::Real(0.0);
96  const MicrophysicsThermoState thermo = diagnose_microphysics_thermo_state(
97  rho_value, rho_theta, qv(i,j,k), rdOcp,
98  use_anelastic_reference_pressure, p0);
99  tabs(i,j,k) = thermo.temperature;
100  // Morrison's native pressure unit is Pa.
101  pres(i,j,k) = thermo.pressure_pa;
102 }
