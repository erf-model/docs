40 // Reference-illumination hydrometeor-only proxy, not a radiative-transfer solution.
41 // The two branches avoid overflow in tau + scale for large finite values.
42 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE amrex::Real
43 cloud_reflectance (amrex::Real tau, amrex::Real scale) noexcept
44 {
45  if (!amrex::Math::isfinite(tau) || tau < zero ||
46  !amrex::Math::isfinite(scale) || scale <= zero) {
47  return cloud_optics_missing;
48  }
49  if (tau == zero) {
50  return zero;
51  }
52  const amrex::Real value = tau > scale ? one / (one + scale / tau)
53  : (tau / scale) / (one + tau / scale);
54  return amrex::min(amrex::Real(one), amrex::max(amrex::Real(zero), value));
55 }
