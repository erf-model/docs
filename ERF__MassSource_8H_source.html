21 /**
22  * A localized source of gas inside the domain (erf.mass_source.*), e.g. the release
23  * from a ruptured pipeline.
24  *
25  * The gas is injected at a total rate mdot(t) [kg/s], spread uniformly over the cells
26  * whose centers lie within erf.mass_source.radius of (center_x, center_y) horizontally
27  * and less than erf.mass_source.height above the local ground. The same source is
28  * added to
29  * - rho, so the mass enters the continuity equation (FullyCompressible only: the
30  * anelastic projection would remove it),
31  * - rho*theta, at erf.mass_source.theta if given and otherwise at the cell's own theta,
32  * so the injection by itself does not change the potential temperature, and
33  * - the first passive scalar (rho*s, plotted as rhoadv_0) when erf.mass_source.tracer
34  * is true, so that s is the mass fraction of the released gas.
35  * No momentum is added: the gas enters at rest. The gas has the properties of dry air
36  * (in a moist run it carries no water, so it dilutes the mixing ratios of the cells it
37  * enters). The cell volumes include the terrain metric and the map factors. In a
38  * periodic direction the region wraps around the domain. The rate table is in the time
39  * since the start of the run. Unknown erf.mass_source keys stop the run.
40  *
41  * With mesh refinement every level adds the same source density S = mdot / V to its own
42  * cells in the region, where V is the volume of the region on the composite grid: the
43  * cells of each level that the next finer level does not cover. The composite rate is
44  * then mdot however the levels cut the region, and a level without source cells adds
45  * nothing. V, and the height of the ground under each column, are recomputed only when
46  * the grids change.
47  *
48  * Inputs:
49  * erf.mass_source.center = x y [m]
50  * erf.mass_source.radius = r [m]
51  * erf.mass_source.height = h [m above the local ground]
52  * erf.mass_source.rate = mdot [kg/s >= 0] constant, or
53  * erf.mass_source.rate_file = file two columns "t mdot" [s, kg/s >= 0], '#' comments;
54  * linear in between, zero outside the table
55  * erf.mass_source.theta = theta [K > 0] optional
56  * erf.mass_source.tracer = true optional
57  */

60 /// Is the center of cell (i,j,k) inside the source region? z is the nodal height z_phys_nd
61 /// and zg the nodal height of the ground under the column (2D, at k = kg). period holds the
62 /// domain length in x and y where the domain is periodic and 0 where it is not: there the
63 /// distance is to the nearest periodic image of the center, so the region wraps around
64 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE bool
65 in_source (int i, int j, int k,
66  amrex::Array4<amrex::Real const> const& z,
67  amrex::Array4<amrex::Real const> const& zg, int kg,
68  amrex::GpuArray<amrex::Real,AMREX_SPACEDIM> const& problo,
69  amrex::GpuArray<amrex::Real,AMREX_SPACEDIM> const& dx,
70  amrex::GpuArray<amrex::Real,2> const& period,
71  amrex::Real x0, amrex::Real y0, amrex::Real r2, amrex::Real hmax) noexcept
72 {
73  amrex::Real ddx = problo[0] + (i + amrex::Real(0.5)) * dx[0] - x0;
74  amrex::Real ddy = problo[1] + (j + amrex::Real(0.5)) * dx[1] - y0;
75  if (period[0] > 0) { ddx -= period[0] * std::floor(ddx / period[0] + amrex::Real(0.5)); }
76  if (period[1] > 0) { ddy -= period[1] * std::floor(ddy / period[1] + amrex::Real(0.5)); }
77  if (ddx * ddx + ddy * ddy > r2) { return false; }
78  const amrex::Real g = amrex::Real(0.25) * (zg(i,j,kg) + zg(i+1,j,kg) + zg(i,j+1,kg) + zg(i+1,j+1,kg));
79  const amrex::Real zc = amrex::Real(0.125) * (z(i,j,k ) + z(i+1,j,k ) + z(i,j+1,k ) + z(i+1,j+1,k ) +
80  z(i,j,k+1) + z(i+1,j,k+1) + z(i,j+1,k+1) + z(i+1,j+1,k+1));
81  return zc - g < hmax;
82 }

177  /// Mass flow rate [kg/s] at time t (since the start of the run)
178  [[nodiscard]] amrex::Real rate (amrex::Real t) const
179  {
180  if (m_constant) { return m_mdot[0]; }
181  if (t < m_t.front() || t > m_t.back()) { return amrex::Real(0); }
182  amrex::Long n = 1;
183  while (n < m_t.size() - 1 && m_t[n] < t) { ++n; }
184  const amrex::Real w = (t - m_t[n-1]) / (m_t[n] - m_t[n-1]);
185  return (amrex::Real(1) - w) * m_mdot[n-1] + w * m_mdot[n];
186  }

188  /// Total mass in the rate table [kg] (trapezoidal)
189  [[nodiscard]] amrex::Real total_mass () const
190  {
191  amrex::Real m = 0;
192  for (amrex::Long n = 1; n < m_t.size(); ++n) {
193  m += amrex::Real(0.5) * (m_mdot[n] + m_mdot[n-1]) * (m_t[n] - m_t[n-1]);
194  }
195  return m;

214  const amrex::Vector<const amrex::MultiFab*>& mf_mx,
215  const amrex::Vector<const amrex::MultiFab*>& mf_my,
216  const std::function<void(int, amrex::MultiFab&)>& fill_ground)
217  {
218  bool same = (finest_level + 1 == static_cast<int>(m_lev.size()));
219  for (int lev = 0; same && lev <= finest_level; ++lev) {
220  same = (m_lev[lev].ba == grids[lev]) && (m_lev[lev].dm == dmap[lev]);
221  }
222  if (same) { return; }
223 
224  m_lev.clear();
225  m_lev.resize(finest_level + 1);
226  m_volume = 0;
227  for (int lev = 0; lev <= finest_level; ++lev) {

229  lg.ba = grids[lev];
230  lg.dm = dmap[lev];
231 
232  // The nodal height of the ground under every column of every box, on the
233  // box's own (i,j) range, so that a box that does not reach the bottom (a patch
234  // aloft, or a grid split in z) can tell how high its cells are
235  const int kg = geom[lev].Domain().smallEnd(2);
236  AMREX_ALWAYS_ASSERT(z_phys_nd[lev] && detJ[lev]);
237  amrex::BoxList bl = z_phys_nd[lev]->boxArray().boxList();
238  for (auto& b : bl) { b.setRange(2, kg); }
239  lg.ground = std::make_unique<amrex::MultiFab>(amrex::BoxArray(std::move(bl)),
240  z_phys_nd[lev]->DistributionMap(), 1, 0);
241  fill_ground(lev, *lg.ground);
242 
243  // Cells not covered by the next finer level count towards the composite volume
244  std::unique_ptr<amrex::iMultiFab> uncovered;
245  if (lev < finest_level) {
246  uncovered = std::make_unique<amrex::iMultiFab>(
247  amrex::makeFineMask(grids[lev], dmap[lev], grids[lev+1], ref_ratio[lev], 1, 0));
248  }
249

251  const auto problo = geom[lev].ProbLoArray();
252  const auto period = erf_mass_source::periodic_lengths(geom[lev]);
253  const amrex::Real x0 = m_x0, y0 = m_y0, r2 = m_radius * m_radius, hmax = m_height;
254  // dx in place of a dy over one cell, and the reverse: on an x-z slice mdot is the
255  // rate per dx of span

257  const amrex::Real dV0 = len[0] * len[1] * len[2];
258 
259  amrex::ReduceOps<amrex::ReduceOpSum, amrex::ReduceOpSum> reduce_op;
260  amrex::ReduceData<amrex::Real, amrex::Real> reduce_data(reduce_op);
261  for (amrex::MFIter mfi(grids[lev], dmap[lev]); mfi.isValid(); ++mfi) {
262  const amrex::Box& bx = mfi.validbox();
263  auto z = z_phys_nd[lev]->const_array(mfi);
264  auto zg = lg.ground->const_array(mfi);
265  auto dj = detJ[lev]->const_array(mfi);
266  auto mx = mf_mx[lev]->const_array(mfi);
267  auto my = mf_my[lev]->const_array(mfi);
268  const bool has_mask = (uncovered != nullptr);
269  auto msk = has_mask ? uncovered->const_array(mfi) : amrex::Array4<int const>{};
270  reduce_op.eval(bx, reduce_data,
271  [=] AMREX_GPU_DEVICE (int i, int j, int k) -> amrex::GpuTuple<amrex::Real, amrex::Real>
272  {
273  if (!erf_mass_source::in_source(i,j,k,z,zg,kg,problo,dx,period,x0,y0,r2,hmax)) {
274  return {amrex::Real(0), amrex::Real(0)};
275  }
276  const amrex::Real dV = dV0 * dj(i,j,k) / (mx(i,j,0) * my(i,j,0));
277  const amrex::Real u = (!has_mask || msk(i,j,k) == 1) ? dV : amrex::Real(0);
278  return {dV, u};
279  });
280  }

283  amrex::ParallelDescriptor::ReduceRealSum(v, 2);
284  lg.volume = v[0];

286  }

290  "make radius or height larger, or check that center is inside the domain");

295  }

299  /**

338  src(i,j,k,Rho_comp) += S;
339  src(i,j,k,RhoTheta_comp) += S * th;

341  });
