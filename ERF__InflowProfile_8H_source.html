63 amrex::Real
64 inflow_profile_value (const amrex::Real* z, const amrex::Real* f, const int n, const amrex::Real zq) noexcept
65 {
66  if (zq <= z[0]) { return f[0]; }
67  if (zq >= z[n-1]) { return f[n-1]; }
68  int lo = 0;
69  int hi = n - 1;
70  while (hi - lo > 1) {
71  const int mid = (lo + hi) / 2;
72  if (z[mid] <= zq) { lo = mid; } else { hi = mid; }
73  }
74  const amrex::Real w = (zq - z[lo]) / (z[hi] - z[lo]);
75  return f[lo] + w * (f[hi] - f[lo]);
76 }

78 /**
79  * Mean height above the ground of the nodes [ilo,ihi] x [jlo,jhi] between
80  * levels k and k+1 of the terrain-fitted mesh: a cell centre (two nodes in each
81  * lateral direction), an x-face (ilo == ihi) or a y-face (jlo == jhi). The
82  * ground is the same nodes on level klo. Indices are clamped to the array.
83  */
84 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
85 amrex::Real
86 height_above_ground (const amrex::Array4<const amrex::Real>& z_nd,
87  int ilo, int ihi, int jlo, int jhi, const int k, const int klo) noexcept
88 {
89  const auto lo = amrex::lbound(z_nd);
90  const auto hi = amrex::ubound(z_nd);
91  ilo = amrex::max(lo.x, amrex::min(ilo, hi.x));
92  ihi = amrex::max(lo.x, amrex::min(ihi, hi.x));
93  jlo = amrex::max(lo.y, amrex::min(jlo, hi.y));
94  jhi = amrex::max(lo.y, amrex::min(jhi, hi.y));
95  const int kc = amrex::max(lo.z, amrex::min(k, hi.z - 1));
96  const int kg = amrex::max(lo.z, amrex::min(klo, hi.z));
97  amrex::Real sum = amrex::Real(0.0);
98  int count = 0;
99  for (int j = jlo; j <= jhi; ++j) {
100  for (int i = ilo; i <= ihi; ++i) {
101  sum += amrex::Real(0.5) * (z_nd(i,j,kc) + z_nd(i,j,kc+1)) - z_nd(i,j,kg);
102  ++count;
103  }
104  }
105  return sum / static_cast<amrex::Real>(count);
106 }
