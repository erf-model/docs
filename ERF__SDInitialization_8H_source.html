76 /*! \brief Inverse error function approximation for GPU
77  * \param[in] x Input value in (-1, 1)
78  * \return Approximate inverse error function value
79  */
80 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
81 amrex::Real SD_erfinv_gpu(const amrex::Real x) {
82  amrex::Real a = amrex::Real(0.147);
83  amrex::Real eps = std::numeric_limits<amrex::Real>::epsilon();
84  amrex::Real term = std::log(amrex::Real(1) - x * x + eps);
85  amrex::Real p1 = amrex::Real(2) / (PI * a) + term / amrex::Real(2);
86  amrex::Real p2 = term / a;
87  amrex::Real sign = (x >= 0) ? amrex::Real(1) : amrex::Real(-1);
88  return sign * std::sqrt(std::sqrt(p1 * p1 - p2) - p1);
89 }

91 /*! \brief Generate mass from distribution parameters on GPU
92  * \param[in] params Distribution parameters
93  * \param[in] engine Random engine
94  * \param[out] mult_contribution Contribution to multiplicity (added, not set)
95  * \return Generated mass value
96  */
97 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
98 amrex::Real SD_sample_mass_gpu(const SDDistributionParams& params,
99  const amrex::RandomEngine& engine,
100  amrex::Real& mult_contribution)
101 {
102  using amrex::Random;
103  amrex::Real mass = 0.0;
104  amrex::Real u = Random(engine);
105 
106  switch (params.dist_type) {
107  case SDDistributionType::mass_constant:
108  mass = params.mass_mean;
109  mult_contribution = u; // Random [0,1] for later rescaling
110  break;
111 
112  case SDDistributionType::mass_exponential: {
113  if (params.sampled_mult) {
114  // Sampled multiplicity: sample uniformly in log-space, weight by exponential PDF
115  amrex::Real lnval = params.lnmin + u * params.lnrng;
116  mass = std::exp(lnval);
117  mult_contribution = (params.numdens * params.cell_volume) * std::exp(-mass / params.delta);
118  } else {
119  // Constant multiplicity: true exponential distribution via inverse transform
120  // If U ~ Uniform(0,1), then -delta*ln(U) ~ Exponential(1/delta)
121  mass = -params.delta * std::log(u) + params.mass_min;
122  mult_contribution = 0.0; // Not used for constant multiplicity
123  }
124  break;
125  }
126 
127  case SDDistributionType::radius_log_normal:
128  case SDDistributionType::radius_lognormal_autorange: {
129  amrex::Real dry_r;
130  if (params.sampled_mult) {
131  // Sampled multiplicity: sample uniformly in log-space, weight by log-normal PDF
132  amrex::Real lnval = params.lnmin + u * params.lnrng;
133  dry_r = std::exp(lnval);
134  // Log-normal PDF contribution for multiplicity weighting
135  amrex::Real term = std::exp(-std::log(dry_r/params.radius_mean)*std::log(dry_r/params.radius_mean)
136  /(amrex::Real(2)*params.sigma*params.sigma));
137  mult_contribution = (params.numdens * params.cell_volume) / (params.sigma * std::sqrt(amrex::Real(2)*PI)) * term;
138  } else {
139  // Constant multiplicity: true log-normal via inverse CDF
140  // Map u from [0,1] to [cdf_min, cdf_max] for truncated distribution
141  amrex::Real u_trunc = params.cdf_min + u * (params.cdf_max - params.cdf_min);
142  // Inverse CDF: r = mu * exp(sigma * sqrt(2) * erfinv(2*u - 1))
143  amrex::Real z = SD_erfinv_gpu(amrex::Real(2) * u_trunc - amrex::Real(1)) * std::sqrt(amrex::Real(2));
144  dry_r = params.radius_mean * std::exp(params.sigma * z);
145  mult_contribution = 0.0; // Not used for constant multiplicity
146  }
147  mass = four_thirds_pi * dry_r * dry_r * dry_r * params.density;
148  break;
149  }
150  }
151  return mass;
152 }

582  it amounts to at least one super-droplet of unit multiplicity. */
583  amrex::Real m_numdens_accum = zero;
584  /*!< Injection-level cell volume [m^3], set from geometry at read time */
585  amrex::Real m_cell_volume = -1;
586  /*!< Number of super-droplets to inject this step (0 = none). Per cell in
587  legacy mode; per box in per-box high-multiplicity mode. */
588  int m_inject_num_sd = 0;
589  /*!< Integer multiplicity of each super-droplet injected this step */
590  amrex::Real m_inject_mult = zero;
591  /*!< Tolerance below 1 at which an accumulated injection fires */
592  amrex::Real m_frac_tol = amrex::Real(1.0e-3);
593

596  /*!< Per-box high-multiplicity injection mode (set when particles_per_cell
597  is not specified) */

599  /*!< True when both sd_rate and min_multiplicity are explicitly specified */

601  /*!< Whether each knob was specified in any readInputs call (the

603  bool m_ppc_specified = false;

607  amrex::Real m_eff_sd_rate = -1;
608  /*!< Injection-box volume [m^3], set from the box at read time */
609  amrex::Real m_box_volume = -1;

611  amrex::Real m_sd_accum = zero;

613  unsigned int m_inject_count = 0;
614  /*!< Reproducible seed for scattering this step's super-droplets */
615  unsigned int m_inject_seed = 0;

619  /*! \brief Update time-dependent quantities for particle injection
620  *
621  * This function updates:
622  * 1. The accumulated physical-particle count, and the integer number of
623  * super-droplets and their integer multiplicity to inject this step
624  * 2. Position of the injection domain based on domain velocity
625  *
626  * The number of super-droplets per cell is the specified count (m_ppc, or
627  * derived from the SD rate). The physical-particle count accumulates from
628  * the injection rate; a step injects only once each of those super-droplets
629  * would carry at least unit multiplicity, with integer multiplicity
630  * max(round(num_par/num_sd), 1). This lets sub-unity per-step injection
631  * rates be honored automatically, with no separate opt-in.
632  *
633  * \param[in] a_dt Timestep size in seconds
634  * \param[in] a_active True while the injection time window is open
635  */
636  inline void updateDt (const double a_dt, const bool a_active)
637  {
638  using amrex::Real;
639 
640  if (m_perbox) {
641  // Per-box high-multiplicity injection: accumulate the box-level
642  // super-droplet and physical-particle counts. Inject floor(SD
643  // count) super-droplets at the implied multiplicity (= rate /
644  // eff_sd_rate) once at least one super-droplet is owed. This
645  // decouples the super-droplet count from the grid and bounds the
646  // population at physical/multiplicity.
647  if (a_active) {
648  m_sd_accum += m_eff_sd_rate * a_dt * m_box_volume;
649  m_numdens_accum += m_inj_rate * a_dt * m_box_volume; // physical count in box
650  }
651  if (a_active && m_sd_accum >= one - m_frac_tol) {
652  const int N = std::max(static_cast<int>(std::floor(m_sd_accum + m_frac_tol)), 1);
653  m_inject_mult = std::max(std::round(m_numdens_accum / N), one);
654  m_inject_num_sd = N;
655  m_sd_accum -= N;
656  m_numdens_accum -= N * m_inject_mult;
657  m_inject_seed = m_inject_count++;
658  } else {
659  m_inject_num_sd = 0;
660  m_inject_mult = zero;
661  }
662  } else {
663 
664  // Specified number of super-droplets per cell (m_ppc) or from the SD rate.
665  Real num_sd;
666  if (m_sd_inj_rate > 0) {
667  num_sd = std::max(Real(std::round(m_sd_inj_rate * a_dt * m_cell_volume)), one);
668  } else {
669  num_sd = std::max(Real(m_ppc), one);
670  }
671 
672  // Physical particles this step (per cell) and the resulting multiplicity.
673  const Real par_step = m_inj_rate * a_dt * m_cell_volume;
674  const Real mult_step = par_step / num_sd;
675 
676  if (!a_active) {
677  m_inject_num_sd = 0;
678  m_inject_mult = zero;
679  } else if (mult_step >= one) {
680  // Each super-droplet already carries at least unit multiplicity:
681  // inject the specified count every step (no accumulation).
682  m_inject_num_sd = static_cast<int>(num_sd);
683  m_inject_mult = std::max(std::round(mult_step), one);
684  m_numdens_accum = zero;
685  } else {
686  // Sub-unity per-step multiplicity: accumulate the physical-particle
687  // count and inject only once it amounts to >= 1 multiplicity.
688  m_numdens_accum += m_inj_rate * a_dt;
689  const Real num_par = m_numdens_accum * m_cell_volume;
690  const Real mult = num_par / num_sd;
691  if (mult >= one - m_frac_tol) {
692  m_inject_mult = std::max(std::round(mult), one); // integer multiplicity
693  m_inject_num_sd = static_cast<int>(num_sd);
694  m_numdens_accum -= (num_sd * m_inject_mult) / m_cell_volume; // carry remainder
695  } else {
696  m_inject_num_sd = 0;
697  m_inject_mult = zero;
698  }
699  }
700 
701  }
702 
703  if (this->m_type == SDInitShape::uniform) {
704  amrex::Vector<amrex::Real> lo = {zero, zero, zero};
705  amrex::Vector<amrex::Real> hi = {zero, zero, zero};
706  for (int dir = 0; dir < AMREX_SPACEDIM; dir++) {
707  lo[dir] = this->m_particle_domain.lo(dir) + m_domain_vel[dir] * a_dt;
708  hi[dir] = this->m_particle_domain.hi(dir) + m_domain_vel[dir] * a_dt;
709  }
710  this->m_particle_domain.setLo(lo);
711  this->m_particle_domain.setHi(hi);
712  } else if (m_type == SDInitShape::bubble) {
713  amrex::Vector<amrex::Real> center = {zero, zero, zero};
714  for (int dir = 0; dir < AMREX_SPACEDIM; dir++) {
715  center[dir] = this->m_particle_domain.lo(dir) + m_domain_vel[dir] * a_dt;
716  }
717  this->m_particle_domain.setLo(center);
718  }
719  }
