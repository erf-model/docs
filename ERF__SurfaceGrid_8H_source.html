26  * @struct SurfaceGrid
27  * @brief 2D horizontal surface grid
28  *
29  * A slab one cell thick with C*nx x C*ny cells, where (nx, ny) is the size of
30  * the region the atmospheric level's boxes cover: the whole domain on level 0,
31  * the refined region on a finer level.
32  */

38  int C = 1; ///< Refinement factor (erf.fire.grid_ratio, erf.flood.grid_ratio)

40  /// the region the grid covers. The grid's own index space starts at zero at
41  /// that corner, as every AMReX Geometry does, so surface cell (i, j) lies in
42  /// atmospheric cell (i / C + atm_lo[0], j / C + atm_lo[1]). Zero on level 0.

47 /**
48  * @brief Construct a refined 2D surface grid over one atmospheric level's boxes
49  *
50  * @param[in] ba_atm BoxArray of the atmospheric level
51  * @param[in] dm_atm DistributionMapping of the atmospheric level
52  * @param[in] geom_atm Geometry of the atmospheric level
53  * @param[in] C Refinement factor (>= 1, aborts otherwise)
54  * @param[in] names The coupling's names, for the messages
55  *
56  * @return SurfaceGrid over the region the boxes cover; its lev is left at 0

59  * 1. Take the k = 0 slab of each box in ba_atm, keeping the box order.

64  * 4. The Geometry spans the region: index space 0 to C * size - 1, the
65  * physical extent of the region (the domain's own bounds where the region
66  * spans the domain), periodic only in a direction the region spans and the
67  * atmosphere is periodic in.

92 /**
93  * True when box n of the surface grid refines the columns of box n of the
94  * atmospheric level (shifted by -atm_lo, refined by {C, C, 1}) for every n,
95  * the two have the same owners, and both are cell-centred (the maps loop over
96  * the valid cells, so a nodal box would reach one column past its pair).

98 inline bool surface_pairs_with_atm (const amrex::MultiFab& surf_mf, const amrex::MultiFab& atm_mf,
99  const SurfaceGrid& sg)
100 {
101  const amrex::BoxArray& sba = surf_mf.boxArray();
102  const amrex::BoxArray& aba = atm_mf.boxArray();
103  if (!sba.ixType().cellCentered() || !aba.ixType().cellCentered()) {
104  return false;
105  }
106  if (sba.size() != aba.size() || surf_mf.DistributionMap() != atm_mf.DistributionMap()) {
107  return false;
108  }
109  for (int n = 0; n < aba.size(); ++n) {
110  amrex::Box ab = aba[n];
111  amrex::Box sb = sba[n];
112  ab.setSmall(2, 0); ab.setBig(2, 0);
113  sb.setSmall(2, 0); sb.setBig(2, 0);
114  ab.shift(-sg.atm_lo);
115  ab.refine(amrex::IntVect(sg.C, sg.C, 1));
116  if (ab != sb) { return false; }
117  }
118  return true;

122  * Exchange the ghost cells of a cell-centred MultiFab on a surface grid (or
123  * on the atmospheric columns it covers) and fill the ones outside a
124  * non-periodic face of the domain by copying the nearest interior value.
125  *
126  * FillBoundary alone leaves those cells untouched, so with an inflow-outflow
127  * atmosphere (the surface grid inherits its periodicity) a stencil that reads
128  * past the region's edge would read whatever the allocator had left there.
129  */
130 inline void surface_fill_boundary (amrex::MultiFab& mf, const amrex::Geometry& geom,
131  const SurfaceCouplingNames& names)
132 {
133  AMREX_ALWAYS_ASSERT_WITH_MESSAGE(mf.ixType().cellCentered(),
134  (names.tag() + " Filling the ghost cells of the " + names.grid()
135  + ": the MultiFab must be cell-centred").c_str());
136  mf.FillBoundary(geom.periodicity());
137  const auto ng = mf.nGrowVect();
138  const bool px = geom.isPeriodic(0) || ng[0] == 0;
139  const bool py = geom.isPeriodic(1) || ng[1] == 0;
140  const bool pz = geom.isPeriodic(2) || ng[2] == 0;
141  if (px && py && pz) { return; } // nothing outside a non-periodic face to fill
142  const amrex::Box& dom = geom.Domain();
143  const auto dlo = dom.smallEnd();
144  const auto dhi = dom.bigEnd();
145  const int ncomp = mf.nComp();
146  for (amrex::MFIter mfi(mf); mfi.isValid(); ++mfi) {
147  const amrex::Box& gbx = mfi.fabbox();
148  if (dom.contains(gbx)) { continue; }
149  auto a = mf.array(mfi);
150  amrex::ParallelFor(gbx, ncomp, [=] AMREX_GPU_DEVICE (int i, int j, int k, int n) noexcept
151  {
152  int ic = i, jc = j, kc = k;
153  if (!px) { ic = amrex::min(amrex::max(i, dlo[0]), dhi[0]); }
154  if (!py) { jc = amrex::min(amrex::max(j, dlo[1]), dhi[1]); }
155  if (!pz) { kc = amrex::min(amrex::max(k, dlo[2]), dhi[2]); }
156  if (ic != i || jc != j || kc != k) { a(i, j, k, n) = a(ic, jc, kc, n); }
157  });

192 /**
193  * Average a surface-grid field over each atmospheric column (the C x C surface
194  * cells under it) into the k = 0 slabs of the grid's level: surface cells
195  * (i - lo_x) C + r, (j - lo_y) C + q for r, q in [0, C) average into
196  * atmospheric cell (i, j). Box n of the output is the k = 0 slab of
197  * atmospheric box n; an output box taller than one cell gets the average at
198  * every k of its valid region. On level 0, with k = 0 slabs, this is
199  * amrex::average_down() with ratio (C, C, 1), summed in the same order.
200  */
201 inline void surface_coarsen_to_atm (amrex::MultiFab& atm_out, const amrex::MultiFab& surf,
202  const SurfaceGrid& sg, const SurfaceCouplingNames& names)
203 {
204  AMREX_ALWAYS_ASSERT_WITH_MESSAGE(
205  surface_pairs_with_atm(surf, atm_out, sg),
206  (names.tag() + " Averaging the " + names.grid()
207  + " onto the atmospheric columns: the two must pair box for box").c_str());
208  AMREX_ALWAYS_ASSERT(atm_out.nComp() >= surf.nComp());
209  const int C = sg.C;
210  const int lo_x = sg.atm_lo[0];
211  const int lo_y = sg.atm_lo[1];
212  const int ncomp = surf.nComp();
213  const amrex::Real inv_cells = amrex::Real(1.0) / amrex::Real(C * C);
214  for (amrex::MFIter mfi(atm_out); mfi.isValid(); ++mfi) {
215  auto crse = atm_out.array(mfi);
216  auto fine = surf.const_array(mfi);
217  const int kf = surf.boxArray()[mfi.index()].smallEnd(2);
218  amrex::ParallelFor(mfi.validbox(), ncomp, [=] AMREX_GPU_DEVICE (int i, int j, int k, int n) noexcept
219  {
220  const int ii = (i - lo_x) * C;
221  const int jj = (j - lo_y) * C;
222  amrex::Real c = amrex::Real(0.0);
223  for (int q = 0; q < C; ++q) {
224  for (int r = 0; r < C; ++r) {
225  c += fine(ii + r, jj + q, kf, n);
226  }
227  }
228  crse(i, j, k, n) = inv_cells * c;
229  });
230  }
231 }
