12 /**
13  * @file ERF_ForceRestoreHydrology.H
14  * @brief Water of the two-stream force-restore surface: rain in, runoff, a prognostic deep
15  * reservoir with gravitational drainage, and a prognostic deep temperature.
16  *
17  * The force-restore balance (ERF_SimplifiedSEB.H) carries q_s, the volumetric water content
18  * [m^3/m^3] of a surface layer d_s deep (erf.radiation.seb_moisture_layer_depth_m), restored
19  * toward q_deep over tau_q. These functions add the terms of Noilhan and Planton (1989,
20  * "NP89") and Mahfouf and Noilhan (1996, "MN96") that it leaves out, each behind its own
21  * input:
22  *
23  * - Rain (erf.radiation.seb_precipitation_source = microphysics). P [kg/m^2/s] enters the
24  * surface layer as in NP89 Eq. (11) and Deardorff (1978) Eq. (12), with depth d_1 = d_s:
25  * dq_s/dt += P / (rho_w d_s).
26  * NP89 Eq. (11) multiplies (P - E) by one coefficient C1. The existing balance takes E out
27  * of q_s with C1 = 1 (prognostic_dqs_dt), so rain uses C1 = 1 too. NP89 Eq. (19) and Table 2
28  * give C1 = 0.08-0.34 at saturation, rising as the soil dries, and D78 Eq. (13) gives
29  * C1 = 0.5-14. With C1 = 1 the 10 cm layer fills faster than with NP89's C1 near saturation,
30  * so heavy rain runs off sooner than in NP89.
31  * - Runoff (with rain). Water above the porosity w_sat runs off (NP89 Sec. 4a: "Runoff occurs
32  * when w_g or w_2 exceeds the saturation value w_sat"). The surface layer's runoff that
33  * leaves the column is at most this step's rain: water above w_sat that was already in q_s
34  * (a restore overshoot, or a restart from a run without hydrology) is clamped off q_s and
35  * does not come out of the column.
36  * - Deep reservoir (erf.radiation.seb_deep_prognostic_enable). q_deep becomes w_2, the mean
37  * water content of the soil column of depth d_2 (erf.radiation.seb_deep_layer_depth_m),
38  * which contains the surface layer (NP89 Eq. 12, MN96 Eq. 4):
39  * dq_deep/dt = (P - E) / (rho_w d_2).
40  * Since w_2 is the mean of the whole column, the restore term of q_s moves water within the
41  * column and does not appear here; the column's water is rho_w d_2 q_deep.
42  * - Deep temperature (with the deep reservoir). T_2 relaxes toward T_s over tau
43  * (erf.radiation.seb_restore_timescale_s), NP89 Eq. (6):
44  * dT_2/dt = (T_s - T_2) / tau.
45  * - Gravitational drainage (erf.radiation.seb_drainage_enable). Water above field capacity
46  * w_fc leaves the bottom of the column, MN96 Eq. (4):
47  * dq_deep/dt += -(C_3 / tau_1d) max(0, q_deep - w_fc),
48  * with C_3 = (C_3 d_2) / d_2 (MN96 Eq. 16 and Table 1, which tabulates C_3 for d_2 = 1 m).
49  * MN96 Eq. (15) makes C_3 proportional to tau, and Table 1 is computed with tau = 1 day
50  * (MN96 appendix), so the drainage uses tau_1d = 86400 s, not seb_restore_timescale_s. The
51  * drainage rate then does not depend on that input.
52  *
53  * The relaxation terms (drainage, deep temperature) are linear in the state, so each is
54  * integrated exactly over the step, x_new - x_eq = (x_old - x_eq) exp(-k dt); the update
55  * cannot overshoot whatever the step. The source terms (rain, evaporation) are forward Euler,
56  * as the existing balance is.
57  *
58  * References:
59  * - Deardorff, J. W., 1978: Efficient prediction of ground surface temperature and moisture,
60  * with inclusion of a layer of vegetation. J. Geophys. Res., 83, 1889-1903.
61  * - Noilhan, J., and S. Planton, 1989: A simple parameterization of land surface processes
62  * for meteorological models. Mon. Wea. Rev., 117, 536-549.
63  * - Mahfouf, J.-F., and J. Noilhan, 1996: Inclusion of gravitational drainage in a land
64  * surface scheme based on the force-restore method. J. Appl. Meteor., 35, 987-992.
65  */

118 /**
119  * @brief One step of the force-restore hydrology for one column, after the balance has
120  * updated T_s and q_s (prognostic_dTs_dt, prognostic_dqs_dt).
121  *
122  * In order: rain into the surface layer and the surface runoff above the porosity (at most
123  * the step's rain: water that never infiltrates); the infiltrated rain minus the evaporation
124  * into the column, then its drainage; the deep temperature toward the updated T_s; runoff of
125  * the column above the porosity; the bounds of q_s and q_deep. The surface runoff is taken
126  * before the drainage, so water that never entered the soil does not drain.
127  *
128  * With the deep reservoir, the column then satisfies exactly
129  * rho_w d_2 (q_deep_new - q_deep_old) = (P - E) dt - drainage - runoff + deficit.
130  *
131  * @param[in,out] q_s surface-layer water content [m^3/m^3]
132  * @param[in,out] q_deep column water content [m^3/m^3]; changed only with prm.deep
133  * @param[in,out] t_deep deep temperature [K]; changed only with prm.deep
134  * @param[in] t_s surface temperature after this step's update [K]
135  * @param[in] p rain reaching the soil [kg/m^2/s]; used only with prm.rain
136  * @param[in] e evaporation LE / L_v [kg/m^2/s], positive upward
137  * @param[in] dt the step [s]
138  * @param[in] prm parameters
139  * @return the water that left the column, and any deficit
140  */
141 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
142 ForceRestoreHydrologyFluxes
143 force_restore_hydrology_step (amrex::Real& q_s, amrex::Real& q_deep, amrex::Real& t_deep,
144  amrex::Real t_s, amrex::Real p, amrex::Real e, amrex::Real dt,
145  const ForceRestoreHydrologyParams& prm) noexcept
146 {
147  ForceRestoreHydrologyFluxes out;
148  const amrex::Real zero(0.0);
149  const amrex::Real one(1.0);
150  if (!prm.rain) { p = zero; }
151  if (!amrex::Math::isfinite(p)) { p = zero; }
152  if (!amrex::Math::isfinite(e)) { e = zero; }
153 
154  // Rain into the surface layer: NP89 Eq. (11), first term, with C1 = 1 and d_1 = d_s.
155  // (The balance already removed E from q_s in prognostic_dqs_dt.)
156  // Runoff above saturation (NP89 Sec. 4a) from the surface layer: at most this step's rain
157  // runs off there, since that water never infiltrated. Water above w_sat that q_s held
158  // before the rain is clamped off q_s and stays in the column.
159  amrex::Real surface_runoff = zero;
160  if (prm.rain) {
161  q_s += dt * p / (rhor * prm.d_s);
162  if (q_s > prm.w_sat) {
163  surface_runoff = amrex::min(rhor * prm.d_s * (q_s - prm.w_sat), p * dt);
164  q_s = prm.w_sat;
165  out.runoff += surface_runoff;
166  }
167  }
168 
169  if (prm.deep) {
170  // The column's water: NP89 Eq. (12), MN96 Eq. (4), first term, with the rain that
171  // infiltrated.
172  q_deep += (dt * (p - e) - surface_runoff) / (rhor * prm.d_2);
173 
174  // Gravitational drainage toward field capacity: MN96 Eq. (4), second term, exact
175  // over the step for the linear relaxation.
176  if (prm.drain && q_deep > prm.w_fc) {
177  const amrex::Real removed = (q_deep - prm.w_fc) *
178  (one - std::exp(-prm.c3 * dt / force_restore_c3_tau));
179  q_deep -= removed;
180  out.drainage = rhor * prm.d_2 * removed;
181  }
182 
183  // Deep temperature: NP89 Eq. (6), exact over the step.
184  t_deep += (t_s - t_deep) * (one - std::exp(-dt / prm.tau));
185 
186  // Runoff of the column above saturation.
187  if (q_deep > prm.w_sat) {
188  out.runoff += rhor * prm.d_2 * (q_deep - prm.w_sat);
189  q_deep = prm.w_sat;
190  }
191  }
192 
193  // NP89 Eq. (12): 0 < w_2. An evaporation the column cannot supply is recorded, not hidden:
194  // the latent heat flux has already taken it, so a deficit is water the coupled system did
195  // not conserve (LE is not limited by q_deep).
196  if (prm.deep && q_deep < zero) {
197  out.deficit = -rhor * prm.d_2 * q_deep;
198  q_deep = zero;
199  }
200 
201  q_s = amrex::max(prm.q_min, amrex::min(prm.q_max, q_s));
202  return out;
203 }
