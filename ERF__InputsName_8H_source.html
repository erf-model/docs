64  * The values on the right of '=': whitespace separates them, and a double-quoted value is one
65  * value without its quotes.

67  inline std::vector<std::string>
68  split_values (const std::string& rhs)
69  {
70  std::vector<std::string> vals;
71  std::size_t i = 0;
72  while (i < rhs.size()) {
73  if (rhs[i] == ' ' || rhs[i] == '\t' || rhs[i] == '\r') { ++i; continue; }
74  if (rhs[i] == '"') {
75  auto e = rhs.find('"', i + 1);
76  if (e == std::string::npos) { e = rhs.size(); }
77  vals.push_back(rhs.substr(i + 1, e - i - 1));
78  i = e + 1;
79  } else {
80  auto e = rhs.find_first_of(" \t\r", i);
81  if (e == std::string::npos) { e = rhs.size(); }
82  vals.push_back(rhs.substr(i, e - i));
83  i = e;
84  }
85  }
86  return vals;
87  }

91  * when the first setting came from a file this one included earlier (the base + override
92  * idiom; ParmParse keeps the last value). A key set twice in one file, in two files included
93  * side by side, or in a file and then again in a file it includes afterwards is a duplicate.

95  inline bool
