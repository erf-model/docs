8 /**
9  * @file ERF_TwoStreamSW.H
10  * @brief Shortwave (solar) radiation kernels: Beer-Lambert direct beam and
11  * the two-stream layer solution for the diffuse field.
12  *
13  * The column model (ERF_TwoStreamColumn.H) treats the solar radiation as a
14  * direct beam plus a diffuse field with upward and downward streams:
15  *
16  * - The direct beam is attenuated by Beer-Lambert,
17  * F_dir(m) = S0 * mu0 * exp(-tau_cum(m) / mu0),
18  * where tau_cum is the optical depth from the top of the atmosphere to
19  * interface m and mu0 the cosine of the solar zenith angle.
20  *
21  * - Each layer converts part of the direct beam into diffuse radiation
22  * (scattering) and reflects/transmits the diffuse streams. For a
23  * homogeneous layer with optical depth tau, single-scattering albedo omega
24  * and asymmetry factor g, compute_sw_layer_two_stream() returns the
25  * two-stream reflectance and transmittance for diffuse incidence (R_dif,
26  * T_dif) and the diffuse reflectance and transmittance generated by direct
27  * incidence (R_dir, T_dir), together with the direct transmittance
28  * T_noscat = exp(-tau/mu0). The gamma coefficients are those of the
29  * practical improved flux method (Zdunkowski et al. 1980), as used by
30  * RRTMGP; the layer solution follows Meador and Weaver (1980), Eqs. 14-15.
31  *
32  * - The layers are combined with the surface by the adding method: the
33  * surface reflects the fraction alpha of the direct and diffuse flux that
34  * reaches it, and the reflected radiation is scattered and absorbed again
35  * on its way up. Heating rates follow from the divergence of the net flux
36  * F_dir + F_diff_down - F_diff_up.
37  *
38  * With omega == 0 in every layer the diffuse field reduces to the reflected
39  * direct beam only, and the absorbed surface flux reduces exactly to
40  * (1 - alpha) times the Beer-Lambert direct beam.
41  *
42  * References:
43  * -----------
44  * - Meador, W. E., and W. R. Weaver, 1980: Two-stream approximations to
45  * radiative transfer in planetary atmospheres: A unified description
46  * of existing methods and a new improvement. J. Atmos. Sci., 37,
47  * 630-643.
48  * - Zdunkowski, W. G., R. M. Welch, and G. Korb, 1980: An investigation of
49  * the structure of typical two-stream methods for the calculation of
50  * solar fluxes and heating rates in clouds. Beitr. Phys. Atmos., 53,
51  * 147-166.
52  * - Toon, O. B., C. P. McKay, T. P. Ackerman, and K. Santhanam, 1989:
53  * Rapid calculation of radiative heating rates and photodissociation
54  * rates in inhomogeneous multiple scattering atmospheres. J. Geophys.
55  * Res., 94, 16287-16301.
56  */

58 /**
59  * @brief Compute Beer-Lambert direct-beam flux at a given optical depth.
60  *
61  * F_dir = S0 * cos_zenith * exp(-tau_cumulative / cos_zenith)
62  *
63  * @param[in] tau_cumulative Optical depth from the top of the atmosphere [unitless].
64  * @param[in] S0 Solar constant at the top of the atmosphere [W/m^2].
65  * @param[in] cos_zenith Cosine of the solar zenith angle [unitless].
66  *
67  * @return Downwelling direct-beam flux [W/m^2], or 0 if cos_zenith <= 0 (night).
68  */
69 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
70 amrex::Real
71 compute_sw_direct_flux(amrex::Real tau_cumulative, amrex::Real S0,
72  amrex::Real cos_zenith)
73 {
74  if (cos_zenith <= 0.0) {
75  return 0.0;
76  }
77  return S0 * cos_zenith * std::exp(-tau_cumulative / cos_zenith);
78 }

80 /**
81  * @brief Compute the shortwave heating rate of a layer from the net
82  * (downward positive) flux at its top and bottom interfaces.
83  *
84  * Q = (F_net_top - F_net_bot) / (dz * rho * cp) [K/s]
85  *
86  * Energy converging into the layer warms it. Returns 0 for unphysical
87  * inputs (dz, rho, cp <= 0) or a non-finite result.
88  *
89  * @param[in] flux_top Net downward flux at the top of the layer [W/m^2].
90  * @param[in] flux_bot Net downward flux at the bottom of the layer [W/m^2].
91  * @param[in] dz Vertical thickness of the layer [m]. Must be positive.
92  * For terrain-aware grids, pass dz = z_cc(k+1) - z_cc(k).
93  * @param[in] rho Density [kg/m^3]. Must be positive.
94  * @param[in] cp Specific heat at constant pressure [J/(kg·K)]. Must be positive.
95  *
96  * @return Heating rate [K/s]. Positive values indicate warming.
97  */
98 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
99 amrex::Real
100 compute_sw_heating_rate(amrex::Real flux_top, amrex::Real flux_bot,
101  amrex::Real dz, amrex::Real rho, amrex::Real cp)
102 {
103  if (dz <= 0.0 || rho <= 0.0 || cp <= 0.0) {
104  return 0.0;
105  }
106  // Flux divergence: (W/m^2) / (m) = W/m^3
107  amrex::Real flux_divergence = (flux_top - flux_bot) / dz;
108  // Heating rate: (W/m^3) / (kg/m^3 * J/(kg*K)) = K/s
109  amrex::Real heating = flux_divergence / (rho * cp);
110  if (!std::isfinite(heating)) {
111  return 0.0;
112  }
113  return heating;
114 }

116 /**
117  * @brief Two-stream reflectance and transmittance of one homogeneous layer.
118  *
119  * All quantities are fractions of the incident flux:
120  * - R_dif, T_dif: reflectance and transmittance for diffuse incidence.
121  * - R_dir, T_dir: diffuse flux reflected upward / transmitted downward per
122  * unit direct-beam flux incident at the top of the layer (the scattered
123  * part only; the surviving direct beam is T_noscat).
124  * - T_noscat = exp(-tau / mu0): direct-beam transmittance.
125  *
126  * For a non-absorbing layer (omega = 1) R_dif + T_dif = 1 and
127  * R_dir + T_dir + T_noscat = 1. For a non-scattering layer (omega = 0)
128  * R_dif = R_dir = T_dir = 0 and T_dif = exp(-2 tau) (diffusivity factor 2).
129  */
130 struct TwoStreamLayerSW
131 {
132  amrex::Real R_dif;
133  amrex::Real T_dif;
134  amrex::Real R_dir;
135  amrex::Real T_dir;
136  amrex::Real T_noscat;
137 };

139 /**
140  * @brief Compute the two-stream layer solution for shortwave radiation.
141  *
142  * Gamma coefficients (practical improved flux method, Zdunkowski et al.
143  * 1980, as in RRTMGP):
144  * gamma1 = (8 - omega * (5 + 3 g)) / 4
145  * gamma2 = 3 omega (1 - g) / 4
146  * gamma3 = (2 - 3 g mu0) / 4
147  * gamma4 = 1 - gamma3
148  * k = sqrt(gamma1^2 - gamma2^2)
149  *
150  * Diffuse incidence (Meador and Weaver 1980):
151  * R_dif = gamma2 (1 - e^{-2 k tau}) / D
152  * T_dif = 2 k e^{-k tau} / D, D = k (1 + e^{-2 k tau}) + gamma1 (1 - e^{-2 k tau})
153  *
154  * Direct incidence (Meador and Weaver 1980, Eqs. 14-15), with
155  * alpha1 = gamma1 gamma4 + gamma2 gamma3, alpha2 = gamma1 gamma3 + gamma2 gamma4:
156  * R_dir = omega / (D (1 - k^2 mu0^2)) *
157  * [ (1 - k mu0)(alpha2 + k gamma3) - (1 + k mu0)(alpha2 - k gamma3) e^{-2 k tau}
158  * - 2 (k gamma3 - alpha2 k mu0) e^{-k tau} T_noscat ]
159  * T_dir = - omega / (D (1 - k^2 mu0^2)) *
160  * [ (1 + k mu0)(alpha1 + k gamma4) T_noscat - (1 - k mu0)(alpha1 - k gamma4) e^{-2 k tau} T_noscat
161  * - 2 (k gamma4 + alpha1 k mu0) e^{-k tau} ]
162  *
163  * The removable singularity at k mu0 = 1 is avoided by nudging mu0 slightly,
164  * and the results are clipped to the physical range (non-negative, and
165  * R_dir + T_dir <= 1 - T_noscat).
166  *
167  * @param[in] tau Optical depth of the layer [unitless].
168  * @param[in] omega Single-scattering albedo of the layer, in [0, 1].
169  * @param[in] g Asymmetry factor of the layer, in [-1, 1].
170  * @param[in] cos_zenith Cosine of the solar zenith angle, in (0, 1].
171  * @return Layer reflectances and transmittances (see TwoStreamLayerSW).
172  */
173 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
174 TwoStreamLayerSW
175 compute_sw_layer_two_stream(amrex::Real tau, amrex::Real omega, amrex::Real g,
176  amrex::Real cos_zenith)
177 {
178  TwoStreamLayerSW L{0.0, 1.0, 0.0, 0.0, 1.0};
179 
180  if (!(tau > 0.0) || !std::isfinite(tau)) {
181  return L; // Empty layer: transparent to everything
182  }
183  if (!(cos_zenith > 0.0)) {
184  // Night: no direct beam; the diffuse properties are still defined but
185  // the caller does not use them.
186  L.T_noscat = 0.0;
187  }
188 
189  amrex::Real w0 = omega;
190  if (!(w0 > 0.0) || !std::isfinite(w0)) w0 = 0.0;
191  if (w0 > 1.0) w0 = 1.0;
192  amrex::Real asym = g;
193  if (!std::isfinite(asym)) asym = 0.0;
194  if (asym > 1.0) asym = 1.0;
195  if (asym < -1.0) asym = -1.0;
196  amrex::Real mu0 = (cos_zenith > 0.0) ? cos_zenith : 1.0;
197  if (mu0 > 1.0) mu0 = 1.0;
198 
199  const amrex::Real gamma1 = (8.0 - w0 * (5.0 + 3.0 * asym)) / 4.0;
200  const amrex::Real gamma2 = 3.0 * w0 * (1.0 - asym) / 4.0;
201  const amrex::Real gamma3 = (2.0 - 3.0 * asym * mu0) / 4.0;
202  const amrex::Real gamma4 = 1.0 - gamma3;
203 
204  const amrex::Real alpha1 = gamma1 * gamma4 + gamma2 * gamma3;
205  const amrex::Real alpha2 = gamma1 * gamma3 + gamma2 * gamma4;
206 
207  // k^2 = gamma1^2 - gamma2^2 >= 0; guard the conservative limit omega = 1.
208  amrex::Real k_sq = (gamma1 - gamma2) * (gamma1 + gamma2);
209  if (k_sq < 1.0e-12) k_sq = 1.0e-12;
210  const amrex::Real k = std::sqrt(k_sq);
211 
212  const amrex::Real E = std::exp(-k * tau);
213  const amrex::Real E2 = E * E;
214  const amrex::Real D = k * (1.0 + E2) + gamma1 * (1.0 - E2);
215  const amrex::Real RT = 1.0 / D;
216 
217  L.R_dif = RT * gamma2 * (1.0 - E2);
218  L.T_dif = RT * 2.0 * k * E;
219  if (L.R_dif < 0.0) L.R_dif = 0.0;
220  if (L.T_dif < 0.0) L.T_dif = 0.0;
221  if (L.R_dif + L.T_dif > 1.0) {
222  const amrex::Real s = 1.0 / (L.R_dif + L.T_dif);
223  L.R_dif *= s;
224  L.T_dif *= s;
225  }
226 
227  if (!(cos_zenith > 0.0)) {
228  return L;
229  }
230 
231  // Direct beam: nudge mu0 away from the removable singularity k mu0 = 1.
232  amrex::Real k_mu = k * mu0;
233  if (std::abs(1.0 - k_mu * k_mu) < 1.0e-4) {
234  k_mu = (k_mu < 1.0) ? (1.0 - 1.0e-2) : (1.0 + 1.0e-2);
235  }
236  const amrex::Real T_noscat = std::exp(-tau / mu0);
237  L.T_noscat = T_noscat;
238 
239  const amrex::Real RT2 = w0 * RT / (1.0 - k_mu * k_mu);
240  const amrex::Real k_g3 = k * gamma3;
241  const amrex::Real k_g4 = k * gamma4;
242 
243  amrex::Real R_dir = RT2 * ((1.0 - k_mu) * (alpha2 + k_g3)
244  - (1.0 + k_mu) * (alpha2 - k_g3) * E2
245  - 2.0 * (k_g3 - alpha2 * k_mu) * E * T_noscat);
246  amrex::Real T_dir = -RT2 * ((1.0 + k_mu) * (alpha1 + k_g4) * T_noscat
247  - (1.0 - k_mu) * (alpha1 - k_g4) * E2 * T_noscat
248  - 2.0 * (k_g4 + alpha1 * k_mu) * E);
249 
250  if (!std::isfinite(R_dir) || R_dir < 0.0) R_dir = 0.0;
251  if (!std::isfinite(T_dir) || T_dir < 0.0) T_dir = 0.0;
252  const amrex::Real budget = 1.0 - T_noscat; // scattered + absorbed fraction
253  if (R_dir + T_dir > budget) {
254  const amrex::Real s = (R_dir + T_dir > 0.0) ? budget / (R_dir + T_dir) : 0.0;
255  R_dir *= s;
256  T_dir *= s;
257  }
258  L.R_dir = R_dir;
259  L.T_dir = T_dir;
260  return L;
261 }
