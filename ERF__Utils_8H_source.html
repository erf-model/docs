426 /**
427  * Time interpolation weights for boundary data at fixed intervals.
428  *
429  * The time must be measured from the first slice, and formed as
430  * elapsed + (start_time - start_bdy_time), not as (elapsed + start_time) - start_bdy_time:
431  * start_time and start_bdy_time are epoch seconds (~1.7e9), where doubles are spaced
432  * ~2.4e-7 s apart, so the second form rounds the elapsed time to that spacing. The
433  * difference of the two epoch times is exact.
434  *
435  * @param[in] time_since_start time since the first slice
436  * @param[in] data_span time from the first slice to the last one
437  * @param[in] dT interval between slices
438  */
439 AMREX_FORCE_INLINE
440 BdyTimeWeights
441 bdy_time_weights (const double time_since_start, const double data_span, const double dT)
442 {
443  BdyTimeWeights w;
444  if (time_since_start >= data_span) {
445  // Do not over run the last slice
446  w.n_lo = static_cast<int>(data_span / dT);
447  w.n_hi = w.n_lo;
448  w.alpha = zero;
449  } else {
450  w.n_lo = static_cast<int>(time_since_start / dT);
451  w.n_hi = w.n_lo + 1;
452  w.alpha = static_cast<amrex::Real>((time_since_start - w.n_lo * dT) / dT);
453  }
454  AMREX_ALWAYS_ASSERT(w.alpha >= zero && w.alpha <= one);
455  return w;
456 }

991 /**
992  * Decide whether data at time "time" should come from the old-time data, the new-time data,
993  * or an interpolation between the two.
994  *
995  * This is done in double precision, before any cast to Real, and with a single tolerance
996  * at both ends. The tolerance is a small fraction of (t_new - t_old), but never smaller than
997  * a few ulps of the times themselves, since those carry roundoff proportional to their
998  * magnitude -- which matters once the time is large (e.g. months measured in seconds).
999  *
1000  * Times outside [t_old, t_new] are clamped to the nearer end rather than extrapolated.
1001  * If t_old == t_new (to within the tolerance), Old is returned.
1002  *
1003  * @param[in] time time at which data are requested
1004  * @param[in] t_old old time of the level
1005  * @param[in] t_new new time of the level
1006  */
1007 inline TimeLevelSel
1008 select_time_level (double time, double t_old, double t_new)
1009 {
1010  const double t_mag = std::max({std::abs(time), std::abs(t_old), std::abs(t_new)});
1011  const double tol = std::max(1.e-8 * std::abs(t_new - t_old),
1012  16.0 * std::numeric_limits<double>::epsilon() * t_mag);
1013 
1014  if (time <= t_old + tol) { return TimeLevelSel::Old; }
1015  if (time >= t_new - tol) { return TimeLevelSel::New; }
1016  return TimeLevelSel::Interp;
1017 }
