479  const bool terrain = (z_cc != nullptr);
480  AMREX_ALWAYS_ASSERT(!terrain || (z_nd && z_ground));
481  const int klo = geom.Domain().smallEnd(2);
482  const auto dx = geom.CellSizeArray();
483  const auto problo = geom.ProbLoArray();
484  for (const auto& face : faces) {
485  if (!face.inflow()) { continue; }
486  const amrex::Real speed = std::hypot(face.u, face.v);
487  for (const auto& isect : pb_ba[lev].intersections(face.region)) {
488  const int b = isect.first;
489  pb_Ug[lev][b] = std::max(pb_Ug[lev][b], face.Ug);
490  if (speed > pb_mag[lev][b]) {
491  pb_mag[lev][b] = speed;
492  pb_dir[lev][b] = std::atan2(std::abs(face.u), std::abs(face.v));
493  }
494  }
495  const amrex::Real height = face.height;
496  const amrex::Box region = amrex::convert(face.region, auto_mask[lev].ixType());
497  for (amrex::MFIter mfi(auto_mask[lev]); mfi.isValid(); ++mfi) {
498  const amrex::Box bx = mfi.validbox() & region;
499  if (!bx.ok()) { continue; }
500  const auto mask = auto_mask[lev].array(mfi);
501  const auto cc = terrain ? z_cc->const_array(mfi) : amrex::Array4<const amrex::Real>{};
502  const auto nd = terrain ? z_nd->const_array(mfi) : amrex::Array4<const amrex::Real>{};
503  const auto ground = terrain ? z_ground->const_array(mfi) : amrex::Array4<const amrex::Real>{};
504  ParallelFor(bx, [=] AMREX_GPU_DEVICE (int i, int j, int k) noexcept {
505  amrex::Real z = problo[2] + (k-klo + (on_w_faces ? zero : myhalf))*dx[2];
506  amrex::Real zs = problo[2];
507  if (terrain) {
508  zs = fourth*(ground(i,j,klo) + ground(i+1,j,klo) +
509  ground(i,j+1,klo) + ground(i+1,j+1,klo));
510  z = on_w_faces ? fourth*(nd(i,j,k) + nd(i+1,j,k) +
511  nd(i,j+1,k) + nd(i+1,j+1,k))
512  : cc(i,j,k);
513  }
514  if (z > zs && z-zs <= height) { mask(i,j,k) = one; }
515  });
516  }
517  }
518  // Reconsider even boxes that previously had calm winds. They must be able
519  // to start again after a later parent update changes the inflow.
520  pb_interval[lev].assign(pb_ba[lev].size(), -one);
521  pb_local_etime[lev].assign(pb_ba[lev].size(), zero);
522  auto_ready[lev] = 1;
523  auto_next_update[lev] = time + period;
524  }

526  /**
527  * @brief Update perturbation amplitudes and intervals when each box is ready.
528  * @param lev AMR level index.
529  * @param dt Time step for elapsed-time accumulation.
530  * @param mf_xvel X-velocity field used to compute box mean velocities.
531  * @param mf_yvel Y-velocity field used to compute box mean velocities.
532  * @param mf_cons Conserved state used for iteration layout.
533  */
534  void calc_tpi_update (const int lev,
535  const double dt,
536  amrex::MultiFab& mf_xvel,
537  amrex::MultiFab& mf_yvel,
538  amrex::MultiFab& mf_cons)
539  {
540  auto m_ixtype = mf_cons.boxArray().ixType(); // safety step
541  if (pt_type[lev] == 3) { m_ixtype = amrex::IndexType(amrex::IntVect(0,0,1)); }
542 
543  const int nboxes = pb_ba[lev].size();
544  if (nboxes == 0 || (auto_mode && !auto_ready[lev])) { return; }
545 
546  // Which boxes refresh this step. pb_interval and pb_local_etime hold the same values on
547  // every rank, so every rank makes the same choice.
548  amrex::Vector<int> update_box(nboxes, 0);
549  for (int boxIdx = 0; boxIdx < nboxes; boxIdx++) {
550  if (pt_type[lev] == 2 || pt_type[lev] == 3) {
551  // for CPM, perturbation boxes/cells refresh after boxes have advected box width/length * num_layers (advective time scale)
552  update_box[boxIdx] = ((pb_local_etime[lev][boxIdx] >= pb_interval[lev][boxIdx]*tpi_layers) && pb_local_etime[lev][boxIdx] != -one);
553  } else {
554  update_box[boxIdx] = ( pb_local_etime[lev][boxIdx] >= pb_interval[lev][boxIdx] && pb_local_etime[lev][boxIdx] != -one );
555  }
556  }
557 
558  // Compute mean velocity of each refreshing perturbation box
559  if (!auto_mode) { calc_tpi_meanMag(lev, update_box, mf_cons, mf_xvel, mf_yvel); }
560 
561  for (int boxIdx = 0; boxIdx < nboxes; boxIdx++) {
562  if ( update_box[boxIdx] ) {
563 
564  if (pb_mag[lev][boxIdx] !=zero) {
565  double interval = 0.0;
566  if (pt_type[lev] == 2 || pt_type[lev] == 3) {
567  // Wind direction correction for angled wind
568  amrex::Real wind_direction = pb_dir[lev][boxIdx];
569  if (wind_direction > PI / 4) { wind_direction = PI / 2 - wind_direction; }
570  // CPM only cares about the side length of a box, min call maintains flexibility.
571  interval = one / std::cos(wind_direction) * std::min(tpi_Lpb[lev], tpi_Wpb[lev]) / pb_mag[lev][boxIdx];
572  } else {
573  interval = tpi_lref[lev] / pb_mag[lev][boxIdx];
574  }
575  // 10% variation
576  const amrex::Real r = erf_hash_rng::hash_uniform(boxIdx, pb_nupdate[lev][boxIdx], 0,
577  hash_comp_interval, lev, tpi_seed);
578  pb_interval[lev][boxIdx] = (amrex::Real(0.9) + amrex::Real(0.2)*r) * interval;
579 
580  // Reset local elapsed time
581  pb_local_etime[lev][boxIdx] = zero;
582  } else {
583  // No mean wind through this box, so no advective time scale: it is not refreshed again
584  pb_local_etime[lev][boxIdx] = -one;
585  }
586 
587  // Trigger amplitude calculation per perturbation box
588  calc_tpi_amp(lev, boxIdx, pb_interval[lev][boxIdx]);
589 
590  // Trigger random amplitude storage per cell within perturbation box
591  pseudoRandomPert(boxIdx, lev, m_ixtype);
592 
593  pb_nupdate[lev][boxIdx]++;
594 
595  } else {
596  // set perturbation amplitudes to 0 for CPM. A little inefficient but leverages as much as existing code as possible.
597  if (pt_type[lev] == 2 || pt_type[lev] == 3) { zero_amp(boxIdx, lev, m_ixtype); }
598 
599  // Increase by timestep of level 0
600  if (pb_local_etime[lev][boxIdx] != -one) { pb_local_etime[lev][boxIdx] += dt; }
601  } // if
602  } // for
603 
604  if (pt_type[lev] < 2) { // box perturbation method only
605  netZeroBuoyant(lev);
606  }

652  /**
653  * @brief Compute the perturbation amplitude for one perturbation box.
654  * @param lev AMR level index.
655  * @param boxIdx Perturbation-box index on the level.
656  * @param interval Current perturbation update interval [s].
657  */
658  void calc_tpi_amp (const int& lev,
659  const int& boxIdx,
660  const double& interval)
661  {
662  pb_amp[lev][boxIdx] = zero; // Safety step
663  if (pt_type[lev] == 2) { // CPM
664  constexpr amrex::Real cp = amrex::Real(1004); // specific heat of air [J/(kg K)]
665  constexpr amrex::Real Ec = amrex::Real(0.2); // Eckert number
666  const amrex::Real Ug = auto_mode ? pb_Ug[lev][boxIdx] : input_Ug;
667  pb_amp[lev][boxIdx] = (Ug * Ug) / (Ec * cp);
668  } else if (pt_type[lev] == 3) {
669  pb_amp[lev][boxIdx] = input_w_amp;
670  } else { // box perturbation
671  amrex::Real Um = pb_mag[lev][boxIdx];
672 
673  // No mean wind: nothing to trip, and with tpi_Ti > 0 the amplitude below would be
674  // 0/0. The box is not refreshed again, so a zero amplitude also clears the
675  // perturbation it held
676  if (Um == zero) { return; }
677 
678  amrex::Real beta = one/tpi_Tinf; // Thermal expansion coefficient
679 
680  // Pseudo Random temperature the ignores scale when mechanically tripping turbulence
681  amrex::Real g = CONST_GRAV;
682  // get total refinement ratio on each level
683  int total_ref_ratio = 1;
684  for (int level = lev; level >= 1; level--) {
685  total_ref_ratio *= ref_ratio[level-1][2];
686  }
687  // calculation needs to be scale-aware since the formulation relies on the physical size of the box
688  if (tpi_Ti > zero) g = (tpi_nonDim * Um * Um) / (tpi_Ti * tpi_Hpb[lev]) * 1 / total_ref_ratio;
689 
690  // Ma and Senocak (2023) Eq. 8, solving for delta phi
691  pb_amp[lev][boxIdx] = (tpi_nonDim * Um * Um) / (g * beta * tpi_Hpb[lev]) * 1 / total_ref_ratio;
692 
693  // Divide by the interval to get a forcing rate [K/s], Ma & Senocak (2023) Eq. 7.
694  // The source perturbation adds this rate as a source term; the direct
695  // perturbation adds it times the time step to the state at every step. Either
696  // way the total over one refresh interval is delta phi, even if dt changes
697  pb_amp[lev][boxIdx] /= static_cast<amrex::Real>(interval);
698  }
699  }

701  /**
702  * @brief Assign pseudo-random perturbations to cells in one perturbation box.
703  * @param boxIdx Perturbation-box index on the level.
704  * @param lev AMR level index.
705  * @param m_ixtype Index type used to align perturbation storage.
706  */
707  void pseudoRandomPert (const int& boxIdx,
708  const int& lev,
709  const amrex::IndexType& m_ixtype)
710  {
711  for (amrex::MFIter mfi(pb_cell[lev],TileNoZ()); mfi.isValid(); ++mfi) {
712  amrex::Box vbx = mfi.tilebox();
713  amrex::Box pbx = amrex::convert(pb_ba[lev][boxIdx], m_ixtype);
714  if (pt_type[lev] == 3) { pbx.setBig(2, pbx.bigEnd(2) - 1); } // prevent double counting after converting to k-faces
715  amrex::Box ubx = pbx & vbx;
716  if (ubx.ok()) {
717  amrex::Real amp_copy = pb_amp[lev][boxIdx];
718  const amrex::Array4<amrex::Real>& pert_cell = pb_cell[lev].array(mfi);
719  const bool use_mask = auto_mode;
720  const auto mask = use_mask ? auto_mask[lev].const_array(mfi)
721  : amrex::Array4<const amrex::Real>{};
722 
723  if (pt_type[lev] == 2 || pt_type[lev] == 3) { // CPM: one value per perturbation box
724  const amrex::Real rand_number_const = erf_hash_rng::hash_symmetric(
725  boxIdx, pb_nupdate[lev][boxIdx], 0, hash_comp_cpm, lev, tpi_seed);
726  ParallelFor(ubx, [=] AMREX_GPU_DEVICE(int i, int j, int k) noexcept {
727  pert_cell(i,j,k) = (use_mask && mask(i,j,k) == zero) ? zero : rand_number_const * amp_copy;
728  });
729  } else { // one value per cell, keyed by its global index and the box's refresh count
730  const std::uint64_t cell_seed = erf_hash_rng::splitmix64(
731  tpi_seed ^ erf_hash_rng::splitmix64(static_cast<std::uint64_t>(pb_nupdate[lev][boxIdx])));
732  ParallelFor(ubx, [=] AMREX_GPU_DEVICE(int i, int j, int k) noexcept {
733  pert_cell(i,j,k) = (use_mask && mask(i,j,k) == zero) ? zero
734  : erf_hash_rng::hash_symmetric(i, j, k, hash_comp_cell, lev, cell_seed) * amp_copy;
735  });
736  }
737  }
738  }

766  /**
767  * @brief Remove the mean of the box-perturbation field, so the net buoyant forcing is zero.
768  *
769  * Each perturbation box's mean is summed over every grid box it overlaps and over all
770  * ranks, so the correction does not depend on the domain decomposition.
771  * @param lev AMR level index.
772  */
773  void netZeroBuoyant (const int& lev)
774  {
775  const int nboxes = pb_ba[lev].size();
776  if (nboxes == 0) { return; }
777 
778  // Sum and cell count of each perturbation box, then one reduction over the ranks
779  amrex::Vector<amrex::Real> sums(2*nboxes, zero);
780  for (amrex::MFIter mfi(pb_cell[lev]); mfi.isValid(); ++mfi) {
781  const amrex::Box& vbx = mfi.validbox();
782  const amrex::Array4<const amrex::Real>& pert_cell = pb_cell[lev].const_array(mfi);
783  for (const auto& isect : pb_ba[lev].intersections(vbx)) {
784  sums[2*isect.first ] += box_sum(pert_cell, isect.second);
785  sums[2*isect.first+1] += auto_mode
786  ? box_sum(auto_mask[lev].const_array(mfi), isect.second)
787  : static_cast<amrex::Real>(isect.second.numPts());
788  }
789  }
790  amrex::ParallelAllReduce::Sum(sums.data(), static_cast<int>(sums.size()),
791  amrex::ParallelContext::CommunicatorSub());
792 
793  // Normalizing the adjustment based on how many boxes there are; each box's
794  // value is already normalized by the number of cells within it
795  tpi_net_buoyant = zero;
796  for (int boxIdx = 0; boxIdx < nboxes; boxIdx++) {
797  pb_netZero[lev][boxIdx] = (sums[2*boxIdx+1] > zero) ? sums[2*boxIdx] / sums[2*boxIdx+1] : zero;
798  tpi_net_buoyant += pb_netZero[lev][boxIdx];
799  }
800  tpi_pert_adjust = tpi_net_buoyant / static_cast<amrex::Real>(nboxes);
801  if (auto_mode) {
802  amrex::Real sum = zero, count = zero;
803  for (int b = 0; b < nboxes; ++b) { sum += sums[2*b]; count += sums[2*b+1]; }
804  tpi_pert_adjust = count > zero ? sum / count : zero;
805  }
806 
807  const amrex::Real adjust = tpi_pert_adjust;
808  for (amrex::MFIter mfi(pb_cell[lev]); mfi.isValid(); ++mfi) {
809  const amrex::Box& vbx = mfi.validbox();
810  const amrex::Array4<amrex::Real>& pert_cell = pb_cell[lev].array(mfi);
811  const bool use_mask = auto_mode;
812  const auto mask = use_mask ? auto_mask[lev].const_array(mfi)
813  : amrex::Array4<const amrex::Real>{};
814  for (const auto& isect : pb_ba[lev].intersections(vbx)) {
815  ParallelFor(isect.second, [=] AMREX_GPU_DEVICE(int i, int j, int k) noexcept {
816  if (!use_mask || mask(i,j,k) > zero) { pert_cell(i,j,k) -= adjust; }
817  });
818  }
819  }
820  }

822  /**
823  * @brief Compute the mean horizontal velocity magnitude and direction of the refreshing perturbation boxes.
824  *
825  * The velocities are summed over every grid box a perturbation box overlaps, each face once,
826  * and over all ranks, so every rank gets the same mean whatever the domain decomposition.
827  * @param lev AMR level index.
828  * @param update_box Nonzero for the perturbation boxes that refresh this step.
829  * @param mf_cons Conserved state used for iteration layout.
830  * @param mf_xvel X-velocity field.
831  * @param mf_yvel Y-velocity field.
832  */
833  void calc_tpi_meanMag (const int& lev,
834  const amrex::Vector<int>& update_box,
835  amrex::MultiFab& mf_cons,
836  amrex::MultiFab& mf_xvel,
837  amrex::MultiFab& mf_yvel)
838  {
839  const amrex::BoxArray& m_pb_ba = pb_ba[lev];
840  const int nboxes = m_pb_ba.size();
841 
842  // Per box: sum of u, number of u faces, sum of v, number of v faces
843  amrex::Vector<amrex::Real> sums(4*nboxes, zero);
844  for (amrex::MFIter mfi(mf_cons); mfi.isValid(); ++mfi) {
845  const amrex::Box& vbx = mfi.validbox(); // cell centered
846  const amrex::Array4<const amrex::Real>& xvel_arry = mf_xvel.const_array(mfi);
847  const amrex::Array4<const amrex::Real>& yvel_arry = mf_yvel.const_array(mfi);
848  for (const auto& isect : m_pb_ba.intersections(vbx)) {
849  const int boxIdx = isect.first;
850  if (!update_box[boxIdx]) { continue; }
851  const amrex::Box& pbx = m_pb_ba[boxIdx];
852 
853  // The faces of the overlap's cells; the high face only where the overlap ends
854  // at the perturbation box's high side, so a face shared by two grid boxes counts once
855  amrex::Box ubx_u = amrex::surroundingNodes(isect.second, 0);
856  if (isect.second.bigEnd(0) < pbx.bigEnd(0)) { ubx_u.growHi(0, -1); }
857  amrex::Box ubx_v = amrex::surroundingNodes(isect.second, 1);
858  if (isect.second.bigEnd(1) < pbx.bigEnd(1)) { ubx_v.growHi(1, -1); }
859 
860  sums[4*boxIdx ] += box_sum(xvel_arry, ubx_u);
861  sums[4*boxIdx+1] += static_cast<amrex::Real>(ubx_u.numPts());
862  sums[4*boxIdx+2] += box_sum(yvel_arry, ubx_v);
863  sums[4*boxIdx+3] += static_cast<amrex::Real>(ubx_v.numPts());
864  }
865  }
866  amrex::ParallelAllReduce::Sum(sums.data(), static_cast<int>(sums.size()),
867  amrex::ParallelContext::CommunicatorSub());
868 
869  for (int boxIdx = 0; boxIdx < nboxes; boxIdx++) {
870  if (!update_box[boxIdx]) { continue; }
871  const amrex::Real u_avg = (sums[4*boxIdx+1] > zero) ? sums[4*boxIdx ] / sums[4*boxIdx+1] : zero;
872  const amrex::Real v_avg = (sums[4*boxIdx+3] > zero) ? sums[4*boxIdx+2] / sums[4*boxIdx+3] : zero;
873  pb_mag[lev][boxIdx] = std::sqrt(u_avg*u_avg + v_avg*v_avg);
874  pb_dir[lev][boxIdx] = std::atan(std::abs(u_avg) / std::abs(v_avg+std::numeric_limits<amrex::Real>::epsilon()));
875  }
876  }

881  * @param bx Box to sum over.
882  * @return Sum of arr over bx.
