308  /**
309  * @brief Update perturbation amplitudes and intervals when each box is ready.
310  * @param lev AMR level index.
311  * @param dt Time step for elapsed-time accumulation.
312  * @param mf_xvel X-velocity field used to compute box mean velocities.
313  * @param mf_yvel Y-velocity field used to compute box mean velocities.
314  * @param mf_cons Conserved state used for iteration layout.
315  */
316  void calc_tpi_update (const int lev,
317  const double dt,
318  amrex::MultiFab& mf_xvel,
319  amrex::MultiFab& mf_yvel,
320  amrex::MultiFab& mf_cons)
321  {
322  auto m_ixtype = mf_cons.boxArray().ixType(); // safety step
323  if (pt_type[lev] == 3) { m_ixtype = amrex::IndexType(amrex::IntVect(0,0,1)); }
324 
325  const int nboxes = pb_ba[lev].size();
326 
327  // Which boxes refresh this step. pb_interval and pb_local_etime hold the same values on
328  // every rank, so every rank makes the same choice.
329  amrex::Vector<int> update_box(nboxes, 0);
330  for (int boxIdx = 0; boxIdx < nboxes; boxIdx++) {
331  if (pt_type[lev] == 2 || pt_type[lev] == 3) {
332  // for CPM, perturbation boxes/cells refresh after boxes have advected box width/length * num_layers (advective time scale)
333  update_box[boxIdx] = ((pb_local_etime[lev][boxIdx] >= pb_interval[lev][boxIdx]*tpi_layers) && pb_local_etime[lev][boxIdx] != -one);
334  } else {
335  update_box[boxIdx] = ( pb_local_etime[lev][boxIdx] >= pb_interval[lev][boxIdx] && pb_local_etime[lev][boxIdx] != -one );
336  }
337  }
338 
339  // Compute mean velocity of each refreshing perturbation box
340  calc_tpi_meanMag(lev, update_box, mf_cons, mf_xvel, mf_yvel);
341 
342  for (int boxIdx = 0; boxIdx < nboxes; boxIdx++) {
343  if ( update_box[boxIdx] ) {
344 
345  if (pb_mag[lev][boxIdx] !=zero) {
346  double interval = 0.0;
347  if (pt_type[lev] == 2 || pt_type[lev] == 3) {
348  // Wind direction correction for angled wind
349  amrex::Real wind_direction = pb_dir[lev][boxIdx];
350  if (wind_direction > PI / 4) { wind_direction = PI / 2 - wind_direction; }
351  // CPM only cares about the side length of a box, min call maintains flexibility.
352  interval = one / std::cos(wind_direction) * std::min(tpi_Lpb[lev], tpi_Wpb[lev]) / pb_mag[lev][boxIdx];
353  } else {
354  interval = tpi_lref[lev] / pb_mag[lev][boxIdx];
355  }
356  // 10% variation
357  const amrex::Real r = erf_hash_rng::hash_uniform(boxIdx, pb_nupdate[lev][boxIdx], 0,
358  hash_comp_interval, lev, tpi_seed);
359  pb_interval[lev][boxIdx] = (amrex::Real(0.9) + amrex::Real(0.2)*r) * interval;
360 
361  // Reset local elapsed time
362  pb_local_etime[lev][boxIdx] = zero;
363  } else {
364  // No mean wind through this box, so no advective time scale: it is not refreshed again
365  pb_local_etime[lev][boxIdx] = -one;
366  }
367 
368  // Trigger amplitude calculation per perturbation box
369  calc_tpi_amp(lev, boxIdx, pb_interval[lev][boxIdx], dt);
370 
371  // Trigger random amplitude storage per cell within perturbation box
372  pseudoRandomPert(boxIdx, lev, m_ixtype);
373 
374  pb_nupdate[lev][boxIdx]++;
375 
376  } else {
377  // set perturbation amplitudes to 0 for CPM. A little inefficient but leverages as much as existing code as possible.
378  if (pt_type[lev] == 2 || pt_type[lev] == 3) { zero_amp(boxIdx, lev, m_ixtype); }
379 
380  // Increase by timestep of level 0
381  if (pb_local_etime[lev][boxIdx] != -one) { pb_local_etime[lev][boxIdx] += dt; }
382  } // if
383  } // for
384 
385  if (pt_type[lev] < 2) { // box perturbation method only
386  netZeroBuoyant(lev);
387  }

429  /**
430  * @brief Compute the perturbation amplitude for one perturbation box.
431  * @param lev AMR level index.
432  * @param boxIdx Perturbation-box index on the level.
433  * @param interval Current perturbation update interval [s].
434  * @param dt Timestep [s].
435  */
436  void calc_tpi_amp (const int& lev,
437  const int& boxIdx,
438  const double& interval,
439  const double& dt)
440  {
441  pb_amp[lev][boxIdx] = zero; // Safety step
442  if (pt_type[lev] == 2) { // CPM
443  amrex::Real cp = 1004; // specific heat of air [J/(kg K)]
444  amrex::Real Ec = amrex::Real(0.2); // Eckert number
445  pb_amp[lev][boxIdx] = (input_Ug * input_Ug) / (Ec * cp);
446  } else if (pt_type[lev] == 3) {
447  pb_amp[lev][boxIdx] = input_w_amp;
448  } else { // box perturbation
449  amrex::Real Um = pb_mag[lev][boxIdx];
450 
451  // No mean wind: nothing to trip, and with tpi_Ti > 0 the amplitude below would be
452  // 0/0. The box is not refreshed again, so a zero amplitude also clears the
453  // perturbation it held
454  if (Um == zero) { return; }
455 
456  amrex::Real beta = one/tpi_Tinf; // Thermal expansion coefficient
457 
458  // Pseudo Random temperature the ignores scale when mechanically tripping turbulence
459  amrex::Real g = CONST_GRAV;
460  // get total refinement ratio on each level
461  int total_ref_ratio = 1;
462  for (int level = lev; level >= 1; level--) {
463  total_ref_ratio *= ref_ratio[level-1][2];
464  }
465  // calculation needs to be scale-aware since the formulation relies on the physical size of the box
466  if (tpi_Ti > zero) g = (tpi_nonDim * Um * Um) / (tpi_Ti * tpi_Hpb[lev]) * 1 / total_ref_ratio;
467 
468  // Ma and Senocak (2023) Eq. 8, solving for delta phi
469  pb_amp[lev][boxIdx] = (tpi_nonDim * Um * Um) / (g * beta * tpi_Hpb[lev]) * 1 / total_ref_ratio;
470 
471  if (pt_type[lev] == 0) {
472  // Source mode: divide by interval to get forcing rate (K/s)
473  // Ma & Senocak (2023) Eq. 7
474  pb_amp[lev][boxIdx] /= static_cast<amrex::Real>(interval);
475  } else if (pt_type[lev] == 1) {
476  // Direct mode: divide by number of timesteps to get per-step increment.
477  // This gives the same total perturbation as source mode and should resolve
478  // instability issues previously associated with direct mode.
479  amrex::Real num_steps = interval / dt;
480  pb_amp[lev][boxIdx] /= num_steps;
481  }
482  }
483  }

486  * @brief Assign pseudo-random perturbations to cells in one perturbation box.
487  * @param boxIdx Perturbation-box index on the level.
488  * @param lev AMR level index.
489  * @param m_ixtype Index type used to align perturbation storage.
490  */
491  void pseudoRandomPert (const int& boxIdx,
492  const int& lev,
493  const amrex::IndexType& m_ixtype)
494  {
495  for (amrex::MFIter mfi(pb_cell[lev],TileNoZ()); mfi.isValid(); ++mfi) {
496  amrex::Box vbx = mfi.tilebox();

498  if (pt_type[lev] == 3) { pbx.setBig(2, pbx.bigEnd(2) - 1); } // prevent double counting after converting to k-faces
499  amrex::Box ubx = pbx & vbx;
500  if (ubx.ok()) {
501  amrex::Real amp_copy = pb_amp[lev][boxIdx];
502  const amrex::Array4<amrex::Real>& pert_cell = pb_cell[lev].array(mfi);
503 
504  if (pt_type[lev] == 2 || pt_type[lev] == 3) { // CPM: one value per perturbation box
505  const amrex::Real rand_number_const = erf_hash_rng::hash_symmetric(
506  boxIdx, pb_nupdate[lev][boxIdx], 0, hash_comp_cpm, lev, tpi_seed);
507  ParallelFor(ubx, [=] AMREX_GPU_DEVICE(int i, int j, int k) noexcept {
508  pert_cell(i,j,k) = rand_number_const * amp_copy;
509  });
510  } else { // one value per cell, keyed by its global index and the box's refresh count
511  const std::uint64_t cell_seed = erf_hash_rng::splitmix64(
512  tpi_seed ^ erf_hash_rng::splitmix64(static_cast<std::uint64_t>(pb_nupdate[lev][boxIdx])));
513  ParallelFor(ubx, [=] AMREX_GPU_DEVICE(int i, int j, int k) noexcept {
514  pert_cell(i,j,k) = erf_hash_rng::hash_symmetric(i, j, k, hash_comp_cell, lev, cell_seed) * amp_copy;
515  });
516  }
517  }
518  }

546  /**
547  * @brief Remove the mean of the box-perturbation field, so the net buoyant forcing is zero.
548  *
549  * Each perturbation box's mean is summed over every grid box it overlaps and over all
550  * ranks, so the correction does not depend on the domain decomposition.
551  * @param lev AMR level index.
552  */
553  void netZeroBuoyant (const int& lev)
554  {
555  const int nboxes = pb_ba[lev].size();
556 
557  // Sum and cell count of each perturbation box, then one reduction over the ranks
558  amrex::Vector<amrex::Real> sums(2*nboxes, zero);
559  for (amrex::MFIter mfi(pb_cell[lev]); mfi.isValid(); ++mfi) {
560  const amrex::Box& vbx = mfi.validbox();
561  const amrex::Array4<const amrex::Real>& pert_cell = pb_cell[lev].const_array(mfi);
562  for (const auto& isect : pb_ba[lev].intersections(vbx)) {
563  sums[2*isect.first ] += box_sum(pert_cell, isect.second);
564  sums[2*isect.first+1] += static_cast<amrex::Real>(isect.second.numPts());
565  }
566  }
567  amrex::ParallelAllReduce::Sum(sums.data(), static_cast<int>(sums.size()),
568  amrex::ParallelContext::CommunicatorSub());
569 
570  // Normalizing the adjustment based on how many boxes there are; each box's
571  // value is already normalized by the number of cells within it
572  tpi_net_buoyant = zero;
573  for (int boxIdx = 0; boxIdx < nboxes; boxIdx++) {
574  pb_netZero[lev][boxIdx] = (sums[2*boxIdx+1] > zero) ? sums[2*boxIdx] / sums[2*boxIdx+1] : zero;
575  tpi_net_buoyant += pb_netZero[lev][boxIdx];
576  }
577  tpi_pert_adjust = tpi_net_buoyant / static_cast<amrex::Real>(nboxes);
578 
579  const amrex::Real adjust = tpi_pert_adjust;
580  for (amrex::MFIter mfi(pb_cell[lev]); mfi.isValid(); ++mfi) {
581  const amrex::Box& vbx = mfi.validbox();
582  const amrex::Array4<amrex::Real>& pert_cell = pb_cell[lev].array(mfi);
583  for (const auto& isect : pb_ba[lev].intersections(vbx)) {
584  ParallelFor(isect.second, [=] AMREX_GPU_DEVICE(int i, int j, int k) noexcept {
585  pert_cell(i,j,k) -= adjust;
586  });
587  }
588  }
589  }

591  /**
592  * @brief Compute the mean horizontal velocity magnitude and direction of the refreshing perturbation boxes.
593  *
594  * The velocities are summed over every grid box a perturbation box overlaps, each face once,
595  * and over all ranks, so every rank gets the same mean whatever the domain decomposition.
596  * @param lev AMR level index.
597  * @param update_box Nonzero for the perturbation boxes that refresh this step.
598  * @param mf_cons Conserved state used for iteration layout.
599  * @param mf_xvel X-velocity field.
600  * @param mf_yvel Y-velocity field.
601  */
602  void calc_tpi_meanMag (const int& lev,
603  const amrex::Vector<int>& update_box,
604  amrex::MultiFab& mf_cons,
605  amrex::MultiFab& mf_xvel,
606  amrex::MultiFab& mf_yvel)
607  {
608  const amrex::BoxArray& m_pb_ba = pb_ba[lev];
609  const int nboxes = m_pb_ba.size();
610 
611  // Per box: sum of u, number of u faces, sum of v, number of v faces
612  amrex::Vector<amrex::Real> sums(4*nboxes, zero);
613  for (amrex::MFIter mfi(mf_cons); mfi.isValid(); ++mfi) {
614  const amrex::Box& vbx = mfi.validbox(); // cell centered
615  const amrex::Array4<const amrex::Real>& xvel_arry = mf_xvel.const_array(mfi);
616  const amrex::Array4<const amrex::Real>& yvel_arry = mf_yvel.const_array(mfi);
617  for (const auto& isect : m_pb_ba.intersections(vbx)) {
618  const int boxIdx = isect.first;
619  if (!update_box[boxIdx]) { continue; }
620  const amrex::Box& pbx = m_pb_ba[boxIdx];
621 
622  // The faces of the overlap's cells; the high face only where the overlap ends
623  // at the perturbation box's high side, so a face shared by two grid boxes counts once
624  amrex::Box ubx_u = amrex::surroundingNodes(isect.second, 0);
625  if (isect.second.bigEnd(0) < pbx.bigEnd(0)) { ubx_u.growHi(0, -1); }
626  amrex::Box ubx_v = amrex::surroundingNodes(isect.second, 1);
627  if (isect.second.bigEnd(1) < pbx.bigEnd(1)) { ubx_v.growHi(1, -1); }
628 
629  sums[4*boxIdx ] += box_sum(xvel_arry, ubx_u);
630  sums[4*boxIdx+1] += static_cast<amrex::Real>(ubx_u.numPts());
631  sums[4*boxIdx+2] += box_sum(yvel_arry, ubx_v);
632  sums[4*boxIdx+3] += static_cast<amrex::Real>(ubx_v.numPts());
633  }
634  }
635  amrex::ParallelAllReduce::Sum(sums.data(), static_cast<int>(sums.size()),
636  amrex::ParallelContext::CommunicatorSub());
637 
638  for (int boxIdx = 0; boxIdx < nboxes; boxIdx++) {
639  if (!update_box[boxIdx]) { continue; }
640  const amrex::Real u_avg = (sums[4*boxIdx+1] > zero) ? sums[4*boxIdx ] / sums[4*boxIdx+1] : zero;
641  const amrex::Real v_avg = (sums[4*boxIdx+3] > zero) ? sums[4*boxIdx+2] / sums[4*boxIdx+3] : zero;
642  pb_mag[lev][boxIdx] = std::sqrt(u_avg*u_avg + v_avg*v_avg);
643  pb_dir[lev][boxIdx] = std::atan(std::abs(u_avg) / std::abs(v_avg+std::numeric_limits<amrex::Real>::epsilon()));
644  }
645  }

647  /**
648  * @brief Sum an array over a box.
649  * @param arr Array to sum.
650  * @param bx Box to sum over.
651  * @return Sum of arr over bx.
652  */
653  static amrex::Real box_sum (const amrex::Array4<const amrex::Real>& arr, const amrex::Box& bx)
654  {

659  {
660  return {arr(i,j,k)};
661  });
662  return amrex::get<0>(reduce_data.value(reduce_op));
663  }
