23 /**
24  * @struct FloodGrid
25  * @brief 2D horizontal flood grid
26  *
27  * A slab one cell thick with C*nx x C*ny cells, where C is erf.flood.grid_ratio
28  * and (nx, ny) is the size of the region the atmospheric level's boxes cover:
29  * the whole domain on level 0, the refined region on a finer level.
30  */
31 struct FloodGrid
32 {
33  amrex::BoxArray ba; ///< 2D BoxArray (C*nx x C*ny x 1), box n refines atmospheric box n
34  amrex::DistributionMapping dm; ///< The atmospheric level's owners
35  amrex::Geometry geom; ///< 2D Geometry in physical coordinates [m]
36  int C = 1; ///< erf.flood.grid_ratio
37  /// Index, on the atmospheric level the grid refines, of the lower corner of
38  /// the region the flood grid covers: flood cell (i, j) lies in atmospheric
39  /// cell (i / C + atm_lo[0], j / C + atm_lo[1]). Zero on level 0.
40  amrex::IntVect atm_lo {0, 0, 0};

44 /**
45  * @brief Construct a refined 2D flood grid over one atmospheric level's boxes
46  *
47  * @param[in] ba_atm BoxArray of the atmospheric level
48  * @param[in] dm_atm DistributionMapping of the atmospheric level
49  * @param[in] geom_atm Geometry of the atmospheric level
50  * @param[in] C Refinement factor (erf.flood.grid_ratio, >= 1)
51  *
52  * @return FloodGrid over the region the boxes cover; its lev is left at 0 for
53  * the caller to set.
54  *
55  * 1. Take the k = 0 slab of each box in ba_atm, keeping the box order.
56  * 2. The region is their bounding rectangle, which must be covered exactly

59  * of atmospheric box n, so dm_atm is reused.
60  * 4. The Geometry spans the region: index space 0 to C * size - 1, the
61  * physical extent of the region (the domain's own bounds where the region
62  * spans the domain), periodic only in a direction the region spans.
63  */

73 inline bool flood_same_columns (const amrex::BoxArray& a, const amrex::BoxArray& b)
74 {

82  }

87  * True when box n of the flood grid refines the columns of box n of the
88  * atmospheric level (shifted by -atm_lo, refined by {C, C, 1}) for every n,
89  * and the two have the same owners.

92  const FloodGrid& fg)
93 {

95  const amrex::BoxArray& aba = atm_mf.boxArray();
96  if (fba.size() != aba.size() || flood_mf.DistributionMap() != atm_mf.DistributionMap()) {
97  return false;
98  }
99  for (int n = 0; n < aba.size(); ++n) {
100  amrex::Box ab = aba[n];
101  amrex::Box fb = fba[n];
102  ab.enclosedCells();
103  fb.enclosedCells();
104  ab.setSmall(2, 0); ab.setBig(2, 0);
105  fb.setSmall(2, 0); fb.setBig(2, 0);
106  ab.shift(-fg.atm_lo);
107  ab.refine(amrex::IntVect(fg.C, fg.C, 1));
108  if (ab != fb) { return false; }
109  }
110  return true;
