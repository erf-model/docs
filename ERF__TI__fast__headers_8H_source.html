15 /**
16  * The coarse/fine face fluxes of the last RK stage, in the units the flux registers hold:
17  * (rho u)/m_uy, (rho v)/m_vx and the time-weighted (rho w)/(m_x m_y) without terrain;
18  * ax (rho u)/m_uy, ay (rho v)/m_vx and the time-weighted Omega/(m_x m_y) with it.
19  *
20  * A coarse level records in rec[dir] the flux each substep applied through every face, and in
21  * rec_w the vertical quantity its z-faces hold at the end of the substep -- (rho w) without
22  * terrain, Omega with it -- and in rec_w_old the same at the start of the substep. A fine level
23  * takes the fluxes in tgt[dir] (component 0) on the faces where component 1 is one -- its
24  * coarse/fine faces -- in place of its own, and tgt_w as that vertical quantity there at the
25  * end of the substep. The fine level's fluxes through its boundary then add up to the coarse
26  * ones.
27  */

29  std::array<amrex::MultiFab*, AMREX_SPACEDIM> rec {{AMREX_D_DECL(nullptr, nullptr, nullptr)}};
30  amrex::MultiFab* rec_w = nullptr;
31  amrex::MultiFab* rec_w_old = nullptr;
32  std::array<const amrex::MultiFab*, AMREX_SPACEDIM> tgt {{AMREX_D_DECL(nullptr, nullptr, nullptr)}};
33  const amrex::MultiFab* tgt_w = nullptr;
34 };
