97  const bool has_total = surface_precip_has_source(sources.total);
98  const bool has_rain = surface_precip_has_source(sources.rain);
99  const bool has_snow = surface_precip_has_source(sources.snow);
100  const bool has_graupel = surface_precip_has_source(sources.graupel);
101  const bool has_hail = surface_precip_has_source(sources.hail);

103  const Real f_rain = sources.rain.native_to_kg_m2;
104  const Real f_snow = sources.snow.native_to_kg_m2;
105  const Real f_graupel = sources.graupel.native_to_kg_m2;
106  const Real f_hail = sources.hail.native_to_kg_m2;

108  // Whether box n of a source holds the plane k = klo.
109  auto holds_klo = [klo] (const SurfacePrecipAccumulationSource& src, int n) {
110  if (!surface_precip_has_source(src)) { return true; }
111  const Box& b = src.accum->boxArray()[n];
112  return b.smallEnd(2) <= klo && klo <= b.bigEnd(2);
113  };
114 
115  for (MFIter mfi(accum_2d, false); mfi.isValid(); ++mfi) {
116  const int n = mfi.index();
117  if (!(holds_klo(sources.total, n) && holds_klo(sources.rain, n) &&
118  holds_klo(sources.snow, n) && holds_klo(sources.graupel, n) &&
119  holds_klo(sources.hail, n))) { continue; }
120  Array4<Real> a = accum_2d.array(mfi);
121  Array4<const Real> total = has_total ? sources.total.accum->const_array(mfi) : Array4<const Real>{};
122  Array4<const Real> rain = has_rain ? sources.rain.accum->const_array(mfi) : Array4<const Real>{};
123  Array4<const Real> snow = has_snow ? sources.snow.accum->const_array(mfi) : Array4<const Real>{};
124  Array4<const Real> graupel = has_graupel ? sources.graupel.accum->const_array(mfi) : Array4<const Real>{};
125  Array4<const Real> hail = has_hail ? sources.hail.accum->const_array(mfi) : Array4<const Real>{};
126  ParallelFor(mfi.tilebox(), [=] AMREX_GPU_DEVICE (const IntVect& iv) noexcept
127  {
128  const int i = iv[0], j = iv[1];
129  Real frozen = Real(0.0);
130  if (phase != SurfacePrecipPhase::Total || frozen_in_comp1) {
131  if (has_snow) { frozen += snow(i, j, klo) * f_snow; }
132  if (has_graupel) { frozen += graupel(i, j, klo) * f_graupel; }
133  if (has_hail) { frozen += hail(i, j, klo) * f_hail; }
134  }
135  Real v;
136  if (phase == SurfacePrecipPhase::Frozen) {
137  v = frozen;
138  } else if (phase == SurfacePrecipPhase::Liquid) {
139  // No bound here: this is an accumulation, and a bound on it would change the
140  // difference of two of its values. The user bounds that difference.
141  if (has_total) {
142  v = total(i, j, klo) * f_total - frozen;
143  } else {
144  v = has_rain ? rain(i, j, klo) * f_rain : Real(0.0);
145  }
146  } else if (has_total) {
147  v = total(i, j, klo) * f_total;
148  } else {
149  v = Real(0.0);
150  if (has_rain) { v += rain(i, j, klo) * f_rain; }
151  if (has_snow) { v += snow(i, j, klo) * f_snow; }
152  if (has_graupel) { v += graupel(i, j, klo) * f_graupel; }
153  if (has_hail) { v += hail(i, j, klo) * f_hail; }
154  }
155  a(i, j, 0, 0) = v;
156  if (frozen_in_comp1) { a(i, j, 0, 1) = frozen; }
157  });
158  }
159 }

161 /**
162  * @brief Total liquid-water-equivalent surface precipitation accumulation [kg/m2]
163  * of every column, from the scheme-native accumulators.
164  *
165  * Fills component 0 of accum_2d, a MultiFab on the level's boxes coarsened to one
166  * cell in z (box n of accum_2d is box n of the accumulators, with the same
167  * owner), from the k = klo plane of the sources. Schemes that provide a total
168  * give it directly; schemes that provide the species give their sum, so
169  * nothing is counted twice (the same rule as the 2D plotfile precipitation
170  * diagnostics). Without any source the field is zero. A box whose 3D box does
171  * not hold the k = klo plane (a fine level that does not span z, or the upper
172  * box of a column split in z) has no ground to read and stays zero, so no
173  * accumulator is read outside its box. Used by the fire layer's
174  * fuel moisture (ERF_FirePrecip.H) and the force-restore precipitation rate
175  * (ERF_ForceRestorePrecipRate.H).
176  *
177  * With phase = Frozen the field is the frozen part, snow + graupel + hail
178  * (zero for a scheme without those accumulators). With phase = Liquid it is
179  * the rest: the total minus the frozen part, or without a total the rain
180  * accumulator. Frozen plus Liquid is then the Total field for every scheme
181  * with a total, including one whose total holds more than its species
182  * (Morrison's also counts the cloud water and cloud ice that fell out). The
183  * Liquid field is not bounded at zero: where the frozen accumulators exceed
184  * the total by round-off it is slightly negative, and a caller that takes the
185  * change over a step bounds that change (force_restore_rain_rate).
186  */
187 inline void surface_precip_accum_k0 (amrex::MultiFab& accum_2d,
188  const SurfacePrecipAccumulationSources& sources,
189  int klo,
190  SurfacePrecipPhase phase = SurfacePrecipPhase::Total)
191 {
192  surface_precip_fill_k0(accum_2d, sources, klo, phase, false);
193 }
