43  * Allocation-free canonical forward transform used by host and device code.
44  * Returns false instead of throwing when the state is invalid or materially
45  * unrealizable. The throwing host API below delegates to this function.
46  */
47 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
48 bool try_two_moment_to_endpoints (const amrex::Real C, const amrex::Real M,
49  const amrex::Real lower, const amrex::Real upper,
50  EndpointTransform& result) noexcept
51 {
52  if (!amrex::Math::isfinite(C) || !amrex::Math::isfinite(M) || !amrex::Math::isfinite(lower) ||
53  !amrex::Math::isfinite(upper) || lower < amrex::Real(0.0) ||
54  !(lower < upper) || C < amrex::Real(0.0)) {
55  return false;
56  }
57  const amrex::Real scale = std::abs(M) + std::abs(lower * C) + std::abs(upper * C);
58  const amrex::Real moment_tolerance = amrex::Real(128.0) *
59  std::numeric_limits<amrex::Real>::epsilon() * scale;
60  const amrex::Real low_numerator = std::fma(upper, C, -M);
61  const amrex::Real high_numerator = std::fma(-lower, C, M);
62  if (!amrex::Math::isfinite(scale) || !amrex::Math::isfinite(moment_tolerance) ||
63  !amrex::Math::isfinite(low_numerator) || !amrex::Math::isfinite(high_numerator) ||
64  low_numerator < -moment_tolerance || high_numerator < -moment_tolerance) {
65  return false;
66  }
67  if (C == amrex::Real(0.0)) {
68  if (std::abs(M) > moment_tolerance) return false;
69  result = {amrex::Real(0.0), amrex::Real(0.0), moment_tolerance,
70  moment_tolerance / (upper - lower), M != amrex::Real(0.0)};
71  return amrex::Math::isfinite(result.endpoint_tolerance);
72  }
73  const amrex::Real denominator = upper - lower;
74  amrex::Real L = low_numerator / denominator;
75  amrex::Real H = high_numerator / denominator;
76  const amrex::Real endpoint_tolerance = moment_tolerance / denominator;
77  if (!amrex::Math::isfinite(L) || !amrex::Math::isfinite(H) || !amrex::Math::isfinite(endpoint_tolerance)) {
78  return false;
79  }
80  const bool normalized = L < amrex::Real(0.0) || H < amrex::Real(0.0);
81  if (L < amrex::Real(0.0)) L = amrex::Real(0.0);
82  if (H < amrex::Real(0.0)) H = amrex::Real(0.0);
83  result = {L, H, moment_tolerance, endpoint_tolerance, normalized};
84  return true;
85 }

88 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
89 bool try_endpoints_to_two_moment (const amrex::Real L, const amrex::Real H,
90  const amrex::Real lower, const amrex::Real upper,
91  amrex::Real& C, amrex::Real& M) noexcept
92 {
93  if (!amrex::Math::isfinite(L) || !amrex::Math::isfinite(H) || !amrex::Math::isfinite(lower) ||
94  !amrex::Math::isfinite(upper) || L < amrex::Real(0.0) || H < amrex::Real(0.0) ||
95  lower < amrex::Real(0.0) || !(lower < upper)) {
96  return false;
97  }
98  const amrex::Real count = L + H;
99  const amrex::Real mass = std::fma(lower, L, upper * H);
100  if (!amrex::Math::isfinite(count) || !amrex::Math::isfinite(mass) ||
101  count < amrex::Real(0.0) || mass < amrex::Real(0.0)) {
102  return false;
103  }
104  C = count;
105  M = mass;
106  return true;
107 }
