90 /**
91  * @brief Leaf area index of @p params on day of year @p day, interpolated between the
92  * monthly values as Noah-MP does for its table-LAI options (PhenologyMainMod): month
93  * index 12 * day / days_in_year, linear between the two nearest months, wrapping December
94  * to January. In the southern hemisphere Noah-MP shifts the day by half a year first;
95  * pass @p southern = true for that.
96  */
97 [[nodiscard]] inline amrex::Real
98 noahmp_leaf_area_index (const NoahMPVegetationParams& params, amrex::Real day,
99  amrex::Real days_in_year, bool southern) noexcept
100 {
101  amrex::Real d = day;
102  if (southern) {
103  d = d + amrex::Real(0.5) * days_in_year;
104  while (d >= days_in_year) { d -= days_in_year; }
105  }
106  const amrex::Real month = amrex::Real(12.0) * d / days_in_year;
107  // Noah-MP: IntpMonth1 = MonthCurrent + 0.5 (an integer, so truncated),
108  // IntpMonth2 = IntpMonth1 + 1, weight of month 1 = (IntpMonth1 + 0.5) - MonthCurrent.
109  int m1 = static_cast<int>(month + amrex::Real(0.5));
110  int m2 = m1 + 1;
111  const amrex::Real w1 = (static_cast<amrex::Real>(m1) + amrex::Real(0.5)) - month;
112  const amrex::Real w2 = amrex::Real(1.0) - w1;
113  if (m1 < 1) { m1 = 12; }
114  if (m2 > 12) { m2 = 1; }
115  return w1 * params.lai[m1 - 1] + w2 * params.lai[m2 - 1];
116 }

118 /**
119  * @brief The day of the year, as Noah-MP counts it, of a date "YYYY-MM-DD[ HH:MM[:SS]]"
120  * (ERF's start_datetime).
121  *
122  * The day is Noah-MP's DayJulianInYear, which PhenologyMainMod takes as
123  * 0 <= day < days_in_year. It is 0 at 00:00 on 1 January, and the time of day is added
124  * as a fraction. @p days_in_year is 365 or 366.
125  *
126  * @return false when the date cannot be read, the month is outside 1-12, the day is
127  * outside the month, or the time is outside 00:00:00-23:59:60.
128  */
129 [[nodiscard]] inline bool
130 noahmp_day_of_year (const std::string& datetime, amrex::Real& day,
131  amrex::Real& days_in_year)
132 {
133  int year = 0, month = 0, mday = 0, hour = 0, minute = 0;
134  double second = 0.0;
135  const int n = std::sscanf(datetime.c_str(), "%d-%d-%d %d:%d:%lf",
136  &year, &month, &mday, &hour, &minute, &second);
137  if (n < 3 || month < 1 || month > 12) { return false; }
138  if (n < 4) { hour = 0; }
139  if (n < 5) { minute = 0; }
140  if (n < 6) { second = 0.0; }
141  const bool leap = (year % 4 == 0 && year % 100 != 0) || year % 400 == 0;
142  const int month_days[12] = {31, leap ? 29 : 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
143  if (mday < 1 || mday > month_days[month - 1] || hour < 0 || hour > 23 ||
144  minute < 0 || minute > 59 || second < 0.0 || second > 60.0) {
145  return false;
146  }
147  int whole = mday - 1;
148  for (int m = 0; m < month - 1; ++m) { whole += month_days[m]; }
149  day = static_cast<amrex::Real>(whole) +
150  static_cast<amrex::Real>((hour * 3600.0 + minute * 60.0 + second) / 86400.0);
151  days_in_year = leap ? amrex::Real(366) : amrex::Real(365);
152  return true;
153 }
