28  * One set of running statistics: the mean, the root mean square, the minimum and the maximum of
29  * each named scalar quantity over every sample added since the caller's averaging start.
30  * write() rewrites <output_root>_stats.csv with the current
31  * values; the sums are part of the checkpoint, so the statistics continue across a restart.
32  * A non-finite sample aborts, naming the statistics and the quantity.
33  */

60  void accumulate (double t, const std::vector<amrex::Real>& values)
61  {
62  AMREX_ALWAYS_ASSERT_WITH_MESSAGE(values.size() == m_names.size(), "RunningStats::accumulate: one value per quantity");
63  for (std::size_t i = 0; i < values.size(); ++i) {
64  if (!std::isfinite(values[i])) {
65  std::ostringstream msg;
66  msg << "RunningStats " << m_name << ": quantity " << m_names[i] << " is not finite at t = " << t << " s";
67  amrex::Abort(msg.str());
68  }
69  }
70  if (m_count == 0) { m_t_first = t; }
71  m_t_last = t;
72  for (std::size_t i = 0; i < values.size(); ++i) {
73  const double v = static_cast<double>(values[i]);
74  m_sum[i] += v;
75  m_sum2[i] += v * v;
76  m_min[i] = std::min(m_min[i], values[i]);
77  m_max[i] = std::max(m_max[i], values[i]);
78  }
79  ++m_count;
80  }
81  amrex::Real mean (std::size_t i) const { return (m_count > 0) ? static_cast<amrex::Real>(m_sum[i] / m_count) : amrex::Real(0.0); }
82  amrex::Real rms (std::size_t i) const
83  {
84  return (m_count > 0) ? static_cast<amrex::Real>(std::sqrt(std::max(m_sum2[i] / m_count, 0.0))) : amrex::Real(0.0);

114  /**
115  * Restore the state written by write_state(); false when there is none. Aborts, changing

117  * value, or quantities other than this object's (but those accepted by renamed_from).
118  */

120  {

124  return false;
125  }
126  amrex::Vector<char> chars;

128  std::istringstream in(std::string(chars.dataPtr(), chars.size()));

130  std::size_t size = 0;
131  int count = 0;
132  double t0 = 0.0, t1 = 0.0;
133  if (!(in >> key >> eq >> count) || key != "count" || !(in >> key >> eq >> size) || key != "size" ||
134  !(in >> key >> eq >> t0) || key != "t_first" || !(in >> key >> eq >> t1) || key != "t_last") {
135  amrex::Abort("malformed statistics checkpoint '" + fname + "'");
136  }
137  if (size != m_names.size()) {

140  }
141  if (count < 0 || !std::isfinite(t0) || !std::isfinite(t1) || (count > 0 && t1 < t0)) {
142  amrex::Abort("malformed statistics checkpoint '" + fname + "': the count must be >= 0 and t_first <= t_last, both finite");
143  }
144  std::vector<double> sum(size), sum2(size);
145  std::vector<amrex::Real> mn(size), mx(size);
146  std::string afresh;

148  if (!(in >> qname >> sum[i] >> sum2[i] >> mn[i] >> mx[i])) {

150  }

153  if (r == m_renamed.end() || r->second.first != m_names[i]) {

161  }
162  }
163  if (!std::isfinite(sum[i]) || !std::isfinite(sum2[i]) || !std::isfinite(mn[i]) || !std::isfinite(mx[i])) {
164  amrex::Abort("malformed statistics checkpoint '" + fname + "': a value of " + m_names[i] + " is not finite");
165  }
