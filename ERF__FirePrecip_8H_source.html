10 /**
11  * @file ERF_FirePrecip.H
12  * @brief Rain per column for the fuel moisture model from the microphysics
13  * surface precipitation accumulators, for an engine that reads the rain
14  * (FireEngine::wants_surface_precip).
15  *
16  * Every Eulerian microphysics scheme with precipitation keeps a cumulative
17  * surface accumulator per atmospheric column (Kessler's rain_accum, SAM's and
18  * Morrison's rain, snow and graupel accumulations, ...), exposed through
19  * Microphysics::Get_Surface_Precip_Accumulation_Ptrs() with a factor that
20  * turns the scheme's native unit into kg/m2 of liquid water. One kg/m2 of
21  * liquid water is one millimetre of depth, so the change of the accumulation
22  * over a step, divided by the step, is the rain rate in mm/hr that the
23  * time-lag and stick moisture updates take. The fire layer keeps the
24  * accumulation it saw at the end of its previous step (checkpointed) and
25  * forms that difference every step; a fresh start begins at zero, which is
26  * where every scheme starts its accumulators.
27  *
28  * Snow, graupel and hail count with their liquid-water equivalent: the fuel
29  * models have no frozen-precipitation branch. There is no canopy interception.
30  */
31 
32 /**
33  * @brief Rain rate [mm/hr] from the change of the surface accumulation over one step.
34  *
35  * @param[in] accum_now_kgm2 accumulation at the end of the step [kg/m2 == mm]
36  * @param[in] accum_prev_kgm2 accumulation at the end of the previous step [kg/m2]
37  * @param[in] dt_s the step [s]
38  * @return (now - prev) * 3600 / dt; zero for a non-positive step or a non-positive
39  * change (an accumulator that was reset, or round-off), and zero for NaN.
40  */
41 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
42 amrex::Real fire_precip_rate_mm_hr (amrex::Real accum_now_kgm2,
43  amrex::Real accum_prev_kgm2,
44  amrex::Real dt_s) noexcept
45 {
46  if (!(dt_s > amrex::Real(0.0))) { return amrex::Real(0.0); }
47  const amrex::Real d = accum_now_kgm2 - accum_prev_kgm2;
48  if (!(d > amrex::Real(0.0))) { return amrex::Real(0.0); }
49  return d * amrex::Real(3600.0) / dt_s;
50 }

52 /**
53  * @brief Total liquid-water-equivalent surface precipitation accumulation [kg/m2]
54  * of every column, from the scheme-native accumulators.
55  *
56  * Fills accum_2d, a one-component MultiFab on the level's boxes coarsened to one

60  * nothing is counted twice (the same rule as the 2D plotfile precipitation
