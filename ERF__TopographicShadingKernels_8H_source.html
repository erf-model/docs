38 // Scan grid-line crossings toward the sun, interpolating the resolved terrain
39 // between mass points (WRF's toposhad construction). dx/dy are physical lengths
40 // at the target, including its map factors; the ray uses that local tangent plane.
41 // The search cutoff is horizontal distance, not a count of grid rows.
42 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
43 bool shadowed (int i, int j, const amrex::Array4<const Real>& terrain,
44  int ilo, int ihi, int jlo, int jhi, bool periodic_x, bool periodic_y,
45  Real dx, Real dy, Sun s, Real search_length)
46 {
47  const Real horizontal = std::sqrt(s.x*s.x + s.y*s.y);
48  constexpr Real real_eps = std::numeric_limits<Real>::epsilon();
49  // A horizontal component below sqrt(eps) cannot be distinguished reliably
50  // from a vertical ray when the unit sun vector is rounded to Real.
51  if (s.z <= Real(1.e-4) || horizontal <= std::sqrt(real_eps)) { return false; }
52  // Trigonometric roundoff at cardinal azimuths must not send a ray out
53  // through a domain edge to which it is parallel (e.g. cos(90 degrees)).
54  constexpr Real angular_eps = Real(8)*real_eps;
55  const Real di = std::abs(s.x) < angular_eps*horizontal ? Real(0) : s.x / (horizontal * dx);
56  const Real dj = std::abs(s.y) < angular_eps*horizontal ? Real(0) : s.y / (horizontal * dy);
57  const Real ds = Real(1) / amrex::max(std::abs(di), std::abs(dj));
58  const Real z = terrain(i,j,0,0);
59  for (Real r = ds; r <= search_length; r += ds) {
60  const Real x = Real(i) + r*di, y = Real(j) + r*dj;
61  if ((!periodic_x && (x < ilo || x > ihi)) ||
62  (!periodic_y && (y < jlo || y > jhi))) { break; }
63  int ix = int(std::floor(x)), jy = int(std::floor(y));
64  const Real wx = x-Real(ix), wy = y-Real(jy);
65  int ip = ix+1, jp = jy+1;
66  if (periodic_x) { ix = wrap(ix,ilo,ihi-ilo+1); ip = wrap(ip,ilo,ihi-ilo+1); }
67  else { ip = amrex::min(ip,ihi); }
68  if (periodic_y) { jy = wrap(jy,jlo,jhi-jlo+1); jp = wrap(jp,jlo,jhi-jlo+1); }
69  else { jp = amrex::min(jp,jhi); }
70  AMREX_ASSERT(terrain.contains(ix,jy,0) && terrain.contains(ip,jp,0));
71  const Real h = (Real(1)-wy)*((Real(1)-wx)*terrain(ix,jy,0,0) + wx*terrain(ip,jy,0,0))
72  + wy*((Real(1)-wx)*terrain(ix,jp,0,0) + wx*terrain(ip,jp,0,0));
73  // Interpolation and ray-height arithmetic can differ by several ulps
74  // at the local terrain height, especially with single precision.
75  const Real ray_height = z + r*s.z/horizontal;
76  // All heights are metres; use a 1 m scale near sea level.
77  const Real height_scale = amrex::max(Real(1),
78  amrex::max(std::abs(z), amrex::max(std::abs(h), std::abs(ray_height))));
79  const Real height_tol = Real(8)*real_eps*height_scale;
80  if (h > ray_height + height_tol) { return true; }
81  }
82  return false;
83 }
