80 /** Full O(nbins + properties^2) validator for untrusted/arbitrary views. */
81 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
82 bool valid_population_remap_view (const PopulationRemapView& population) noexcept
83 {
84  if (!valid_population_remap_context(population)) return false;
85  for (int edge = 0; edge <= population.nbins; ++edge) {
86  const amrex::Real value = population.edges[edge];
87  if (!amrex::Math::isfinite(value) || value < amrex::Real(0.0) ||
88  (edge > 0 && !(population.edges[edge - 1] < value))) {
89  return false;
90  }
91  }
92  if (population.moment_mode == MomentMode::OneMoment) {
93  for (int bin = 0; bin < population.nbins; ++bin) {
94  const amrex::Real pivot = population.pivots[bin];
95  if (!amrex::Math::isfinite(pivot) || pivot <= amrex::Real(0.0) ||
96  pivot < population.edges[bin] || pivot > population.edges[bin + 1] ||
97  (bin > 0 && !(population.pivots[bin - 1] < pivot))) {
98  return false;
99  }
100  }
101  }
102  for (int property = 0; property < population.property_count; ++property) {
103  const int offset = population.property_component_offsets[property];
104  if (offset < 0 || offset > population.state_components - population.nbins ||
105  (offset < population.mass_offset + population.nbins &&
106  population.mass_offset < offset + population.nbins)) {
107  return false;
108  }
109  if (population.moment_mode == MomentMode::TwoMoment &&
110  offset < population.number_offset + population.nbins &&
111  population.number_offset < offset + population.nbins) {
112  return false;
113  }
114  for (int previous = 0; previous < property; ++previous) {
115  const int previous_offset = population.property_component_offsets[previous];
116  if (offset < previous_offset + population.nbins &&
117  previous_offset < offset + population.nbins) {
118  return false;
119  }
120  }
121  }
122  return true;
123 }

217 /** Accept at most four representable steps outward from a global endpoint. */
218 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
219 bool within_boundary_roundoff (const amrex::Real value, const amrex::Real endpoint,
220  const bool lower_boundary) noexcept
221 {
222  const amrex::Real direction = lower_boundary ?
223  -std::numeric_limits<amrex::Real>::infinity() :
224  std::numeric_limits<amrex::Real>::infinity();
225  amrex::Real allowance = endpoint;
226  for (int step = 0; step < 4; ++step) {
227  allowance = std::nextafter(allowance, direction);
228  }
229  return lower_boundary ? value >= allowance : value <= allowance;
230 }

232 /** Validate a route against its local claimed bin(s), without searching. */
233 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
234 bool route_plan_is_consistent (const PopulationRemapView& population,
235  const PacketRoutingPlan& plan) noexcept
236 {
237  if (!valid_population_remap_context(population) ||
238  !same_population_context(population, plan.population) ||
239  !amrex::Math::isfinite(plan.packet_number) ||
240  !amrex::Math::isfinite(plan.packet_particle_mass) ||
241  plan.packet_number < amrex::Real(0.0) || plan.packet_particle_mass < amrex::Real(0.0) ||
242  plan.destination_count < 0 || plan.destination_count > 2 ||
243  !finite_nonnegative(plan.residual_number_weight) ||
244  plan.residual_number_weight > amrex::Real(1.0)) return false;
245 
246  if (plan.packet_number == amrex::Real(0.0)) {
247  return plan.status == RemapStatus::Ok && plan.destination_count == 0 &&
248  plan.residual_number_weight == amrex::Real(0.0) &&
249  !plan.normalized_roundoff;
250  }
251 
252  if (population.moment_mode == MomentMode::TwoMoment) {
253  if (plan.packet_particle_mass == amrex::Real(0.0)) {
254  return plan.status == RemapStatus::ZeroWaterResidual &&
255  plan.destination_count == 0 &&
256  plan.residual_number_weight == amrex::Real(1.0) &&
257  !plan.normalized_roundoff;
258  }
259  const amrex::Real lower = population.edges[0];
260  const amrex::Real top = population.edges[population.nbins];
261  int required_bin = -1;
262  amrex::Real expected_mass = plan.packet_particle_mass;
263  bool normalized = false;
264  if (plan.packet_particle_mass < lower) {
265  if (!within_boundary_roundoff(plan.packet_particle_mass, lower, true)) return false;
266  required_bin = 0;
267  expected_mass = lower;
268  normalized = true;
269  } else if (plan.packet_particle_mass > top) {
270  if (!within_boundary_roundoff(plan.packet_particle_mass, top, false)) return false;
271  required_bin = population.nbins - 1;
272  expected_mass = top;
273  normalized = true;
274  }
275  if (plan.status != RemapStatus::Ok || plan.destination_count != 1 ||
276  plan.residual_number_weight != amrex::Real(0.0) ||
277  plan.normalized_roundoff != normalized) return false;
278  const auto& route = plan.destinations[0];
279  return (required_bin < 0 || route.bin == required_bin) &&
280  route.number_weight == amrex::Real(1.0) &&
281  route.particle_mass == expected_mass && route.bin >= 0 &&
282  route.bin < population.nbins && expected_mass >= population.edges[route.bin] &&
283  (expected_mass < population.edges[route.bin + 1] ||
284  (route.bin == population.nbins - 1 &&
285  expected_mass <= population.edges[route.bin + 1]));
286  }
287 
288  const amrex::Real first = population.pivots[0];
289  const amrex::Real last = population.pivots[population.nbins - 1];
290  if (plan.packet_particle_mass == amrex::Real(0.0)) {
291  return plan.status == RemapStatus::ZeroWaterResidual &&
292  plan.destination_count == 0 &&
293  plan.residual_number_weight == amrex::Real(1.0) &&
294  !plan.normalized_roundoff;
295  }
296  if (plan.packet_particle_mass < first) {
297  amrex::Real wet_weight = amrex::Real(0.0);
298  if (checked_quotient(plan.packet_particle_mass, first, wet_weight) !=
299  QuotientStatus::Ok) return false;
300  return plan.status == RemapStatus::ZeroWaterResidual && !plan.normalized_roundoff &&
301  plan.destination_count == 1 && plan.residual_number_weight ==
302  amrex::Real(1.0) - wet_weight && plan.destinations[0].bin == 0 &&
303  plan.destinations[0].number_weight == wet_weight &&
304  plan.destinations[0].particle_mass == first;
305  }
306  if (plan.packet_particle_mass > last || plan.status == RemapStatus::ZeroWaterResidual) {
307  return false;
308  }
309  if (plan.destination_count == 1) {
310  const int pivot_bin = plan.destinations[0].bin;
311  if (pivot_bin < 0 || pivot_bin >= population.nbins) return false;
312  return plan.status == RemapStatus::Ok && !plan.normalized_roundoff &&
313  plan.destination_count == 1 && plan.residual_number_weight == amrex::Real(0.0) &&
314  plan.packet_particle_mass == population.pivots[pivot_bin] &&
315  plan.destinations[0].bin == pivot_bin &&
316  plan.destinations[0].number_weight == amrex::Real(1.0) &&
317  plan.destinations[0].particle_mass == population.pivots[pivot_bin];
318  }
319  if (plan.destination_count != 2) return false;
320  const int lower_bin = plan.destinations[0].bin;
321  if (lower_bin < 0 || lower_bin + 1 >= population.nbins ||
322  plan.destinations[1].bin != lower_bin + 1) return false;
323  const amrex::Real lower = population.pivots[lower_bin];
324  const amrex::Real upper = population.pivots[lower_bin + 1];
325  if (!(plan.packet_particle_mass > lower && plan.packet_particle_mass < upper)) return false;
326  const amrex::Real width = upper - lower;
327  amrex::Real lower_weight = amrex::Real(0.0);
328  amrex::Real upper_weight = amrex::Real(0.0);
329  if (checked_quotient(upper - plan.packet_particle_mass, width, lower_weight) !=
330  QuotientStatus::Ok ||
331  checked_quotient(plan.packet_particle_mass - lower, width, upper_weight) !=
332  QuotientStatus::Ok) return false;
333  return plan.status == RemapStatus::Ok && !plan.normalized_roundoff &&
334  plan.destination_count == 2 && plan.residual_number_weight == amrex::Real(0.0) &&
335  plan.destinations[0].bin == lower_bin &&
336  plan.destinations[0].number_weight == lower_weight &&
337  plan.destinations[0].particle_mass == lower &&
338  plan.destinations[1].bin == lower_bin + 1 &&
339  plan.destinations[1].number_weight == upper_weight &&
340  plan.destinations[1].particle_mass == upper;
341 }

433 /**
434  * Classify one packet without allocation or state mutation. Two-moment routing
435  * uses half-open intervals, includes the global top edge, and admits only four
436  * representable steps of outer-boundary roundoff. The plan keeps the original
437  * packet mass while its destination records the exact boundary mass. The
438  * immutable metadata must have been fully qualified once before this call;
439  * the per-packet path performs only constant-time context checks and a
440  * logarithmic interval search.
441  */
442 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
443 PacketRoutingPlan plan_packet_routing (const PopulationRemapView population,
444  const amrex::Real packet_number,
445  const amrex::Real particle_mass) noexcept
446 {
447  PacketRoutingPlan plan;
448  plan.population = population;
449  plan.packet_number = packet_number;
450  plan.packet_particle_mass = particle_mass;
451  if (!valid_population_remap_context(population) ||
452  !amrex::Math::isfinite(packet_number) || !amrex::Math::isfinite(particle_mass) ||
453  packet_number < amrex::Real(0.0) || particle_mass < amrex::Real(0.0)) {
454  return plan;
455  }
456  if (packet_number == amrex::Real(0.0)) {
457  plan.status = RemapStatus::Ok;
458  return plan;
459  }
460 
461  if (population.moment_mode == MomentMode::TwoMoment) {
462  // A dry positive-number packet is residual material, not a populated
463  // zero-water liquid interval, even when the first edge is exactly zero.
464  if (particle_mass == amrex::Real(0.0)) {
465  plan.status = RemapStatus::ZeroWaterResidual;
466  plan.residual_number_weight = amrex::Real(1.0);
467  return plan;
468  }
469  if (particle_mass < population.edges[0]) {
470  if (!remap_detail::within_boundary_roundoff(
471  particle_mass, population.edges[0], true)) {
472  plan.status = RemapStatus::BelowSupportedGrid;
473  return plan;
474  }
475  plan.status = RemapStatus::Ok;
476  plan.destination_count = 1;
477  plan.destinations[0] = {0, amrex::Real(1.0), population.edges[0]};
478  plan.normalized_roundoff = true;
479  return plan;
480  }
481  if (particle_mass > population.edges[population.nbins]) {
482  if (!remap_detail::within_boundary_roundoff(
483  particle_mass, population.edges[population.nbins], false)) {
484  plan.status = RemapStatus::Overflow;
485  return plan;
486  }
487  plan.status = RemapStatus::Ok;
488  plan.destination_count = 1;
489  plan.destinations[0] = {population.nbins - 1, amrex::Real(1.0),
490  population.edges[population.nbins]};
491  plan.normalized_roundoff = true;
492  return plan;
493  }
494  const int bin = particle_mass == population.edges[population.nbins] ?
495  population.nbins - 1 :
496  amrex::bisect(population.edges, 0, population.nbins, particle_mass);
497  plan.status = RemapStatus::Ok;
498  plan.destination_count = 1;
499  plan.destinations[0] = {bin, amrex::Real(1.0), particle_mass};
500  return plan;
501  }
502 
503  if (particle_mass == amrex::Real(0.0)) {
504  plan.status = RemapStatus::ZeroWaterResidual;
505  plan.residual_number_weight = amrex::Real(1.0);
506  return plan;
507  }
508  if (particle_mass < population.pivots[0]) {
509  amrex::Real wet_weight = amrex::Real(0.0);
510  const auto quotient_status = remap_detail::checked_quotient(
511  particle_mass, population.pivots[0], wet_weight);
512  if (quotient_status != remap_detail::QuotientStatus::Ok) {
513  plan.status = remap_detail::remap_quotient_status(quotient_status);
514  return plan;
515  }
516  plan.status = RemapStatus::ZeroWaterResidual;
517  plan.destination_count = 1;
518  plan.destinations[0] = {0, wet_weight, population.pivots[0]};
519  plan.residual_number_weight = amrex::Real(1.0) - wet_weight;
520  return plan;
521  }
522  if (particle_mass > population.pivots[population.nbins - 1]) {
523  plan.status = RemapStatus::Overflow;
524  return plan;
525  }
526  const int lower_bin = particle_mass == population.pivots[population.nbins - 1] ?
527  population.nbins - 1 :
528  amrex::bisect(population.pivots, 0, population.nbins - 1, particle_mass);
529  if (particle_mass == population.pivots[lower_bin]) {
530  plan.status = RemapStatus::Ok;
531  plan.destination_count = 1;
532  plan.destinations[0] = {lower_bin, amrex::Real(1.0), population.pivots[lower_bin]};
533  return plan;
534  }
535  if (lower_bin + 1 < population.nbins) {
536  const amrex::Real lower = population.pivots[lower_bin];
537  const amrex::Real upper = population.pivots[lower_bin + 1];
538  const amrex::Real width = upper - lower;
539  amrex::Real lower_weight = amrex::Real(0.0);
540  amrex::Real upper_weight = amrex::Real(0.0);
541  const auto lower_status = remap_detail::checked_quotient(
542  upper - particle_mass, width, lower_weight);
543  const auto upper_status = remap_detail::checked_quotient(
544  particle_mass - lower, width, upper_weight);
545  if (lower_status != remap_detail::QuotientStatus::Ok ||
546  upper_status != remap_detail::QuotientStatus::Ok) {
547  const auto quotient_status =
548  lower_status != remap_detail::QuotientStatus::Ok ? lower_status : upper_status;
549  plan.status = remap_detail::remap_quotient_status(quotient_status);
550  return plan;
551  }
552  plan.status = RemapStatus::Ok;
553  plan.destination_count = 2;
554  plan.destinations[0] = {lower_bin, lower_weight, lower};
555  plan.destinations[1] = {lower_bin + 1, upper_weight, upper};
556  return plan;
557  }
558  return plan;
559 }

569 /**
570  * Allocation-free packet application. The first pass validates the complete
571  * plan, residual outputs, and every candidate increment. The second pass
572  * commits only after all failure paths have been ruled out. Input/output and
573  * property/state storage supplied by the caller must not overlap.
574  */
575 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
576 PacketApplicationCoreResult apply_packet_routing_core (
577  const PopulationRemapView population, const PacketRoutingPlan plan,
578  const amrex::Real* property_packet_amounts, const int property_count,
579  amrex::Real* residual_properties, const int residual_capacity,
580  amrex::Real* candidate_state, const int state_count) noexcept
581 {
582  PacketApplicationCoreResult result;
583  result.status = plan.status;
584  if (plan.status != RemapStatus::Ok && plan.status != RemapStatus::ZeroWaterResidual) {
585  return result;
586  }
587  if (!valid_population_remap_context(population) ||
588  !same_population_context(population, plan.population) ||
589  !amrex::Math::isfinite(plan.packet_number) ||
590  !amrex::Math::isfinite(plan.packet_particle_mass) ||
591  plan.packet_number < amrex::Real(0.0) || plan.packet_particle_mass < amrex::Real(0.0) ||
592  property_count != population.property_count ||
593  (property_count > 0 && (property_packet_amounts == nullptr || residual_properties == nullptr)) ||
594  residual_capacity < property_count || candidate_state == nullptr ||
595  state_count != population.state_components || plan.destination_count < 0 ||
596  plan.destination_count > 2 || !amrex::Math::isfinite(plan.residual_number_weight) ||
597  plan.residual_number_weight < amrex::Real(0.0) ||
598  plan.residual_number_weight > amrex::Real(1.0)) {
599  result.status = RemapStatus::Invalid;
600  return result;
601  }
602  if (!remap_detail::route_plan_is_consistent(population, plan)) {
603  result.status = RemapStatus::Invalid;
604  return result;
605  }
606  for (int property = 0; property < property_count; ++property) {
607  if (!remap_detail::finite_nonnegative(property_packet_amounts[property])) {
608  result.status = RemapStatus::Invalid;
609  return result;
610  }
611  }
612  if (plan.packet_number == amrex::Real(0.0)) {
613  for (int property = 0; property < property_count; ++property) {
614  if (property_packet_amounts[property] != amrex::Real(0.0)) {
615  result.status = RemapStatus::Invalid;
616  return result;
617  }
618  }
619  }
620  amrex::Real weight_sum = plan.residual_number_weight;
621  for (int destination = 0; destination < plan.destination_count; ++destination) {
622  const auto& route = plan.destinations[destination];
623  if (route.bin < 0 || route.bin >= population.nbins ||
624  !amrex::Math::isfinite(route.number_weight) || route.number_weight < amrex::Real(0.0) ||
625  route.number_weight > amrex::Real(1.0) ||
626  !remap_detail::finite_nonnegative(route.particle_mass)) {
627  result.status = RemapStatus::Invalid;
628  return result;
629  }
630  for (int previous = 0; previous < destination; ++previous) {
631  if (plan.destinations[previous].bin == route.bin) {
632  result.status = RemapStatus::Invalid;
633  return result;
634  }
635  }
636  weight_sum += route.number_weight;
637  }
638  if (plan.packet_number > amrex::Real(0.0)) {
639  const amrex::Real weight_tolerance = amrex::Real(16.0) *
640  std::numeric_limits<amrex::Real>::epsilon();
641  if (!amrex::Math::isfinite(weight_sum) ||
642  std::abs(weight_sum - amrex::Real(1.0)) > weight_tolerance ||
643  ((plan.status == RemapStatus::ZeroWaterResidual) !=
644  (plan.residual_number_weight > amrex::Real(0.0)))) {
645  result.status = RemapStatus::Invalid;
646  return result;
647  }
648  } else if (plan.destination_count != 0 ||
649  plan.residual_number_weight != amrex::Real(0.0)) {
650  result.status = RemapStatus::Invalid;
651  return result;
652  }
653 
654  amrex::Real residual_number = amrex::Real(0.0);
655  auto product_status = remap_detail::checked_product(
656  plan.packet_number, plan.residual_number_weight, residual_number);
657  if (product_status != remap_detail::ProductStatus::Ok) {
658  result.status = remap_detail::remap_product_status(product_status);
659  return result;
660  }
661  amrex::Real roundoff_water_mass_correction = amrex::Real(0.0);
662  // Validate all outputs and candidate increments before changing either.
663  for (int property = 0; property < property_count; ++property) {
664  amrex::Real residual = amrex::Real(0.0);
665  product_status = remap_detail::checked_product(
666  property_packet_amounts[property], plan.residual_number_weight, residual);
667  if (product_status != remap_detail::ProductStatus::Ok) {
668  result.status = remap_detail::remap_product_status(product_status);
669  return result;
670  }
671  }
672  for (int destination = 0; destination < plan.destination_count; ++destination) {
673  const auto& route = plan.destinations[destination];
674  amrex::Real number_increment = amrex::Real(0.0);
675  amrex::Real mass_increment = amrex::Real(0.0);
676  const amrex::Real deposited_mass = route.particle_mass;
677  product_status = remap_detail::checked_product(
678  plan.packet_number, route.number_weight, number_increment);
679  if (product_status == remap_detail::ProductStatus::Ok) {
680  product_status = remap_detail::checked_product(
681  number_increment, deposited_mass, mass_increment);
682  }
683  if (product_status != remap_detail::ProductStatus::Ok) {
684  result.status = remap_detail::remap_product_status(product_status);
685  return result;
686  }
687  if (population.moment_mode == MomentMode::TwoMoment && plan.normalized_roundoff) {
688  const amrex::Real mass_delta =
689  route.particle_mass - plan.packet_particle_mass;
690  if (!amrex::Math::isfinite(mass_delta) ||
691  mass_delta == amrex::Real(0.0)) {
692  result.status = RemapStatus::Invalid;
693  return result;
694  }
695  const amrex::Real mass_delta_magnitude = mass_delta < amrex::Real(0.0) ?
696  -mass_delta : mass_delta;
697  amrex::Real correction_magnitude = amrex::Real(0.0);
698  product_status = remap_detail::checked_product(
699  plan.packet_number, mass_delta_magnitude, correction_magnitude);
700  if (product_status != remap_detail::ProductStatus::Ok) {
701  result.status = remap_detail::remap_product_status(product_status);
702  return result;
703  }
704  const amrex::Real correction = mass_delta < amrex::Real(0.0) ?
705  -correction_magnitude : correction_magnitude;
706  if (!amrex::Math::isfinite(correction) ||
707  !amrex::Math::isfinite(roundoff_water_mass_correction + correction)) {
708  result.status = RemapStatus::Invalid;
709  return result;
710  }
711  roundoff_water_mass_correction += correction;
712  }
713  const int mass_component = population.mass_offset + route.bin;
714  const amrex::Real old_mass = candidate_state[mass_component];
715  if (!remap_detail::finite_nonnegative(old_mass) ||
716  !remap_detail::finite_nonnegative(old_mass + mass_increment)) {
717  result.status = RemapStatus::Invalid;
718  return result;
719  }
720  const amrex::Real next_mass = old_mass + mass_increment;
721  amrex::Real next_number = amrex::Real(0.0);
722  if (population.number_offset >= 0) {
723  const int number_component = population.number_offset + route.bin;
724  const amrex::Real old_number = candidate_state[number_component];
725  if (!remap_detail::finite_nonnegative(old_number)) {
726  result.status = RemapStatus::Invalid;
727  return result;
728  }
729  next_number = old_number + number_increment;
730  if (!remap_detail::finite_nonnegative(next_number)) {
731  result.status = RemapStatus::Invalid;
732  return result;
733  }
734  if (next_number == amrex::Real(0.0)) {
735  if (next_mass != amrex::Real(0.0)) {
736  result.status = RemapStatus::Invalid;
737  return result;
738  }
739  } else {
740  amrex::Real mean = amrex::Real(0.0);
741  const auto quotient_status = remap_detail::checked_quotient(
742  next_mass, next_number, mean);
743  if (quotient_status != remap_detail::QuotientStatus::Ok) {
744  result.status = remap_detail::remap_quotient_status(quotient_status);
745  return result;
746  }
747  if (!remap_detail::canonical_two_moment_moments(
748  population, route.bin, next_number, next_mass)) {
749  result.status = RemapStatus::Invalid;
750  return result;
751  }
752  }
753  } else {
754  const auto quotient_status = remap_detail::checked_quotient(
755  next_mass, population.pivots[route.bin], next_number);
756  if (quotient_status != remap_detail::QuotientStatus::Ok) {
757  result.status = remap_detail::remap_quotient_status(quotient_status);
758  return result;
759  }
760  }
761  for (int property = 0; property < property_count; ++property) {
762  amrex::Real increment = amrex::Real(0.0);
763  const int component = population.property_component_offsets[property] + route.bin;
764  const amrex::Real old_value = candidate_state[component];
765  product_status = remap_detail::checked_product(
766  property_packet_amounts[property], route.number_weight, increment);
767  if (product_status != remap_detail::ProductStatus::Ok) {
768  result.status = remap_detail::remap_product_status(product_status);
769  return result;
770  }
771  if (!remap_detail::finite_nonnegative(old_value)) {
772  result.status = RemapStatus::Invalid;
773  return result;
774  }
775  const amrex::Real next_value = old_value + increment;
776  if (!remap_detail::finite_nonnegative(next_value)) {
777  result.status = RemapStatus::Invalid;
778  return result;
779  }
780  if (next_number == amrex::Real(0.0)) {
781  if (next_value != amrex::Real(0.0)) {
782  result.status = RemapStatus::Invalid;
783  return result;
784  }
785  } else {
786  amrex::Real per_particle = amrex::Real(0.0);
787  const auto quotient_status = remap_detail::checked_quotient(
788  next_value, next_number, per_particle);
789  if (quotient_status != remap_detail::QuotientStatus::Ok) {
790  result.status = remap_detail::remap_quotient_status(quotient_status);
791  return result;
792  }
793  }
794  }
795  }
796 
797  for (int destination = 0; destination < plan.destination_count; ++destination) {
798  const auto& route = plan.destinations[destination];
799  const amrex::Real number_increment = plan.packet_number * route.number_weight;
800  const amrex::Real deposited_mass = route.particle_mass;
801  candidate_state[population.mass_offset + route.bin] += number_increment * deposited_mass;
802  if (population.number_offset >= 0) {
803  candidate_state[population.number_offset + route.bin] += number_increment;
804  }
805  for (int property = 0; property < property_count; ++property) {
806  candidate_state[population.property_component_offsets[property] + route.bin] +=
807  property_packet_amounts[property] * route.number_weight;
808  }
809  }
810  for (int property = 0; property < property_count; ++property) {
811  residual_properties[property] =
812  property_packet_amounts[property] * plan.residual_number_weight;
813  }
814  result.residual_number = residual_number;
815  result.residual_water_mass = amrex::Real(0.0);
816  result.roundoff_water_mass_correction = roundoff_water_mass_correction;
817  result.normalized_roundoff = plan.normalized_roundoff;
818  return result;
819 }

833 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
834 ReconstructionStatus reconstruct_bin_core (
835  const PopulationRemapView population, const int bin,
836  const amrex::Real* persisted_state, const int state_count,
837  amrex::Real* property_per_particle, const int property_capacity,
838  ReconstructionDeltaView& reconstruction) noexcept
839 {
840  if (!valid_population_remap_context(population) || bin < 0 || bin >= population.nbins ||
841  persisted_state == nullptr || state_count != population.state_components ||
842  property_capacity < population.property_count ||
843  (population.property_count > 0 && property_per_particle == nullptr)) {
844  return ReconstructionStatus::Invalid;
845  }
846  if (!remap_detail::canonical_persisted_bin_state(
847  population, bin, persisted_state, state_count)) {
848  return ReconstructionStatus::Invalid;
849  }
850  const amrex::Real mass = persisted_state[population.mass_offset + bin];
851  if (!remap_detail::finite_nonnegative(mass)) return ReconstructionStatus::Invalid;
852  ReconstructionDeltaView next;
853  next.population = population;
854  next.bin = bin;
855  next.property_per_particle = property_per_particle;
856  next.property_count = population.property_count;
857  if (population.moment_mode == MomentMode::OneMoment) {
858  next.particle_mass = population.pivots[bin];
859  if (remap_detail::checked_quotient(
860  mass, next.particle_mass, next.number) != remap_detail::QuotientStatus::Ok) {
861  return ReconstructionStatus::Invalid;
862  }
863  } else {
864  const amrex::Real number = persisted_state[population.number_offset + bin];
865  if (!remap_detail::finite_nonnegative(number)) return ReconstructionStatus::Invalid;
866  if (!remap_detail::canonical_two_moment_bin_state(
867  population, bin, persisted_state, state_count)) {
868  return ReconstructionStatus::Invalid;
869  }
870  const amrex::Real lower = population.edges[bin];
871  const amrex::Real upper = population.edges[bin + 1];
872  EndpointTransform endpoints;
873  if (!try_two_moment_to_endpoints(number, mass, lower, upper, endpoints)) {
874  return ReconstructionStatus::Invalid;
875  }
876  next.number = number;
877  // Endpoint tolerance is only a realizability aid for the transform.
878  // Persisted state has already passed strict canonical ownership and is
879  // never normalized by no-process reconstruction.
880  next.normalized_roundoff = false;
881  if (number > amrex::Real(0.0)) {
882  if (remap_detail::checked_quotient(
883  mass, number, next.particle_mass) != remap_detail::QuotientStatus::Ok) {
884  return ReconstructionStatus::Invalid;
885  }
886  }
887  }
888  next.empty = next.number == amrex::Real(0.0);
889  if (next.empty) next.particle_mass = amrex::Real(0.0);
890  for (int property = 0; property < population.property_count; ++property) {
891  const amrex::Real amount = persisted_state[
892  population.property_component_offsets[property] + bin];
893  if (!remap_detail::finite_nonnegative(amount)) return ReconstructionStatus::Invalid;
894  if (next.empty) {
895  if (amount != amrex::Real(0.0)) return ReconstructionStatus::Invalid;
896  } else {
897  amrex::Real per_particle = amrex::Real(0.0);
898  if (remap_detail::checked_quotient(
899  amount, next.number, per_particle) != remap_detail::QuotientStatus::Ok) {
900  return ReconstructionStatus::Invalid;
901  }
902  }
903  }
904  for (int property = 0; property < population.property_count; ++property) {
905  const amrex::Real amount = persisted_state[
906  population.property_component_offsets[property] + bin];
907  if (next.empty) {
908  property_per_particle[property] = amrex::Real(0.0);
909  } else if (remap_detail::checked_quotient(
910  amount, next.number, property_per_particle[property]) !=
911  remap_detail::QuotientStatus::Ok) {
912  return ReconstructionStatus::Invalid;
913  }
914  }
915  reconstruction = next;
916  return next.empty ? ReconstructionStatus::Empty : ReconstructionStatus::Populated;
917 }

924 /** Integrate one reconstructed delta over [lower, upper), optionally including upper. */
925 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
926 bool integrate_interval_core (const ReconstructionDeltaView reconstruction,
927  const amrex::Real lower, const amrex::Real upper,
928  const bool include_upper, amrex::Real* attached_properties,
929  const int property_capacity,
930  IntegratedMomentsCoreResult& integral) noexcept
931 {
932  if (!amrex::Math::isfinite(lower) || !amrex::Math::isfinite(upper) || !(lower < upper) ||
933  !remap_detail::finite_nonnegative(reconstruction.number) ||
934  !remap_detail::finite_nonnegative(reconstruction.particle_mass) ||
935  reconstruction.property_count < 0 || property_capacity < reconstruction.property_count ||
936  (reconstruction.property_count > 0 &&
937  (reconstruction.property_per_particle == nullptr || attached_properties == nullptr)) ||
938  (!reconstruction.empty && reconstruction.number <= amrex::Real(0.0)) ||
939  (reconstruction.empty && (reconstruction.number != amrex::Real(0.0) ||
940  reconstruction.particle_mass != amrex::Real(0.0)))) {
941  return false;
942  }
943  for (int property = 0; property < reconstruction.property_count; ++property) {
944  const amrex::Real value = reconstruction.property_per_particle[property];
945  if (!remap_detail::finite_nonnegative(value) ||
946  (reconstruction.empty && value != amrex::Real(0.0))) {
947  return false;
948  }
949  }
950  const bool contains = !reconstruction.empty && reconstruction.particle_mass >= lower &&
951  (reconstruction.particle_mass < upper ||
952  (include_upper && reconstruction.particle_mass == upper));
953  IntegratedMomentsCoreResult next;
954  if (contains) {
955  next.number = reconstruction.number;
956  if (!remap_detail::finite_product(reconstruction.number,
957  reconstruction.particle_mass, next.water_mass)) {
958  return false;
959  }
960  for (int property = 0; property < reconstruction.property_count; ++property) {
961  amrex::Real amount = amrex::Real(0.0);
962  if (!remap_detail::finite_product(reconstruction.number,
963  reconstruction.property_per_particle[property], amount)) {
964  return false;
965  }
966  }
967  }
968  for (int property = 0; property < reconstruction.property_count; ++property) {
969  amrex::Real amount = amrex::Real(0.0);
970  if (contains && !remap_detail::finite_product(reconstruction.number,
971  reconstruction.property_per_particle[property], amount)) {
972  return false;
973  }
974  attached_properties[property] = amount;
975  }
976  integral = next;
977  return true;
978 }

980 /** Project one reconstructed bin into caller-owned state after full validation. */
981 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
982 bool project_reconstruction_bin_core (const PopulationRemapView population,
983  const ReconstructionDeltaView reconstruction,
984  amrex::Real* candidate_state,
985  const int state_count) noexcept
986 {
987  if (!valid_population_remap_context(population) ||
988  !same_population_context(population, reconstruction.population) ||
989  reconstruction.bin < 0 || reconstruction.bin >= population.nbins ||
990  candidate_state == nullptr || state_count != population.state_components ||
991  reconstruction.property_count != population.property_count ||
992  (reconstruction.property_count > 0 && reconstruction.property_per_particle == nullptr) ||
993  !remap_detail::finite_nonnegative(reconstruction.number) ||
994  !remap_detail::finite_nonnegative(reconstruction.particle_mass) ||
995  (!reconstruction.empty && reconstruction.number <= amrex::Real(0.0)) ||
996  (reconstruction.empty && (reconstruction.number != amrex::Real(0.0) ||
997  reconstruction.particle_mass != amrex::Real(0.0)))) {
998  return false;
999  }
1000  if (!reconstruction.empty) {
1001  if (population.moment_mode == MomentMode::OneMoment) {
1002  if (reconstruction.particle_mass != population.pivots[reconstruction.bin]) return false;
1003  }
1004  }
1005  amrex::Real mass = amrex::Real(0.0);
1006  if (!remap_detail::finite_product(reconstruction.number,
1007  reconstruction.particle_mass, mass)) {
1008  return false;
1009  }
1010  amrex::Real projected_number = reconstruction.number;
1011  if (population.moment_mode == MomentMode::OneMoment) {
1012  if (remap_detail::checked_quotient(
1013  mass, population.pivots[reconstruction.bin], projected_number) !=
1014  remap_detail::QuotientStatus::Ok) return false;
1015  } else if (!remap_detail::canonical_two_moment_moments(
1016  population, reconstruction.bin, projected_number, mass)) {
1017  return false;
1018  }
1019  for (int property = 0; property < population.property_count; ++property) {
1020  const amrex::Real value = reconstruction.property_per_particle[property];
1021  amrex::Real amount = amrex::Real(0.0);
1022  if (!remap_detail::finite_nonnegative(value) ||
1023  (reconstruction.empty && value != amrex::Real(0.0)) ||
1024  !remap_detail::finite_product(reconstruction.number, value, amount)) {
1025  return false;
1026  }
1027  if (projected_number == amrex::Real(0.0)) {
1028  if (amount != amrex::Real(0.0)) return false;
1029  } else {
1030  amrex::Real per_particle = amrex::Real(0.0);
1031  if (remap_detail::checked_quotient(
1032  amount, projected_number, per_particle) !=
1033  remap_detail::QuotientStatus::Ok) return false;
1034  }
1035  }
1036  const int bin = reconstruction.bin;
1037  candidate_state[population.mass_offset + bin] = mass;
1038  if (population.number_offset >= 0) {
1039  candidate_state[population.number_offset + bin] = reconstruction.number;
1040  }
1041  for (int property = 0; property < population.property_count; ++property) {
1042  candidate_state[population.property_component_offsets[property] + bin] =
1043  reconstruction.number * reconstruction.property_per_particle[property];
1044  }
1045  return true;
1046 }
