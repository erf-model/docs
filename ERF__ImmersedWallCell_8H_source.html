14 /**
15  * The wall cell of immersed-forcing terrain and the fraction-stress wall law
16  * (erf.if_wall_form = fraction_stress), shared by the momentum sources, the wall-face
17  * masks of the diffusion and the RANS wall distance so they cannot disagree.
18  *
19  * A column of solid fractions beta (terrain_blanking) meets the air in its WALL CELL: a
20  * partial cell with fluid above it, or a fluid cell sitting on a solid cell. The surface is
21  * at h = z_bot + beta dz of that cell (beta = 0 for the fluid cell), the wall cell's fluid
22  * centroid at d1 = (1 - beta) dz / 2 above it and the centre of the cell above at
23  * d2 = (3/2 - beta) dz. The fraction-stress law takes the friction velocity
24  * u*^2 = (1 - beta) u*_own^2 + beta u*_ref^2,
25  * u*_own = kappa |u_t| / ln(d1/z0), u*_ref = kappa |u_t,ref| / ln(d2/z0),
26  * so a nearly full fluid cell uses its own velocity (on a face-aligned surface, beta = 0,
27  * that is the plain surface-layer law of flat ground at dz/2) and a thin fluid sliver the
28  * better resolved cell above, and applies it as a wall stress spread over the cell height,
29  * integrated exactly over the step:
30  * F = -rho C u_t, C = (1 - exp(-r dt)) / dt, r = u*^2 / (dz |u_t|).
31  * The resolved diffusive flux through the wall face (the bottom face of the wall cell) is
32  * switched off, so the column loses exactly u*^2 through the wall. This is form 7 of the
33  * kynema-sgf immersed terrain, written for the vertical columns of ERF's immersed forcing
34  * (horizontal velocity, vertical distances) and neutral stratification.
35  */

70 /**
71  * Friction velocity of the fraction-stress law.
72  * @param ut horizontal speed in the wall cell
73  * @param ut_ref horizontal speed in the cell above
74  * @param beta wall_beta of the wall cell
75  * @param dz cell height
76  * @param z0 roughness length
77  * The distances are floored at 2 z0 so that ln(d/z0) stays above ln 2 in a thin sliver,
78  * where the (1 - beta) weight of the own estimate is small anyway.
79  */
80 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
81 amrex::Real friction_velocity (amrex::Real ut, amrex::Real ut_ref, amrex::Real beta,
82  amrex::Real dz, amrex::Real z0) noexcept
83 {
84  const amrex::Real d_min = amrex::Real(2.0) * z0;
85  const amrex::Real d1 = amrex::max(amrex::Real(0.5) * (amrex::Real(1.0) - beta) * dz, d_min);
86  const amrex::Real d2 = amrex::max((amrex::Real(1.5) - beta) * dz, d1);
87  const amrex::Real us_own = KAPPA * ut / std::log(d1 / z0);
88  const amrex::Real us_ref = KAPPA * ut_ref / std::log(d2 / z0);
89  return std::sqrt((amrex::Real(1.0) - beta) * us_own * us_own + beta * us_ref * us_ref);
90 }

93 /**
94  * Factor on the diffusive coupling through the face above the wall cell (the face between the
95  * wall cell, whose fluid centroid sits d1 = (1-beta) dz/2 above the surface, and the cell above,
96  * at d2 = (3/2-beta) dz). The solver differences across that face over dz; the factor dz/dz_eff
97  * replaces dz by
98  * spacing = 0 (grid): dz (factor 1)
99  * spacing = 1 (physical): d2 - d1 = (1 - beta/2) dz
100  * spacing = 2 (log): (d1 + d2) ln(d2/d1) / (2 ln 3), the spacing that gives the face flux,
101  * with the face viscosity the mean of the two cells' kappa u* z, the
102  * same relative error for a log profile as the first interior face of
103  * flat ground (d1 = dz/2, d2 = 3 dz/2), where it equals dz: ln 3 u*^2
104  * for every beta
105  * All three are 1 for beta = 0, so a surface on a cell face stays flat ground. The factor is
106  * bounded only while z0 < dz/4 (d2 > d1 for every beta): the inputs require that.
107  */
108 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
109 amrex::Real face_above_factor (amrex::Real beta, amrex::Real dz, amrex::Real z0, int spacing) noexcept
110 {
111  const amrex::Real d1 = amrex::max(amrex::Real(0.5) * (amrex::Real(1.0) - beta) * dz, amrex::Real(2.0) * z0);
112  const amrex::Real d2 = amrex::max((amrex::Real(1.5) - beta) * dz, d1);
113  const amrex::Real dz_phys = d2 - d1;
114  const amrex::Real dz_log = (d1 + d2) * std::log(d2 / d1) / (amrex::Real(2.0) * amrex::Real(1.0986122886681098));
115  const amrex::Real dz_eff = (spacing == 1) ? dz_phys : ((spacing == 2) ? dz_log : dz);
116  return dz / amrex::max(dz_eff, std::numeric_limits<amrex::Real>::min());
117 }

181 /**
182  * Wall state of the fraction-stress law with MOST stability functions (the functions of the
183  * flat surface layer, similarity_funs::calc_psi_m / calc_psi_h with Dyer's or QNSE's stable
184  * side as erf.most.stable_functions says, zeta clamped to +-100):
185  * u*^2 = (1-b) (kappa ut / phi_m(d1))^2 + b (kappa ut_ref / phi_m(d2))^2
186  * theta* = -q / u* (heat_flux)
187  * = (1-b) kappa (theta - theta_s) / phi_h(d1)
188  * + b kappa (theta_ref - theta_s) / phi_h(d2) (surface_temp)
189  * = theta u*^2 / (kappa g L) (obukhov)
190  * q* = -F_q / u* (moist_flux)
191  * = (1-b) kappa (q - q_s) / phi_h(d1)
192  * + b kappa (q_ref - q_s) / phi_h(d2) (surface_moist)
193  * 1/L = kappa g theta* / (u*^2 theta) (heat_flux, surface_temp: iterated)
194  * = kappa g theta_v* / (u*^2 theta_v) (the same with a moisture model, c.moist:
195  * theta_v = theta (1 + epsv q),
196  * theta_v* = theta* (1 + epsv q) + epsv theta q*,
197  * iterated also for a moisture condition alone)
198  * with d1 = (1-b) dz/2, d2 = (3/2-b) dz floored at 2 z0 and phi_{m,h}(d) = ln(d/z0) - psi(d/L).
199  * With c.moist a given L (obukhov) sets theta_v*, and theta* = (theta_v* - epsv theta q*) / (1 + epsv q).
200  * q and q_ref are the water vapour mixing ratios of the wall cell and the cell above (ignored
201  * without c.moist). The iteration starts from neutral and runs a fixed number of passes, so it is
202  * deterministic and has no data-dependent trip count.
203  */
204 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
205 WallState wall_state (amrex::Real ut, amrex::Real ut_ref, amrex::Real theta, amrex::Real theta_ref,
206  amrex::Real q, amrex::Real q_ref,
207  amrex::Real beta, amrex::Real dz, amrex::Real z0, const WallCond& c) noexcept
208 {
209  const similarity_funs& sfuns = c.sfuns;
210  const amrex::Real one_r = amrex::Real(1.0);
211  const amrex::Real d_min = amrex::Real(2.0) * z0;
212  const amrex::Real d1 = amrex::max(amrex::Real(0.5) * (one_r - beta) * dz, d_min);
213  const amrex::Real d2 = amrex::max((amrex::Real(1.5) - beta) * dz, d1);
214  const amrex::Real zmax = amrex::Real(100.0);
215  const amrex::Real us_min = amrex::Real(1.0e-3);
216  const amrex::Real kg = KAPPA * CONST_GRAV;
217  // 1/L is solved for (iterated) with a heat condition that is not a given length, and with a
218  // moisture condition alone when the moisture enters the buoyancy
219  const bool moist_cond = c.moist && (c.moist_type != WallCond::moist_none);
220  const bool solve_L = (c.type == WallCond::heat_flux || c.type == WallCond::surface_temp ||
221  (moist_cond && c.type != WallCond::obukhov));
222  const int n_iter = solve_L ? 12 : 1;
223 
224  WallState w;
225  w.Linv = (c.type == WallCond::obukhov) ? c.Linv : amrex::Real(0.0);
226  // 1/L is kept where d1/L stays within the zeta clamp: the stable given-flux problem has no
227  // solution below a wind threshold (about 3 m/s at q = -0.02 K m/s, d1 = 10 m, z0 = 0.1 m)
228  // and the iteration would otherwise run 1/L away while u* collapses to the floor
229  const amrex::Real Linv_max = zmax / d1;
230  for (int it = 0; it < n_iter; ++it) {
231  w.Linv = amrex::max(amrex::min(w.Linv, Linv_max), -Linv_max);
232  const amrex::Real z1 = amrex::max(amrex::min(d1 * w.Linv, zmax), -zmax);
233  const amrex::Real z2 = amrex::max(amrex::min(d2 * w.Linv, zmax), -zmax);
234  const amrex::Real us_own = KAPPA * ut / phi(d1, z0, sfuns.calc_psi_m(z1));
235  const amrex::Real us_ref = KAPPA * ut_ref / phi(d2, z0, sfuns.calc_psi_m(z2));
236  w.ustar = std::sqrt((one_r - beta) * us_own * us_own + beta * us_ref * us_ref);
237  const amrex::Real us_f = amrex::max(w.ustar, us_min);
238  if (c.moist_type == WallCond::moist_flux) {
239  w.qstar = -c.qflux / us_f;
240  } else if (c.moist_type == WallCond::surface_moist) {
241  w.qstar = (one_r - beta) * KAPPA * (q - c.q_s) / phi(d1, z0, sfuns.calc_psi_h(z1))
242  + beta * KAPPA * (q_ref - c.q_s) / phi(d2, z0, sfuns.calc_psi_h(z2));
243  } else {
244  w.qstar = amrex::Real(0.0);
245  }
246  if (c.type == WallCond::heat_flux) {
247  w.thetastar = -c.q / us_f;
248  } else if (c.type == WallCond::surface_temp) {
249  w.thetastar = (one_r - beta) * KAPPA * (theta - c.theta_s) / phi(d1, z0, sfuns.calc_psi_h(z1))
250  + beta * KAPPA * (theta_ref - c.theta_s) / phi(d2, z0, sfuns.calc_psi_h(z2));
251  } else if (c.type == WallCond::obukhov) {
252  if (c.moist) {
253  // the given length sets theta_v*; theta* is what is left after the moisture part
254  const amrex::Real thv_star = virtual_theta(theta, q) * us_f * us_f * c.Linv / kg;
255  w.thetastar = (thv_star - epsv * theta * w.qstar) / (one_r + epsv * q);
256  } else {
257  w.thetastar = theta * us_f * us_f * c.Linv / kg;
258  }
259  } else {
260  w.thetastar = amrex::Real(0.0);
261  }
262  if (solve_L) {
263  w.Linv = c.moist ? kg * virtual_thetastar(w, theta, q) / (us_f * us_f * virtual_theta(theta, q))
264  : kg * w.thetastar / (us_f * us_f * theta);
265  }
266  }
267  // the stress uses the same floored u* as theta* and 1/L
268  w.ustar = amrex::max(w.ustar, us_min);
269  w.Linv = amrex::max(amrex::min(w.Linv, Linv_max), -Linv_max);
270  return w;
271 }

281 /**
282  * A wall flux (per unit mass, already divided by the cell height) that is proportional to
283  * delta = value - surface value, integrated exactly over dt as a decay so the cell cannot
284  * overshoot the surface value; a flux that is not a decay (rate <= 0) is applied as it stands.
285  */
286 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
287 amrex::Real decay_toward (amrex::Real flux, amrex::Real delta, amrex::Real dt) noexcept
288 {
289  const amrex::Real eps = std::numeric_limits<amrex::Real>::epsilon();
290  const amrex::Real rate = -flux / (delta + std::copysign(eps, delta));
291  const amrex::Real dt_safe = amrex::max(dt, std::numeric_limits<amrex::Real>::min());
292  // a decay (rate > 0) is integrated exactly; anything else is applied as it stands
293  const amrex::Real exact = std::expm1(-amrex::max(rate, amrex::Real(0.0)) * dt_safe) / dt_safe * delta;
294  return (rate > amrex::Real(0.0) && std::abs(delta) > eps) ? exact : flux;
295 }

297 /**
298  * Source of the wall heat flux in the wall cell, per unit mass (d theta / dt): the kinematic
299  * flux -u* theta* spread over the cell height. With a given surface temperature theta* is
300  * proportional to theta - theta_s and the term is a decay, integrated exactly over dt so it
301  * cannot overshoot the surface temperature; otherwise the flux is applied as it stands.
302  */
303 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
304 amrex::Real heat_source (const WallState& w, amrex::Real theta, const WallCond& c,
305  amrex::Real dz, amrex::Real dt) noexcept
306 {
307  const amrex::Real flux = -w.ustar * w.thetastar / dz;
308  if (c.type != WallCond::surface_temp) { return flux; }
309  return decay_toward(flux, theta - c.theta_s, dt);

314  * flux -u* q* spread over the cell height; with a given surface mixing ratio the term is a
315  * decay toward q_s, integrated exactly as heat_source does toward theta_s (as there, the

317  * never passes q_s). Zero without a moisture condition (q* = 0).

319 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE

328 /**
329  * Effective rate C of the wall stress u*^2 spread over the cell height dz, integrated exactly
330  * over dt: the momentum source is -rho C u_t.
331  */
332 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
333 amrex::Real stress_rate (amrex::Real ustar, amrex::Real ut, amrex::Real dz, amrex::Real dt) noexcept
334 {
335  // dt is floored rather than branched on, so no guarded 0/0 can be speculated (FPE traps);
336  // (1 - exp(-r dt)) / dt tends to r as dt -> 0
337  const amrex::Real r = ustar * ustar / (dz * (ut + std::numeric_limits<amrex::Real>::epsilon()));
338  const amrex::Real dt_safe = amrex::max(dt, std::numeric_limits<amrex::Real>::min());
339  return -std::expm1(-r * dt_safe) / dt_safe;
340 }

342 /**
343  * Wall value of the k-eqn TKE in the wall cell (erf.if_wall_tke), the value erf.dirichlet_k holds in
344  * the first cell over flat ground (Axell & Liungman 2001, Eq. 16):
345  * k = (u*^3 + kappa B d1)^(2/3) / Cmu0^2, B = -g u* bstar / btheta (unstable only),
346  * with u* from the wall state, the buoyancy scale bstar and its temperature btheta (theta* and
347  * theta; with a moisture model theta_v* and theta_v, see virtual_thetastar) and d1 the
348  * fluid-centroid height of the wall cell (floored at 2 z0, as the wall state does). kynema-sgf
349  * relaxes its immersed wall cells toward the same value.
350  */
351 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
352 amrex::Real wall_tke (const WallState& w, amrex::Real beta, amrex::Real dz, amrex::Real z0,
353  amrex::Real Cmu0, amrex::Real bstar, amrex::Real btheta) noexcept
354 {
355  const amrex::Real d1 = amrex::max(amrex::Real(0.5) * (amrex::Real(1.0) - beta) * dz,
356  amrex::Real(2.0) * z0);
357  const amrex::Real B = (bstar < amrex::Real(-1.e-8))
358  ? -CONST_GRAV * w.ustar * bstar / btheta : amrex::Real(0.0);
359  return std::pow(w.ustar * w.ustar * w.ustar + KAPPA * B * d1, amrex::Real(2.0) / amrex::Real(3.0))
360  / (Cmu0 * Cmu0);
361 }
