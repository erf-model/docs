14 /**
15  * The wall cell of immersed-forcing terrain and the fraction-stress wall law
16  * (erf.if_wall_form = fraction_stress), shared by the momentum sources, the wall-face
17  * masks of the diffusion and the RANS wall distance so they cannot disagree.
18  *
19  * A column of solid fractions beta (terrain_blanking) meets the air in its WALL CELL: a
20  * partial cell with fluid above it, or a fluid cell sitting on a solid cell. The surface is
21  * at h = z_bot + beta dz of that cell (beta = 0 for the fluid cell), the wall cell's fluid
22  * centroid at d1 = (1 - beta) dz / 2 above it and the centre of the cell above at
23  * d2 = (3/2 - beta) dz. The fraction-stress law takes the friction velocity
24  * u*^2 = (1 - beta) u*_own^2 + beta u*_ref^2,
25  * u*_own = kappa |u_t| / ln(d1/z0), u*_ref = kappa |u_t,ref| / ln(d2/z0),
26  * so a nearly full fluid cell uses its own velocity (on a face-aligned surface, beta = 0,
27  * that is the plain surface-layer law of flat ground at dz/2) and a thin fluid sliver the
28  * better resolved cell above, and applies it as a wall stress spread over the cell height,
29  * integrated exactly over the step:
30  * F = -rho C u_t, C = (1 - exp(-r dt)) / dt, r = u*^2 / (dz |u_t|).
31  * The resolved diffusive flux through the wall face (the bottom face of the wall cell) is
32  * switched off, so the column loses exactly u*^2 through the wall. This is form 7 of the
33  * kynema-sgf immersed terrain, written for the vertical columns of ERF's immersed forcing
34  * (horizontal velocity, vertical distances) and neutral stratification.
35  */

70 /**
71  * Friction velocity of the fraction-stress law.
72  * @param ut horizontal speed in the wall cell
73  * @param ut_ref horizontal speed in the cell above
74  * @param beta wall_beta of the wall cell
75  * @param dz cell height
76  * @param z0 roughness length
77  * The distances are floored at 2 z0 so that ln(d/z0) stays above ln 2 in a thin sliver,
78  * where the (1 - beta) weight of the own estimate is small anyway.
79  */
80 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
81 amrex::Real friction_velocity (amrex::Real ut, amrex::Real ut_ref, amrex::Real beta,
82  amrex::Real dz, amrex::Real z0) noexcept
83 {
84  const amrex::Real d_min = amrex::Real(2.0) * z0;
85  const amrex::Real d1 = amrex::max(amrex::Real(0.5) * (amrex::Real(1.0) - beta) * dz, d_min);
86  const amrex::Real d2 = amrex::max((amrex::Real(1.5) - beta) * dz, d1);
87  const amrex::Real us_own = KAPPA * ut / std::log(d1 / z0);
88  const amrex::Real us_ref = KAPPA * ut_ref / std::log(d2 / z0);
89  return std::sqrt((amrex::Real(1.0) - beta) * us_own * us_own + beta * us_ref * us_ref);
90 }

93 /**
94  * Factor on the diffusive coupling through the face above the wall cell (the face between the
95  * wall cell, whose fluid centroid sits d1 = (1-beta) dz/2 above the surface, and the cell above,
96  * at d2 = (3/2-beta) dz). The solver differences across that face over dz; the factor dz/dz_eff
97  * replaces dz by
98  * spacing = 0 (grid): dz (factor 1)
99  * spacing = 1 (physical): d2 - d1 = (1 - beta/2) dz
100  * spacing = 2 (log): (d1 + d2) ln(d2/d1) / (2 ln 3), the spacing that gives the face flux,
101  * with the face viscosity the mean of the two cells' kappa u* z, the
102  * same relative error for a log profile as the first interior face of
103  * flat ground (d1 = dz/2, d2 = 3 dz/2), where it equals dz: ln 3 u*^2
104  * for every beta
105  * All three are 1 for beta = 0, so a surface on a cell face stays flat ground. The factor is
106  * bounded only while z0 < dz/4 (d2 > d1 for every beta): the inputs require that.
107  */
108 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
109 amrex::Real face_above_factor (amrex::Real beta, amrex::Real dz, amrex::Real z0, int spacing) noexcept
110 {
111  const amrex::Real d1 = amrex::max(amrex::Real(0.5) * (amrex::Real(1.0) - beta) * dz, amrex::Real(2.0) * z0);
112  const amrex::Real d2 = amrex::max((amrex::Real(1.5) - beta) * dz, d1);
113  const amrex::Real dz_phys = d2 - d1;
114  const amrex::Real dz_log = (d1 + d2) * std::log(d2 / d1) / (amrex::Real(2.0) * amrex::Real(1.0986122886681098));
115  const amrex::Real dz_eff = (spacing == 1) ? dz_phys : ((spacing == 2) ? dz_log : dz);
116  return dz / amrex::max(dz_eff, std::numeric_limits<amrex::Real>::min());
117 }

147 /**
148  * Wall state of the fraction-stress law with MOST stability functions (the Dyer functions of
149  * the flat surface layer, similarity_funs::calc_psi_m / calc_psi_h, zeta clamped to +-100):
150  * u*^2 = (1-b) (kappa ut / phi_m(d1))^2 + b (kappa ut_ref / phi_m(d2))^2
151  * theta* = -q / u* (heat_flux)
152  * = (1-b) kappa (theta - theta_s) / phi_h(d1)
153  * + b kappa (theta_ref - theta_s) / phi_h(d2) (surface_temp)
154  * = theta u*^2 / (kappa g L) (obukhov)
155  * 1/L = kappa g theta* / (u*^2 theta) (heat_flux, surface_temp: iterated)
156  * with d1 = (1-b) dz/2, d2 = (3/2-b) dz floored at 2 z0 and phi_{m,h}(d) = ln(d/z0) - psi(d/L).
157  * The iteration starts from neutral and runs a fixed number of passes, so it is deterministic
158  * and has no data-dependent trip count.
159  */
160 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
161 WallState wall_state (amrex::Real ut, amrex::Real ut_ref, amrex::Real theta, amrex::Real theta_ref,
162  amrex::Real beta, amrex::Real dz, amrex::Real z0, const WallCond& c) noexcept
163 {
164  similarity_funs sfuns;
165  const amrex::Real one_r = amrex::Real(1.0);
166  const amrex::Real d_min = amrex::Real(2.0) * z0;
167  const amrex::Real d1 = amrex::max(amrex::Real(0.5) * (one_r - beta) * dz, d_min);
168  const amrex::Real d2 = amrex::max((amrex::Real(1.5) - beta) * dz, d1);
169  const amrex::Real zmax = amrex::Real(100.0);
170  const amrex::Real us_min = amrex::Real(1.0e-3);
171  const amrex::Real kg = KAPPA * CONST_GRAV;
172  const int n_iter = (c.type == WallCond::heat_flux || c.type == WallCond::surface_temp) ? 12 : 1;
173 
174  WallState w;
175  w.Linv = (c.type == WallCond::obukhov) ? c.Linv : amrex::Real(0.0);
176  // 1/L is kept where d1/L stays within the zeta clamp: the stable given-flux problem has no
177  // solution below a wind threshold (about 3 m/s at q = -0.02 K m/s, d1 = 10 m, z0 = 0.1 m)
178  // and the iteration would otherwise run 1/L away while u* collapses to the floor
179  const amrex::Real Linv_max = zmax / d1;
180  for (int it = 0; it < n_iter; ++it) {
181  w.Linv = amrex::max(amrex::min(w.Linv, Linv_max), -Linv_max);
182  const amrex::Real z1 = amrex::max(amrex::min(d1 * w.Linv, zmax), -zmax);
183  const amrex::Real z2 = amrex::max(amrex::min(d2 * w.Linv, zmax), -zmax);
184  const amrex::Real us_own = KAPPA * ut / phi(d1, z0, sfuns.calc_psi_m(z1));
185  const amrex::Real us_ref = KAPPA * ut_ref / phi(d2, z0, sfuns.calc_psi_m(z2));
186  w.ustar = std::sqrt((one_r - beta) * us_own * us_own + beta * us_ref * us_ref);
187  const amrex::Real us_f = amrex::max(w.ustar, us_min);
188  if (c.type == WallCond::heat_flux) {
189  w.thetastar = -c.q / us_f;
190  } else if (c.type == WallCond::surface_temp) {
191  w.thetastar = (one_r - beta) * KAPPA * (theta - c.theta_s) / phi(d1, z0, sfuns.calc_psi_h(z1))
192  + beta * KAPPA * (theta_ref - c.theta_s) / phi(d2, z0, sfuns.calc_psi_h(z2));
193  } else if (c.type == WallCond::obukhov) {
194  w.thetastar = theta * us_f * us_f * c.Linv / kg;
195  } else {
196  w.thetastar = amrex::Real(0.0);
197  }
198  if (c.type == WallCond::heat_flux || c.type == WallCond::surface_temp) {
199  w.Linv = kg * w.thetastar / (us_f * us_f * theta);
200  }
201  }
202  // the stress uses the same floored u* as theta* and 1/L
203  w.ustar = amrex::max(w.ustar, us_min);
204  w.Linv = amrex::max(amrex::min(w.Linv, Linv_max), -Linv_max);
205  return w;
206 }

208 /**
209  * Source of the wall heat flux in the wall cell, per unit mass (d theta / dt): the kinematic
210  * flux -u* theta* spread over the cell height. With a given surface temperature theta* is
211  * proportional to theta - theta_s and the term is a decay, integrated exactly over dt so it
212  * cannot overshoot the surface temperature; otherwise the flux is applied as it stands.
213  */
214 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
215 amrex::Real heat_source (const WallState& w, amrex::Real theta, const WallCond& c,
216  amrex::Real dz, amrex::Real dt) noexcept
217 {
218  const amrex::Real flux = -w.ustar * w.thetastar / dz;
219  if (c.type != WallCond::surface_temp) { return flux; }
220  const amrex::Real dth = theta - c.theta_s;
221  const amrex::Real eps = std::numeric_limits<amrex::Real>::epsilon();
222  const amrex::Real rate = -flux / (dth + std::copysign(eps, dth));
223  const amrex::Real dt_safe = amrex::max(dt, std::numeric_limits<amrex::Real>::min());
224  // a decay (rate > 0) is integrated exactly; anything else is applied as it stands
225  const amrex::Real exact = std::expm1(-amrex::max(rate, amrex::Real(0.0)) * dt_safe) / dt_safe * dth;
226  return (rate > amrex::Real(0.0) && std::abs(dth) > eps) ? exact : flux;
227 }

229 /**
230  * Effective rate C of the wall stress u*^2 spread over the cell height dz, integrated exactly
231  * over dt: the momentum source is -rho C u_t.
232  */
233 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
234 amrex::Real stress_rate (amrex::Real ustar, amrex::Real ut, amrex::Real dz, amrex::Real dt) noexcept
235 {
236  // dt is floored rather than branched on, so no guarded 0/0 can be speculated (FPE traps);
237  // (1 - exp(-r dt)) / dt tends to r as dt -> 0
238  const amrex::Real r = ustar * ustar / (dz * (ut + std::numeric_limits<amrex::Real>::epsilon()));
239  const amrex::Real dt_safe = amrex::max(dt, std::numeric_limits<amrex::Real>::min());
240  return -std::expm1(-r * dt_safe) / dt_safe;
241 }

243 /**
244  * Wall value of the k-eqn TKE in the wall cell (erf.if_wall_tke), the value erf.dirichlet_k holds in
245  * the first cell over flat ground (Axell & Liungman 2001, Eq. 16):
246  * k = (u*^3 + kappa B d1)^(2/3) / Cmu0^2, B = -g u* theta* / theta (unstable only),
247  * with u*, theta* from the wall state and d1 the fluid-centroid height of the wall cell (floored
248  * at 2 z0, as the wall state does). kynema-sgf relaxes its immersed wall cells toward the same value.
249  */
250 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
251 amrex::Real wall_tke (const WallState& w, amrex::Real beta, amrex::Real dz, amrex::Real z0,
252  amrex::Real theta, amrex::Real Cmu0) noexcept
253 {
254  const amrex::Real d1 = amrex::max(amrex::Real(0.5) * (amrex::Real(1.0) - beta) * dz,
255  amrex::Real(2.0) * z0);
256  const amrex::Real B = (w.thetastar < amrex::Real(-1.e-8))
257  ? -CONST_GRAV * w.ustar * w.thetastar / theta : amrex::Real(0.0);
258  return std::pow(w.ustar * w.ustar * w.ustar + KAPPA * B * d1, amrex::Real(2.0) / amrex::Real(3.0))
259  / (Cmu0 * Cmu0);
260 }
