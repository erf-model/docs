72 AMREX_FORCE_INLINE
73 AMREX_GPU_HOST_DEVICE
74 amrex::Real
75 deterministic_perturbation (amrex::Real x, amrex::Real y, amrex::Real z,
76  const amrex::GpuArray<amrex::Real, AMREX_SPACEDIM>& lo,
77  const amrex::GpuArray<amrex::Real, AMREX_SPACEDIM>& length,
78  amrex::Real amplitude) noexcept
79 {
80  constexpr amrex::Real pi = amrex::Real(3.1415926535897932384626433832795);
81  const amrex::Real sx = std::sin(amrex::Real(2.0) * pi * (x - lo[0]) / length[0]);
82  const amrex::Real sy = std::sin(amrex::Real(2.0) * pi * (y - lo[1]) / length[1]);
83  const amrex::Real sz = std::sin(pi * (z - lo[2]) / length[2]);
84  return amplitude * sx * sy * sz;
85 }

91  amrex::Real z) noexcept
92 {

102  config.temperature_perturbation_amplitude : config.theta_perturbation_amplitude;
103  return linear_profile(bottom, top, z,
104  config.prob_lo[2], length[2]) +
105  deterministic_perturbation(x, y, z, config.prob_lo, length,
106  amplitude);

119 AMREX_FORCE_INLINE

121 amrex::Real
122 vapor_mixing_ratio_from_relative_humidity (amrex::Real temperature_K,
123  amrex::Real pressure_Pa,
124  amrex::Real relative_humidity) noexcept
125 {
126  // erf_esatw returns hPa; convert saturation and vapor pressure to Pa before
127  // applying the dry-air vapor mixing-ratio definition.
128  const amrex::Real es_Pa = erf_esatw(temperature_K) * amrex::Real(100.0);
129  const amrex::Real ev_Pa = relative_humidity * es_Pa;
130  return RdoRv * ev_Pa / (pressure_Pa - ev_Pa);
131 }
