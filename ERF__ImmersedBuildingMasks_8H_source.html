28 /**
29  * The face blanking the explicit path zeroes on after the projection (ERF_SlowRhsPost.cpp): the
30  * face-centred fraction when the build has it, else the mean of the two cells the face joins.
31  * (io, jo, ko) is the unit vector of the face's direction.

33 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
34 amrex::Real raw_face_blank (const amrex::Array4<const amrex::Real>& t, const amrex::Array4<const amrex::Real>& tf,
35  int i, int j, int k, int io, int jo, int ko) noexcept
36 {
37  return (tf) ? tf(i,j,k) : myhalf * (t(i,j,k) + t(i-io,j-jo,k-ko));

40 /**
41  * The face blanking as the building kernels read it: with the snap, from the snapped cells (solid
42  * when either is), so both EB builds give the same staircase; otherwise raw_face_blank; then
43  * thresholded at small (eb2.small_volfrac), below which the face counts as fluid.
44  */
45 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
46 amrex::Real face_blank (const amrex::Array4<const amrex::Real>& t, const amrex::Array4<const amrex::Real>& tf,
47  int i, int j, int k, int io, int jo, int ko, bool snap, amrex::Real small) noexcept
48 {
49  const amrex::Real b = snap ? amrex::max(snapped(t(i,j,k), snap), snapped(t(i-io,j-jo,k-ko), snap))
50  : raw_face_blank(t, tf, i, j, k, io, jo, ko);
51  return (b < small) ? amrex::Real(zero) : b;
52 }

65 /**
66  * @param t Cell blanking (with at least one ghost cell in each direction)
67  * @param tf Face-centred blanking of this direction, or an empty Array4
68  * @param klo Lowest cell index of the domain (the face below the domain bottom counts as fluid)
69  * @param snap erf.if_snap_partial_cells
70  * @param small eb2.small_volfrac
71  */
72 template <int dir>
73 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
74 HorizFace horiz_face (const amrex::Array4<const amrex::Real>& t, const amrex::Array4<const amrex::Real>& tf,
75  int i, int j, int k, int klo, bool snap, amrex::Real small) noexcept
76 {
77  static_assert(dir == 0 || dir == 1, "horiz_face: x- or y-faces only");
78  constexpr int io = (dir == 0) ? 1 : 0;
79  constexpr int jo = (dir == 1) ? 1 : 0;
80  HorizFace f;
81  f.t = face_blank(t, tf, i, j, k, io, jo, 0, snap, small);
82  f.below = (k == klo) ? amrex::Real(zero) : face_blank(t, tf, i, j, k-1, io, jo, 0, snap, small);
83  f.above = face_blank(t, tf, i, j, k+1, io, jo, 0, snap, small);
84  f.lo = (dir == 0) ? face_blank(t, tf, i, j-1, k, io, jo, 0, snap, small)
85  : face_blank(t, tf, i-1, j, k, io, jo, 0, snap, small);
86  f.hi = (dir == 0) ? face_blank(t, tf, i, j+1, k, io, jo, 0, snap, small)
87  : face_blank(t, tf, i+1, j, k, io, jo, 0, snap, small);
88  f.raw = raw_face_blank(t, tf, i, j, k, io, jo, 0);
89  f.normal = snap && (snapped(t(i,j,k), snap) != snapped(t(i-io,j-jo,k), snap));
90  return f;
91 }

115 /**
116  * Masks of a horizontal-momentum face (x or y). lo/hi are the neighbouring faces across the
117  * other horizontal direction: south/north for an x-face, west/east for a y-face.
118  */
119 struct HorizMasks {
120  amrex::Real roof, lo, hi, wall, side, interior;
121  //! faces carrying the MOST wall law (a target velocity from the log law)
122  [[nodiscard]] AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
123  amrex::Real most () const noexcept { return roof + lo + hi; }
124  //! weight of the linear drag CdM |u| toward zero on this face, with t its blanking
125  [[nodiscard]] AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
126  amrex::Real linear (amrex::Real t) const noexcept { return (wall + side) * t + interior; }
127 };

129 /**
130  * @param t Blanking of the face
131  * @param below Blanking of the face below (zero at the domain bottom)
132  * @param above Blanking of the face above
133  * @param lo Blanking of the neighbouring face on the low side across the other direction
134  * @param hi Blanking of the neighbouring face on the high side across the other direction
135  * @param normal_face With the snap, a face joining a solid and a fluid cell (wall-normal)
136  * @param use_most erf.if_use_most
137  * @param snap erf.if_snap_partial_cells
138  */
139 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
140 HorizMasks horiz_masks (amrex::Real t, amrex::Real below, amrex::Real above,
141  amrex::Real lo, amrex::Real hi,
142  bool normal_face, bool use_most, bool snap) noexcept
143 {
144  HorizMasks m{};
145  // With the snap every solid face has t = 1: a roof face lies in the top solid row
146  // (t <= below, the face above fluid), and a face carrying a wall law is not also an
147  // interior face. A wall-normal face gets the interior drag (no penetration), not a wall law.
148  m.roof = (!normal_face && t > zero && (snap ? t <= below : t < below) && above == zero && use_most) ? one : zero;
149  m.lo = (!normal_face && t > zero && t <= hi && lo == zero && use_most) ? one : zero;
150  m.hi = (!normal_face && t > zero && t <= lo && hi == zero && use_most) ? one : zero;
151  m.wall = (t > zero && t < one && !use_most) ? one : zero; // all walls without MOST
152  m.side = (t > zero && t < one && use_most && m.most() == zero) ? one : zero; // partial faces not under MOST
153  m.interior = (t == one && !(snap && m.most() > zero)) ? one : zero;
154  return m;
155 }

179  * @param above_floor The face is not the domain bottom (k >= 1)
180  * @param normal_face With the snap, a face joining a solid and a fluid cell (wall-normal)

183  */
184 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
185 VertMasks vert_masks (amrex::Real t, amrex::Real above,
186  amrex::Real south, amrex::Real north, amrex::Real west, amrex::Real east,
187  bool above_floor, bool normal_face, bool use_most, bool snap) noexcept
188 {
189  VertMasks m{};
190  m.south = (!normal_face && t > zero && t <= north && south == zero && use_most && above_floor) ? one : zero;
191  m.north = (!normal_face && t > zero && t <= south && north == zero && use_most && above_floor) ? one : zero;
192  m.west = (!normal_face && t > zero && t <= east && west == zero && use_most && above_floor) ? one : zero;
193  m.east = (!normal_face && t > zero && t <= west && east == zero && use_most && above_floor) ? one : zero;
194  m.wall = (t > zero && t < one && !use_most) ? one : zero;
195  m.roof = (!normal_face && t > zero && above == zero && use_most) ? one : zero;
196  // with the snap a face carrying a wall law or the roof drag is not also an interior face
197  m.interior = (t == one && !(snap && (m.most() + m.roof) > zero)) ? one : zero;
198  return m;
199 }

207 /**
208  * Weight of the drag the implicit projection applies on a face (erf.if_implicit_projection):
209  * the linear-drag weight of the kernel, and the full rate on every face the explicit path zeroes
210  * after the projection (raw blanking 1, which the kernels leave without a wall law). The latter
211  * also caps the vertical roof face of an exact box without the snap, where the explicit kernel
212  * adds the roof and the interior drag together (weight 2 t), at one.
213  */
214 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
215 amrex::Real implicit_weight (amrex::Real linear, amrex::Real raw) noexcept
216 {
217  return (raw == one) ? amrex::Real(one) : linear;
