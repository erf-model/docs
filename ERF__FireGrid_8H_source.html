24  * refinement factor and (nx, ny) is the size of the region the atmospheric

71 /**
72  * Exchange the ghost cells of a fire-grid MultiFab and fill the ones outside a
73  * non-periodic face of the domain by copying the nearest interior value.
74  *
75  * FillBoundary alone leaves those cells untouched, so with an inflow-outflow
76  * atmosphere (the fire grid inherits its periodicity) the gradient stencil of
77  * the level set read whatever the allocator had left there. Whether that was
78  * benign depended on the box layout: one box per rank happened to run, two
79  * boxes on a rank blew the level set up at the inflow face within three steps.
80  * Every fire-grid exchange goes through here.
81  */
82 inline void fire_fill_boundary (amrex::MultiFab& mf, const amrex::Geometry& geom)
83 {
84  mf.FillBoundary(geom.periodicity());
85  const auto ng = mf.nGrowVect();
86  const bool px = geom.isPeriodic(0) || ng[0] == 0;
87  const bool py = geom.isPeriodic(1) || ng[1] == 0;
88  const bool pz = geom.isPeriodic(2) || ng[2] == 0;
89  if (px && py && pz) { return; } // nothing outside a non-periodic face to fill
90  const amrex::Box& dom = geom.Domain();
91  const auto dlo = dom.smallEnd();
92  const auto dhi = dom.bigEnd();
93  const int ncomp = mf.nComp();
94  for (amrex::MFIter mfi(mf); mfi.isValid(); ++mfi) {
95  const amrex::Box& gbx = mfi.fabbox();
96  if (dom.contains(gbx)) { continue; }
97  auto a = mf.array(mfi);
98  amrex::ParallelFor(gbx, ncomp, [=] AMREX_GPU_DEVICE (int i, int j, int k, int n) noexcept
99  {
100  int ic = i, jc = j, kc = k;
101  if (!px) { ic = amrex::min(amrex::max(i, dlo[0]), dhi[0]); }
102  if (!py) { jc = amrex::min(amrex::max(j, dlo[1]), dhi[1]); }
103  if (!pz) { kc = amrex::min(amrex::max(k, dlo[2]), dhi[2]); }
104  if (ic != i || jc != j || kc != k) { a(i, j, k, n) = a(ic, jc, kc, n); }
105  });
106  }
107 }
