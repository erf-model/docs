23 /**
24  * @brief Fill the halo of a coarse 2D surface field before it is interpolated.
25  *
26  * FillBoundary sets the interior and periodic ghost columns. Across a non-periodic face a
27  * ghost column takes the nearest column inside the domain: the slope stencil of the
28  * interpolater reads it for every boundary type. A periodic ghost column is left as
29  * FillBoundary set it. InterpFromCoarseLevel reads it twice, from this halo and from the
30  * periodic image of the valid column, and the two must agree.
31  */
32 inline void
33 seb_extend_halo (amrex::MultiFab& mf, const amrex::Geometry& geom)
34 {
35  using amrex::Real;
36  mf.FillBoundary(geom.periodicity());
37  const bool clamp_x = !geom.isPeriodic(0);
38  const bool clamp_y = !geom.isPeriodic(1);
39  const auto dlo = amrex::lbound(geom.Domain());
40  const auto dhi = amrex::ubound(geom.Domain());
41  for (amrex::MFIter mfi(mf, amrex::TilingIfNotGPU()); mfi.isValid(); ++mfi) {
42  const amrex::Box& gbx = mfi.growntilebox();
43  const amrex::Array4<Real>& a = mf.array(mfi);
44  amrex::ParallelFor(gbx, mf.nComp(), [=] AMREX_GPU_DEVICE (int i, int j, int k, int n) noexcept
45  {
46  const int ic = clamp_x ? amrex::max(dlo.x, amrex::min(dhi.x, i)) : i;
47  const int jc = clamp_y ? amrex::max(dlo.y, amrex::min(dhi.y, j)) : j;
48  if (ic != i || jc != j) { a(i,j,k,n) = a(ic,jc,k,n); }
49  });
50  }

53 /**
54  * @brief Fill @p fine from @p crse when the coarse land mask holds water.
55  *
56  * A water column holds a value the balance never updated. The fill keeps it out of the
57  * land columns:
58  *
59  * - A fine column whose parent and the parent's eight neighbors are all land takes the
60  * conservative linear interpolation, as on a level with no water.
61  * - Elsewhere no slope is used: the slope of a coarse column next to water is formed with
62  * the held value. A fine land column over a coarse land column takes the parent's value.
63  * A fine land column over a coarse water column takes the mean of the land columns among
64  * the parent's eight neighbors.
65  * - A fine land column whose parent and all eight neighbors are water has no land source.
66  * It takes the parent's held value and is counted.
67  * - A fine water column in that region takes its parent's value.
68  *
69  * Neighbors beyond a non-periodic domain face are left out. RemakeLevel copies the old
70  * fine values on top afterwards, so only columns new to the level keep this fill.
71  *
72  * @param fine the fine field; valid and ghost columns inside the domain are set
73  * @param crse the coarse field, after seb_extend_halo()
74  * @param crse_mask the coarse land mask, ghost columns filled, at least one ghost column
75  * @param fine_mask the fine land mask with the ghost columns of @p fine, or nullptr (all land)
76  * @param bcs one boundary record per component of @p fine
77  * @return the number of fine land columns with no land source on this rank (not reduced)
78  */
