39 /**
40  * Read one number from line n of a text file that holds one number per line.
41  * Aborts, naming the file and the line, on a line that is missing, is not a
42  * number (a whole number for T = int), or has text after the number.
43  */
44 template <typename T>
45 T erf_get_single_value (std::istream& is, int n, const std::string& fname)
46 {
47  T r = 0;
48  std::string line;
49  if (std::getline(is, line)) {
50  std::stringstream ss(line);
51  ss >> r;
52  if (ss.fail()) {
53  amrex::Abort("Terrain file " + fname + ": line " + std::to_string(n)
54  + " is not a number: '" + line + "'");
55  }
56  std::string extra;
57  if (ss >> extra) {
58  if (std::is_integral_v<T> && (extra[0] == '.' || extra[0] == 'e' || extra[0] == 'E')) {
59  amrex::Abort("Terrain file " + fname + ": line " + std::to_string(n) + " must be a whole"
60  + " number (nx or ny): '" + line + "'");
61  }
62  amrex::Abort("Terrain file " + fname + ": line " + std::to_string(n) + " has text after"
63  + " the value: '" + line + "'; the file has one value per line, with no"
64  + " other text");
65  }
66  } else if (n <= 2) {
67  amrex::Abort("Terrain file " + fname + ": the file ends at line " + std::to_string(n-1)
68  + "; lines 1 and 2 must give nx and ny");
69  } else {
70  amrex::Abort("Terrain file " + fname + ": the file ends at line " + std::to_string(n-1)
71  + ", but nx and ny ask for nx + ny + nx*ny values after them");
72  }
73  return r;
74 }
