7 /**
8  * Terrain-metric part of the vertical diffusion on a terrain-fitted mesh, for
9  * erf.implicit_terrain_metric.
10  *
11  * The flux through a zeta face projects the horizontal fluxes onto the face
12  * normal. Their h*d/dz parts add a vertical-diffusion term K_h * M * d/dz to the
13  * zeta flux, with the factors below. The explicit operator forms it from
14  * averaged horizontal fluxes; with the option on, the same term in compact form
15  * (face differences in zeta) is also put into the implicit vertical solve and
16  * subtracted explicitly at the state the explicit terms see, so the stages
17  * with an implicit solve treat the K_h * M * d2/dz2 stiffness implicitly. The
18  * factors are the exact du/dz (dphi/dz) content of the projected fluxes, so for
19  * smooth fields on a uniform slope the explicit remainder (averaged minus
20  * compact) is small; at the grid scale it is the whole compact term with the
21  * opposite sign, cancelled within the stage by the implicit solve, which is why
22  * every stage needs the solve (SolverChoice::check_params).
23  *
24  * h_xi and h_eta are the terrain slopes dz/dxi and dz/deta at the face, mx and my
25  * the map factors there.
26  */
27 
28 /**
29  * Momentum: the zeta flux of u holds -0.5 * mu_h_tot * M_u * du/dz with
30  * M_u = (4/3) (h_xi mx)^2 + (h_eta my)^2, from h_xi (S11 - er/3) and h_eta S12:
31  * S11 carries -h_xi mx du/dz, and the expansion rate er carries the same term
32  * through Omega, so the deviatoric S11 - er/3 carries -(2/3) h_xi mx du/dz. For
33  * v, M_v = (h_xi mx)^2 + (4/3) (h_eta my)^2, from h_xi S12 and h_eta (S22 - er/3).
34  *
35  * @param[in] stagdir 0 for u, 1 for v
36  */
37 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
38 amrex::Real
39 TerrainMetricMomFactor (const int stagdir,
40  const amrex::Real h_xi, const amrex::Real h_eta,
41  const amrex::Real mx, const amrex::Real my) noexcept
42 {
43  const amrex::Real ax = h_xi * mx;
44  const amrex::Real ay = h_eta * my;
45  const amrex::Real four_thirds = amrex::Real(4.0) / amrex::Real(3.0);
46  return (stagdir == 0) ? four_thirds*ax*ax + ay*ay
47  : ax*ax + four_thirds*ay*ay;
48 }
