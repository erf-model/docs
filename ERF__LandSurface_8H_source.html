321  Plot_Lsm_Data (const PlotFileType plotfile_type,

325  {
326  // Note we must use finest_level here rather than m_lsm_model.size(), which is
327  // (max_level+1) -- the models at levels that have not (yet) been created
328  // have been allocated but not Define'd or Init'ed, so their data pointers

332  int nvar = this->Get_Data_Size();

334 
335  // Only write if we have valid pointers
336  if (this->Get_Data_Ptr(0,0)) {
337  varnames.resize(nvar);
338  m_lsm_geom_lev.resize(nlev);
339  m_lsm_data_lev.resize(nlev);
340  std::string plotfilename = amrex::Concatenate(plot_file_lsm, level_steps[0], 5);
341  for (int lev = 0; lev <= finest_level; ++lev) {
342  m_lsm_geom_lev[lev] = this->Get_Lsm_Geom(lev);
343 
344  if (m_lsm_geom_lev[lev].Domain().isEmpty()) {
345  nlev_valid--;
346  continue;
347  }
348 
349  amrex::MultiFab* mf_lsm = this->Get_Data_Ptr(lev,0);
350  amrex::IntVect ng(0,0,0);
351  amrex::BoxArray ba = mf_lsm->boxArray();
352  amrex::DistributionMapping dm = mf_lsm->DistributionMap();
353  m_lsm_data_lev[lev].define(ba, dm, nvar, ng);
354  for (int n(0); n<nvar; ++n) {
355  mf_lsm = this->Get_Data_Ptr(lev,n);
356  amrex::MultiFab::Copy(m_lsm_data_lev[lev],*(mf_lsm),0,n,1,0);
357  if (lev==0) varnames[n] = this->Get_DataName(n);
358  }
359  }
360  if (plotfile_type == PlotFileType::Amrex
361 #ifdef ERF_USE_NETCDF
362  || plotfile_type == PlotFileType::Netcdf
363 #endif
364  ) {
365 #ifdef ERF_USE_NETCDF
366  // NetCDF output of the land-surface fields is not implemented yet, so fall back
367  // to the AMReX format -- which is what these plotfiles were before the plotfile
368  // type was consulted here -- rather than silently writing nothing at all.
369  warn_nc_lsm_plot_fallback(plotfile_type);
370 #endif
371  WriteMultiLevelPlotfile (plotfilename, nlev_valid, GetVecOfConstPtrs(m_lsm_data_lev),
372  varnames, m_lsm_geom_lev, static_cast<amrex::Real>(time),
373  level_steps, ref_ratio);
374  } else {

379  m_lsm_data_lev.clear();
380  }
381

383  {

390  }
