42  const amrex::Array4<const amrex::Real>& states,

44  const amrex::Array4<amrex::Real>& rho,
45  const amrex::Array4<amrex::Real>& theta,
46  const amrex::Array4<amrex::Real>& qv,
47  const amrex::Array4<amrex::Real>& qc,
48  const amrex::Array4<amrex::Real>& qp,
49  const amrex::Array4<amrex::Real>& qt,
50  const amrex::Array4<amrex::Real>& tabs,
51  const amrex::Array4<amrex::Real>& pres,
52  const amrex::Real rdOcp,
53  const bool use_anelastic_reference_pressure,
54  const int i, const int j, const int k) noexcept
55 {
56  const amrex::Real rho_value = states(i,j,k,Rho_comp);
57  const amrex::Real rho_theta = states(i,j,k,RhoTheta_comp);
58  const amrex::Real qv_value = states(i,j,k,RhoQ1_comp) / rho_value;
59 
60  rho(i,j,k) = rho_value;
61  theta(i,j,k) = rho_theta / rho_value;
62  qv(i,j,k) = qv_value;
63  qc(i,j,k) = states(i,j,k,RhoQ2_comp) / rho_value;
64  qp(i,j,k) = states(i,j,k,RhoQ3_comp) / rho_value;
65  qt(i,j,k) = qv(i,j,k) + qc(i,j,k);
66 
67  const amrex::Real p0 = use_anelastic_reference_pressure
68  ? base(i,j,k,BaseState::p0_comp) : amrex::Real(0.0);
69  const MicrophysicsThermoState thermo = diagnose_microphysics_thermo_state(
70  rho_value, rho_theta, qv_value, rdOcp,
71  use_anelastic_reference_pressure, p0);
72  tabs(i,j,k) = thermo.temperature;
73  // Kessler's saturation helpers use mbar / hPa.
74  pres(i,j,k) = amrex::Real(0.01) * thermo.pressure_pa;
75 }
