Detailed Description

The two-stream radiation model (erf.radiation_model = TwoStream).

Owns everything the model keeps between steps: the per-level surface property fields it falls back on when no land-surface model supplies them, the surface-energy-balance (SEB) fields and force-restore state, and the domain-mean flux diagnostics of the last column sweep. ERF holds one instance and calls it only after testing solverChoice.rad_type == RadiationType::TwoStream; every method here checks the same predicate and returns without acting otherwise.

Lifecycle:

  • resize(nlevs) once at construction (before the inputs are read);
  • define_level(lev, radChoice, ba2d, dm) whenever a level is (re)built;
  • advance(...) twice per step: "pre_dycore" runs the column sweep on the old state and writes the heating rates, "post_dycore" reuses that sweep's flux diagnostics and advances the SEB state by dt_step;
  • write_checkpoint / read_checkpoint for the prognostic surface state.

The heating-rate MultiFab itself stays in ERF, since RRTMGP writes the same 2-component (SW, LW) array and the RhoTheta source term and the plotfile read it without caring which model filled it.

671  Array4<amrex::Real> lw_sfc_out;

758  amrex::Real sw_flux_cloudy = 0.0;
759  amrex::Real sw_up_cloudy = 0.0;

764  i, j, bx, dz_uniform_lev, state_arr, ts_params, /*cloudy=*/true,

787  // into qheating_clear_arr (which is the real output

794  amrex::Real q_cloudy_val = qheating_cloudy_arr(i, j, k, comp);
795  qheating_clear_arr(i, j, k, comp) =

797  cloud_fraction * q_cloudy_val;
798  }
799  }

801  // nlev + 1 levels: the top interface sits at kmax + 1
802  for (int k = kmin; k <= kmax + 1; ++k) {
803  for (int comp = 0; comp < 4; ++comp) {
804  rad_flux_clear_arr(i, j, k, comp) =
805  (1.0 - cloud_fraction) * rad_flux_clear_arr(i, j, k, comp) +
806  cloud_fraction * rad_flux_cloudy_arr(i, j, k, comp);
807  }
808  }
809  }

815  if (lw_flux_from_rad) lw_sfc_out(i, j, 0) = -lw_net_col;

817  // The incident TOA flux is the same for both evaluations.
818  // Return tuple for reduction
819  return {max_heating_col, sw_flux_col, sw_up_col, lw_net_col, lw_up_col, sw_toa_clear};
820  }

826  max_heating_box = amrex::get<0>(reduce_tuple);
827  sw_sum_box = amrex::get<1>(reduce_tuple);
828  sw_up_sum_box = amrex::get<2>(reduce_tuple);
829  lw_sum_box = amrex::get<3>(reduce_tuple);
830  lw_up_sum_box = amrex::get<4>(reduce_tuple);
831  sw_toa_sum_box = amrex::get<5>(reduce_tuple);
832

835  sw_surface_sum += sw_sum_box;
836  sw_up_toa_sum += sw_up_sum_box;
837  lw_net_sum += lw_sum_box;
838  lw_up_toa_sum += lw_up_sum_box;
839  sw_toa_sum += sw_toa_sum_box;
840  }

843  // not change with the decomposition. These are collective calls; the
844  // conditions around them are input-driven and identical on all ranks.
845  if (do_sweep) {

847  ParallelDescriptor::ReduceRealSum(sums, 5);
848  sw_surface_sum = sums[0]; sw_up_toa_sum = sums[1];
849  lw_net_sum = sums[2]; lw_up_toa_sum = sums[3];
850  sw_toa_sum = sums[4];
851  ParallelDescriptor::ReduceLongSum(n_columns_total);
852  ParallelDescriptor::ReduceRealMax(max_heating_global);
853  }
854  } // do_sweep

857  // non-finite value here becomes a non-finite state one step later.
858  if (do_sweep && qheating_mf != nullptr &&

860  amrex::Abort("TwoStreamRadiation: non-finite heating rate after the column sweep at level " +
861  std::to_string(lev) + ", step " + std::to_string(nstep));

909  amrex::Real lw_net = lw_flux_arr(i, j, 0);

1008  amrex::Real lw_net = lw_flux_arr(i, j, 0);

1023

1029  t_s_new = amrex::max(t_min, amrex::min(t_max, t_s_new));

1031

1033  t_s_arr(i, j, 0) = t_s_new;
1034  q_s_arr(i, j, 0) = q_s_new;

1045  q_s_sum_box = amrex::get<2>(prog_reduce_tuple);
1046  q_s_max_box = amrex::get<3>(prog_reduce_tuple);
1047

1049  t_s_sum += t_s_sum_box;
1050  t_s_max_val = std::max(t_s_max_val, t_s_max_box);
1051  q_s_sum += q_s_sum_box;
1052  q_s_max_val = std::max(q_s_max_val, q_s_max_box);
1053  n_prog_columns += n_cols_box;
1054  }

1058  // corrupted (a bad restart file, or an overwrite elsewhere).
