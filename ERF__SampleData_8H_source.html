475  /**
476  * Turn the ends of the lines given by latitude and longitude into domain coordinates,
477  * with the level-0 latitude/longitude map of the run, and check them. A vertical line
478  * gives one lat/lon at both ends. The two ends of a horizontal line are in general not
479  * on one row of cells: they must lie within one cell of each other across the line,
480  * which is then sampled along the line of cells through its lo end. Collective.
481  */
482  void
483  resolve_latlon (const LatLonMap& latlon_map, const amrex::Geometry& geom0)
484  {
485  const int nline = static_cast<int>(m_bnd_rbx.size());
486  for (int iline(0); iline<nline; ++iline) {
487  const int dir = m_dir[iline];
488  const std::string line = "Line " + std::to_string(iline) + " of erf.sample_line_lo/hi_latlon";
489  const amrex::Real* req[2] = {&m_latlon_lo[3*iline], &m_latlon_hi[3*iline]};
490 
491  if (dir == 2) {
492  if (req[0][0] != req[1][0] || req[0][1] != req[1][1]) {
493  amrex::Abort(line + " is vertical (erf.sample_line_dir = 2), so its two ends"
494  " must give the same latitude and longitude");
495  }
496  if (req[0][2] > req[1][2]) {
497  amrex::Abort(line + " has its lo end above its hi end");
498  }
499  } else if (req[0][2] != req[1][2]) {
500  amrex::Abort(line + " is horizontal, so its two ends must give the same z");
501  }
502 
503  // A vertical line has one lat/lon
504  const int nend = (dir == 2) ? 1 : 2;
505  LatLonLocation where[2];
506  for (int e(0); e<nend; ++e) {
507  where[e] = latlon_map.locate_or_abort(req[e][0], req[e][1],
508  line + (e == 0 ? ", lo end," : ", hi end,"));
509  }
510  if (dir == 2) { where[1] = where[0]; }
511 
512  amrex::Real lo[AMREX_SPACEDIM] = {where[0].x, where[0].y, req[0][2]};
513  amrex::Real hi[AMREX_SPACEDIM] = {where[1].x, where[1].y, req[1][2]};
514  amrex::Real snap = amrex::Real(0.0);
515  if (dir != 2) {
516  const int across = 1 - dir;
517  snap = hi[across] - lo[across];
518  const amrex::Real gap = std::abs(snap) / geom0.CellSize(across);
519  if (gap > amrex::Real(1.0)) {
520  amrex::Abort(line + " is not along erf.sample_line_dir: its ends are " +
521  std::to_string(gap) + " cells apart in direction " +
522  std::to_string(across) + ", and a line between two lat/lon"
523  " points is sampled along one grid direction, so its ends may"
524  " be at most one cell apart across it");
525  }
526  if (hi[dir] < lo[dir]) {
527  amrex::Abort(line + " has its lo end beyond its hi end in direction " +
528  std::to_string(dir) + "; swap the two ends");
529  }
530  hi[across] = lo[across];
531  }
532  m_bnd_rbx[iline] = amrex::RealBox(lo, hi);
533  if (m_interp) { check_interp_heights(iline); }
534 
535  amrex::Print() << line << ": lo end at lat=" << where[0].lat << " long=" << where[0].lon
536  << " (x=" << lo[0] << " y=" << lo[1] << ")";
537  if (dir != 2) {
538  amrex::Print() << ", hi end at lat=" << where[1].lat << " long=" << where[1].lon
539  << " (x=" << hi[0] << " y=" << hi[1] << "), sampled along the line of cells"
540  << " through the lo end, " << std::abs(snap) << " m from the hi end";
541  }
542  amrex::Print() << std::endl;
543  }
544  }

639  /**
640  * The samples of line iline on a level (erf.sample_line_interp). Each sample is in a column of
641  * cells of the level: the one that holds the line (vertical), or each one it crosses
642  * (horizontal). Its height is the requested one, above sea level or, with
643  * erf.sample_line_z_agl, above the ground under the column. The ground and the top of a
644  * column are those of the level, the means of the four nodes under and over it, where the
645  * level holds the column's bottom or top cell; elsewhere the ground is the level-0 ground,
646  * bilinear at the column's centre (a level that stops above the ground). Its value is
647  * linear in height between the two cell centres of the column that bracket it; between the
648  * ground and the lowest cell centre with fluid it is that cell's, and between the top cell
649  * centre and the top the top cell's. A sample below the ground or above the top is
650  * missing. The level holds the line if it holds every cell the samples read; level 0
651  * always does. Collective: every rank gets the same plan.
652  *
653  * @param[in] z_nd the nodal heights of the level, or nullptr (the top is then prob_hi)
654  * @param[in] volfrac the volume fraction of the level with immersed terrain, or nullptr
655  * @param[in] ground the ground of the level (z_nd, or the immersed terrain surface), or nullptr
656  * @param[in] ground0 the ground of level 0, or nullptr (the ground is then prob_lo)
657  */
658  InterpPlan
659  make_interp_plan (int iline, int lev, const amrex::Geometry& geom, const amrex::Geometry& geom0,
660  const amrex::MultiFab& z_cc, const amrex::MultiFab* z_nd,
661  const amrex::MultiFab* volfrac, const amrex::MultiFab* ground,
662  const amrex::MultiFab* ground0) const
663  {
664  InterpPlan plan;
665  plan.ba = z_cc.boxArray();
666  plan.cols = lineIndexBoxXY(m_bnd_rbx[iline], geom);
667  const amrex::Box& dom = geom.Domain();
668  const int kdlo = dom.smallEnd(2);
669  const int nz = dom.length(2);
670  const int ncol = static_cast<int>(plan.cols.numPts());
671  const int dir = m_dir[iline];
672  const int along = (dir == 2) ? 0 : dir;
673  const amrex::IntVect clo = plan.cols.smallEnd();
674 
675  // The heights of the cell centres of the columns, which cells the level holds, and their
676  // volume fractions, from the bottom of the domain to its top
677  amrex::Box colbox(plan.cols);
678  colbox.setSmall(2, kdlo);
679  colbox.setBig(2, dom.bigEnd(2));
680  amrex::MultiFab t(z_cc.boxArray(), z_cc.DistributionMap(), 3, 0);
681  amrex::MultiFab::Copy(t, z_cc, 0, 0, 1, 0);
682  t.setVal(amrex::Real(1.0), 1, 1, 0);
683  if (volfrac != nullptr) {
684  amrex::MultiFab::Copy(t, *volfrac, 0, 2, 1, 0);
685  } else {
686  t.setVal(amrex::Real(1.0), 2, 1, 0);
687  }
688  const amrex::Vector<amrex::Real> c = gather_box(t, 0, 3, colbox, true);
689  const auto ccol = static_cast<std::size_t>(colbox.numPts());
690  auto at = [&] (int icol, int k, int n) {
691  return c[static_cast<std::size_t>(icol) + static_cast<std::size_t>(ncol) * k + ccol * n];
692  };
693  auto held_at = [&] (int icol, int k) { return at(icol, k, 1) > amrex::Real(0.5); };
694 
695  // The mean of the four nodes at node level k of mf under or over each column of the level
696  auto column_node_mean = [&] (const amrex::MultiFab& mf, int k) {
697  amrex::IntVect nlo(clo); nlo[2] = k;
698  amrex::IntVect nhi(plan.cols.bigEnd() + amrex::IntVect(1,1,0)); nhi[2] = k;
699  const amrex::Box nbx(nlo, nhi, amrex::IntVect::TheNodeVector());
700  const amrex::Vector<amrex::Real> g = gather_box(mf, 0, 1, nbx, true);
701  const int nnx = nbx.length(0);
702  amrex::Vector<amrex::Real> mean(ncol);
703  for (int icol(0); icol<ncol; ++icol) {
704  amrex::IntVect cell(clo);
705  cell[along] += icol;
706  const int i = cell[0] - nlo[0], j = cell[1] - nlo[1];
707  mean[icol] = amrex::Real(0.25) * (g[i + nnx*j] + g[i+1 + nnx*j] + g[i + nnx*(j+1)] + g[i+1 + nnx*(j+1)]);
708  }
709  return mean;
710  };
711 
712  // The ground under the centre of each column, bilinear between the level-0 nodes around
713  // it: for the columns whose bottom the level does not hold
714  amrex::Vector<amrex::Real> ground_col(ncol, geom0.ProbLo(2));
715  if (ground0 != nullptr) {
716  const amrex::Box& dom0 = geom0.Domain();
717  amrex::Vector<int> i0(ncol), j0(ncol);
718  amrex::Vector<amrex::Real> fx(ncol), fy(ncol);
719  for (int icol(0); icol<ncol; ++icol) {
720  amrex::IntVect cell(clo);
721  cell[along] += icol;
722  int* idx[2] = {&i0[icol], &j0[icol]};
723  amrex::Real* f[2] = {&fx[icol], &fy[icol]};
724  for (int d(0); d<2; ++d) {
725  const amrex::Real xc = geom.ProbLo(d) + (cell[d] + amrex::Real(0.5)) * geom.CellSize(d);
726  const amrex::Real s = (xc - geom0.ProbLo(d)) / geom0.CellSize(d);
727  *idx[d] = std::min(std::max(static_cast<int>(std::floor(s)), dom0.smallEnd(d)), dom0.bigEnd(d));
728  *f[d] = std::min(std::max(s - *idx[d], amrex::Real(0.0)), amrex::Real(1.0));
729  }
730  }
731  const amrex::IntVect nlo(*std::min_element(i0.begin(), i0.end()), *std::min_element(j0.begin(), j0.end()),
732  dom0.smallEnd(2));
733  const amrex::IntVect nhi(*std::max_element(i0.begin(), i0.end()) + 1,
734  *std::max_element(j0.begin(), j0.end()) + 1, dom0.smallEnd(2));
735  const amrex::Box nbx(nlo, nhi, amrex::IntVect::TheNodeVector());
736  const amrex::Vector<amrex::Real> g = gather_box(*ground0, 0, 1, nbx, true);
737  const int nnx = nbx.length(0);
738  auto node = [&] (int i, int j) { return g[(i - nlo[0]) + nnx * (j - nlo[1])]; };
739  for (int icol(0); icol<ncol; ++icol) {
740  const int i = i0[icol], j = j0[icol];
741  const amrex::Real wx = fx[icol], wy = fy[icol];
742  ground_col[icol] = (amrex::Real(1.0) - wy) * ((amrex::Real(1.0) - wx) * node(i,j) + wx * node(i+1,j))
743  + wy * ((amrex::Real(1.0) - wx) * node(i,j+1) + wx * node(i+1,j+1));
744  }
745  }
746  // the level's own ground where it holds the column's bottom cell
747  if (ground != nullptr) {
748  const amrex::Vector<amrex::Real> own = column_node_mean(*ground, kdlo);
749  for (int icol(0); icol<ncol; ++icol) {
750  if (held_at(icol, 0)) { ground_col[icol] = own[icol]; }
751  }
752  }
753  // The top of each column, where the level holds its top cell; elsewhere no sample above
754  // the level's highest cell is bracketed, so the level does not hold it
755  amrex::Vector<amrex::Real> top_col(ncol, geom.ProbHi(2));
756  if (z_nd != nullptr) {
757  const amrex::Vector<amrex::Real> own = column_node_mean(*z_nd, dom.bigEnd(2) + 1);
758  for (int icol(0); icol<ncol; ++icol) {
759  if (held_at(icol, nz-1)) { top_col[icol] = own[icol]; }
760  }
761  }
762 
763  // The samples: the heights of a vertical line in its column, or one height in each column
764  const amrex::Vector<amrex::Real> h = interp_heights(iline);
765  const int nsamp = (dir == 2) ? static_cast<int>(h.size()) : ncol;
766  plan.zt.resize(nsamp);
767  plan.col.resize(nsamp);
768  plan.k0.assign(nsamp, 0);
769  plan.k1.assign(nsamp, 0);
770  plan.w.assign(nsamp, amrex::Real(0.0));
771  plan.missing.assign(nsamp, 0);
772  plan.holds = true;
773  int kmin = nz, kmax = -1;
774  for (int s(0); s<nsamp; ++s) {
775  const int icol = (dir == 2) ? 0 : s;
776  const amrex::Real hs = (dir == 2) ? h[s] : h[0];
777  plan.col[s] = icol;
778  plan.zt[s] = hs + (m_z_agl ? ground_col[icol] : amrex::Real(0.0));
779  const amrex::Real z = plan.zt[s];
780  if (z < ground_col[icol] || z > top_col[icol]) { plan.missing[s] = 1; continue; }
781 
782  auto held = [&] (int k) { return held_at(icol, k); };
783  auto fluid = [&] (int k) { return held(k) && at(icol, k, 2) > amrex::Real(0.0); };
784  auto zc = [&] (int k) { return at(icol, k, 0); };
785  int k0 = -1, k1 = -1;
786  amrex::Real w = amrex::Real(0.0);
787  // between two cell centres with fluid
788  for (int k(0); k<nz-1 && k0 < 0; ++k) {
789  if (fluid(k) && fluid(k+1) && zc(k) <= z && z < zc(k+1)) {
790  k0 = k; k1 = k+1; w = (z - zc(k)) / (zc(k+1) - zc(k));
791  }
792  }
793  // between the top cell centre and the top
794  if (k0 < 0 && fluid(nz-1) && z >= zc(nz-1)) { k0 = k1 = nz-1; }
795  // between the ground and the lowest cell centre with fluid, if the level holds the
796  // column up to it
797  if (k0 < 0) {
798  int kf = 0;
799  while (kf < nz && held(kf) && !fluid(kf)) { ++kf; }
800  if (kf < nz && fluid(kf) && z < zc(kf)) { k0 = k1 = kf; }
801  }
802  if (k0 < 0) {
803  // Level 0 holds every cell, so there no cells with fluid bracket the height (a
804  // column without fluid, or a height under an overhang of immersed terrain); a
805  // finer level does not hold the cells around it
806  if (lev == 0) { plan.missing[s] = 1; continue; }
807  plan.holds = false;
808  return plan;
809  }
810  plan.k0[s] = k0;
811  plan.k1[s] = k1;
812  plan.w[s] = w;
813  kmin = std::min(kmin, k0);
814  kmax = std::max(kmax, k1);
815  }
816  if (kmax >= kmin) {
817  plan.cells = plan.cols;
818  plan.cells.setSmall(2, kdlo + kmin);
819  plan.cells.setBig(2, kdlo + kmax);
820  }
821  return plan;
822  }

850  * one box (interp_sample_box) on the I/O rank. z is the height of each sample above sea
851  * level; a missing sample has volume fraction 0 and erf.sampling_covered_value elsewhere.
852  * Collective.
853  */
854  amrex::MultiFab
855  interp_line_data (int iline, const InterpPlan& plan, const amrex::MultiFab& data, int z_comp, int vf_comp) const
856  {
857  const int ncomp = data.nComp();
858  const int nsamp = static_cast<int>(plan.zt.size());
859  amrex::Vector<amrex::Real> d;
860  if (plan.cells.ok()) { d = gather_box(data, 0, ncomp, plan.cells, false); }
861 
862  const amrex::Box sbx = interp_sample_box(iline, plan.cols);
863  const int io = amrex::ParallelDescriptor::IOProcessorNumber();
864  amrex::MultiFab out(amrex::BoxArray(sbx), amrex::DistributionMapping(amrex::Vector<int>{io}), ncomp, 0);
865  if (amrex::ParallelDescriptor::IOProcessor()) {
866  const int ncol = static_cast<int>(plan.cols.numPts());
867  const auto ncell = static_cast<std::size_t>(plan.cells.ok() ? plan.cells.numPts() : 0);
868  const int kc0 = plan.cells.ok() ? plan.cells.smallEnd(2) - plan.cols.smallEnd(2) : 0;
869  auto cell = [&] (int icol, int k, int n) {
870  return d[static_cast<std::size_t>(icol) + static_cast<std::size_t>(ncol) * (k - kc0) + ncell * n];
871  };
872  // in the order of the box's FArrayBox, whose samples run along its one long direction
873  amrex::Vector<amrex::Real> v(static_cast<std::size_t>(nsamp) * ncomp);
874  for (int n(0); n<ncomp; ++n) {
875  for (int s(0); s<nsamp; ++s) {
876  amrex::Real val;
877  if (n == z_comp) {
878  val = plan.zt[s];
879  } else if (plan.missing[s] != 0) {
880  val = (n == vf_comp) ? amrex::Real(0.0) : m_covered_value;
881  } else {
882  const amrex::Real w = plan.w[s];
883  val = (amrex::Real(1.0) - w) * cell(plan.col[s], plan.k0[s], n);
884  if (w > amrex::Real(0.0)) { val += w * cell(plan.col[s], plan.k1[s], n); }
885  }
886  v[static_cast<std::size_t>(s) + static_cast<std::size_t>(nsamp) * n] = val;
887  }
888  }
889  amrex::Gpu::copy(amrex::Gpu::hostToDevice, v.begin(), v.end(), out[0].dataPtr());
890  }
891  return out;

903  * The heights of the faces of the column through cell (i,j) on a level, from the nodal
904  * heights z_nd: the lower face of each cell, its upper face, and 1 where the level holds
905  * the cell (0 where it does not). Each is indexed from the bottom of the domain.
906  */
907  static amrex::Vector<amrex::Gpu::HostVector<amrex::Real>>
908  column_face_heights (const amrex::MultiFab& z_nd, const amrex::Geometry& geom,
909  const amrex::IntVect& cell)
910  {
911  const amrex::Box& dom = geom.Domain();
912  amrex::IntVect lo(cell); lo[2] = dom.smallEnd(2);
913  amrex::IntVect hi(cell); hi[2] = dom.bigEnd(2);
914  const amrex::Box col(lo, hi);
915 
916  amrex::MultiFab f(amrex::convert(z_nd.boxArray(), amrex::IntVect(0)),
917  z_nd.DistributionMap(), 3, 0);
918  f.setVal(amrex::Real(0.0));

935  * The height of the ground under cell (i,j) of level 0: the mean of the four nodes at the
936  * bottom of its column in ground_nd, which is z_phys_nd or, with immersed terrain, the
937  * terrain surface (ERF::station_ground)
938  */
939  static amrex::Real
940  column_ground (const amrex::MultiFab& ground_nd, const amrex::Geometry& geom,
941  const amrex::IntVect& cell)
942  {

944  amrex::ReduceOps<amrex::ReduceOpSum> reduce_op;

946  for (amrex::MFIter mfi(ground_nd); mfi.isValid(); ++mfi) {
947  // The nodal box that holds the cell, so that a node shared by two boxes counts once
948  const amrex::Box& vb = mfi.validbox();

950  j < vb.smallEnd(1) || j >= vb.bigEnd(1) ||

954  [=] AMREX_GPU_DEVICE (int ii, int jj, int kk) -> amrex::GpuTuple<amrex::Real> {
955  return { amrex::Real(0.25) * (z(ii,jj,kk) + z(ii+1,jj,kk) + z(ii,jj+1,kk) + z(ii+1,jj+1,kk)) };
956  });
957  }
958  amrex::Real ground = amrex::get<0>(reduce_data.value(reduce_op));
959  amrex::ParallelAllReduce::Sum(ground, amrex::ParallelContext::CommunicatorSub());

1007  /**
1008  * The cells of a line given by real coordinates on a level, from the face heights of the
1009  * column through its first point. A vertical line runs from the cell that holds the lower
1010  * z to the cell that holds the upper z (each clamped to the column); a horizontal line is
1011  * the row of cells at the level of cells that holds z in that column, so it follows the
1012  * terrain-fitted mesh. With erf.sample_line_z_agl, ground is added to z. The box is
1013  * empty if the level does not hold these cells of the column.
1014  */
1015  amrex::Box
1016  lineIndexBox (int iline, const amrex::Geometry& geom, const amrex::MultiFab& z_nd,
1017  amrex::Real ground)
1018  {
1019  const amrex::RealBox& rb = m_bnd_rbx[iline];
1020  const amrex::Box ibx = lineIndexBoxXY(rb, geom);
1021  const amrex::IntVect cell = ibx.smallEnd();
1022  const auto h = column_face_heights(z_nd, geom, cell);
1023  const auto& zl = h[0];
1024  const auto& zu = h[1];
1025  const auto& held = h[2];
1026  const int n = static_cast<int>(zl.size());
1027  const int kdlo = geom.Domain().smallEnd(2);
1028  const amrex::Real offset = m_z_agl ? ground : amrex::Real(0.0);
1029 
1030  // The cell (from the bottom) that holds height z, or -1 where the level does not hold it
1031  auto find_cell = [&] (amrex::Real z) -> int {
1032  if (held[0] > amrex::Real(0.5) && z < zl[0]) { return 0; }
1033  if (held[n-1] > amrex::Real(0.5) && z >= zu[n-1]) { return n-1; }
1034  for (int m = 0; m < n; ++m) {
1035  if (held[m] > amrex::Real(0.5) && zl[m] <= z && z < zu[m]) { return m; }
1036  }
1037  return -1;
1038  };
1039  const int mlo = find_cell(rb.lo(2) + offset);
1040  const int mhi = (m_dir[iline] == 2) ? find_cell(rb.hi(2) + offset) : mlo;
1041  if (mlo < 0 || mhi < 0 || mhi < mlo) { return amrex::Box(); }
1042  for (int m = mlo; m <= mhi; ++m) {
1043  if (held[m] < amrex::Real(0.5)) { return amrex::Box(); }
1044  }
1045  amrex::IntVect lo(ibx.smallEnd()); lo[2] = kdlo + mlo;
1046  amrex::IntVect hi(ibx.bigEnd()); hi[2] = kdlo + mhi;
1047  return amrex::Box(lo, hi);
1048  }

1050  //! The cells of a real box in x and y on a level, with k = 0: its k comes from heights
1051  static amrex::Box
1052  lineIndexBoxXY (const amrex::RealBox& rb, const amrex::Geometry& geom)
1053  {
1054  amrex::IntVect lo, hi;
1055  for (int d(0); d<2; ++d) {
1056  lo[d] = static_cast<int>(std::floor((rb.lo(d) - geom.ProbLo(d))/geom.CellSize(d)));
1057  hi[d] = static_cast<int>(std::floor((rb.hi(d) - geom.ProbLo(d))/geom.CellSize(d)));
1058  lo[d] = std::max(lo[d], geom.Domain().smallEnd(d));
1059  hi[d] = std::min(hi[d], geom.Domain().bigEnd(d));
1060  }
1061  lo[2] = hi[2] = geom.Domain().smallEnd(2);
1062  return amrex::Box(lo, hi);
1063  }
