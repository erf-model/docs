22  * Factor by which to scale erf.fixed_dt (and erf.fixed_fast_dt) on a level so that the fixed
23  * step respects the CFL limit (erf-model/ERF#5238). The fixed step exceeds the limit when its
24  * Courant number fixed_dt * cfl_rate is above FixedDtCFLLimitValue(erf.cfl) = min(erf.cfl, 0.8),
25  * and is then scaled to exactly that Courant number, so the scaled step is continuous in the
26  * Courant number. Returns 1 when the fixed step is within the limit, or when there is no fixed
27  * step, no CFL limit (cfl <= 0) or no constraining rate on the level.
28  *
29  * @param[in] fixed_dt fixed time step of the level
30  * @param[in] cfl_rate inverse time step at a Courant number of one on the level (the rate the
31  * adaptive dt of estTimeStep is based on)
32  * @param[in] cfl erf.cfl
33  */
34 inline double
35 FixedDtCFLScale (double fixed_dt, double cfl_rate, double cfl)
36 {
37  if (fixed_dt <= 0.0 || cfl_rate <= 0.0 || cfl <= 0.0) { return 1.0; }
38  const double limit = FixedDtCFLLimitValue(cfl);
39  const double courant = fixed_dt * cfl_rate;
40  // Relative tolerance, so a step already set to the limit is not scaled again by round-off
41  if (courant <= limit * (1.0 + 1.e-10)) { return 1.0; }
42  return limit / courant;
43 }
