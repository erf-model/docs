130 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
131 NativeAggregate aggregate_native_bundle (const NativeBundle& bundle) noexcept
132 {
133  const amrex::Real f = bundle.vegetation_fraction;
134  // At the endpoints take the one tile in use, so that an unused tile's
135  // placeholder (which may be a sentinel) cannot leak in through 0 * value.
136  if (!native_vegetated_tile_is_used(f)) {
137  return {bundle.temperature_bare, bundle.mixing_ratio_bare};
138  }
139  if (!native_bare_tile_is_used(f)) {
140  return {bundle.temperature_vegetated, bundle.mixing_ratio_vegetated};
141  }
142  return {
143  f * bundle.temperature_vegetated + (amrex::Real(1.0)-f) * bundle.temperature_bare,
144  f * bundle.mixing_ratio_vegetated + (amrex::Real(1.0)-f) * bundle.mixing_ratio_bare
145  };
146 }

154 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
155 bool evaluate_most_factor (const MostState& state,
156  amrex::Real height,
157  amrex::Real& factor) noexcept
158 {
159  if (!valid_real(state.roughness_height) ||
160  !valid_real(state.obukhov_length) ||
161  state.roughness_height <= amrex::Real(0.0) ||
162  height <= state.roughness_height ||
163  state.obukhov_length == amrex::Real(0.0)) {
164  return false;
165  }
166  const similarity_funs sfuns{};
167  const amrex::Real zeta = amrex::min(height / state.obukhov_length,
168  most_profile_zeta_max());
169  factor = std::log(height / state.roughness_height) - sfuns.calc_psi_h(zeta);
170  return valid_real(factor);
171 }

173 // The momentum counterpart of evaluate_most_factor: ln(z/z0) - psi_m(z/L), so that
174 // the wind speed at height z is u_star / KAPPA times it.
175 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
176 bool evaluate_most_momentum_factor (amrex::Real roughness_height,
177  amrex::Real obukhov_length,
178  amrex::Real height,
179  amrex::Real& factor) noexcept
180 {
181  if (!valid_real(roughness_height) || !valid_real(obukhov_length) ||
182  roughness_height <= amrex::Real(0.0) ||
183  height <= roughness_height ||
184  obukhov_length == amrex::Real(0.0)) {
185  return false;
186  }
187  const similarity_funs sfuns{};
188  const amrex::Real zeta = amrex::min(height / obukhov_length,
189  most_profile_zeta_max());
190  factor = std::log(height / roughness_height) - sfuns.calc_psi_m(zeta);
191  return valid_real(factor);

194 // Evaluate ERF's integrated scalar profile without changing the production
195 // surface-layer state. The inputs use theta and mixing-ratio scales.
196 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
197 bool evaluate_most_profile (const MostState& state,
198  amrex::Real height,
199  MostProfile& profile) noexcept
200 {
201  if (!valid_real(state.theta_surface) || !valid_real(state.theta_star) ||
202  !valid_real(state.mixing_ratio_surface) || !valid_real(state.mixing_ratio_star)) {
203  return false;
204  }
205 
206  amrex::Real profile_factor = amrex::Real(0.0);
207  if (!evaluate_most_factor(state, height, profile_factor)) {
208  return false;
209  }
210 
211  profile.potential_temperature = state.theta_surface +
212  state.theta_star / KAPPA * profile_factor;
213  profile.mixing_ratio = state.mixing_ratio_surface +
214  state.mixing_ratio_star / KAPPA * profile_factor;
215  return valid_real(profile.potential_temperature) &&
216  valid_real(profile.mixing_ratio) &&
217  profile.potential_temperature > amrex::Real(0.0) &&
218  profile.mixing_ratio >= amrex::Real(0.0);
219 }
