43 /**
44  * Reconstruct a nodal field S from data T that lives at cell centers in (x,y),
45  * such that the four-node average of S matches T as closely as a *bounded*,
46  * smooth nodal field can.
47  *
48  * ---------------------------------------------------------------------------
49  * Why we do not simply invert the averaging operator
50  * ---------------------------------------------------------------------------
51  * The four-node averaging operator A,
52  *
53  * (A S)(i,j) = 1/4 [ S(i,j) + S(i+1,j) + S(i,j+1) + S(i+1,j+1) ] ,
54  *
55  * has symbol cos(kx/2) cos(ky/2), which vanishes at the grid Nyquist mode.
56  * Inverting A exactly -- e.g. with the back substitution
57  *
58  * S(i+1,j+1) = 4 T(i,j) - S(i,j) - S(i+1,j) - S(i,j+1)
59  *
60  * -- therefore amplifies grid-scale content of T without bound. The Green's
61  * function of that recursion is 4 / [(1+x)(1+y)], a doubly alternating
62  * cumulative sum, so a checkerboard component of T of amplitude a produces a
63  * nodal response of amplitude a*i*j. On a 128^2 grid an O(10 m) grid-scale
64  * roughness in the WRF terrain thus yields nodal heights that are kilometers
65  * away from the WRF terrain, including physically impossible negative
66  * elevations (ERF issue #3709). Adding a free homogeneous solution cannot
67  * repair this: the null space of A is spanned by (-1)^i f(j) + (-1)^j g(i),
68  * which cannot represent the (-1)^(i+j) i j growth of the particular solution.
69  * The failure is silent, because checking that avg4(S) reproduces T is then
70  * satisfied by construction.
71  *
72  * ---------------------------------------------------------------------------
73  * What we do instead
74  * ---------------------------------------------------------------------------
75  * Let S_ref be the direct interpolation of T to the nodes (makeReference) and
76  * R = T - A S_ref the part of T that interpolation fails to reproduce. We
77  * solve for a *correction* E to the interpolant,
78  *
79  * min_E || A E - R ||^2 + lambda || V E ||^2 + mu || E ||^2 , S = S_ref + E,
80  *
81  * where V is a roughness operator (first derivative or Laplacian). The normal
82  * equations
83  *
84  * ( A^T A + lambda V^T V + mu I ) E = A^T R
85  *
86  * are symmetric positive definite for any mu > 0, so E exists, is unique, and
87  * -- unlike the exact inverse -- is bounded. Specifically, writing
88  * ( A^T A + lambda V^T V + mu I ) E = A^T R and pairing with E gives
89  *
90  * || A E ||^2 + lambda || V E ||^2 + mu || E ||^2 <= || A E || || R || ,
91  *
92  * and since 2 sqrt(mu) ||A E|| ||E|| <= ||A E||^2 + mu ||E||^2, we get the
93  * a priori bound
94  *
95  * || E || <= || R || / ( 2 sqrt(mu) ) .
96  *
97  * The correction is therefore controlled by how badly interpolation already
98  * fails, never by the conditioning of A.
99  *
100  * ---------------------------------------------------------------------------
101  * Choosing mu
102  * ---------------------------------------------------------------------------
103  * A single fixed mu cannot serve both smooth and rough terrain: small mu
104  * recovers well-resolved features almost exactly but lets rough fields drift
105  * far from the interpolant, and large mu does the reverse. We therefore start
106  * from a small mu and raise it until the correction satisfies the pointwise cap
107  *
108  * max |E| <= min( 2 max|R| , 1/4 (max T - min T) ) ,
109  *
110  * i.e. the nodes may not move away from the direct interpolation by more than
111  * twice the interpolation misfit, nor by more than a quarter of the relief of
112  * the data. Because ||E|| -> 0 as mu -> infinity the loop always terminates,
113  * and on failure we fall back to the interpolant itself. Smooth terrain keeps
114  * the smallest mu and is reproduced to well below the interpolation error;
115  * rough terrain is pulled back toward the interpolant instead of blowing up.
116  *
117  * The averaging error max |avg4(S) - T| returned in SolveInfo is now a genuine
118  * diagnostic -- it is no longer satisfied by construction -- and is reported
119  * alongside the same quantity for the plain interpolant and the range of S, so
120  * callers can sanity check the result.
121  */

256  t_min = std::min(t_min, T_arr(i,j,0));
257  t_max = std::max(t_max, T_arr(i,j,0));
258  }
259  }

263  relief_factor * (t_max - t_min));

266  /**
267  * @brief Local storage for the right-hand side of the normal equations.
268  */

270  applyAT(misfit,rhs);

282  bool accepted = false;

284  SolveInfo attempt_info = conjugateGradient(rhs,E,relative_tolerance,max_iterations);
285  info.iterations += attempt_info.iterations;
286  info.final_residual = attempt_info.final_residual;
287  info.converged = attempt_info.converged;
288

292  m_mu *= mu_growth;
293  ++info.refinements;
294  }
295 
296  if (accepted) {
297  auto const S_arr = S.array();
298  auto const r_arr = reference.const_array();
299  auto const E_arr = E.const_array();
300  for (int j=m_jlo; j<=m_jhi+1; ++j) {
301  for (int i=m_ilo; i<=m_ihi+1; ++i) {
302  S_arr(i,j,0) = r_arr(i,j,0) + E_arr(i,j,0);
303  }
304  }
305  } else {
306  // Should not happen -- ||E|| -> 0 as mu -> infinity -- but if it
307  // does, the interpolant is a safe, bounded answer.
308  S.copy<amrex::RunOn::Host>(reference,0,0,1);
309  info.deviation = zero;
310  }
311  info.regularization = m_mu;
312 
313  computeDiagnostics(T,S,info);
314  return {std::move(S),info};

597  //! R = ( A^T A + lambda V^T V + mu I ) S
598  /**
599  * @brief Apply the full system operator M = A^T A + lambda V^T V + mu I.
600  * @param[in] S Nodal field.
601  * @param[out] R Resulting nodal field.
602  */
603  void applyM (amrex::FArrayBox const& S,
604  amrex::FArrayBox& R) const
605  {
606  applyA(S,m_cc_scratch);
607  applyAT(m_cc_scratch,R);
608  applyVariationOperator(S,m_nd_scratch);
609 
610  auto const R_arr = R.array();
611  auto const v_arr = m_nd_scratch.const_array();
612  auto const S_arr = S.const_array();
613  for (int j=m_jlo; j<=m_jhi+1; ++j) {
614  for (int i=m_ilo; i<=m_ihi+1; ++i) {
615  R_arr(i,j,0) += m_lambda * v_arr(i,j,0) + m_mu * S_arr(i,j,0);
616  }
617  }
618  }

620  /**
621  * Diagonal of M, assembled by accumulating the square of every stencil
622  * coefficient that touches a node. Used as a Jacobi preconditioner.
623  */
624  void computeDiagonal (amrex::FArrayBox& diag) const
625  {
626  diag.setVal<amrex::RunOn::Host>(m_mu);
627  auto const d_arr = diag.array();
628 
629  // A^T A: each incident cell contributes (1/4)^2
630  Real const qa = fourth * fourth;
631  for (int j=m_jlo; j<=m_jhi; ++j) {
632  for (int i=m_ilo; i<=m_ihi; ++i) {
633  d_arr(i ,j ,0) += qa;
634  d_arr(i+1,j ,0) += qa;
635  d_arr(i ,j+1,0) += qa;
636  d_arr(i+1,j+1,0) += qa;
637  }
638  }
639 
640  // lambda V^T V
641  if (m_var_op == VariationOperator::FirstDeriv) {
642  Real const qx = m_lambda * m_gx * m_gx;
643  Real const qy = m_lambda * m_gy * m_gy;
644  for (int j=m_jlo; j<=m_jhi+1; ++j) {
645  for (int i=m_ilo; i<=m_ihi; ++i) {
646  d_arr(i ,j,0) += qx;
647  d_arr(i+1,j,0) += qx;
648  }
649  }
650  for (int j=m_jlo; j<=m_jhi; ++j) {
651  for (int i=m_ilo; i<=m_ihi+1; ++i) {
652  d_arr(i,j ,0) += qy;
653  d_arr(i,j+1,0) += qy;
654  }
655  }
656  } else {
657  Real const qx = m_lambda * m_gxx * m_gxx;
658  Real const qy = m_lambda * m_gyy * m_gyy;
659  Real const qc = m_lambda * four * (m_gxx + m_gyy) * (m_gxx + m_gyy);
660  for (int j=m_jlo+1; j<=m_jhi; ++j) {
661  for (int i=m_ilo+1; i<=m_ihi; ++i) {
662  d_arr(i-1,j ,0) += qx;
663  d_arr(i+1,j ,0) += qx;
664  d_arr(i ,j-1,0) += qy;
665  d_arr(i ,j+1,0) += qy;
666  d_arr(i ,j ,0) += qc;
667  }
668  }
669  }
670  }

708  //! Jacobi-preconditioned conjugate gradient on the SPD operator M.
709  //! x is both the initial guess and the result; it is reset to zero here
710  //! because every attempt restarts the correction from the interpolant.
711  [[nodiscard]] SolveInfo conjugateGradient (amrex::FArrayBox const& rhs,
712  amrex::FArrayBox& x,
713  Real relative_tolerance,
714  int max_iterations) const
715  {
716  /**
717  * @brief Local storage for the operator diagonal.
718  */
719  amrex::FArrayBox diag(m_nodal_box,1,amrex::The_Managed_Arena());
720  computeDiagonal(diag);
721 
722  /**
723  * @brief Local storage for the residual.
724  */
725  amrex::FArrayBox r (m_nodal_box,1,amrex::The_Managed_Arena());
726  /**
727  * @brief Local storage for the preconditioned residual.
728  */
729  amrex::FArrayBox z (m_nodal_box,1,amrex::The_Managed_Arena());
730  /**
731  * @brief Local storage for the search direction.
732  */
733  amrex::FArrayBox p (m_nodal_box,1,amrex::The_Managed_Arena());
734  /**
735  * @brief Local storage for the product of the operator and the search direction.
736  */
737  amrex::FArrayBox Mp(m_nodal_box,1,amrex::The_Managed_Arena());
738 
739  x.setVal<amrex::RunOn::Host>(zero);
740  r.copy<amrex::RunOn::Host>(rhs,0,0,1); // r = rhs - M*0
741  applyJacobi(diag,r,z);
742  p.copy<amrex::RunOn::Host>(z,0,0,1);
743 
744  Real rz = dot(r,z);
745  Real const initial = std::sqrt(dot(r,r));
746  Real const tol = relative_tolerance * std::max({initial,one});
747 
748  SolveInfo info;
749  info.final_residual = initial;
750  if (initial <= tol) {
751  info.converged = true;
752  return info;
753  }
754 
755  for (int it=0; it<max_iterations; ++it) {
756  applyM(p,Mp);
757  Real const pMp = dot(p,Mp);
758  if (!(pMp > zero) || !std::isfinite(pMp)) {
759  throw std::runtime_error("CG curvature failure.");
760  }
761 
762  Real const alpha = rz/pMp;
763  {
764  auto const x_arr = x.array();
765  auto const r_arr = r.array();
766  auto const p_arr = p.const_array();
767  auto const Mp_arr = Mp.const_array();
768  for (int j=m_jlo; j<=m_jhi+1; ++j) {
769  for (int i=m_ilo; i<=m_ihi+1; ++i) {
770  x_arr(i,j,0) += alpha * p_arr(i,j,0);
771  r_arr(i,j,0) -= alpha * Mp_arr(i,j,0);
772  }
773  }
774  }
775 
776  info.iterations = it+1;
777  info.final_residual = std::sqrt(dot(r,r));
778  if (info.final_residual <= tol) {
779  info.converged = true;
780  return info;
781  }
782 
783  applyJacobi(diag,r,z);
784  Real const rz_new = dot(r,z);
785  Real const beta = rz_new/rz;
786  {
787  auto const p_arr = p.array();
788  auto const z_arr = z.const_array();
789  for (int j=m_jlo; j<=m_jhi+1; ++j) {
790  for (int i=m_ilo; i<=m_ihi+1; ++i) {
791  p_arr(i,j,0) = z_arr(i,j,0) + beta * p_arr(i,j,0);
792  }
793  }
794  }
795  rz = rz_new;
796  }
797  return info;
798  }
