52 /**
53  * Decide the face-level spectral advection semantics without performing ERF's
54  * deferred physical-boundary ghost-fill lifecycle. At a physical face whose
55  * high-order stencil is not qualified, the admissible donor proposal remains
56  * available and high-order transport falls back to it.
57  */
58 [[nodiscard]] constexpr AdvectionFacePolicy
59 make_advection_face_policy (const AdvectionFacePolicyInput& input) noexcept
60 {
61  using Action = AdvectionFaceAction;
62  if (input.kind == AdvectionFaceKind::Periodic) {
63  return {Action::PeriodicSharedFace, false, false, true};
64  }
65  if (input.kind == AdvectionFaceKind::ImpermeableWall ||
66  input.outward_carrier == amrex::Real(0.0)) {
67  return {Action::ZeroNormalFlux, false, false, false};
68  }
69 
70  if (input.outward_carrier > amrex::Real(0.0)) {
71  // Outward flow uses the interior spectrum. The donor path is always
72  // defined; high order is used only with a qualified physical stencil.
73  return {Action::InteriorDonor, true, false,
74  input.high_order_stencil_available};
75  }
76 
77  // Reverse flow contradicts a face declared as advective outflow. External
78  // spectral data must not silently change that declared boundary policy.
79  if (input.kind == AdvectionFaceKind::AdvectiveOutflow) {
80  return {Action::RejectInwardOutflow, false, false, false};
81  }
82 
83  // The remaining negative-carrier boundary category is prescribed inflow.
84  // A complete explicit atomic-group spectrum is required; qc/qr cannot
85  // supply missing bin or attached-property composition.
86  if (!input.explicit_spectral_state_present) {
87  return {Action::RejectMissingInflowState, false, false, false};
88  }
89  if (!input.explicit_spectral_group_complete) {
90  return {Action::RejectIncompleteInflowState, false, false, false};
91  }
92  return {Action::ExplicitSpectralInflow, false, true,
93  input.high_order_stencil_available};
94 }
