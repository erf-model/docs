16 /**
17  * @brief Interface between ERF and land-surface and urban models.
18  *
19  * SurfaceModel collects data and fluxes from the LSM and urban models,
20  * performs the land-urban weight averaging and exposes the resulting fields
21  * for use by ERF boundary-condition and physics components.
22  *
23  * Models may register data and mappings independently, without requiring a
24  * fixed ordering or one-to-one correspondence between their fields.
25  *
26  * Model-specific field names can also be registered under a common name for
27  * consumers such as RRTMGP and MOST.
28  *
29  * This class supports its own checkpoint state, ensuring the mappings are
30  * restored correctly on restart. This class also provides a surface plotfile
31  * for weight-averaged model data for the fields registered.
32  */

294  }
295  } else if (type == SurfaceModelType::URBAN) {
296  urban_data_lev[lev].resize(model_data.size());

298  weighted_urban_data_lev[lev].resize(model_data.size());

301  m_urban_names.resize(model_data.size());
302  AMREX_ALWAYS_ASSERT(data_names.size() == model_data.size());

363  /**
364  * @brief Selects the model fields that participate in weighted averaging.
365  *
366  * When using fluxes, the first five elements map to u-stress, v-stress,
367  * heat flux, vapor flux, and surface temperature. Otherwise, the first
368  * four elements map to ustar, tstar, qstar, and surface temperature. A
369  * field index of -1 skips that field.
370  *
371  * @param type Whether the fields belong to the land or urban model.
372  * @param field_indices Indices of the fields to average.
373  * @param use_fluxes Whether the selected fields are fluxes.
374  */
375  void set_model_fields(SurfaceModelType type, const amrex::Vector<int> &field_indices, const bool use_fluxes = true)
376  {
377  const int nfields = static_cast<int>(field_indices.size());
378  if (!use_fluxes) {
379  AMREX_ALWAYS_ASSERT_WITH_MESSAGE(nfields >= 4, "Must have at least four fields to average (ustar, tstar, qstar, tsurf)");
380  } else {
381  AMREX_ALWAYS_ASSERT_WITH_MESSAGE(nfields >= 5, "Must have at least five fields to average (u-stress, v-stress, hfx, qfx, tsurf)");
382  }
383 
384  // Latch the export mode off the first model that registers, then make sure every
385  // subsequent model agrees: either all of them supply fluxes, or all of them supply
386  // u*, t*, q* directly. Taking the mode from the caller rather than assuming the
387  // default is what makes the non-flux path reachable at all.
388  if (!m_export_fluxes_set) {
389  m_export_fluxes = use_fluxes;
390  m_export_fluxes_set = true;
391  }
392  AMREX_ALWAYS_ASSERT_WITH_MESSAGE(use_fluxes == m_export_fluxes,
393  "All registered surface models must export either fluxes or u*/t*/q*, not a mix");
394 
395  const int nregistered = (type == SurfaceModelType::LAND)
396  ? static_cast<int>(m_lsm_names.size())
397  : static_cast<int>(m_urban_names.size());
398  for (const int field : field_indices) {
399  AMREX_ALWAYS_ASSERT_WITH_MESSAGE(
400  field == -1 || (field >= 0 && field < nregistered),
401  "Model field index must be -1 or within the registered provider data range");
402  }
403 
404  if (type == SurfaceModelType::LAND) {
405  lsm_fields = amrex::Vector<int>(field_indices);
406  for (auto& fields_at_level : weighted_lsm_data_lev) {
407  fields_at_level.clear();
408  }
409  } else if (type == SurfaceModelType::URBAN) {
410  urban_fields = amrex::Vector<int>(field_indices);
411  for (auto& fields_at_level : weighted_urban_data_lev) {
412  fields_at_level.clear();
413  }
414  }
415  }

679  /**
680  * @brief Builds the ordered radiation-field list for one AMR level.
681  * @param lev AMR level to update.
682  */
683  void build_radiation_fieldlist(int lev) {
684  amrex::Print() << " --- building radiation field list" << std::endl;
685 
686  // loop over fields and check against names of radiation variables
687  for (int i = 0; i < static_cast<int>(radnames.size()); ++i) {
688  amrex::Print() << " - Checking for radiation variable '" << radnames[i] << "'" << std::endl;
689 
690  amrex::MultiFab* mf = nullptr;
691  auto rad_it = radiation_input_map.find(radnames[i]);
692  if (rad_it != radiation_input_map.end()) {
693  const int land_idx = rad_it->second.map.first;
694  const int urban_idx = rad_it->second.map.second;
695  const bool valid_land = m_use_land && land_idx >= 0 &&
696  lev < static_cast<int>(lsm_data_lev.size()) &&
697  land_idx < static_cast<int>(lsm_data_lev[lev].size()) &&
698  lsm_data_lev[lev][land_idx] != nullptr;
699  const bool valid_urban = m_use_urban && urban_idx >= 0 &&
700  lev < static_cast<int>(urban_data_lev.size()) &&
701  urban_idx < static_cast<int>(urban_data_lev[lev].size()) &&
702  urban_data_lev[lev][urban_idx] != nullptr;
703  if (valid_land && valid_urban) {
704  if (!rad_it->second.weighted[lev]) {
705  rad_it->second.weighted[lev] = std::make_unique<amrex::MultiFab>(
706  m_ba2d[lev], m_dmap[lev], 1, amrex::IntVect(1,1,0));
707  }
708  auto& weighted = *rad_it->second.weighted[lev];
709  const auto& land = *lsm_data_lev[lev][land_idx];
710  const auto& urban = *urban_data_lev[lev][urban_idx];
711  for (amrex::MFIter mfi(weighted); mfi.isValid(); ++mfi) {
712  const amrex::Box bx = mfi.tilebox();
713  const auto weights = wavg[lev]->const_array(mfi);
714  const auto land_arr = land.const_array(mfi);
715  const auto urban_arr = urban.const_array(mfi);
716  // LSM surface fields may be stored at k <= 0; use the top valid LSM
717  // plane, which matches k=0 when the LSM keeps those fields synchronized
718  // at the surface (such as SLM). Urban surface fields are stored at k=0.
719  const int land_k = land.box(mfi.index()).bigEnd(2);
720  const int urban_k = urban.box(mfi.index()).smallEnd(2);
721  auto output = weighted.array(mfi);
722  amrex::ParallelFor(bx, [=] AMREX_GPU_DEVICE (int i, int j, int k) {
723  output(i,j,k) = land_arr(i,j,land_k) *
724  weights(i,j,0,SurfaceModelType::LAND) +
725  urban_arr(i,j,urban_k) *
726  weights(i,j,0,SurfaceModelType::URBAN);
727  });
728  }
729  weighted.FillBoundary(m_geom[lev].periodicity());
730  mf = rad_it->second.weighted[lev].get();
731  } else if (valid_land) {
732  mf = lsm_data_lev[lev][land_idx];
733  } else if (valid_urban) {
734  mf = urban_data_lev[lev][urban_idx];
735  }
736  }
737  if (mf) {
738  rad_fields[lev].push_back(mf);
739  } else {
740  amrex::Print() << " - WARNING: did not find radiation variable, setting to null to use default RRTMGP value!" << std::endl;
741  rad_fields[lev].push_back(nullptr);
742  }
743  }
744  AMREX_ALWAYS_ASSERT_WITH_MESSAGE(
745  static_cast<int>(rad_fields[lev].size()) == static_cast<int>(radnames.size()),
746  "Did not find all radiation fields");
747  }
